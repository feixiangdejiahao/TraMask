+// defaultParamInfo records information about a single function parameter while
+// scanning source before it is handed to the generated parser.
+type defaultParamInfo struct {
+	ExprText   string
+	HasExpr    bool
+	IsVariadic bool
+}
+
+// defaultRewrite describes a region of the source that should be removed so
+// that "name = expression" becomes just "name".
+type defaultRewrite struct {
+	eqOffset  int
+	endOffset int
+}
+
+// funcDefaultInfo records a function definition that contains default parameter
+// values.
+type funcDefaultInfo struct {
+	origOffset int
+	infos      []defaultParamInfo
+	rews       []defaultRewrite
+}

+// defaultScanner is a small scanner used to find function definitions and
+// rewrite default parameter declarations before parsing.
+type defaultScanner struct {
+	src    []rune
+	pos    int
+	starts []int
+}
+
+func newDefaultScanner(src string) *defaultScanner {
+	runes := []rune(src)
+	return &defaultScanner{src: runes, starts: lineStarts(runes)}
+}
+
+func (s *defaultScanner) positionAt(offset int) ast.Position {
+	return positionAtStarts(s.starts, offset)
+}
+
+func lineStarts(runes []rune) []int {
+	starts := []int{0}
+	for i, r := range runes {
+		if r == '\n' {
+			starts = append(starts, i+1)
+		}
+	}
+	return starts
+}
+
+func positionAtStarts(starts []int, offset int) ast.Position {
+	lo, hi := 0, len(starts)-1
+	for lo < hi {
+		mid := (lo + hi + 1) / 2
+		if starts[mid] <= offset {
+			lo = mid
+		} else {
+			hi = mid - 1
+		}
+	}
+	return ast.Position{Line: lo + 1, Column: offset - starts[lo] + 1}
+}
+
+func (s *defaultScanner) eof() bool {
+	return s.pos >= len(s.src)
+}
+
+func (s *defaultScanner) peek() rune {
+	return s.peekAt(0)
+}
+
+func (s *defaultScanner) peekAt(offset int) rune {
+	i := s.pos + offset
+	if i < 0 || i >= len(s.src) {
+		return -1
+	}
+	return s.src[i]
+}

+func (s *defaultScanner) next() rune {
+	if s.eof() {
+		return -1
+	}
+	r := s.src[s.pos]
+	s.pos++
+	return r
+}

+func (s *defaultScanner) skipSpace() {
+	for !s.eof() {
+		switch s.peek() {
+		case ' ', '\t', '\r', '\n':
+			s.next()
+		default:
+			return
+		}
+	}
+}
+
+func (s *defaultScanner) skipComment() bool {
+	if s.eof() {
+		return false
+	}
+	switch {
+	case s.peek() == '#':
+		for !s.eof() && s.peek() != '\n' {
+			s.next()
+		}
+		return true
+	case s.peek() == '/' && s.peekAt(1) == '/':
+		for !s.eof() && s.peek() != '\n' {
+			s.next()
+		}
+		return true
+	case s.peek() == '/' && s.peekAt(1) == '*':
+		s.next()
+		s.next()
+		for !s.eof() {
+			if s.peek() == '*' && s.peekAt(1) == '/' {
+				s.next()
+				s.next()
+				return true
+			}
+			s.next()
+		}
+		return true
+	}
+	return false
+}
+
+func (s *defaultScanner) skipSpaceAndComments() {
+	for {
+		s.skipSpace()
+		if !s.skipComment() {
+			return
+		}
+	}
+}
+
+func (s *defaultScanner) skipString() {
+	if s.eof() {
+		return
+	}
+	quote := s.peek()
+	if quote != '"' && quote != '\'' && quote != '`' {
+		return
+	}
+	s.next()
+	if quote == '`' {
+		for !s.eof() {
+			if s.next() == '`' {
+				return
+			}
+		}
+		return
+	}
+	for !s.eof() {
+		r := s.next()
+		if r == '\\' {
+			if !s.eof() {
+				s.next()
+			}
+			continue
+		}
+		if r == quote {
+			return
+		}
+	}
+}

+func (s *defaultScanner) readIdent() string {
+	start := s.pos
+	for !s.eof() && (isLetter(s.peek()) || isDigit(s.peek())) {
+		s.next()
+	}
+	return string(s.src[start:s.pos])
+}

+// preprocessDefaultParams rewrites default parameter declarations in function
+// parameter lists into plain parameter lists and returns the rewritten source
+// together with the parsed default expressions keyed by the position of the
+// enclosing function expression in the rewritten source.
+func preprocessDefaultParams(src string) (string, map[ast.Position][]ast.Expr, error) {
+	if !strings.Contains(src, "func") {
+		return src, nil, nil
+	}
+	s := newDefaultScanner(src)
+	runes := s.src
+	var funcs []funcDefaultInfo
+
+	for !s.eof() {
+		s.skipSpaceAndComments()
+		if s.eof() {
+			break
+		}
+		ch := s.peek()
+		if ch == '"' || ch == '\'' || ch == '`' {
+			s.skipString()
+			continue
+		}
+		if isLetter(ch) {
+			idStart := s.pos
+			id := s.readIdent()
+			if id == "func" {
+				funcPos := s.positionAt(idStart)
+				handled, infos, rews, err := s.processFunc(funcPos)
+				if err != nil {
+					return "", nil, err
+				}
+				if handled && len(rews) > 0 {
+					funcs = append(funcs, funcDefaultInfo{origOffset: idStart, infos: infos, rews: rews})
+				}
+			}
+			continue
+		}
+		s.next()
+	}
+
+	// Remove "= expression" text from the source so the existing parser can
+	// parse the parameter list.
+	var allRanges []defaultRewrite
+	for _, f := range funcs {
+		allRanges = append(allRanges, f.rews...)
+	}
+	sort.Slice(allRanges, func(i, j int) bool {
+		return allRanges[i].eqOffset < allRanges[j].eqOffset
+	})
+	newRunes := make([]rune, 0, len(runes))
+	pos := 0
+	for _, rw := range allRanges {
+		if rw.eqOffset > pos {
+			newRunes = append(newRunes, runes[pos:rw.eqOffset]...)
+		}
+		pos = rw.endOffset
+	}
+	if pos < len(runes) {
+		newRunes = append(newRunes, runes[pos:]...)
+	}
+	newSrc := string(newRunes)
+
+	// Compute the rewritten position of each function and parse its defaults.
+	newStarts := lineStarts(newRunes)
+	defaults := make(map[ast.Position][]ast.Expr)
+	for _, f := range funcs {
+		delta := 0
+		for _, rw := range allRanges {
+			if rw.endOffset <= f.origOffset {
+				delta += rw.endOffset - rw.eqOffset
+			}
+		}
+		newOffset := f.origOffset - delta
+		newPos := positionAtStarts(newStarts, newOffset)
+		defs := make([]ast.Expr, len(f.infos))
+		for i, info := range f.infos {
+			if info.HasExpr {
+				expr, err := parseDefaultExpr(info.ExprText)
+				if err != nil {
+					return "", nil, err
+				}
+				defs[i] = expr
+			}
+		}
+		defaults[newPos] = defs
+	}
+
+	return newSrc, defaults, nil
+}

+func parseDefaultExpr(text string) (ast.Expr, error) {
+	stmt, err := ParseSrc(text)
+	if err != nil {
+		return nil, err
+	}
+	switch stmt := stmt.(type) {
+	case *ast.StmtsStmt:
+		if len(stmt.Stmts) == 1 {
+			if es, ok := stmt.Stmts[0].(*ast.ExprStmt); ok {
+				return es.Expr, nil
+			}
+		}
+	case *ast.ExprStmt:
+		return stmt.Expr, nil
+	}
+	return nil, &Error{Message: "invalid default argument declaration"}
+}

+// processFunc parses the parameter list of a function definition. s.pos must be
+// immediately after the "func" keyword. It returns whether a parameter list was
+// successfully processed, the parameter information, and the rewrite ranges.
+func (s *defaultScanner) processFunc(funcPos ast.Position) (bool, []defaultParamInfo, []defaultRewrite, error) {
+	s.skipSpaceAndComments()
+	if !s.eof() && isLetter(s.peek()) {
+		// named function: skip the name
+		s.readIdent()
+		s.skipSpaceAndComments()
+	}
+	if s.eof() || s.peek() != '(' {
+		return false, nil, nil, nil
+	}
+	s.next() // consume '('
+
+	var infos []defaultParamInfo
+	var rews []defaultRewrite
+	seenDefault := false
+
+	for {
+		s.skipSpaceAndComments()
+		if s.eof() {
+			return false, nil, nil, nil
+		}
+		if s.peek() == ')' {
+			s.next()
+			break
+		}
+		if !isLetter(s.peek()) {
+			return false, nil, nil, nil
+		}
+
+		s.readIdent()
+		infos = append(infos, defaultParamInfo{})
+		idx := len(infos) - 1
+
+		s.skipSpaceAndComments()
+
+		// variadic "..."
+		if s.peek() == '.' && s.peekAt(1) == '.' && s.peekAt(2) == '.' {
+			infos[idx].IsVariadic = true
+			s.next()
+			s.next()
+			s.next()
+			s.skipSpaceAndComments()
+			if s.eof() {
+				return false, nil, nil, nil
+			}
+			if s.peek() == ')' {
+				s.next()
+				break
+			}
+			// variadic parameter must be last; leave malformed source to the parser
+			return false, nil, nil, nil
+		}
+
+		// default value "= expression"
+		if s.peek() == '=' && s.peekAt(1) != '=' {
+			eqOffset := s.pos
+			s.next() // consume '='
+			exprStart := s.pos
+			depth := 0
+		exprLoop:
+			for !s.eof() {
+				c := s.peek()
+				if c == '"' || c == '\'' || c == '`' {
+					s.skipString()
+					continue
+				}
+				if c == '#' || (c == '/' && (s.peekAt(1) == '/' || s.peekAt(1) == '*')) {
+					s.skipComment()
+					continue
+				}
+				switch c {
+				case '(', '[', '{':
+					depth++
+				case ')', ']', '}':
+					if depth == 0 && c == ')' {
+						break exprLoop
+					}
+					depth--
+					if depth < 0 {
+						return false, nil, nil, nil
+					}
+				case ',':
+					if depth == 0 {
+						break exprLoop
+					}
+				}
+				s.next()
+			}
+			exprEnd := s.pos
+			exprText := strings.TrimSpace(string(s.src[exprStart:exprEnd]))
+			if exprText == "" {
+				return false, nil, nil, nil
+			}
+			// A default value followed by "..." means the variadic parameter
+			// itself declares a default, which is not allowed.
+			if strings.HasSuffix(exprText, "...") {
+				return false, nil, nil, &Error{Message: "invalid default argument declaration", Pos: funcPos}
+			}
+			infos[idx].HasExpr = true
+			infos[idx].ExprText = exprText
+			rews = append(rews, defaultRewrite{eqOffset: eqOffset, endOffset: exprEnd})
+		}
+
+		s.skipSpaceAndComments()
+		if s.eof() {
+			return false, nil, nil, nil
+		}
+		if s.peek() == ',' {
+			s.next()
+			continue
+		}
+		if s.peek() == ')' {
+			s.next()
+			break
+		}
+		return false, nil, nil, nil
+	}
+
+	// Validate declarations.
+	for _, info := range infos {
+		if info.IsVariadic {
+			if info.HasExpr {
+				return false, nil, nil, &Error{Message: "invalid default argument declaration", Pos: funcPos}
+			}
+			continue
+		}
+		if info.HasExpr {
+			seenDefault = true
+		} else if seenDefault {
+			return false, nil, nil, &Error{Message: "invalid default argument declaration", Pos: funcPos}
+		}
+	}
+
+	return true, infos, rews, nil
+}

+// attachFuncDefaults attaches parsed default expressions to FuncExpr nodes that
+// were found during preprocessing.
+func attachFuncDefaults(stmt ast.Stmt, defaults map[ast.Position][]ast.Expr) error {
+	if len(defaults) == 0 {
+		return nil
+	}
+	walkStmtForDefaults(stmt, func(funcExpr *ast.FuncExpr) {
+		if defs, ok := defaults[funcExpr.Position()]; ok {
+			funcExpr.Defaults = defs
+		}
+	})
+	return nil
+}

+func walkStmtForDefaults(stmt ast.Stmt, f func(*ast.FuncExpr)) {
+	if stmt == nil || f == nil {
+		return
+	}
+	switch stmt := stmt.(type) {
+	case *ast.StmtsStmt:
+		for _, s := range stmt.Stmts {
+			walkStmtForDefaults(s, f)
+		}
+	case *ast.ExprStmt:
+		walkExprForDefaults(stmt.Expr, f)
+	case *ast.VarStmt:
+		walkExprsForDefaults(stmt.Exprs, f)
+	case *ast.LetsStmt:
+		walkExprsForDefaults(stmt.RHSS, f)
+		walkExprsForDefaults(stmt.LHSS, f)
+	case *ast.LetMapItemStmt:
+		walkExprForDefaults(stmt.RHS, f)
+		walkExprsForDefaults(stmt.LHSS, f)
+	case *ast.IfStmt:
+		walkExprForDefaults(stmt.If, f)
+		walkStmtForDefaults(stmt.Then, f)
+		for _, s := range stmt.ElseIf {
+			walkStmtForDefaults(s, f)
+		}
+		walkStmtForDefaults(stmt.Else, f)
+	case *ast.TryStmt:
+		walkStmtForDefaults(stmt.Try, f)
+		walkStmtForDefaults(stmt.Catch, f)
+		walkStmtForDefaults(stmt.Finally, f)
+	case *ast.ForStmt:
+		walkExprForDefaults(stmt.Value, f)
+		walkStmtForDefaults(stmt.Stmt, f)
+	case *ast.CForStmt:
+		walkStmtForDefaults(stmt.Stmt1, f)
+		walkExprForDefaults(stmt.Expr2, f)
+		walkExprForDefaults(stmt.Expr3, f)
+		walkStmtForDefaults(stmt.Stmt, f)
+	case *ast.LoopStmt:
+		walkExprForDefaults(stmt.Expr, f)
+		walkStmtForDefaults(stmt.Stmt, f)
+	case *ast.ReturnStmt:
+		walkExprsForDefaults(stmt.Exprs, f)
+	case *ast.ThrowStmt:
+		walkExprForDefaults(stmt.Expr, f)
+	case *ast.ModuleStmt:
+		walkStmtForDefaults(stmt.Stmt, f)
+	case *ast.SwitchStmt:
+		walkExprForDefaults(stmt.Expr, f)
+		for _, c := range stmt.Cases {
+			if cc, ok := c.(*ast.SwitchCaseStmt); ok {
+				walkExprsForDefaults(cc.Exprs, f)

+			}
+		}

+	case *ast.DeleteStmt:
+		walkExprForDefaults(stmt.Item, f)

+	case *ast.CloseStmt:

+	case *ast.ChanStmt:

+		walkExprForDefaults(stmt.OkExpr, f)

+	}

+	case *ast.AddrExpr:
+		walkExprForDefaults(expr.Expr, f)
+	case *ast.DerefExpr:
+		walkExprForDefaults(expr.Expr, f)
+	case *ast.ParenExpr:
+		walkExprForDefaults(expr.SubExpr, f)
+	case *ast.NilCoalescingOpExpr:
+		walkExprForDefaults(expr.LHS, f)
+		walkExprForDefaults(expr.RHS, f)
+	case *ast.TernaryOpExpr:

+		walkExprForDefaults(expr.Expr, f)

+		walkExprForDefaults(expr.Item, f)

+		walkExprForDefaults(expr.LenExpr, f)

+		walkExprForDefaults(expr.Type, f)
+	case *ast.LenExpr:

+		walkExprForDefaults(expr.ItemExpr, f)

+	}

+func walkOperatorForDefaults(op ast.Operator, f func(*ast.FuncExpr)) {
+	if op == nil {
+		return
+	}
+	switch op := op.(type) {
+	case *ast.BinaryOperator:

+		walkExprForDefaults(op.RHS, f)

+		walkExprForDefaults(op.RHS, f)
+	case *ast.AddOperator:
+		walkExprForDefaults(op.LHS, f)
+		walkExprForDefaults(op.RHS, f)
+	case *ast.MultiplyOperator:
+		walkExprForDefaults(op.LHS, f)
+		walkExprForDefaults(op.RHS, f)
+	}

+// providedArgCount returns the number of arguments explicitly provided to a VM
+// function call. Omitted parameters are represented by invalid reflect.Value
+// placeholders, and variadic elements are counted individually.
+func providedArgCount(in []reflect.Value, numParams int, varArg bool) int {
+	count := 0
+	for i := 0; i < numParams; i++ {
+		if varArg && i == numParams-1 {
+			if in[i+1].IsValid() {
+				count += in[i+1].Len()
+			}
+			continue
+		}
+		if in[i+1].Interface().(reflect.Value).IsValid() {
+			count++
+		}
+	}
+	return count
+}
