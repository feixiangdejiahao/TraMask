+## Initializing the container
+
+Registrations created with `asFunction()` or `asClass()` can be given an
+`initializer` function. The container can then be initialized asynchronously;
+all initializers are run in dependency order, organized into levels where all
+services at a level complete before the next level begins. Within a level,
+initializers run in parallel (bounded by the `concurrency` option).
+
+```js
+const container = createContainer()
+
+container.register({
+  database: asClass(DatabasePool)
+    .singleton()
+    .initializer(async (instance) => {
+      await instance.connect()
+      return instance
+    }),
+})
+
+const result = await container.initialize({ concurrency: 5 })
+console.log(result.totalDuration)
+console.log(result.metrics.database.duration)
+console.log(result.metrics.database.level)
+```
+
+Once initialized, the container is considered initialized and calling
+`initialize()` again returns immediately. Resolving a registration with an
+`initializer` before it has been initialized throws an
+`AwilixNotInitializedError`. If an initializer throws or rejects, all already
+initialized services are disposed in reverse order and an
+`AwilixInitializationError` is thrown (the original error is available via
+`err.cause`).
+
+Scoped containers can be initialized independently; parent container singletons
+are not reinitialized.
+

+  /**

+   * into dependency levels. Throws AwilixResolutionError if a cycle is detected.
+   */
+  function buildInitializationPlan(): Array<InitializationLevel> {

+    const allNames = [

+      ...Object.getOwnPropertySymbols(allRegistrations),
+    ]

+    // re-initialized when a scoped container is initialized.
+    const isParentOwnedSingleton = (
+      name: string | symbol,
+      resolver: Resolver<any>,

+      !!parentContainer &&

+      !Object.prototype.hasOwnProperty.call(registrations, name)

+      if (
+        resolver &&
+        typeof (resolver as any)[INITIALIZER] === 'function' &&
+        !isParentOwnedSingleton(name, resolver)
+      ) {
+        entries.push({ name, resolver })
+      }
+    }
+
+    if (entries.length === 0) {
+      return []
+    }
+
+    const byName = new Map<string | symbol, InitializationEntry>()
+    const initializedNames = new Set<string | symbol>()
+    for (const entry of entries) {
+      byName.set(entry.name, entry)
+      initializedNames.add(entry.name)
+    }
+
+    // Build a graph over all registrations visible to this container. Edges go
+    // from a registration to the registrations it depends on.
+    const graph = new Map<string | symbol, Array<string | symbol>>()
+    for (const name of allNames) {
+      const resolver = allRegistrations[name as any] as Resolver<any>
+      graph.set(
+        name,
+        resolver
+          ? getResolverDependencies(resolver).filter((depName) =>
+              Object.prototype.hasOwnProperty.call(allRegistrations, depName),
+            )
+          : [],
+      )
+    }
+
+    // For each initialized registration, compute the set of initialized
+    // registrations it depends on (directly or through non-initialized
+    // registrations). Cycles are detected while traversing.
+    const initializedAncestors = new Map<
+      string | symbol,
+      Set<string | symbol>
+    >()
+
+    const collectAncestors = (
+      name: string | symbol,
+      visiting: Set<string | symbol>,
+    ): Set<string | symbol> => {
+      const cached = initializedAncestors.get(name)
+      if (cached) {
+        return cached
+      }
+      if (visiting.has(name)) {
+        throw new AwilixResolutionError(
+          name,
+          [],
+          'Cyclic dependencies detected.',
+        )
+      }
+      visiting.add(name)
+      const result = new Set<string | symbol>()
+      for (const depName of graph.get(name) || []) {
+        if (visiting.has(depName)) {
+          throw new AwilixResolutionError(
+            depName,
+            [],
+            'Cyclic dependencies detected.',
+          )
+        }
+        if (initializedNames.has(depName)) {
+          result.add(depName)
+        }
+        const depAncestors = collectAncestors(depName, visiting)
+        for (const ancestor of depAncestors) {
+          result.add(ancestor)
+        }
+      }
+      visiting.delete(name)
+      initializedAncestors.set(name, result)
+      return result
+    }
+
+    for (const entry of entries) {
+      collectAncestors(entry.name, new Set<string | symbol>())
+    }
+
+    // Compute levels: an initialized registration is in the current level once
+    // all of its initialized ancestors have been processed.
+    const levels: Array<InitializationLevel> = []
+    const processed = new Set<string | symbol>()
+    const remaining = new Set<string | symbol>(entries.map((e) => e.name))
+
+    while (remaining.size > 0) {
+      const levelEntries: Array<InitializationEntry> = []
+      for (const name of remaining) {
+        const ancestors = initializedAncestors.get(name) || new Set()
+        let ready = true
+        for (const ancestor of ancestors) {
+          if (!processed.has(ancestor)) {
+            ready = false
+            break
+          }
+        }
+        if (ready) {
+          levelEntries.push(byName.get(name)!)
+        }
+      }
+      if (levelEntries.length === 0) {
+        // Defensive; cycles are already detected above.
+        const first = remaining.values().next().value as string | symbol
+        throw new AwilixResolutionError(
+          first,
+          [],
+          'Cyclic dependencies detected.',
+        )
+      }
+      for (const entry of levelEntries) {
+        processed.add(entry.name)
+        remaining.delete(entry.name)
+      }
+      levels.push({ entries: levelEntries })
+    }
+
+    return levels
+  }

+  /**
+   * Initializes the container. Resolves all registrations with an `initializer`
+   * in dependency order, running initializers within a level concurrently.
+   */
+  function initialize(
+    options: InitializeOptions = {},
+  ): Promise<InitializeResult> {
+    if (initializationState.status === 'success') {
+      return Promise.resolve(initializationState.result)
+    }

+      )
+    }

+      return initializationState.promise
+    }

+      // the container as failed, allowing initialize() to be retried.
+      return Promise.reject(err)

+        throw err
+      },

+  }

+  /**
+   * Runs a level of initializers, waiting for all in-flight initializers to
+   * complete before throwing. If any initializer fails, the first error is
+   * rethrown after all tasks have settled.
+   */
+  async function runLevel<T>(
+    entries: Array<T>,
+    concurrency: number,
+    task: (entry: T) => Promise<void>,
+  ): Promise<void> {
+    if (entries.length === 0) {
+      return
+    }
+    let index = 0
+    const errors: unknown[] = []
+    const safeConcurrency =
+      Number.isFinite(concurrency) && concurrency > 0 ? concurrency : 1
+    const workerCount = Math.max(1, Math.min(safeConcurrency, entries.length))
+    const workers = Array.from({ length: workerCount }, async () => {
+      while (index < entries.length) {
+        const current = index++
+        try {
+          await task(entries[current])
+        } catch (err) {
+          errors.push(err)
+        }
+      }
+    })
+    await Promise.all(workers)
+    if (errors.length > 0) {
+      throw errors[0]
+    }
+  }

+  /**

+   * are ignored so they never override the original initialization error.
+   */

+    stack: Array<InitializationEntry & { value: any }>,
+  ): Promise<void> {
+    for (let i = stack.length - 1; i >= 0; i--) {
+      const { name, resolver, value } = stack[i]

+        }

+      clearInitializedCacheValue(name)
+    }
+  }

+              value = result === undefined ? instance : result

+              resolver: entry.resolver,
+              value,
+            })
+            setInitializedCacheValue(entry.name, entry.resolver, value)

+            metrics[entry.name] = {
+              duration: performance.now() - metricStart,
+              level: levelIndex,
+            }
+          },
+        )
+      }

+        metrics,
+      }
+    } catch (err) {
+      await rollbackInitialized(initializedStack)
+      throw err
+    }
+  }
