+export type ErrorStackMode = 'off' | 'string' | 'frames';
+export type StripInternalFrames =
+  | 'none'
+  | 'node'
+  | 'superjson'
+  | 'node_and_superjson';
+export type RedactPaths = 'none' | 'basename' | 'strip_cwd';
+export type IncludeCauses = 'none' | 'direct' | 'deep';
+
+export interface ErrorStackOptions {
+  mode?: ErrorStackMode;
+  normalizeNewlines?: boolean;
+  trimLeadingWhitespace?: boolean;
+  maxStackLines?: number;
+  stripInternalFrames?: StripInternalFrames;
+  redactPaths?: RedactPaths;
+  includeCauses?: IncludeCauses;
+  maxCauseDepth?: number;
+  sanitizeMessage?: boolean;
+  classFilter?: string | string[];
+}
+
+export interface NormalizedErrorStackOptions {
+  mode: ErrorStackMode;
+  normalizeNewlines: boolean;
+  trimLeadingWhitespace: boolean;
+  maxStackLines: number | undefined;
+  stripInternalFrames: StripInternalFrames;
+  redactPaths: RedactPaths;
+  includeCauses: IncludeCauses;
+  maxCauseDepth: number;
+  sanitizeMessage: boolean;
+  classFilter: string[];
+}
+
+const MODES = new Set<string>(['off', 'string', 'frames']);
+const STRIP_MODES = new Set<string>([
+  'none',
+  'node',
+  'superjson',
+  'node_and_superjson',
+]);
+const REDACT_MODES = new Set<string>(['none', 'basename', 'strip_cwd']);
+const INCLUDE_CAUSES_MODES = new Set<string>(['none', 'direct', 'deep']);
+
+export function normalizeErrorStackOptions(
+  options: unknown
+): NormalizedErrorStackOptions | undefined {
+  if (
+    typeof options !== 'object' ||
+    options === null ||
+    Array.isArray(options)
+  ) {
+    return undefined;
+  }
+
+  const raw = options as Record<string, unknown>;
+
+  let mode: ErrorStackMode = 'off';
+  if (typeof raw.mode === 'string' && MODES.has(raw.mode)) {
+    mode = raw.mode as ErrorStackMode;
+  }
+
+  let maxStackLines: number | undefined;
+  if (raw.maxStackLines !== undefined) {
+    if (
+      typeof raw.maxStackLines !== 'number' ||
+      !Number.isInteger(raw.maxStackLines) ||
+      raw.maxStackLines <= 0
+    ) {
+      // zero, negative, or non-integer values make the config behave like mode=off
+      mode = 'off';
+    } else {
+      maxStackLines = raw.maxStackLines as number;
+    }
+  }
+
+  let includeCauses: IncludeCauses = 'none';
+  if (
+    typeof raw.includeCauses === 'string' &&
+    INCLUDE_CAUSES_MODES.has(raw.includeCauses)
+  ) {
+    includeCauses = raw.includeCauses as IncludeCauses;
+  }
+
+  let maxCauseDepth = 16;
+  if (raw.maxCauseDepth !== undefined) {
+    if (
+      typeof raw.maxCauseDepth !== 'number' ||
+      !Number.isInteger(raw.maxCauseDepth)
+    ) {
+      includeCauses = 'none';
+    } else {
+      maxCauseDepth = raw.maxCauseDepth as number;
+    }
+  }
+
+  let stripInternalFrames: StripInternalFrames = 'none';
+  if (
+    typeof raw.stripInternalFrames === 'string' &&
+    STRIP_MODES.has(raw.stripInternalFrames)
+  ) {
+    stripInternalFrames = raw.stripInternalFrames as StripInternalFrames;
+  }
+
+  let redactPaths: RedactPaths = 'none';
+  if (
+    typeof raw.redactPaths === 'string' &&
+    REDACT_MODES.has(raw.redactPaths)
+  ) {
+    redactPaths = raw.redactPaths as RedactPaths;
+  }
+
+  let classFilter: string[] = [];
+  if (Array.isArray(raw.classFilter)) {
+    classFilter = raw.classFilter.filter(
+      (value): value is string => typeof value === 'string'
+    );
+  } else if (typeof raw.classFilter === 'string') {
+    classFilter = [raw.classFilter];
+  }
+
+  return Object.freeze({
+    mode,
+    normalizeNewlines:
+      typeof raw.normalizeNewlines === 'boolean'
+        ? raw.normalizeNewlines
+        : false,
+    trimLeadingWhitespace:
+      typeof raw.trimLeadingWhitespace === 'boolean'
+        ? raw.trimLeadingWhitespace
+        : true,
+    maxStackLines,
+    stripInternalFrames,
+    redactPaths,
+    includeCauses,
+    maxCauseDepth,
+    sanitizeMessage:
+      typeof raw.sanitizeMessage === 'boolean' ? raw.sanitizeMessage : false,
+    classFilter,
+  });
+}

+import {

+export function normalizeStackNewlines(stack: string): string {
+  return stack.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
+}
+
+const defaultOptions = normalizeErrorStackOptions({})!;
+
+const SUPERJSON_INTERNAL_MARKERS = [
+  'src/transformer.ts',
+  'src/plainer.ts',
+  'src/index.ts',
+];
+
+function trimLeadingWhitespace(line: string): string {
+  return line.replace(/^\s+/, '');
+}
+
+function redactPath(line: string, mode: RedactPaths): string {
+  if (mode === 'none') {
+    return line;
+  }
+
+  if (mode === 'basename') {
+    // Replace "path/to/file.ts:1:2" with "file.ts:1:2" inside parentheses.
+    return line.replace(/\(([^()]*?)([^/\\()]+)\)/g, '($2)');
+  }
+
+  // strip_cwd - remove the current working directory prefix.
+  const cwd = process.cwd();
+  return line.split(cwd).join('');
+}
+
+function stripInternalFrames(
+  frames: string[],
+  mode: StripInternalFrames
+): string[] {
+  if (mode === 'none') {
+    return frames;
+  }
+
+  return frames.filter((frame) => {
+    if (
+      (mode === 'node' || mode === 'node_and_superjson') &&
+      frame.includes('node:internal')
+    ) {
+      return false;
+    }
+
+    if (
+      (mode === 'superjson' || mode === 'node_and_superjson') &&
+      SUPERJSON_INTERNAL_MARKERS.some((marker) => frame.includes(marker))
+    ) {
+      return false;
+    }
+
+    return true;
+  });
+}
+
+function applyMaxStackLines(
+  frames: string[],
+  maxStackLines: number | undefined
+): string[] {
+  if (maxStackLines === undefined) {
+    return frames;
+  }
+
+  // maxStackLines counts the header line, which is kept separately.
+  const maxFrames = Math.max(0, maxStackLines - 1);
+  return frames.slice(0, maxFrames);
+}
+
+function splitStack(stack: string, normalizeNewlines: boolean): string[] {
+  const normalized = normalizeNewlines ? normalizeStackNewlines(stack) : stack;
+  const lines = normalized.split('\n');
+
+  // Drop trailing empty lines so a trailing CRLF/CR/LF does not create an
+  // artificial empty frame.
+  while (lines.length > 1 && lines[lines.length - 1] === '') {
+    lines.pop();
+  }
+
+  return lines;
+}
+
+export function processStackString(
+  stack: string,
+  options: ErrorStackOptions
+): string {
+  const normalizedOptions =
+    normalizeErrorStackOptions(options) ?? defaultOptions;
+  const lines = splitStack(stack, normalizedOptions.normalizeNewlines);
+  const header = lines[0] ?? '';
+  let frames = lines.slice(1);
+
+  if (normalizedOptions.trimLeadingWhitespace) {
+    frames = frames.map(trimLeadingWhitespace);
+  }
+
+  if (normalizedOptions.redactPaths !== 'none') {
+    frames = frames.map((frame) => redactPath(frame, normalizedOptions.redactPaths));
+  }
+
+  frames = applyMaxStackLines(frames, normalizedOptions.maxStackLines);
+
+  frames = stripInternalFrames(frames, normalizedOptions.stripInternalFrames);
+
+  return [header, ...frames].join('\n');
+}
+
+export function processStackFrames(
+  stack: string,
+  options: ErrorStackOptions
+): { raw: string }[] {
+  const normalizedOptions =
+    normalizeErrorStackOptions(options) ?? defaultOptions;
+  const lines = splitStack(stack, normalizedOptions.normalizeNewlines);
+  const header = lines[0] ?? '';
+  let frames = lines.slice(1);
+
+  if (normalizedOptions.trimLeadingWhitespace) {
+    frames = frames.map(trimLeadingWhitespace);
+  }
+
+  frames = stripInternalFrames(frames, normalizedOptions.stripInternalFrames);
+
+  if (normalizedOptions.redactPaths !== 'none') {
+    frames = frames.map((frame) => redactPath(frame, normalizedOptions.redactPaths));
+  }
+
+  frames = applyMaxStackLines(frames, normalizedOptions.maxStackLines);
+
+  return [{ raw: header }, ...frames.map((frame) => ({ raw: frame }))];
+}
