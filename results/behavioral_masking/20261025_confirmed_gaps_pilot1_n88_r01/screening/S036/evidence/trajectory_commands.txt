type (
	// UDPSession defines a KCP session implemented by UDP
	UDPSession struct {
		conn    net.PacketConn // the underlying packet connection
		ownConn bool           // true if we created conn internally, false if provided by caller
		kcp     *KCP           // KCP ARQ protocol
		l       *Listener      // pointing to the Listener object if it's been accepted by a Listener
		block   BlockCrypt     // block encryption object

		// kcp receiving is based on packets
		// recvbuf turns packets into stream
		recvbuf []byte
		bufptr  []byte

		// FEC codec
		fecDecoder *fecDecoder
		fecEncoder *fecEncoder

		// settings
		remote     net.Addr     // remote peer address
		rd         atomic.Value // read deadline
		wd         atomic.Value // write deadline
		headerSize int          // the header size additional to a KCP frame
		ackNoDelay bool         // send ack immediately for each incoming packet(testing purpose)
		writeDelay bool         // delay kcp.flush() for Write() for bulk transfer
		dup        int          // duplicate udp packets(testing purpose)

		// notifications
		die          chan struct{} // notify current session has Closed
		dieOnce      sync.Once
		chReadEvent  chan struct{} // notify Read() can be called without blocking
		chWriteEvent chan struct{} // notify Write() can be called without blocking

		// socket error handling
		socketReadError      atomic.Value
		socketWriteError     atomic.Value
		chSocketReadError    chan struct{}
		chSocketWriteError   chan struct{}
		socketReadErrorOnce  sync.Once
		socketWriteErrorOnce sync.Once

		// packets waiting to be sent on wire
		chPostProcessing chan sendRequest

		// platform-dependent optimizations
		platform platform

		// rate limiter (bytes per second)
		rateLimiter atomic.Value

		mu sync.Mutex

		// callbackForOOB is an optional callback for handling received out-of-band (OOB) data.
		//
		// OOB data bypasses the KCP reliable data path and is delivered unreliably.
		// The callback is invoked synchronously from the KCP input processing path.
		callbackForOOB atomic.Value
	}

	setReadBuffer interface {
		SetReadBuffer(bytes int) error
	}

	setWriteBuffer interface {
		SetWriteBuffer(bytes int) error
	}

	setDSCP interface {
		SetDSCP(int) error
	}
)

// newUDPSession create a new udp session for client or server
func newUDPSession(conv uint32, dataShards, parityShards int, l *Listener, conn net.PacketConn, ownConn bool, remote net.Addr, block BlockCrypt) *UDPSession {
	sess := new(UDPSession)
	sess.die = make(chan struct{})
	sess.chReadEvent = make(chan struct{}, 1)
	sess.chWriteEvent = make(chan struct{}, 1)
	sess.chSocketReadError = make(chan struct{})
	sess.chSocketWriteError = make(chan struct{})
	sess.chPostProcessing = make(chan sendRequest, devBacklog)
	sess.remote = remote
	sess.conn = conn
	sess.ownConn = ownConn
	sess.l = l

const (
	RINGBUFFER_MIN = 8
	RINGBUFFER_EXP = 1024
)

// RingBuffer is a generic ring (circular) buffer that supports dynamic resizing.
// It provides efficient FIFO queue behavior with amortized constant time operations.
type RingBuffer[T any] struct {
	head     int // Index of the next element to be popped
	tail     int // Index of the next empty slot to push into
	elements []T // Underlying slice storing elements in circular fashion
}

// NewRingBuffer creates a new Ring with a specified initial capacity.
// If the provided size is <= 8, it defaults to 8.
func NewRingBuffer[T any](size int) *RingBuffer[T] {
	if size <= RINGBUFFER_MIN {
		size = RINGBUFFER_MIN // Ensure a minimum size
	}
	return &RingBuffer[T]{
		head:     0,
		tail:     0,
		elements: make([]T, size),
	}
}

// Len returns the number of elements currently in the ring.
func (r *RingBuffer[T]) Len() int {
	if r.head <= r.tail {
		return r.tail - r.head
	}
	// Wrapped case: elements from head to end + elements from start to tail
	return len(r.elements) - r.head + r.tail
}

// Push adds an element to the tail of the ring.
// If the ring is full, it will grow automatically.
func (r *RingBuffer[T]) Push(v T) {
	if r.IsFull() {
		r.grow()
	}
	r.elements[r.tail] = v
	r.tail = (r.tail + 1) % len(r.elements)
}

// Pop removes and returns the element from the head of the ring.
// It returns the zero value and false if the ring is empty.
func (r *RingBuffer[T]) Pop() (T, bool) {
	var zero T
	if r.Len() == 0 {
		return zero, false
	}
	value := r.elements[r.head]
	// Optional: clear the slot to avoid retaining references
	r.elements[r.head] = zero
	r.head = (r.head + 1) % len(r.elements)

// chanSignal is a close-and-recreate channel used to wake waiters.
type chanSignal struct {
	mu sync.Mutex
	ch chan struct{}
}

func newChanSignal() *chanSignal {
	return &chanSignal{ch: make(chan struct{})}
}

func (c *chanSignal) notify() {
	c.mu.Lock()
	if c.ch == nil {
		c.ch = make(chan struct{})
	}
	close(c.ch)
	c.ch = make(chan struct{})
	c.mu.Unlock()
}

func (c *chanSignal) wait() <-chan struct{} {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.ch == nil {
		c.ch = make(chan struct{})
	}
	return c.ch
}

type MuxSession struct {

// NewMuxSession creates a new multiplexing session over conn.
func NewMuxSession(conn net.Conn, cfg *MuxConfig) (*MuxSession, error) {
	if conn == nil {
		return nil, errors.New("kcp: nil conn")
	}

	c := DefaultMuxConfig()
	if cfg != nil {
		c = *cfg
	}

		c.MaxFrameSize = DefaultMuxConfig().MaxFrameSize
	}
	if c.SendWindow == 0 {
		c.SendWindow = DefaultMuxConfig().SendWindow
	}
	if c.RecvWindow == 0 {
		c.RecvWindow = DefaultMuxConfig().RecvWindow
	}
	if c.Side != MuxSideClient && c.Side != MuxSideServer {
		return nil, errors.New("kcp: invalid mux side")
	}

	s := &MuxSession{
		conn:          conn,
		cfg:           c,
		done:          make(chan struct{}),
		streams:       make(map[uint32]*MuxStream),

func (s *MuxSession) isClosed() bool {
	select {
	case <-s.done:
		return true
	default:
		return false
	}
}

// Close shuts down the session and unblocks all readers and writers.
// It returns promptly and does not wait for background goroutines.
func (s *MuxSession) Close() error {
	var called bool
	s.closeOnce.Do(func() {
		called = true
		s.closeErr = io.ErrClosedPipe
		close(s.done)

		s.mu.Lock()
		streams := make([]*MuxStream, 0, len(s.streams))
		for _, st := range s.streams {
			streams = append(streams, st)
		}
		s.mu.Unlock()

		for _, st := range streams {
			st.notifySessionClose()
		}
		s.acceptSig.notify()
		s.signalSend()
		_ = s.conn.Close()
	})
	if !called {
		return io.ErrClosedPipe
	}
	return nil
}

// NumStreams returns the number of currently tracked streams.
func (s *MuxSession) NumStreams() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return len(s.streams)
}

// OpenStream opens a new stream on this session. Client streams get odd IDs,
// server streams get even IDs.
func (s *MuxSession) OpenStream(priority uint8) (*MuxStream, error) {
	if s.isClosed() {
		return nil, io.ErrClosedPipe
	}

	s.mu.Lock()
	if s.isClosed() {
		s.mu.Unlock()
		return nil, io.ErrClosedPipe
	}
	if s.nextID > uint32(^uint32(0))-2 {
		s.mu.Unlock()
		return nil, errors.New("kcp: stream id exhausted")
	}
	id := s.nextID
	s.nextID += 2
	st := newMuxStream(s, id, priority)
	s.streams[id] = st
	s.mu.Unlock()

	binary.BigEndian.PutUint64(payload, uint64(s.cfg.RecvWindow))
	s.enqueueControl(&muxFrame{cmd: muxCmdSYN, sid: id, priority: priority, payload: payload})
	return st, nil
}

// AcceptStream returns the next stream opened by the remote peer.
func (s *MuxSession) AcceptStream() (*MuxStream, error) {
	for {
		s.mu.Lock()
		if len(s.acceptQueue) > 0 {
			st := s.acceptQueue[0]
			s.acceptQueue = s.acceptQueue[1:]
			s.mu.Unlock()
			return st, nil
		}
		if s.isClosed() {
			s.mu.Unlock()
			return nil, io.ErrClosedPipe
		}
		ch := s.acceptSig.wait()
		s.mu.Unlock()

		select {
		case <-ch:
		case <-s.done:
			return nil, io.ErrClosedPipe
		}
	}
}

type MuxStream struct {

func newMuxStream(sess *MuxSession, id uint32, priority uint8) *MuxStream {

// Read reads data from the stream. It blocks until data is available,
// the remote side closes its write side, or the session is closed.
func (st *MuxStream) Read(p []byte) (int, error) {
	if len(p) == 0 {
		return 0, nil
	}

	for {
		st.mu.Lock()
		if len(st.recvBuf) > 0 {
			n := copy(p, st.recvBuf)
			st.recvBuf = st.recvBuf[n:]
			st.freeWindow += uint64(n)
			bothClosed := st.localClosed && st.remoteClosed
			drained := len(st.recvBuf) == 0
			st.mu.Unlock()

			st.sendWindowUpdate()
			if bothClosed && drained {
				st.sess.removeStream(st)
			}
			return n, nil
		}
		if st.remoteClosed {
			st.mu.Unlock()
			return 0, io.EOF
		}
		if st.sess.isClosed() {
			st.mu.Unlock()
			return 0, io.ErrClosedPipe
		}
		ch := st.readSig.wait()
		st.mu.Unlock()

		var timer *time.Timer
		var c <-chan time.Time
		if rd, ok := st.readDeadline.Load().(time.Time); ok && !rd.IsZero() {
			d := time.Until(rd)
			if d <= 0 {
				return 0, errTimeout
			}
			timer = time.NewTimer(d)
			c = timer.C
		}

		select {
		case <-ch:
			if timer != nil {
				timer.Stop()
			}
		case <-c:
			return 0, errTimeout
		case <-st.sess.done:
			return 0, io.ErrClosedPipe
		}
	}
}

// Write writes all of p to the stream. It blocks until the whole slice has
// been accepted by the send pipeline; it never returns a short write except
// on error.
func (st *MuxStream) Write(p []byte) (int, error) {
	if len(p) == 0 {
		return 0, nil
	}

	maxFrame := int(st.sess.cfg.MaxFrameSize)
	remaining := p
	for len(remaining) > 0 {
		chunkLen := len(remaining)
		if chunkLen > maxFrame {
			chunkLen = maxFrame
		}

		st.sendMu.Lock()
		for st.sendWindow == 0 {
			if st.sendClosed || st.sess.isClosed() {
				st.sendMu.Unlock()
				return 0, io.ErrClosedPipe
			}
			st.sendCond.Wait()
		}
		n := chunkLen
		if uint64(n) > st.sendWindow {
			n = int(st.sendWindow)
		}
		st.sendWindow -= uint64(n)

		frame := &muxFrame{cmd: muxCmdPSH, sid: st.id, priority: st.priority, payload: remaining[:n]}
		st.sess.sendMu.Lock()
		st.sess.appendToStream(st, frame)
		st.sess.sendMu.Unlock()
		st.sendMu.Unlock()

		st.sess.signalSend()
		remaining = remaining[n:]
	}
	return len(p), nil
}

// Close half-closes the local write side. Already-buffered inbound data
// remains readable until drained.
func (st *MuxStream) Close() error {
	if st.sess.isClosed() {
		return io.ErrClosedPipe
	}

	st.sendMu.Lock()
	if st.sendClosed {
		st.sendMu.Unlock()
		return io.ErrClosedPipe
	}
	st.mu.Lock()
	st.localClosed = true
	st.mu.Unlock()

	st.sendClosed = true
	st.sess.sendMu.Lock()
	st.sess.appendToStream(st, &muxFrame{cmd: muxCmdFIN, sid: st.id, priority: st.priority})
	st.sess.sendMu.Unlock()
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.sess.signalSend()

	st.mu.Lock()
	remoteClosed := st.remoteClosed
	drained := len(st.recvBuf) == 0
	st.mu.Unlock()
	if remoteClosed && drained {
		st.sess.removeStream(st)
	}

	atomic.AddUint64(&DefaultSnmp.MuxStreamsClosed, 1)
	return nil
}

func (st *MuxStream) notifySessionClose() {
	st.mu.Lock()
	st.localClosed = true
	st.remoteClosed = true
	st.mu.Unlock()

	st.sendMu.Lock()
	st.sendClosed = true
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.readSig.notify()
}

func (st *MuxStream) notifyRemoteClosed() {
	st.mu.Lock()
	st.remoteClosed = true
	localClosed := st.localClosed
	drained := len(st.recvBuf) == 0
	st.mu.Unlock()

	st.sendMu.Lock()
	st.sendClosed = true
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.readSig.notify()

	if localClosed && drained {
		st.sess.removeStream(st)
	}
	atomic.AddUint64(&DefaultSnmp.MuxStreamsClosed, 1)
}

func (st *MuxStream) setSendWindow(w uint64) {
	st.sendMu.Lock()
	st.sendWindow = w
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
}

func (st *MuxStream) sendWindowUpdate() {
	st.mu.Lock()
	free := st.freeWindow
	if free > uint64(st.sess.cfg.RecvWindow) {
		free = uint64(st.sess.cfg.RecvWindow)
	}
	st.mu.Unlock()

	payload := make([]byte, 8)
	binary.BigEndian.PutUint64(payload, free)
	st.sess.enqueueControl(&muxFrame{cmd: muxCmdUPD, sid: st.id, priority: st.priority, payload: payload})
}

// appendToStream appends a frame to a stream's ordered send queue. The caller
// must hold sess.sendMu.
func (s *MuxSession) appendToStream(st *MuxStream, f *muxFrame) {
	st.sendQ = append(st.sendQ, f)
	if !st.pendingInSession {
		st.pendingInSession = true
		switch priorityBucket(st.priority) {
		case 0:
			s.pendingHigh[st.id] = st
		case 1:
			s.pendingNormal[st.id] = st
		default:
			s.pendingLow[st.id] = st
		}
	}
}

func priorityBucket(priority uint8) int {
	if priority == MuxPriorityHigh {
		return 0
	}
	if priority == MuxPriorityNormal {
		return 1
	}
	return 2
}

func (s *MuxSession) enqueueControl(f *muxFrame) {
	s.sendMu.Lock()
	s.ctrlQ = append(s.ctrlQ, f)
	s.sendMu.Unlock()
	s.signalSend()
}

func (s *MuxSession) signalSend() {
	select {
	case s.sendSignal <- struct{}{}:
	default:
	}
}

// pickStream returns the highest-priority stream with queued frames. The
// caller must hold s.sendMu.
func (s *MuxSession) pickStream() *MuxStream {
	if len(s.pendingHigh) > 0 {
		for _, st := range s.pendingHigh {
			return st
		}
	}
	if len(s.pendingNormal) > 0 {
		for _, st := range s.pendingNormal {
			return st
		}
	}
	if len(s.pendingLow) > 0 {
		for _, st := range s.pendingLow {
			return st
		}
	}
	return nil
}

func (s *MuxSession) nextFrame() *muxFrame {
	for {
		s.sendMu.Lock()
		if len(s.ctrlQ) > 0 {
			f := s.ctrlQ[0]
			s.ctrlQ = s.ctrlQ[1:]
			s.sendMu.Unlock()
			return f
		}
		if st := s.pickStream(); st != nil {
			f := st.sendQ[0]
			st.sendQ = st.sendQ[1:]
			if len(st.sendQ) == 0 {
				st.pendingInSession = false
				delete(s.pendingHigh, st.id)
				delete(s.pendingNormal, st.id)
				delete(s.pendingLow, st.id)
			}
			s.sendMu.Unlock()
			return f
		}
		s.sendMu.Unlock()

		select {
		case <-s.sendSignal:
		case <-s.done:
			return nil
		}
	}
}

func (s *MuxSession) writeLoop() {
	for {
		f := s.nextFrame()
		if f == nil {
			return
		}
		if err := s.writeFrame(f); err != nil {
			// If the session is already closed we simply exit. Otherwise the
			// connection has failed; tear down the session.
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
	}
}

func (s *MuxSession) writeFrame(f *muxFrame) error {
	header := make([]byte, muxHeaderSize)
	header[0] = muxVersion
	header[1] = byte(f.cmd)
	header[2] = f.priority
	binary.BigEndian.PutUint32(header[4:8], uint32(len(f.payload)))
	binary.BigEndian.PutUint32(header[8:12], f.sid)

	buf := header
	if len(f.payload) > 0 {
		buf = make([]byte, muxHeaderSize+len(f.payload))
		copy(buf, header)
		copy(buf[muxHeaderSize:], f.payload)
	}
	if _, err := writeFull(s.conn, buf); err != nil {
		return err
	}

	atomic.AddUint64(&DefaultSnmp.MuxFramesSent, 1)
	if f.cmd == muxCmdPSH {
		atomic.AddUint64(&DefaultSnmp.MuxBytesSent, uint64(len(f.payload)))
	}
	return nil
}

func (s *MuxSession) readLoop() {
	header := make([]byte, muxHeaderSize)
	for {
		if s.isClosed() {
			return
		}
		if _, err := io.ReadFull(s.conn, header); err != nil {
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
		if header[0] != muxVersion {
			if !s.isClosed() {
				s.fail(errors.New("kcp: invalid mux version"))
			}
			return
		}
		cmd := muxCmd(header[1])
		priority := header[2]
		length := binary.BigEndian.Uint32(header[4:8])
		sid := binary.BigEndian.Uint32(header[8:12])

		payload := make([]byte, length)
		if _, err := io.ReadFull(s.conn, payload); err != nil {
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
		atomic.AddUint64(&DefaultSnmp.MuxFramesReceived, 1)

		switch cmd {
		case muxCmdSYN:
			s.handleSYN(sid, priority, payload)
		case muxCmdFIN:
			s.handleFIN(sid)
		case muxCmdPSH:
			s.handlePSH(sid, payload)
		case muxCmdUPD:
			s.handleUPD(sid, payload)
		}
	}
}

func (s *MuxSession) handleSYN(sid uint32, priority uint8, payload []byte) {
	if sid == 0 {
		return
	}
	s.mu.Lock()
	st, ok := s.streams[sid]
	if !ok {
		st = newMuxStream(s, sid, priority)
		s.streams[sid] = st
		s.acceptQueue = append(s.acceptQueue, st)
		if len(payload) >= 8 {
			st.setSendWindow(binary.BigEndian.Uint64(payload))
		}
		s.mu.Unlock()

		atomic.AddUint64(&DefaultSnmp.MuxStreamsOpened, 1)
		s.acceptSig.notify()
		st.sendWindowUpdate()
		return
	}
	if len(payload) >= 8 {
		st.setSendWindow(binary.BigEndian.Uint64(payload))
	}
	s.mu.Unlock()
	st.sendWindowUpdate()
}

func (s *MuxSession) handleFIN(sid uint32) {
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if ok {
		st.notifyRemoteClosed()
	}
}

func (s *MuxSession) handlePSH(sid uint32, payload []byte) {
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if !ok {
		return
	}
	st.mu.Lock()
	st.recvBuf = append(st.recvBuf, payload...)
	st.freeWindow -= uint64(len(payload))
	st.mu.Unlock()
	st.readSig.notify()
	atomic.AddUint64(&DefaultSnmp.MuxBytesReceived, uint64(len(payload)))
}

func (s *MuxSession) handleUPD(sid uint32, payload []byte) {
	if len(payload) < 8 {
		return
	}
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if ok {
		st.setSendWindow(binary.BigEndian.Uint64(payload))
	}
}

func (s *MuxSession) removeStream(st *MuxStream) {
	s.mu.Lock()
	if cur, ok := s.streams[st.id]; ok && cur == st {
		delete(s.streams, st.id)
	}
	s.mu.Unlock()
}

func (s *MuxSession) fail(err error) {
	var called bool
	s.closeOnce.Do(func() {
		called = true
		s.closeErr = err
		close(s.done)

		s.mu.Lock()
		streams := make([]*MuxStream, 0, len(s.streams))
		for _, st := range s.streams {
			streams = append(streams, st)
		}
		s.mu.Unlock()

		for _, st := range streams {
			st.notifySessionClose()
		}
		s.acceptSig.notify()
		s.signalSend()
		_ = s.conn.Close()
	})
	_ = called
}

func writeFull(w io.Writer, b []byte) (int, error) {
	total := 0
	for len(b) > 0 {
		n, err := w.Write(b)
		total += n
		b = b[n:]
		if err != nil {
			return total, err
		}
		if n == 0 {
			return total, io.ErrShortWrite
		}
	}
	return total, nil
}

func TestMuxMultipleStreamsOrdering(t *testing.T) {
	c1, c2 := net.Pipe()
	defer c1.Close()
	defer c2.Close()

	cfg1 := DefaultMuxConfig()
	cfg1.Side = MuxSideClient
	cfg1.MaxFrameSize = 64
	cfg2 := DefaultMuxConfig()
	cfg2.Side = MuxSideServer
	cfg2.MaxFrameSize = 64

	s1, s2 := newMuxPair(t, c1, c2, &cfg1, &cfg2)
	defer s1.Close()
	defer s2.Close()

	// open 3 client streams
	var clientStreams []*MuxStream
	for i := 0; i < 3; i++ {
		st, err := s1.OpenStream(MuxPriorityNormal)
		if err != nil {
			t.Fatal(err)
		}
		clientStreams = append(clientStreams, st)
	}
	var serverStreams []*MuxStream
	for i := 0; i < 3; i++ {
		serverStreams = append(serverStreams, acceptWithTimeout(t, s2))
	}

	// write different patterns to each stream and verify they stay independent
	for i := 0; i < 3; i++ {
		payload := make([]byte, 300)
		for j := range payload {
			payload[j] = byte(i)
		}
		go func(st *MuxStream, p []byte) {
			if _, err := st.Write(p); err != nil {
				t.Errorf("write: %v", err)
			}
		}(clientStreams[i], payload)
	}

	for i := 0; i < 3; i++ {
		got := make([]byte, 0, 300)
		buf := make([]byte, 37)
		for len(got) < 300 {
			n, err := serverStreams[i].Read(buf)
			if err != nil {
				t.Fatalf("stream %d read: %v", i, err)
			}
			got = append(got, buf[:n]...)
		}
		for j := range got {
			if got[j] != byte(i) {
				t.Fatalf("stream %d byte %d = %d, want %d", i, j, got[j], i)
			}
		}
	}
}

func TestMuxHalfClose(t *testing.T) {
	c1, c2 := net.Pipe()
	defer c1.Close()
	defer c2.Close()

	cfg1 := DefaultMuxConfig()
	cfg1.Side = MuxSideClient
	cfg2 := DefaultMuxConfig()
	cfg2.Side = MuxSideServer

	s1, s2 := newMuxPair(t, c1, c2, &cfg1, &cfg2)
	defer s1.Close()
	defer s2.Close()

	st1, err := s1.OpenStream(MuxPriorityNormal)
	if err != nil {
		t.Fatal(err)
	}
	st2 := acceptWithTimeout(t, s2)

	// write then close (half-close)
	go func() {
		if _, err := st1.Write([]byte("payload")); err != nil {
			t.Errorf("write: %v", err)
		}
		if err := st1.Close(); err != nil {
			t.Errorf("close: %v", err)
		}
	}()

	buf := make([]byte, 64)
	n, err := st2.Read(buf)
	if err != nil {
		t.Fatal(err)
	}
	if string(buf[:n]) != "payload" {
		t.Fatalf("got %q", buf[:n])
	}

	// after data drained and remote FIN received, read returns EOF
	_, err = st2.Read(buf)
	if err != io.EOF {
		t.Fatalf("second read err = %v, want io.EOF", err)
	}

	// remote close unblocks local writers
	if _, err := st2.Write([]byte("x")); err != io.ErrClosedPipe {
		t.Fatalf("write after remote close err = %v, want io.ErrClosedPipe", err)
	}
}

func TestMuxBlockedStreamDoesNotStallOthers(t *testing.T) {
	c1, c2 := net.Pipe()
	defer c1.Close()
	defer c2.Close()

	cfg1 := DefaultMuxConfig()
	cfg1.Side = MuxSideClient
	cfg1.MaxFrameSize = 8192
	cfg1.SendWindow = 512
	cfg1.RecvWindow = 512
	cfg2 := cfg1
	cfg2.Side = MuxSideServer

	s1, s2 := newMuxPair(t, c1, c2, &cfg1, &cfg2)
	defer s1.Close()
	defer s2.Close()

	slow1, err := s1.OpenStream(MuxPriorityNormal)
	if err != nil {
		t.Fatal(err)
	}
	slow2 := acceptWithTimeout(t, s2)
	fast1, err := s1.OpenStream(MuxPriorityNormal)
	if err != nil {
		t.Fatal(err)
	}
	fast2 := acceptWithTimeout(t, s2)

	// block slow stream with a large write
	slowDone := make(chan error, 1)
	go func() {
		_, err := slow1.Write(make([]byte, 4096))
		slowDone <- err
	}()

	time.Sleep(100 * time.Millisecond)

	// fast stream should still be writable and readable
	fastDone := make(chan error, 1)
	go func() {
		_, err := fast1.Write([]byte("fast"))
		fastDone <- err
	}()
	buf := make([]byte, 16)
	n, err := fast2.Read(buf)
	if err != nil {
		t.Fatal(err)
	}
	if string(buf[:n]) != "fast" {
		t.Fatalf("got %q", buf[:n])
	}
	select {
	case err := <-fastDone:
		if err != nil {
			t.Fatalf("fast write err: %v", err)
		}
	default:
		t.Fatal("fast write did not complete")
	}

	// drain slow stream to unblock it
	readDone := make(chan struct{})
	go func() {
		total := 0
		buf := make([]byte, 512)
		for total < 4096 {
			n, err := slow2.Read(buf)
			if err != nil {
				break
			}
			total += n
		}
		close(readDone)
	}()
	select {
	case err := <-slowDone:
		if err != nil {
			t.Fatalf("slow write err: %v", err)
		}
	case <-time.After(10 * time.Second):
		t.Fatal("slow write did not unblock")
	}
	<-readDone
}

	st1, err := s1.OpenStream(MuxPriorityNormal)
	if err != nil {
		t.Fatal(err)
	}
	st2 := acceptWithTimeout(t, s2)

	readDone := make(chan error, 1)
	go func() {
		buf := make([]byte, 16)
		_, err := st2.Read(buf)
		readDone <- err
	}()
	writeDone := make(chan error, 1)
	go func() {
		_, err := st1.Write(make([]byte, 1<<20))
		writeDone <- err
	}()

	time.Sleep(100 * time.Millisecond)
	if err := s2.Close(); err != nil {
		t.Fatal(err)
	}

	select {
	case err := <-readDone:
		if err != io.ErrClosedPipe {
			t.Fatalf("read err = %v, want io.ErrClosedPipe", err)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("read not unblocked")
	}
	select {
	case err := <-writeDone:
		if err != io.ErrClosedPipe {
			t.Fatalf("write err = %v, want io.ErrClosedPipe", err)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("write not unblocked")
	}
	if _, err := s1.OpenStream(MuxPriorityNormal); err != io.ErrClosedPipe {
		t.Fatalf("open after close err = %v, want io.ErrClosedPipe", err)
	}
	if _, err := s2.AcceptStream(); err != io.ErrClosedPipe {
		t.Fatalf("accept after close err = %v, want io.ErrClosedPipe", err)
	}
}

type gateConn struct {
	net.Conn
	mu      sync.Mutex
	enabled bool
	gate    chan struct{}
}

func (c *gateConn) setEnabled() {
	c.mu.Lock()
	c.enabled = true
	c.gate = make(chan struct{})
	c.mu.Unlock()
}

func (c *gateConn) release() {
	c.mu.Lock()
	if c.gate != nil {
		close(c.gate)
		c.gate = nil
	}
	c.mu.Unlock()
}

func (c *gateConn) Write(b []byte) (int, error) {
	c.mu.Lock()
	enabled := c.enabled
	gate := c.gate
	c.mu.Unlock()
	if enabled && gate != nil {
		<-gate
	}
	return c.Conn.Write(b)
}

func TestMuxPriorityScheduling(t *testing.T) {
	c1, c2 := net.Pipe()
	defer c1.Close()
	defer c2.Close()
	gc := &gateConn{Conn: c1}

	cfg1 := DefaultMuxConfig()
	cfg1.Side = MuxSideClient
	cfg1.MaxFrameSize = 16
	cfg2 := DefaultMuxConfig()
	cfg2.Side = MuxSideServer
	cfg2.MaxFrameSize = 16

	s1, _ := NewMuxSession(gc, &cfg1)
	s2, _ := NewMuxSession(c2, &cfg2)
	defer s1.Close()
	defer s2.Close()

	low1, err := s1.OpenStream(MuxPriorityLow)
	if err != nil {
		t.Fatal(err)
	}
	low2 := acceptWithTimeout(t, s2)
	high1, err := s1.OpenStream(MuxPriorityHigh)
	if err != nil {
		t.Fatal(err)
	}
	high2 := acceptWithTimeout(t, s2)

	// block the client writer
	gc.setEnabled()

	// queue three low-priority frames
	for i := 0; i < 3; i++ {
		if _, err := low1.Write([]byte("low")); err != nil {
			t.Fatal(err)
		}
	}
	// queue one high-priority frame
	if _, err := high1.Write([]byte("HI!")); err != nil {
		t.Fatal(err)
	}

	// let the writer proceed
	gc.release()

	// the first low frame may already be in-flight; the high frame must jump
	// ahead of the remaining queued low frames
	var received []string
	buf := make([]byte, 16)
	for len(received) < 4 {
		n, err := low2.Read(buf)
		if err != nil {
			t.Fatal(err)
		}
		received = append(received, string(buf[:n]))
	}
	t.Logf("received: %v", received)

	if len(received) != 4 {
		t.Fatalf("received %d frames", len(received))
	}
	if received[0] != "low" {
		t.Fatalf("first frame = %q, want low (in-flight)", received[0])
	}
	if received[1] != "HI!" {
		t.Fatalf("second frame = %q, want high before queued low", received[1])
	}
	// and the high stream should have received the high frame
	hbuf := make([]byte, 16)
	n, err := high2.Read(hbuf)
	if err != nil {
		t.Fatal(err)
	}
	if string(hbuf[:n]) != "HI!" {
		t.Fatalf("high stream got %q", hbuf[:n])
	}
}

// blockingWriteConn never completes writes. Close returns without unblocking
// pending writes, so it exercises the requirement that MuxSession.Close must
// not wait for background writer work.
type blockingWriteConn struct {
	net.Conn
	writeBlock chan struct{}
}

func (c *blockingWriteConn) Write(b []byte) (int, error) {
	<-c.writeBlock
	return 0, io.ErrClosedPipe
}

func TestMuxCloseDoesNotWaitForBlockedWrite(t *testing.T) {
	c1, c2 := net.Pipe()
	defer c1.Close()
	defer c2.Close()
	bwc := &blockingWriteConn{Conn: c1, writeBlock: make(chan struct{})}

	cfg := DefaultMuxConfig()
	cfg.Side = MuxSideClient
	s, err := NewMuxSession(bwc, &cfg)
	if err != nil {
		t.Fatal(err)
	}

	st, err := s.OpenStream(MuxPriorityNormal)
	if err != nil {
		t.Fatal(err)
	}
	// Block a reader so we can also verify it is unblocked.
	readDone := make(chan error, 1)
	go func() {
		buf := make([]byte, 16)
		_, err := st.Read(buf)
		readDone <- err
	}()

	// Give the writer goroutine time to block on the stuck conn.
	time.Sleep(50 * time.Millisecond)

	closeDone := make(chan error, 1)
	go func() {
		closeDone <- s.Close()
	}()
	select {
	case err := <-closeDone:
		if err != nil {
			t.Fatalf("close err = %v", err)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("Close blocked")
	}

	select {
	case err := <-readDone:
		if err != io.ErrClosedPipe {
			t.Fatalf("read err = %v, want io.ErrClosedPipe", err)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("reader not unblocked")
	}

	if s.NumStreams() != 0 {
		t.Fatalf("NumStreams after close = %d, want 0", s.NumStreams())
	}
	if err := s.Close(); err != io.ErrClosedPipe {
		t.Fatalf("second close err = %v, want io.ErrClosedPipe", err)
	}
}

// MuxSide indicates the role of a MuxSession on a connection.
type MuxSide uint8

const (
	MuxSideClient MuxSide = iota
	MuxSideServer
)

// Stream priorities. Lower numeric value means higher priority.
const (
	MuxPriorityHigh   uint8 = 0
	MuxPriorityNormal uint8 = 1
	MuxPriorityLow    uint8 = 2
)

// MuxConfig holds the configuration for a MuxSession.
type MuxConfig struct {
	Side         MuxSide
	MaxFrameSize uint32
	SendWindow   uint32
	RecvWindow   uint32
}

// DefaultMuxConfig returns a MuxConfig with sensible defaults.
func DefaultMuxConfig() MuxConfig {
	return MuxConfig{
		Side:         MuxSideClient,
		MaxFrameSize: 8192,
		SendWindow:   1 << 20,
		RecvWindow:   1 << 20,
	}
}

type muxCmd uint8

const (
	muxCmdSYN muxCmd = iota
	muxCmdFIN
	muxCmdPSH
	muxCmdUPD
)

// muxFrame is an internal representation of one protocol frame.
type muxFrame struct {
	cmd      muxCmd
	sid      uint32
	priority uint8
	payload  []byte
}

// chanSignal is a close-and-recreate channel used to wake waiters.
type chanSignal struct {
	mu sync.Mutex
	ch chan struct{}
}

func newChanSignal() *chanSignal {
	return &chanSignal{ch: make(chan struct{})}
}

func (c *chanSignal) notify() {
	c.mu.Lock()
	if c.ch == nil {
		c.ch = make(chan struct{})
	}
	close(c.ch)
	c.ch = make(chan struct{})
	c.mu.Unlock()
}

func (c *chanSignal) wait() <-chan struct{} {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.ch == nil {
		c.ch = make(chan struct{})
	}
	return c.ch
}

// MuxSession multiplexes many independent ordered streams over one net.Conn.
type MuxSession struct {
	conn net.Conn
	cfg  MuxConfig

	done      chan struct{}
	closeOnce sync.Once
	closeErr  error

	mu          sync.Mutex
	streams     map[uint32]*MuxStream
	nextID      uint32
	acceptQueue []*MuxStream
	acceptSig   *chanSignal

	sendMu        sync.Mutex
	ctrlQ         []*muxFrame
	pendingHigh   map[uint32]*MuxStream
	pendingNormal map[uint32]*MuxStream
	pendingLow    map[uint32]*MuxStream
	sendSignal    chan struct{}
}

// NewMuxSession creates a new multiplexing session over conn.
func NewMuxSession(conn net.Conn, cfg *MuxConfig) (*MuxSession, error) {
	if conn == nil {
		return nil, errors.New("kcp: nil conn")
	}

	c := DefaultMuxConfig()
	if cfg != nil {
		c = *cfg
	}
	if c.MaxFrameSize == 0 {
		c.MaxFrameSize = DefaultMuxConfig().MaxFrameSize
	}
	if c.SendWindow == 0 {
		c.SendWindow = DefaultMuxConfig().SendWindow
	}
	if c.RecvWindow == 0 {
		c.RecvWindow = DefaultMuxConfig().RecvWindow
	}
	if c.Side != MuxSideClient && c.Side != MuxSideServer {
		return nil, errors.New("kcp: invalid mux side")
	}

	s := &MuxSession{
		conn:          conn,
		cfg:           c,
		done:          make(chan struct{}),
		streams:       make(map[uint32]*MuxStream),
		acceptQueue:   make([]*MuxStream, 0, muxAcceptQueue),
		acceptSig:     newChanSignal(),
		pendingHigh:   make(map[uint32]*MuxStream),
		pendingNormal: make(map[uint32]*MuxStream),
		pendingLow:    make(map[uint32]*MuxStream),
		sendSignal:    make(chan struct{}, 1),
	}
	if c.Side == MuxSideServer {
		s.nextID = 2
	} else {
		s.nextID = 1
	}

	go s.readLoop()
	go s.writeLoop()
	return s, nil
}

func (s *MuxSession) isClosed() bool {
	select {
	case <-s.done:
		return true
	default:
		return false
	}
}

// OpenStream opens a new stream on this session. Client streams get odd IDs,
// server streams get even IDs.
func (s *MuxSession) OpenStream(priority uint8) (*MuxStream, error) {
	if s.isClosed() {
		return nil, io.ErrClosedPipe
	}

	s.mu.Lock()
	if s.isClosed() {
		s.mu.Unlock()
		return nil, io.ErrClosedPipe
	}
	if s.nextID > uint32(^uint32(0))-2 {
		s.mu.Unlock()
		return nil, errors.New("kcp: stream id exhausted")
	}
	id := s.nextID
	s.nextID += 2
	st := newMuxStream(s, id, priority)
	s.streams[id] = st
	s.mu.Unlock()

	atomic.AddUint64(&DefaultSnmp.MuxStreamsOpened, 1)

	payload := make([]byte, 8)
	binary.BigEndian.PutUint64(payload, uint64(s.cfg.RecvWindow))
	s.enqueueControl(&muxFrame{cmd: muxCmdSYN, sid: id, priority: priority, payload: payload})
	return st, nil
}

// AcceptStream returns the next stream opened by the remote peer.
func (s *MuxSession) AcceptStream() (*MuxStream, error) {
	for {
		s.mu.Lock()
		if len(s.acceptQueue) > 0 {
			st := s.acceptQueue[0]
			s.acceptQueue = s.acceptQueue[1:]
			s.mu.Unlock()
			return st, nil
		}
		if s.isClosed() {
			s.mu.Unlock()
			return nil, io.ErrClosedPipe
		}
		ch := s.acceptSig.wait()
		s.mu.Unlock()

		select {
		case <-ch:
		case <-s.done:
			return nil, io.ErrClosedPipe
		}
	}
}

// MuxStream is an independent ordered stream inside a MuxSession.
type MuxStream struct {
	sess     *MuxSession
	id       uint32
	priority uint8

	mu           sync.Mutex
	recvBuf      []byte
	freeWindow   uint64
	localClosed  bool
	remoteClosed bool
	readSig      *chanSignal
	readDeadline atomic.Value

	sendMu     sync.Mutex
	sendCond   *sync.Cond
	sendWindow uint64
	sendClosed bool

	// protected by sess.sendMu
	sendQ            []*muxFrame
	pendingInSession bool
}

func newMuxStream(sess *MuxSession, id uint32, priority uint8) *MuxStream {
	st := &MuxStream{
		sess:       sess,
		id:         id,
		priority:   priority,
		freeWindow: uint64(sess.cfg.RecvWindow),
		sendWindow: uint64(sess.cfg.SendWindow),
		readSig:    newChanSignal(),
	}
	st.sendCond = sync.NewCond(&st.sendMu)
	return st
}

// ID returns the stream identifier.
func (st *MuxStream) ID() uint32 {
	return st.id
}

// SetReadDeadline sets the read deadline for this stream.
func (st *MuxStream) SetReadDeadline(t time.Time) error {
	st.readDeadline.Store(t)
	st.readSig.notify()
	return nil
}

// Read reads data from the stream. It blocks until data is available,
// the remote side closes its write side, or the session is closed.
func (st *MuxStream) Read(p []byte) (int, error) {
	if len(p) == 0 {
		return 0, nil
	}

	for {
		st.mu.Lock()
		if len(st.recvBuf) > 0 {
			n := copy(p, st.recvBuf)
			st.recvBuf = st.recvBuf[n:]
			st.freeWindow += uint64(n)
			bothClosed := st.localClosed && st.remoteClosed
			drained := len(st.recvBuf) == 0
			st.mu.Unlock()

			st.sendWindowUpdate()
			if bothClosed && drained {
				st.sess.removeStream(st)
			}
			return n, nil
		}
		if st.remoteClosed {
			st.mu.Unlock()
			return 0, io.EOF
		}
		if st.sess.isClosed() {
			st.mu.Unlock()
			return 0, io.ErrClosedPipe
		}
		ch := st.readSig.wait()
		st.mu.Unlock()

		var timer *time.Timer
		var c <-chan time.Time
		if rd, ok := st.readDeadline.Load().(time.Time); ok && !rd.IsZero() {
			d := time.Until(rd)
			if d <= 0 {
				return 0, errTimeout
			}
			timer = time.NewTimer(d)
			c = timer.C
		}

		select {
		case <-ch:
			if timer != nil {
				timer.Stop()
			}
		case <-c:
			return 0, errTimeout
		case <-st.sess.done:
			return 0, io.ErrClosedPipe
		}
	}
}

// Write writes all of p to the stream. It blocks until the whole slice has
// been accepted by the send pipeline; it never returns a short write except
// on error.
func (st *MuxStream) Write(p []byte) (int, error) {
	if len(p) == 0 {
		return 0, nil
	}

	maxFrame := int(st.sess.cfg.MaxFrameSize)
	remaining := p
	for len(remaining) > 0 {
		chunkLen := len(remaining)
		if chunkLen > maxFrame {
			chunkLen = maxFrame
		}

		st.sendMu.Lock()
		for {
			if st.sendClosed || st.sess.isClosed() {
				st.sendMu.Unlock()
				return 0, io.ErrClosedPipe
			}
			if st.sendWindow > 0 {
				break
			}
			st.sendCond.Wait()
		}
		n := chunkLen
		if uint64(n) > st.sendWindow {
			n = int(st.sendWindow)
		}
		st.sendWindow -= uint64(n)

		frame := &muxFrame{cmd: muxCmdPSH, sid: st.id, priority: st.priority, payload: remaining[:n]}
		st.sess.sendMu.Lock()
		st.sess.appendToStream(st, frame)
		st.sess.sendMu.Unlock()
		st.sendMu.Unlock()

		st.sess.signalSend()
		remaining = remaining[n:]
	}
	return len(p), nil
}

// Close half-closes the local write side. Already-buffered inbound data
// remains readable until drained.
func (st *MuxStream) Close() error {
	if st.sess.isClosed() {
		return io.ErrClosedPipe
	}

	st.mu.Lock()
	if st.localClosed {
		st.mu.Unlock()
		return io.ErrClosedPipe
	}
	st.localClosed = true
	st.mu.Unlock()

	st.sendMu.Lock()
	st.sendClosed = true
	st.sess.sendMu.Lock()
	st.sess.appendToStream(st, &muxFrame{cmd: muxCmdFIN, sid: st.id, priority: st.priority})
	st.sess.sendMu.Unlock()
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.sess.signalSend()

	st.mu.Lock()
	remoteClosed := st.remoteClosed
	drained := len(st.recvBuf) == 0
	st.mu.Unlock()
	if remoteClosed && drained {
		st.sess.removeStream(st)
	}
	return nil
}

func (st *MuxStream) notifySessionClose() {
	st.mu.Lock()
	st.localClosed = true
	st.remoteClosed = true
	st.mu.Unlock()

	st.sendMu.Lock()
	st.sendClosed = true
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.readSig.notify()
}

func (st *MuxStream) notifyRemoteClosed() {
	st.mu.Lock()
	st.remoteClosed = true
	localClosed := st.localClosed
	drained := len(st.recvBuf) == 0
	st.mu.Unlock()

	st.sendMu.Lock()
	st.sendClosed = true
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
	st.readSig.notify()

	if localClosed && drained {
		st.sess.removeStream(st)
	}
}

func (st *MuxStream) setSendWindow(w uint64) {
	st.sendMu.Lock()
	st.sendWindow = w
	st.sendMu.Unlock()
	st.sendCond.Broadcast()
}

	st.mu.Lock()
	free := st.freeWindow
	if free > uint64(st.sess.cfg.RecvWindow) {
		free = uint64(st.sess.cfg.RecvWindow)
	}

// appendToStream appends a frame to a stream's ordered send queue. The caller
// must hold sess.sendMu.
func (s *MuxSession) appendToStream(st *MuxStream, f *muxFrame) {
	st.sendQ = append(st.sendQ, f)
	if !st.pendingInSession {
		st.pendingInSession = true
		switch priorityBucket(st.priority) {
		case 0:
			s.pendingHigh[st.id] = st
		case 1:
			s.pendingNormal[st.id] = st
		default:
			s.pendingLow[st.id] = st
		}
	}
}

func priorityBucket(priority uint8) int {
	if priority == MuxPriorityHigh {
		return 0
	}
	if priority == MuxPriorityNormal {
		return 1
	}
	return 2
}

func (s *MuxSession) enqueueControl(f *muxFrame) {
	s.sendMu.Lock()
	s.ctrlQ = append(s.ctrlQ, f)
	s.sendMu.Unlock()
	s.signalSend()
}

func (s *MuxSession) signalSend() {
	select {
	case s.sendSignal <- struct{}{}:
	default:
	}
}

// pickStream returns the highest-priority stream with queued frames. The
// caller must hold s.sendMu.
func (s *MuxSession) pickStream() *MuxStream {
	if len(s.pendingHigh) > 0 {
		for _, st := range s.pendingHigh {
			return st
		}
	}
	if len(s.pendingNormal) > 0 {
		for _, st := range s.pendingNormal {
			return st
		}
	}
	if len(s.pendingLow) > 0 {
		for _, st := range s.pendingLow {
			return st
		}
	}
	return nil
}

func (s *MuxSession) nextFrame() *muxFrame {
	for {
		s.sendMu.Lock()
		if len(s.ctrlQ) > 0 {
			f := s.ctrlQ[0]
			s.ctrlQ = s.ctrlQ[1:]
			s.sendMu.Unlock()
			return f
		}
		if st := s.pickStream(); st != nil {
			f := st.sendQ[0]
			st.sendQ = st.sendQ[1:]
			if len(st.sendQ) == 0 {
				st.pendingInSession = false
				delete(s.pendingHigh, st.id)
				delete(s.pendingNormal, st.id)
				delete(s.pendingLow, st.id)
			}
			s.sendMu.Unlock()
			return f
		}
		s.sendMu.Unlock()

		select {
		case <-s.sendSignal:
		case <-s.done:
			return nil
		}
	}
}

func (s *MuxSession) writeLoop() {
	for {
		f := s.nextFrame()
		if f == nil {
			return
		}
		if err := s.writeFrame(f); err != nil {
			// If the session is already closed we simply exit. Otherwise the
			// connection has failed; tear down the session.
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
	}
}

func (s *MuxSession) writeFrame(f *muxFrame) error {
	header := make([]byte, muxHeaderSize)
	header[0] = muxVersion
	header[1] = byte(f.cmd)
	header[2] = f.priority
	binary.BigEndian.PutUint32(header[4:8], uint32(len(f.payload)))
	binary.BigEndian.PutUint32(header[8:12], f.sid)

	buf := header
	if len(f.payload) > 0 {
		buf = make([]byte, muxHeaderSize+len(f.payload))
		copy(buf, header)
		copy(buf[muxHeaderSize:], f.payload)
	}
	if _, err := writeFull(s.conn, buf); err != nil {
		return err
	}

	atomic.AddUint64(&DefaultSnmp.MuxFramesSent, 1)
	if f.cmd == muxCmdPSH {
		atomic.AddUint64(&DefaultSnmp.MuxBytesSent, uint64(len(f.payload)))
	}
	return nil
}

func (s *MuxSession) readLoop() {
	header := make([]byte, muxHeaderSize)
	for {
		if s.isClosed() {
			return
		}
		if _, err := io.ReadFull(s.conn, header); err != nil {
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
		if header[0] != muxVersion {
			if !s.isClosed() {
				s.fail(errors.New("kcp: invalid mux version"))
			}
			return
		}
		cmd := muxCmd(header[1])
		priority := header[2]
		length := binary.BigEndian.Uint32(header[4:8])
		sid := binary.BigEndian.Uint32(header[8:12])
		if length > muxMaxPayload {
			if !s.isClosed() {
				s.fail(errors.New("kcp: mux frame too large"))
			}
			return
		}

		payload := make([]byte, length)
		if _, err := io.ReadFull(s.conn, payload); err != nil {
			if !s.isClosed() {
				s.fail(err)
			}
			return
		}
		atomic.AddUint64(&DefaultSnmp.MuxFramesReceived, 1)

		switch cmd {
		case muxCmdSYN:
			s.handleSYN(sid, priority, payload)
		case muxCmdFIN:
			s.handleFIN(sid)
		case muxCmdPSH:
			s.handlePSH(sid, payload)
		case muxCmdUPD:
			s.handleUPD(sid, payload)
		}
	}
}

func (s *MuxSession) handleSYN(sid uint32, priority uint8, payload []byte) {
	if sid == 0 {
		return
	}
	s.mu.Lock()
	if s.isClosed() {
		s.mu.Unlock()
		return
	}
	st, ok := s.streams[sid]
	if !ok {
		st = newMuxStream(s, sid, priority)
		s.streams[sid] = st
		s.acceptQueue = append(s.acceptQueue, st)
		if len(payload) >= 8 {
			st.setSendWindow(binary.BigEndian.Uint64(payload))
		}
		s.mu.Unlock()

		atomic.AddUint64(&DefaultSnmp.MuxStreamsOpened, 1)
		s.acceptSig.notify()
		st.sendWindowUpdate()
		return
	}
	if len(payload) >= 8 {
		st.setSendWindow(binary.BigEndian.Uint64(payload))
	}
	s.mu.Unlock()
	st.sendWindowUpdate()
}

func (s *MuxSession) handleFIN(sid uint32) {
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if ok {
		st.notifyRemoteClosed()
	}
}

func (s *MuxSession) handlePSH(sid uint32, payload []byte) {
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if !ok {
		return
	}
	st.mu.Lock()
	st.recvBuf = append(st.recvBuf, payload...)
	st.freeWindow -= uint64(len(payload))
	st.mu.Unlock()
	st.readSig.notify()
	atomic.AddUint64(&DefaultSnmp.MuxBytesReceived, uint64(len(payload)))
}

func (s *MuxSession) handleUPD(sid uint32, payload []byte) {
	if len(payload) < 8 {
		return
	}
	s.mu.Lock()
	st, ok := s.streams[sid]
	s.mu.Unlock()
	if ok {
		st.setSendWindow(binary.BigEndian.Uint64(payload))
	}
}

func (s *MuxSession) removeStream(st *MuxStream) {
	s.mu.Lock()
	if cur, ok := s.streams[st.id]; ok && cur == st {
		delete(s.streams, st.id)
		atomic.AddUint64(&DefaultSnmp.MuxStreamsClosed, 1)
	}
	s.mu.Unlock()
}

func (s *MuxSession) fail(err error) {
	var called bool
	s.closeOnce.Do(func() {
		called = true
		s.closeErr = err
		close(s.done)

		s.mu.Lock()
		streams := make([]*MuxStream, 0, len(s.streams))
		for _, st := range s.streams {
			streams = append(streams, st)
		}
		s.streams = make(map[uint32]*MuxStream)
		s.mu.Unlock()

		for _, st := range streams {
			st.notifySessionClose()
			atomic.AddUint64(&DefaultSnmp.MuxStreamsClosed, 1)
		}
		s.acceptSig.notify()
		s.signalSend()
		_ = s.conn.Close()
	})
	_ = called
}

func writeFull(w io.Writer, b []byte) (int, error) {
	total := 0
	for len(b) > 0 {
		n, err := w.Write(b)
		total += n
		b = b[n:]
		if err != nil {
			return total, err
		}
		if n == 0 {
			return total, io.ErrShortWrite
		}
	}
	return total, nil
}
