+func read(t *testing.T, in string, structured bool) *model.Value {
+	t.Helper()
+	options := parsing.DefaultReaderOptions()
+	if structured {
+		options.Ext["html-mode"] = "structured"
+	}
+	r, err := html.HTML.NewReader(options)
+	if err != nil {
+		t.Fatalf("unexpected error: %s", err)
+	}
+	v, err := r.Read([]byte(in))
+	if err != nil {
+		t.Fatalf("unexpected error: %s", err)
+	}
+	return v
+}

+func mustGetMapKey(t *testing.T, v *model.Value, key string) *model.Value {
+	t.Helper()
+	got, err := v.GetMapKey(key)
+	if err != nil {
+		t.Fatalf("error getting key %q: %s", key, err)
+	}
+	return got
+}

+func mustString(t *testing.T, v *model.Value) string {
+	t.Helper()
+	s, err := v.StringValue()
+	if err != nil {
+		t.Fatalf("error getting string value: %s", err)
+	}
+	return s

+func write(t *testing.T, v *model.Value, compact bool) string {
+	t.Helper()
+	options := parsing.DefaultWriterOptions()
+	if compact {
+		options = parsing.WriterOptions{Compact: true}
+	}
+	w, err := html.HTML.NewWriter(options)
+	if err != nil {
+		t.Fatalf("unexpected error: %s", err)
+	}
+	out, err := w.Write(v)
+	if err != nil {
+		t.Fatalf("unexpected error: %s", err)
+	}
+	return string(out)
+}

+func TestWriter_RoundTrip(t *testing.T) {
+	r, err := html.HTML.NewReader(parsing.DefaultReaderOptions())
+	if err != nil {
+		t.Fatal(err)
+	}
+	w, err := html.HTML.NewWriter(parsing.DefaultWriterOptions())
+	if err != nil {
+		t.Fatal(err)
+	}
+
+	in := `<html><head><title>T</title></head><body><div class="a">Hello <b>world</b></div><ul><li>one<li>two</ul><br><img src="x.png"><script>if (a < b) x = "&amp;"</script></body></html>`
+	read, err := r.Read([]byte(in))
+	if err != nil {
+		t.Fatal(err)
+	}
+	out, err := w.Write(read)
+	if err != nil {
+		t.Fatal(err)
+	}
+
+	// Re-read the output and ensure it matches the original model.
+	readAgain, err := r.Read(out)
+	if err != nil {
+		t.Fatal(err)
+	}
+	equal, err := read.Equal(readAgain)
+	if err != nil {
+		t.Fatal(err)
+	}
+	eqBool, err := equal.BoolValue()
+	if err != nil {
+		t.Fatal(err)
+	}
+	if !eqBool {
+		t.Fatalf("round trip mismatch:\nfirst:\n%s\nsecond:\n%s", read, readAgain)
+	}
