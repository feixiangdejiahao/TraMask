+#[test]
+fn parent_child_cancellation() {
+    let mut context = Context::default();
+    let parent = context.new_evaluation_handle();
+    let child = parent.child();
+    let grandchild = child.child();
+
+    // Child cancellation does not cancel the parent.
+    assert!(child.cancel_with_reason(js_string!("child reason")));
+    assert!(child.is_cancelled());
+    assert!(!parent.is_cancelled());
+    assert!(grandchild.is_cancelled());
+
+    // The child has its own first effective reason.
+    assert_eq!(
+        child
+            .cancellation_reason(&mut context)
+            .expect("child must have a reason"),
+        JsValue::from(js_string!("child reason"))
+    );
+    // The grandchild inherits the child's reason.
+    assert_eq!(
+        grandchild
+            .cancellation_reason(&mut context)
+            .expect("grandchild must have a reason"),
+        JsValue::from(js_string!("child reason"))
+    );
+    assert!(parent.cancellation_reason(&mut context).is_none());
+
+    // Parent cancellation cascades to descendants.
+    assert!(parent.cancel_with_reason(js_string!("parent reason")));
+    assert!(parent.is_cancelled());
+    assert!(!child.cancel());
+    assert!(!grandchild.cancel_with_reason(js_string!("grandchild reason")));
+
+    // A descendant with its own first effective reason keeps it.
+    assert_eq!(
+        child
+            .cancellation_reason(&mut context)
+            .expect("child must have a reason"),
+        JsValue::from(js_string!("child reason"))
+    );
+
+    // Other descendants inherit the parent reason.
+    let sibling = parent.child();
+    assert!(sibling.is_cancelled());
+    assert_eq!(
+        sibling
+            .cancellation_reason(&mut context)
+            .expect("sibling must have a reason"),
+        JsValue::from(js_string!("parent reason"))
+    );
+}
