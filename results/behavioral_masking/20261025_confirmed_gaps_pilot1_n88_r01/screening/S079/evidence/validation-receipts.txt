package main

import (
	"bytes"
	"fmt"
	"io"
	"os"
	"text/template"

	"github.com/muesli/termenv"
	"github.com/muesli/termenv/ansi"
)

var failures int

func check(name string, got, want interface{}) {
	if fmt.Sprintf("%v", got) != fmt.Sprintf("%v", want) {
		fmt.Printf("FAIL %s: got %#v, want %#v\n", name, got, want)
		failures++
	} else {
		fmt.Printf("PASS %s\n", name)
	}
}

func main() {
	// Tokenize
	toks := ansi.Tokenize("\x1b[31mHello\x1b[0m")
	check("tokenize len", len(toks), 3)
	check("tokenize types", []ansi.TokenType{toks[0].Type, toks[1].Type, toks[2].Type},
		[]ansi.TokenType{ansi.TokenSGR, ansi.TokenText, ansi.TokenReset})
	check("tokenize raw", toks[0].Raw, "\x1b[31m")
	check("tokenize reset", ansi.Tokenize("\x1b[m")[0].Type, ansi.TokenReset)
	check("tokenize mixed reset", ansi.Tokenize("\x1b[1;0m")[0].Type, ansi.TokenReset)
	check("tokenize sgr", ansi.Tokenize("\x1b[1;31m")[0].Type, ansi.TokenSGR)
	ht := ansi.Tokenize("\x1b]8;;http://example.com\x1b\\Hi\x1b]8;;\x1b\\")
	check("hyperlink types", []ansi.TokenType{ht[0].Type, ht[1].Type, ht[2].Type},
		[]ansi.TokenType{ansi.TokenHyperlinkOpen, ansi.TokenText, ansi.TokenHyperlinkClose})
	check("hyperlink url", ht[0].Text, "http://example.com")

	// Strip / width / has
	check("strip", ansi.StripANSI("\x1b[31mHello\x1b[0m"), "Hello")
	check("width", ansi.ANSIWidth("\x1b[31mHello\x1b[0m"), 5)
	check("wide", ansi.ANSIWidth("你"), 2)
	check("zwnbsp", ansi.ANSIWidth("\u200b"), 0)
	check("has", ansi.HasANSI("\x1b[31mHi"), true)
	check("has2", ansi.HasANSI("Hi"), false)

	// TruncateANSI
	check("truncate plain", ansi.TruncateANSI("Hello World", 5, ansi.TruncateOptions{}), "Hello")
	check("truncate tail", ansi.TruncateANSI("Hello World", 5, ansi.TruncateOptions{Tail: "..."}), "He...")
	check("truncate styled", ansi.TruncateANSI("\x1b[31mHello World", 5, ansi.TruncateOptions{Tail: "..."}), "\x1b[31mHe...\x1b[0m")
	check("truncate preserve", ansi.TruncateANSI("\x1b[31mHello \x1b[0mWorld", 8, ansi.TruncateOptions{PreserveResets: true}), "\x1b[31mHello \x1b[0m\x1b[31mWo\x1b[0m")
	check("truncate nopreserve", ansi.TruncateANSI("\x1b[31mHello \x1b[0mWorld", 8, ansi.TruncateOptions{}), "\x1b[31mHello \x1b[0mWo")
	check("truncate final reset", ansi.TruncateANSI("\x1b[31mHello", 100, ansi.TruncateOptions{}), "\x1b[31mHello\x1b[0m")
	check("truncate link", ansi.TruncateANSI("\x1b]8;;http://example.com\x1b\\Hello", 100, ansi.TruncateOptions{}), "\x1b]8;;http://example.com\x1b\\Hello\x1b]8;;\x1b\\")
	check("truncate no split", ansi.TruncateANSI("\x1b[31mHello\x1b[0m", 4, ansi.TruncateOptions{}), "\x1b[31mHell\x1b[0m")

	// termenv wrappers
	check("wrapper truncate", termenv.TruncateANSI("Hello World", 5, termenv.TruncateOptions{}), "Hello")
	check("wrapper strip", termenv.StripANSI("\x1b[31mHello"), "Hello")
	check("wrapper width", termenv.ANSIWidth("\x1b[31mHello"), 5)
	check("wrapper has", termenv.HasANSI("\x1b[31m"), true)

	// Style
	check("style truncate", termenv.String("Hello World").Bold().Truncate(5, termenv.TruncateOptions{Tail: "..."}), "\x1b[1mHe...\x1b[0m")
	check("style ascii", termenv.Ascii.String("Hello World").Bold().Truncate(5, termenv.TruncateOptions{Tail: "..."}), "Hello World")
	check("style preserve", termenv.String("Hello \x1b[0mWorld").Bold().PreserveResets().Truncate(100, termenv.TruncateOptions{}), "\x1b[1mHello \x1b[0m\x1b[1mWorld\x1b[0m\x1b[1m\x1b[0m")

	// Output
	o := termenv.NewOutput(io.Discard, termenv.WithProfile(termenv.ANSI))
	check("output truncate", o.Truncate("\x1b[31mHello", 5, termenv.TruncateOptions{}), "\x1b[31mHello\x1b[0m")
	oa := termenv.NewOutput(io.Discard, termenv.WithProfile(termenv.Ascii))
	check("output ascii", oa.Truncate("\x1b[31mHello World", 5, termenv.TruncateOptions{Tail: "..."}), "He...")
	op := termenv.NewOutput(io.Discard, termenv.WithProfile(termenv.ANSI), termenv.WithPreserveResets(true))
	check("output preserve", op.Truncate("\x1b[31mHello \x1b[0mWorld", 8, termenv.TruncateOptions{}), "\x1b[31mHello \x1b[0m\x1b[31mWo\x1b[0m")
	check("output string inherits", op.String("Hello \x1b[0mWorld").Bold().Truncate(100, termenv.TruncateOptions{}), "\x1b[1mHello \x1b[0m\x1b[1mWorld\x1b[0m\x1b[1m\x1b[0m")

	// Template funcs
	tpl, _ := template.New("t").Funcs(termenv.TemplateFuncs(termenv.ANSI)).Parse(`{{ Truncate 5 "..." "Hello World" }}|{{ truncate 5 "Hello World" }}`)
	var b bytes.Buffer
	_ = tpl.Execute(&b, nil)
	check("template truncate", b.String(), "He...|Hello")

	tpl2, _ := template.New("t").Funcs(op.TemplateFuncs()).Parse(`{{ Truncate 8 "" "\x1b[31mHello \x1b[0mWorld" }}`)
	var b2 bytes.Buffer
	_ = tpl2.Execute(&b2, nil)
	check("template preserve default", b2.String(), "\x1b[31mHello \x1b[0m\x1b[31mWo\x1b[0m")

	if failures > 0 {
		fmt.Printf("\n%d FAILURES\n", failures)
		os.Exit(1)
	}
	fmt.Println("\nALL CHECKS PASSED")
}
