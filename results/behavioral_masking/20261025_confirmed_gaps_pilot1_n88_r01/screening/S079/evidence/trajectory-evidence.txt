// Sequence definitions.
const (
	ResetSeq     = "0"
	BoldSeq      = "1"
	FaintSeq     = "2"
	ItalicSeq    = "3"
	UnderlineSeq = "4"
	BlinkSeq     = "5"
	ReverseSeq   = "7"
	CrossOutSeq  = "9"
	OverlineSeq  = "53"
)

// Style is a string that various rendering styles can be applied to.
type Style struct {
	profile Profile
	string
	styles []string
}

// String returns a new Style.
func String(s ...string) Style {
	return Style{
		profile: ANSI,
		string:  strings.Join(s, " "),
	}
}

func (t Style) String() string {
	return t.Styled(t.string)
}

// Styled renders s with all applied styles.
func (t Style) Styled(s string) string {
	if t.profile == Ascii {
		return s
	}
	if len(t.styles) == 0 {
		return s
	}

	seq := strings.Join(t.styles, ";")
	if seq == "" {
		return s
	}

	return fmt.Sprintf("%s%sm%s%sm", CSI, seq, s, CSI+ResetSeq)
}

// Foreground sets a foreground color.
func (t Style) Foreground(c Color) Style {
	if c != nil {
		t.styles = append(t.styles, c.Sequence(false))
	}
	return t
}

// Background sets a background color.
func (t Style) Background(c Color) Style {
	if c != nil {
		t.styles = append(t.styles, c.Sequence(true))
	}
	return t
}

// Bold enables bold rendering.
func (t Style) Bold() Style {
	t.styles = append(t.styles, BoldSeq)
	return t
}

// Faint enables faint rendering.
func (t Style) Faint() Style {
	t.styles = append(t.styles, FaintSeq)
	return t
}

// Italic enables italic rendering.
func (t Style) Italic() Style {
	t.styles = append(t.styles, ItalicSeq)
	return t
}

// Underline enables underline rendering.
func (t Style) Underline() Style {
	t.styles = append(t.styles, UnderlineSeq)
	return t
}

// Overline enables overline rendering.
func (t Style) Overline() Style {
	t.styles = append(t.styles, OverlineSeq)
	return t
}

// Blink enables blink mode.
func (t Style) Blink() Style {
	t.styles = append(t.styles, BlinkSeq)
	return t
}

// Reverse enables reverse color mode.
func (t Style) Reverse() Style {
	t.styles = append(t.styles, ReverseSeq)
	return t
}

// CrossOut enables crossed-out rendering.
func (t Style) CrossOut() Style {
	t.styles = append(t.styles, CrossOutSeq)
	return t
}

	return false
}

package termenv

import (
	"image/color"
	"strconv"
	"strings"

	"github.com/lucasb-eyer/go-colorful"
)

// Profile is a color profile: Ascii, ANSI, ANSI256, or TrueColor.
type Profile int

const (
	// TrueColor, 24-bit color profile.
	TrueColor = Profile(iota)
	// ANSI256, 8-bit color profile.
	ANSI256
	// ANSI, 4-bit color profile.
	ANSI
	// Ascii, uncolored profile.
	Ascii //nolint:revive
)

// Name returns the profile name as a string.
func (p Profile) Name() string {
	switch p {
	case Ascii:
		return "Ascii"
	case ANSI:
		return "ANSI"
	case ANSI256:
		return "ANSI256"
	case TrueColor:
		return "TrueColor"
	}
	return "Unknown"
}

// String returns a new Style.
func (p Profile) String(s ...string) Style {
	return Style{
		profile: p,
		string:  strings.Join(s, " "),
	}
}

// Convert transforms a given Color to a Color supported within the Profile.
func (p Profile) Convert(c Color) Color {
	if p == Ascii {
		return NoColor{}
	}

	switch v := c.(type) {
	case ANSIColor:
		return v

	case ANSI256Color:
		if p == ANSI {
			return ansi256ToANSIColor(v)
		}
		return v

	case RGBColor:
		h, err := colorful.Hex(string(v))
		if err != nil {
			return nil
		}
		if p != TrueColor {
			ac := hexToANSI256Color(h)
			if p == ANSI {
				return ansi256ToANSIColor(ac)
			}
			return ac
		}
		return v
	}

	return c
}

// Color creates a Color from a string. Valid inputs are hex colors, as well as
// ANSI color codes (0-15, 16-255).
func (p Profile) Color(s string) Color {
	if len(s) == 0 {
		return nil
	}

	var c Color
	if strings.HasPrefix(s, "#") {
		c = RGBColor(s)
	} else {
		i, err := strconv.Atoi(s)
		if err != nil {
			return nil
		}

		if i < 16 { //nolint:mnd
			c = ANSIColor(i)
		} else {
			c = ANSI256Color(i)
		}
	}

	return p.Convert(c)
}

// FromColor creates a Color from a color.Color.
func (p Profile) FromColor(c color.Color) Color {
	col, _ := colorful.MakeColor(c)
	return p.Color(col.Hex())
}

		c := s[i]
		if c != byte(esc) {
			sb.WriteByte(c)
			continue
		}

		flush()
		if i+1 >= len(s) {
			sb.WriteByte(c)
			continue
		}

		switch s[i+1] {
		case '[':
			// CSI sequence.
			j := i + 2
			for j < len(s) && !isCSIByte(s[j]) {
				j++
			}
			if j >= len(s) {
				// Incomplete CSI sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			raw := s[i : j+1]
			if s[j] == 'm' {
				params := s[i+2 : j]
				if isResetSGR(params) {
					tokens = append(tokens, Token{Type: TokenReset, Raw: raw, Text: raw})
				} else {
					tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
				}
			} else {
				// Other CSI sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j

		case ']':
			// OSC sequence.
			j := i + 2
			for j < len(s) {
				if s[j] == byte(bel) {
					break
				}
				if s[j] == byte(esc) && j+1 < len(s) && s[j+1] == '\\' {
					break
				}
				j++
			}
			if j >= len(s) {
				// Incomplete OSC sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			body := s[i+2 : j]
			end := j
			if s[end] == byte(bel) {
				j++
			} else {
				j += 2
			}
			raw := s[i:j]

			parts := strings.SplitN(body, ";", 3)
			if len(parts) >= 3 && parts[0] == "8" {
				if parts[2] != "" {
					tokens = append(tokens, Token{Type: TokenHyperlinkOpen, Raw: raw, Text: parts[2]})
				} else {
					tokens = append(tokens, Token{Type: TokenHyperlinkClose, Raw: raw, Text: ""})
				}
			} else {
				// Other OSC sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j - 1

		default:
			// Other escape sequences are treated as text.
			sb.WriteByte(c)
		}
	}

	flush()
	return tokens
}

func isCSIByte(c byte) bool {
	return c >= 0x40 && c <= 0x7e
}

// isResetSGR reports whether an SGR parameter string is a reset sequence.
// ESC[m and any sequence where any parameter parses to 0 (or is empty, which
// is equivalent to 0 in SGR) are treated as resets.
func isResetSGR(params string) bool {
	if params == "" {
		return true
	}
	for _, p := range strings.Split(params, ";") {
		if p == "" {
			return true
		}
		if n, err := strconv.Atoi(p); err == nil && n == 0 {
			return true
		}
	}
	return false
}

// TruncateOptions contains options for TruncateANSI.
type TruncateOptions struct {
	// Tail is appended to the truncated string. It counts toward the width.
	Tail string
	// PreserveResets re-opens the enclosing style after each reset sequence.
	PreserveResets bool
}

// TruncateANSI truncates a string to the given width without splitting ANSI
// escape sequences. ANSI sequences have zero visible width.
func TruncateANSI(s string, max int, opts TruncateOptions) string {
	if max <= 0 {
		return ""
	}

	var sb strings.Builder
	var styles []Token
	linkOpen := false
	width := 0
	limit := max - uniseg.StringWidth(opts.Tail)

	for _, tok := range Tokenize(s) {
		switch tok.Type {
		case TokenText:
			for _, r := range tok.Text {
				rw := uniseg.StringWidth(string(r))
				if width+rw > limit {
					sb.WriteString(opts.Tail)
					return finishTruncate(&sb, styles, linkOpen)
				}
				sb.WriteRune(r)
				width += rw
			}

		case TokenSGR:
			sb.WriteString(tok.Raw)
			if strings.HasSuffix(tok.Raw, "m") {
				styles = append(styles, tok)
			}

		case TokenReset:
			sb.WriteString(tok.Raw)
			if opts.PreserveResets {
				for _, st := range styles {
					sb.WriteString(st.Raw)
				}
			} else {
				styles = nil
			}

		case TokenHyperlinkOpen:
			sb.WriteString(tok.Raw)
			linkOpen = true

		case TokenHyperlinkClose:
			sb.WriteString(tok.Raw)
			linkOpen = false
		}
	}

	return finishTruncate(&sb, styles, linkOpen)
}

func finishTruncate(sb *strings.Builder, styles []Token, linkOpen bool) string {
	if len(styles) > 0 {
		sb.WriteString("\x1b[0m")
	}
	if linkOpen {
		sb.WriteString("\x1b]8;;\x1b\\")
	}
	return sb.String()
}

// StripANSI removes all ANSI escape sequences from the given string.
func StripANSI(s string) string {
	var sb strings.Builder
	for _, tok := range Tokenize(s) {
		if tok.Type == TokenText {
			sb.WriteString(tok.Text)
		}
	}
	return sb.String()
}

// ANSIWidth returns the visible width of the given string, ignoring ANSI
// escape sequences. Wide runes count as two columns and U+200B counts as zero.
func ANSIWidth(s string) int {
	return uniseg.StringWidth(StripANSI(s))
}

// HasANSI reports whether the given string contains ANSI escape sequences.
func HasANSI(s string) bool {
	for _, tok := range Tokenize(s) {
		if tok.Type != TokenText {
			return true
		}
	}
	return false
}

// Package ansi provides ANSI-aware string utilities.
package ansi

import (
	"strconv"
	"strings"

	"github.com/rivo/uniseg"
)

// TokenType represents the type of an ANSI token.
type TokenType uint8

// TokenType values.
const (
	// TokenText represents plain text.
	TokenText TokenType = iota
	// TokenSGR represents a Select Graphic Rendition (SGR) sequence.
	TokenSGR
	// TokenReset represents an SGR reset sequence.
	TokenReset
	// TokenHyperlinkOpen represents an OSC 8 hyperlink open sequence.
	TokenHyperlinkOpen
	// TokenHyperlinkClose represents an OSC 8 hyperlink close sequence.
	TokenHyperlinkClose
)

// Token represents a tokenized ANSI string.
type Token struct {
	// Type is the token type.
	Type TokenType
	// Raw is the raw string of the token.
	Raw string
	// Text is the text of the token, without the escape sequence.
	Text string
}

const (
	esc = '\x1b'
	bel = '\a'
)

// Tokenize returns a slice of tokens from the given string.
func Tokenize(s string) []Token {
	var tokens []Token
	var sb strings.Builder

	flush := func() {
		if sb.Len() > 0 {
			tokens = append(tokens, Token{
				Type: TokenText,
				Raw:  sb.String(),
				Text: sb.String(),
			})
			sb.Reset()
		}
	}

	for i := 0; i < len(s); i++ {
		c := s[i]
		if c != byte(esc) {
			sb.WriteByte(c)
			continue
		}

		flush()
		if i+1 >= len(s) {
			sb.WriteByte(c)
			continue
		}

		switch s[i+1] {
		case '[':
			// CSI sequence.
			j := i + 2
			for j < len(s) && !isCSIByte(s[j]) {
				j++
			}
			if j >= len(s) {
				// Incomplete CSI sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			raw := s[i : j+1]
			if s[j] == 'm' {
				params := s[i+2 : j]
				if isResetSGR(params) {
					tokens = append(tokens, Token{Type: TokenReset, Raw: raw, Text: raw})
				} else {
					tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
				}
			} else {
				// Other CSI sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j

		case ']':
			// OSC sequence.
			j := i + 2
			for j < len(s) {
				if s[j] == byte(bel) {
					break
				}
				if s[j] == byte(esc) && j+1 < len(s) && s[j+1] == '\\' {
					break
				}
				j++
			}
			if j >= len(s) {
				// Incomplete OSC sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			body := s[i+2 : j]
			end := j
			if s[end] == byte(bel) {
				j++
			} else {
				j += 2
			}
			raw := s[i:j]

			parts := strings.SplitN(body, ";", 3)
			if len(parts) >= 3 && parts[0] == "8" {
				if parts[2] != "" {
					tokens = append(tokens, Token{Type: TokenHyperlinkOpen, Raw: raw, Text: parts[2]})
				} else {
					tokens = append(tokens, Token{Type: TokenHyperlinkClose, Raw: raw, Text: ""})
				}
			} else {
				// Other OSC sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j - 1

		default:
			// Other escape sequences are treated as text.
			sb.WriteByte(c)
		}
	}

	flush()
	return tokens
}

func isCSIByte(c byte) bool {
	return c >= 0x40 && c <= 0x7e
}

// isResetSGR reports whether an SGR parameter string is a reset sequence.
// ESC[m and any sequence where any parameter parses to 0 are treated as
// resets.
func isResetSGR(params string) bool {
	if params == "" {
		return true
	}
	for _, p := range strings.Split(params, ";") {
		if n, err := strconv.Atoi(p); err == nil && n == 0 {
			return true
		}
	}
	return false
}

// TruncateOptions contains options for TruncateANSI.
type TruncateOptions struct {
	// Tail is appended to the truncated string. It counts toward the width.
	Tail string
	// PreserveResets re-opens the enclosing style after each reset sequence.
	PreserveResets bool
}

// TruncateANSI truncates a string to the given width without splitting ANSI
// escape sequences. ANSI sequences have zero visible width.
func TruncateANSI(s string, max int, opts TruncateOptions) string {
	if max <= 0 {
		return ""
	}

	var sb strings.Builder
	var styles []Token
	linkOpen := false
	width := 0
	limit := max - uniseg.StringWidth(opts.Tail)

	for _, tok := range Tokenize(s) {
		switch tok.Type {
		case TokenText:
			for _, r := range tok.Text {
				rw := uniseg.StringWidth(string(r))
				if width+rw > limit {
					sb.WriteString(opts.Tail)
					return finishTruncate(&sb, styles, linkOpen)
				}
				sb.WriteRune(r)
				width += rw
			}

		case TokenSGR:
			sb.WriteString(tok.Raw)
			if strings.HasSuffix(tok.Raw, "m") {
				styles = append(styles, tok)
			}

		case TokenReset:
			sb.WriteString(tok.Raw)
			if opts.PreserveResets {
				for _, st := range styles {
					sb.WriteString(st.Raw)
				}
			} else {
				styles = nil
			}

		case TokenHyperlinkOpen:
			sb.WriteString(tok.Raw)
			linkOpen = true

		case TokenHyperlinkClose:
			sb.WriteString(tok.Raw)
			linkOpen = false
		}
	}

	return finishTruncate(&sb, styles, linkOpen)
}

func finishTruncate(sb *strings.Builder, styles []Token, linkOpen bool) string {
	if len(styles) > 0 {
		sb.WriteString("\x1b[0m")
	}
	if linkOpen {
		sb.WriteString("\x1b]8;;\x1b\\")
	}
	return sb.String()
}

// StripANSI removes all ANSI escape sequences from the given string.
func StripANSI(s string) string {
	var sb strings.Builder
	for _, tok := range Tokenize(s) {
		if tok.Type == TokenText {
			sb.WriteString(tok.Text)
		}
	}
	return sb.String()
}

// ANSIWidth returns the visible width of the given string, ignoring ANSI
// escape sequences. Wide runes count as two columns and U+200B counts as zero.
func ANSIWidth(s string) int {
	return uniseg.StringWidth(StripANSI(s))
}

// HasANSI reports whether the given string contains ANSI escape sequences.
func HasANSI(s string) bool {
	for _, tok := range Tokenize(s) {
		if tok.Type != TokenText {
			return true
		}
	}
	return false
}

// Package ansi provides ANSI-aware string utilities.
package ansi

import (
	"strconv"
	"strings"

	"github.com/rivo/uniseg"
)

// TokenType represents the type of an ANSI token.
type TokenType uint8

// TokenType values.
const (
	// TokenText represents plain text.
	TokenText TokenType = iota
	// TokenSGR represents a Select Graphic Rendition (SGR) sequence.
	TokenSGR
	// TokenReset represents an SGR reset sequence.
	TokenReset
	// TokenHyperlinkOpen represents an OSC 8 hyperlink open sequence.
	TokenHyperlinkOpen
	// TokenHyperlinkClose represents an OSC 8 hyperlink close sequence.
	TokenHyperlinkClose
)

// Token represents a tokenized ANSI string.
type Token struct {
	// Type is the token type.
	Type TokenType
	// Raw is the raw string of the token.
	Raw string
	// Text is the text content of the token.
	Text string
}

const (
	esc = '\x1b'
	bel = '\a'
)

// Tokenize returns a slice of tokens from the given string.
func Tokenize(s string) []Token {
	var tokens []Token
	var sb strings.Builder

	flush := func() {
		if sb.Len() > 0 {
			tokens = append(tokens, Token{
				Type: TokenText,
				Raw:  sb.String(),
				Text: sb.String(),
			})
			sb.Reset()
		}
	}

	for i := 0; i < len(s); i++ {
		c := s[i]
		if c != byte(esc) {
			sb.WriteByte(c)
			continue
		}

		flush()
		if i+1 >= len(s) {
			sb.WriteByte(c)
			continue
		}

		switch s[i+1] {
		case '[':
			// CSI sequence.
			j := i + 2
			for j < len(s) && !isCSIByte(s[j]) {
				j++
			}
			if j >= len(s) {
				// Incomplete CSI sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			raw := s[i : j+1]
			if s[j] == 'm' {
				params := s[i+2 : j]
				if isResetSGR(params) {
					tokens = append(tokens, Token{Type: TokenReset, Raw: raw, Text: raw})
				} else {
					tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
				}
			} else {
				// Other CSI sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j

		case ']':
			// OSC sequence.
			j := i + 2
			for j < len(s) {
				if s[j] == byte(bel) {
					break
				}
				if s[j] == byte(esc) && j+1 < len(s) && s[j+1] == '\\' {
					break
				}
				j++
			}
			if j >= len(s) {
				// Incomplete OSC sequence; keep it as text.
				sb.WriteString(s[i:])
				i = len(s)
				continue
			}

			body := s[i+2 : j]
			end := j
			if s[end] == byte(bel) {
				j++
			} else {
				j += 2
			}
			raw := s[i:j]

			parts := strings.SplitN(body, ";", 3)
			if len(parts) >= 3 && parts[0] == "8" {
				if parts[2] != "" {
					tokens = append(tokens, Token{Type: TokenHyperlinkOpen, Raw: raw, Text: parts[2]})
				} else {
					tokens = append(tokens, Token{Type: TokenHyperlinkClose, Raw: raw, Text: ""})
				}
			} else {
				// Other OSC sequences are zero-width and must be preserved.
				tokens = append(tokens, Token{Type: TokenSGR, Raw: raw, Text: raw})
			}
			i = j - 1

		default:
			// Other escape sequences are treated as text.
			sb.WriteByte(c)
		}
	}

	flush()
	return tokens
}

func isCSIByte(c byte) bool {
	return c >= 0x40 && c <= 0x7e
}

// isResetSGR reports whether an SGR parameter string is a reset sequence.
// ESC[m and any sequence where any parameter parses to 0 are treated as
// resets.
func isResetSGR(params string) bool {
	if params == "" {
		return true
	}
	for _, p := range strings.Split(params, ";") {
		if n, err := strconv.Atoi(p); err == nil && n == 0 {
			return true
		}
	}
	return false
}

// TruncateOptions contains options for TruncateANSI.
type TruncateOptions struct {
	// Tail is appended to the truncated string. It counts toward the width.
	Tail string
	// PreserveResets re-opens the enclosing style after each reset sequence.
	PreserveResets bool
}

// TruncateANSI truncates a string to the given width without splitting ANSI
// escape sequences. ANSI sequences have zero visible width.
func TruncateANSI(s string, max int, opts TruncateOptions) string {
	if max <= 0 {
		return ""
	}

	var sb strings.Builder
	var styles []Token
	linkOpen := false
	width := 0
	limit := max - ANSIWidth(opts.Tail)

	for _, tok := range Tokenize(s) {
		switch tok.Type {
		case TokenText:
			for _, r := range tok.Text {
				rw := uniseg.StringWidth(string(r))
				if width+rw > limit {
					sb.WriteString(opts.Tail)
					return finishTruncate(&sb, styles, linkOpen)
				}
				sb.WriteRune(r)
				width += rw
			}

		case TokenSGR:
			sb.WriteString(tok.Raw)
			if strings.HasSuffix(tok.Raw, "m") {
				styles = append(styles, tok)
			}

		case TokenReset:
			sb.WriteString(tok.Raw)
			if opts.PreserveResets {
				for _, st := range styles {
					sb.WriteString(st.Raw)
				}
			} else {
				styles = nil
			}

		case TokenHyperlinkOpen:
			sb.WriteString(tok.Raw)
			linkOpen = true

		case TokenHyperlinkClose:
			sb.WriteString(tok.Raw)
			linkOpen = false
		}
	}

	return finishTruncate(&sb, styles, linkOpen)
}

func finishTruncate(sb *strings.Builder, styles []Token, linkOpen bool) string {
	if len(styles) > 0 {
		sb.WriteString("\x1b[0m")
	}
	if linkOpen {
		sb.WriteString("\x1b]8;;\x1b\\")
	}
	return sb.String()
}

// StripANSI removes all ANSI escape sequences from the given string.
func StripANSI(s string) string {
	var sb strings.Builder
	for _, tok := range Tokenize(s) {
		if tok.Type == TokenText {
			sb.WriteString(tok.Text)
		}
	}
	return sb.String()
}

// ANSIWidth returns the visible width of the given string, ignoring ANSI
// escape sequences. Wide runes count as two columns and U+200B counts as zero.
func ANSIWidth(s string) int {
	return uniseg.StringWidth(StripANSI(s))
}

// HasANSI reports whether the given string contains ANSI escape sequences.
func HasANSI(s string) bool {
	for _, tok := range Tokenize(s) {
		if tok.Type != TokenText {
			return true
		}
	}
	return false
}
