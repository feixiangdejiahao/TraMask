    @contextmanager
    def _tee(self, channel: Literal["stdout", "stderr"]):
        """Capture output of a given standard stream and store it in history.

        Uses patching of write method for maximal compatibility,
        because ipykernel checks for instances of the stream class,
        and stream classes in ipykernel implement more complex logic.
        """
        stream = getattr(sys, channel)
        original_write = stream.write
        execution_count = self.execution_count

        def write(data, *args, **kwargs):
            """Write data to both the original destination and the capture dictionary."""
            result = original_write(data, *args, **kwargs)
            if any(
                [
                    self.display_pub.is_publishing,
                    self.displayhook.is_active,
                    self.showing_traceback,
                ]
            ):
                return result
            if not data:
                return result
            output_stream = None
            outputs_by_counter = self.history_manager.outputs
            output_type = "out_stream" if channel == "stdout" else "err_stream"
            if execution_count in outputs_by_counter:
                outputs = outputs_by_counter[execution_count]
                if outputs[-1].output_type == output_type:
                    output_stream = outputs[-1]
            if output_stream is None:
                output_stream = HistoryOutput(
                    output_type=output_type, bundle={"stream": []}
                )
                outputs_by_counter[execution_count].append(output_stream)

            output_stream.bundle["stream"].append(data)  # Append to existing stream
            return result

        stream.write = write
        yield
        stream.write = original_write

    def run_cell(
        self,
        raw_cell,
        store_history=False,
        silent=False,
        shell_futures=True,
        cell_id=None,
    ):
        """Run a complete IPython cell.

        Parameters
        ----------
        raw_cell : str
            The code (including IPython code such as %magic functions) to run.
        store_history : bool
            If True, the raw and translated cell will be stored in IPython's
            history. For user code calling back into IPython's machinery, this
            should be set to False.
        silent : bool
            If True, avoid side-effects, such as implicit displayhooks and
            and logging.  silent=True forces store_history=False.
        shell_futures : bool
            If True, the code will share future statements with the interactive
            shell. It will both be affected by previous __future__ imports, and
            any __future__ imports in the code will affect the shell. If False,
            __future__ imports are not shared in either direction.
        cell_id : str, optional
            A unique identifier for the cell. This is used in the messaging system
            to match output with execution requests and for tracking cell execution
            history across kernel restarts. In notebook contexts, this is typically
            a UUID generated by the frontend. If None, the kernel may generate an
            internal identifier or proceed without cell tracking capabilities.
        Returns
        -------
        result : :class:`ExecutionResult`
        """
        result = None
        with self._tee(channel="stdout"), self._tee(channel="stderr"):
            try:
                result = self._run_cell(
                    raw_cell, store_history, silent, shell_futures, cell_id
                )
            finally:
                self.events.trigger("post_execute")
                if not silent:
                    self.events.trigger("post_run_cell", result)
        return result

range_re = re.compile(r"""
(?P<start>\d+)?
((?P<sep>[\-:])
 (?P<end>\d+)?)?

def extract_code_ranges(ranges_str):
    """Turn a string of range for %%load into 2-tuples of (start, stop)
    ready to use as a slice of the content split by lines.

    Examples
    --------
    list(extract_input_ranges("5-10 2"))
    [(4, 10), (1, 2)]
    """
    for range_str in ranges_str.split():
        rmatch = range_re.match(range_str)
        if not rmatch:
            continue
        sep = rmatch.group("sep")
        start = rmatch.group("start")
        end = rmatch.group("end")

        if sep == '-':
            start = int(start) - 1 if start else None
            end = int(end) if end else None
        elif sep == ':':
            start = int(start) - 1 if start else None
            end = int(end) - 1 if end else None
        else:
            end = int(start)
            start = int(start) - 1
        yield (start, end)


def extract_symbols(code, symbols):
    """
    Return a tuple  (blocks, not_found)
    where ``blocks`` is a list of code fragments
    for each symbol parsed from code, and ``not_found`` are
    symbols not found in the code.

    For example::

        In [1]: code = '''a = 10
           ...: def b(): return 42
           ...: class A: pass'''

        In [2]: extract_symbols(code, 'A,b,z')
        Out[2]: (['class A: pass\\n', 'def b(): return 42\\n'], ['z'])
    """
    symbols = symbols.split(',')

    # this will raise SyntaxError if code isn't valid Python
    py_code = ast.parse(code)

    marks = [(getattr(s, 'name', None), s.lineno) for s in py_code.body]
    code = code.split('\n')

    symbols_lines = {}
    
    # we already know the start_lineno of each symbol (marks). 
    # To find each end_lineno, we traverse in reverse order until each 
    # non-blank line
    end = len(code)  
    for name, start in reversed(marks):
        while not code[end - 1].strip():
            end -= 1
        if name:
            symbols_lines[name] = (start - 1, end)
        end = start - 1

    # Now symbols_lines is a map
    # {'symbol_name': (start_lineno, end_lineno), ...}
    
    # fill a list with chunks of codes for each requested symbol
    blocks = []
    not_found = []
    for symbol in symbols:
        if symbol in symbols_lines:
            start, end = symbols_lines[symbol]
            blocks.append('\n'.join(code[start:end]) + '\n')
        else:
            not_found.append(symbol)

    return blocks, not_found

def strip_initial_indent(lines):
    """For %load, strip indent from lines until finding an unindented line.

    https://github.com/ipython/ipython/issues/9775
    """
    indent_re = re.compile(r'\s+')

    it = iter(lines)
    first_line = next(it)
    indent_match = indent_re.match(first_line)

    if indent_match:
        # First line was indented
        indent = indent_match.group()
        yield first_line[len(indent):]

        for line in it:
            if line.startswith(indent):
                yield line[len(indent) :]
            elif line in ("\n", "\r\n") or len(line) == 0:
                yield line
            else:
                # Less indented than the first line - stop dedenting
                yield line
                break
    else:
        yield first_line

    # Pass the remaining lines through without dedenting
    for line in it:
        yield line

class DisplayTrap(Configurable):
    """Object to manage sys.displayhook.

    This came from IPython.core.kernel.display_hook, but is simplified
    (no callbacks or formatters) until more of the core is refactored.
    """

    hook = Any()

    def __init__(self, hook=None):
        super(DisplayTrap, self).__init__(hook=hook, config=None)
        self.old_hook = None
        # We define this to track if a single BuiltinTrap is nested.
        # Only turn off the trap when the outermost call to __exit__ is made.
        self._nested_level = 0

    def __enter__(self):
        if self._nested_level == 0:
            self.set()
        self._nested_level += 1
        return self

    def __exit__(self, type, value, traceback):
        if self._nested_level == 1:
            self.unset()
        self._nested_level -= 1
        # Returning False will cause exceptions to propagate
        return False

    @property
    def is_active(self) -> bool:
        return self._nested_level != 0

    def set(self):
        """Set the hook."""
        if sys.displayhook is not self.hook:
            self.old_hook = sys.displayhook
            sys.displayhook = self.hook

    def unset(self):
        """Unset the hook."""
        sys.displayhook = self.old_hook

@undoc
class MagicHelpFormatter(argparse.RawDescriptionHelpFormatter):
    """A HelpFormatter with a couple of changes to meet our needs.
    """
    # Modified to dedent text.
    def _fill_text(self, text, width, indent):
        return argparse.RawDescriptionHelpFormatter._fill_text(self, dedent(text), width, indent)

    # Modified to wrap argument placeholders in <> where necessary.
    def _format_action_invocation(self, action):
        if not action.option_strings:
            metavar, = self._metavar_formatter(action, action.dest)(1)
            return metavar

        else:
            parts = []

            # if the Optional doesn't take a value, format is:
            #    -s, --long
            if action.nargs == 0:
                parts.extend(action.option_strings)

            # if the Optional takes a value, format is:
            #    -s ARGS, --long ARGS
            else:
                default = action.dest.upper()
                args_string = self._format_args(action, default)
                # IPYTHON MODIFICATION: If args_string is not a plain name, wrap
                # it in <> so it's valid RST.
                if not NAME_RE.match(args_string):
                    args_string = "<%s>" % args_string
                for option_string in action.option_strings:
                    parts.append('%s %s' % (option_string, args_string))

            return ', '.join(parts)

    # Override the default prefix ('usage') to our % magic escape,
    # in a code block.
    def add_usage(self, usage, actions, groups, prefix="::\n\n  %"):
        super(MagicHelpFormatter, self).add_usage(usage, actions, groups, prefix)

    """ Construct an argument parser using the function decorations.
    """

"""This module contains classes - NamedFileInTemporaryDirectory, TemporaryWorkingDirectory.

These classes add extra features such as creating a named file in temporary directory and
creating a context manager for the working directory which is also temporary.

import os as _os
from pathlib import Path
from tempfile import TemporaryDirectory


class NamedFileInTemporaryDirectory:
    def __init__(self, filename, mode, bufsize=-1, add_to_syspath=False, **kwds):
        """
        Open a file named `filename` in a temporary directory.

        This context manager is preferred over `NamedTemporaryFile` in
        stdlib `tempfile` when one needs to reopen the file.

        Arguments `mode` and `bufsize` are passed to `open`.
        Rest of the arguments are passed to `TemporaryDirectory`.

        """
        self._tmpdir = TemporaryDirectory(**kwds)
        path = Path(self._tmpdir.name) / filename
        encoding = None if "b" in mode else "utf-8"
        self.file = open(path, mode, bufsize, encoding=encoding)

    def cleanup(self):
        self.file.close()
        self._tmpdir.cleanup()

    __del__ = cleanup

    def __enter__(self):
        return self.file

    def __exit__(self, type, value, traceback):
        self.cleanup()


class TemporaryWorkingDirectory(TemporaryDirectory):
    """
    Creates a temporary directory and sets the cwd to that directory.
    Automatically reverts to previous cwd upon cleanup.
    Usage example:

        with TemporaryWorkingDirectory() as tmpdir:
            ...
    """

    def __enter__(self):
        self.old_wd = Path.cwd()
        _os.chdir(self.name)
        return super(TemporaryWorkingDirectory, self).__enter__()

    def __exit__(self, exc, value, tb):
        _os.chdir(self.old_wd)
        return super(TemporaryWorkingDirectory, self).__exit__(exc, value, tb)

"""Session bundle recording and replay for IPython.

A session bundle is a ZIP archive (``.ipybundle``) containing two files:

- ``metadata.json``: information about the session and how it was recorded.
- ``events.jsonl``: one JSON object per executed cell.

The module exposes helpers to save/load/validate bundles, to start/stop
recording on a running :class:`~IPython.core.interactiveshell.InteractiveShell`,
and to replay a bundle back into a shell.

FORMAT = "ipython-session-bundle"
FORMAT_VERSION = 1
BUNDLE_SUFFIX = ".ipybundle"
REDACTED = "<redacted>"


class SessionBundleValidationError(Exception):
    """Raised when a session bundle fails validation in strict mode."""

    def __init__(self, bundle_path: Path | str, errors: list[str]) -> None:
        self.bundle_path = Path(bundle_path)
        self.errors = list(errors)
        super().__init__(
            "Invalid session bundle at {}: {}".format(
                self.bundle_path, "; ".join(self.errors)

def _resolve_bundle_path(path: Path | str) -> Path:
    """Normalize a bundle path, appending ``.ipybundle`` if needed."""
    path = Path(path)
    if path.suffix != BUNDLE_SUFFIX:
        path = Path(str(path) + BUNDLE_SUFFIX)
    return path


def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def _redact_text(text: str, patterns: list[str]) -> str:
    for pattern in patterns:
        if pattern:
            text = text.replace(pattern, REDACTED)
    return text


def _redact_object(obj: Any, patterns: list[str]) -> Any:
    """Recursively redact literal patterns from strings in ``obj``."""
    if isinstance(obj, str):
        return _redact_text(obj, patterns)
    if isinstance(obj, list):
        return [_redact_object(item, patterns) for item in obj]
    if isinstance(obj, dict):
        return {
            key: _redact_object(value, patterns)
            for key, value in obj.items()
        }
    return obj


def _redact_json_line(line: str, patterns: list[str]) -> str:
    """Redact patterns from a serialized JSON line, including keys."""
    for pattern in patterns:
        if pattern:
            line = line.replace(pattern, REDACTED)
    return line

def save_session_bundle(
    path: Path | str,
    meta: dict[str, Any],
    events: list[dict[str, Any]],
    *,
    overwrite: bool = False,
) -> Path:
    """Write ``metadata.json`` and ``events.jsonl`` into a bundle at ``path``.

    The final bundle ``Path`` is returned.  If ``path`` does not already end
    with ``.ipybundle`` the suffix is appended.

    Parameters
    ----------
    path : path-like
        Destination for the bundle.
    meta : dict
        Metadata dictionary written as ``metadata.json``.
    events : list of dict
        Cell events written as ``events.jsonl``.
    overwrite : bool
        If ``True`` replace an existing bundle; otherwise raise
        :class:`FileExistsError` when the target already exists.
    """
    path = _resolve_bundle_path(path)
    if path.exists() and not overwrite:
        raise FileExistsError(f"Session bundle already exists: {path}")

    meta = dict(meta)
    if "event_count" not in meta:
        meta["event_count"] = len(events)

    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp_name = tempfile.mkstemp(
        prefix=path.name + ".", suffix=".tmp", dir=str(path.parent)
    )
    os.close(fd)
    tmp_path = Path(tmp_name)
    try:
        with zipfile.ZipFile(tmp_path, "w", zipfile.ZIP_DEFLATED) as zf:
            zf.writestr("metadata.json", json.dumps(meta, indent=2) + "\n")
            events_text = "".join(
                json.dumps(event, ensure_ascii=False) + "\n" for event in events
            )
            zf.writestr("events.jsonl", events_text)
        os.replace(tmp_path, path)
    finally:
        if tmp_path.exists():
            tmp_path.unlink()
    return path


def load_session_bundle(path: Path | str) -> tuple[dict[str, Any], list[dict[str, Any]]]:
    """Load a session bundle and return ``(metadata, events)``.

    This function does not execute any recorded code.
    """
    path = Path(path)
    with zipfile.ZipFile(path) as zf:
        with zf.open("metadata.json") as f:
            metadata = json.load(f)
        with zf.open("events.jsonl") as f:
            raw_events = f.read().decode("utf-8")
    events = []
    for line in raw_events.splitlines():
        line = line.strip()
        if line:
            events.append(json.loads(line))
    return metadata, events

def _is_int(value: Any) -> bool:
    return isinstance(value, int) and not isinstance(value, bool)


def _is_iso8601(value: Any) -> bool:
    if not isinstance(value, str):
        return False
    try:
        datetime.fromisoformat(value)
    except ValueError:
        return False
    return True


def _validate_metadata(meta: Any) -> list[str]:
    errors: list[str] = []
    if not isinstance(meta, dict):
        errors.append("metadata.json must contain a JSON object")
        return errors
    if meta.get("format") != FORMAT:
        errors.append(
            f"metadata['format'] must be {FORMAT!r}, got {meta.get('format')!r}"
        )
    if not _is_int(meta.get("format_version")) or meta["format_version"] < 1:
        errors.append(
            "metadata['format_version'] must be an integer >= 1, got "
            f"{meta.get('format_version')!r}"
        )
    if not _is_iso8601(meta.get("created_at")):
        errors.append(
            "metadata['created_at'] must be an ISO-8601 timestamp, got "
            f"{meta.get('created_at')!r}"
        )
    for field in ("ipython_version", "python_version", "platform"):
        if not isinstance(meta.get(field), str):
            errors.append(f"metadata[{field!r}] must be a string")
    redactions = meta.get("redactions")
    if not isinstance(redactions, list) or not all(
        isinstance(item, str) for item in redactions
    ):
        errors.append("metadata['redactions'] must be a list of strings")
    event_count = meta.get("event_count")
    if event_count is not None and not _is_int(event_count):
        errors.append("metadata['event_count'] must be an integer when present")
    return errors


def _validate_events(raw_events: str) -> tuple[list[dict[str, Any]], list[str]]:
    errors: list[str] = []
    events: list[dict[str, Any]] = []
    expected_seq = 1
    for lineno, line in enumerate(raw_events.splitlines(), start=1):
        line = line.strip()
        if not line:
            continue
        try:
            event = json.loads(line)
        except Exception as exc:
            errors.append(f"events.jsonl line {lineno} is not valid JSON: {exc}")
            continue
        if not isinstance(event, dict):
            errors.append(f"events.jsonl line {lineno} must be a JSON object")
            continue
        events.append(event)

        if event.get("type") != "cell":
            errors.append(f"event {lineno}: 'type' must be 'cell', got {event.get('type')!r}")
        seq = event.get("seq")
        if not _is_int(seq):
            errors.append(f"event {lineno}: 'seq' must be an integer")
        elif seq != expected_seq:
            errors.append(
                f"event {lineno}: 'seq' must be {expected_seq}, got {seq}"
            )
        expected_seq += 1

        if not _is_iso8601(event.get("recorded_at")):
            errors.append(
                f"event {lineno}: 'recorded_at' must be an ISO-8601 timestamp"
            )
        exec_count = event.get("execution_count")
        if exec_count is not None and not _is_int(exec_count):
            errors.append(f"event {lineno}: 'execution_count' must be int or null")
        if not isinstance(event.get("code"), str):
            errors.append(f"event {lineno}: 'code' must be a string")
        if not isinstance(event.get("success"), bool):
            errors.append(f"event {lineno}: 'success' must be a boolean")
        if not isinstance(event.get("stdout"), str):
            errors.append(f"event {lineno}: 'stdout' must be a string")
        if not isinstance(event.get("stderr"), str):
            errors.append(f"event {lineno}: 'stderr' must be a string")
        execute_result = event.get("execute_result")
        if not isinstance(execute_result, dict):
            errors.append(f"event {lineno}: 'execute_result' must be an object")
        elif execute_result and not isinstance(execute_result.get("text/plain"), str):
            errors.append(
                f"event {lineno}: non-empty 'execute_result' must include "
                "'text/plain' as a string"
            )
        success = event.get("success")
        if success is False:
            error = event.get("error")
            if not isinstance(error, dict):
                errors.append(
                    f"event {lineno}: failed events must include an 'error' object"
                )
            else:
                if not isinstance(error.get("ename"), str):
                    errors.append(f"event {lineno}: error['ename'] must be a string")
                if not isinstance(error.get("evalue"), str):
                    errors.append(f"event {lineno}: error['evalue'] must be a string")
                traceback = error.get("traceback")
                if not isinstance(traceback, list) or not traceback or not all(
                    isinstance(item, str) for item in traceback
                ):
                    errors.append(
                        f"event {lineno}: error['traceback'] must be a non-empty "
                        "list of strings"
                    )
    return events, errors


def validate_session_bundle(path: Path | str, *, strict: bool = True) -> list[str]:
    """Validate a session bundle, returning human-readable error strings.

    Parameters
    ----------
    path : path-like
        Path to the bundle to validate.
    strict : bool
        If ``True``, raise :class:`SessionBundleValidationError` when errors are
        found.  If ``False``, return the list of errors instead of raising.

    Returns
    -------
    list of str
        The validation errors found (empty when the bundle is valid).
    """
    path = Path(path)
    errors: list[str] = []
    if not path.exists():
        errors.append(f"Bundle does not exist: {path}")
    elif not path.is_file():
        errors.append(f"Bundle path is not a file: {path}")
    else:
        try:
            with zipfile.ZipFile(path) as zf:
                names = zf.namelist()
                metadata: Optional[dict[str, Any]] = None
                if "metadata.json" not in names:
                    errors.append("Bundle is missing metadata.json")
                else:
                    try:
                        metadata = json.loads(
                            zf.read("metadata.json").decode("utf-8")
                        )
                    except Exception as exc:
                        errors.append(f"metadata.json is not valid JSON: {exc}")
                    if isinstance(metadata, dict):
                        errors.extend(_validate_metadata(metadata))
                    elif metadata is not None:
                        errors.append("metadata.json must contain a JSON object")

                events: list[dict[str, Any]] = []
                if "events.jsonl" not in names:
                    errors.append("Bundle is missing events.jsonl")
                else:
                    raw_events = zf.read("events.jsonl").decode("utf-8")
                    events, event_errors = _validate_events(raw_events)
                    errors.extend(event_errors)
                    if isinstance(metadata, dict):
                        redactions = metadata.get("redactions")
                        if isinstance(redactions, list):
                            for pattern in redactions:
                                if isinstance(pattern, str) and pattern and pattern in raw_events:
                                    errors.append(
                                        f"Redacted string {pattern!r} appears in events.jsonl"
                                    )
                        event_count = metadata.get("event_count")
                        if _is_int(event_count) and event_count != len(events):
                            errors.append(
                                "metadata['event_count'] ({}) does not match the "
                                "number of events ({})".format(event_count, len(events))
                            )
        except zipfile.BadZipFile as exc:
            errors.append(f"Not a valid ZIP archive: {exc}")
        except Exception as exc:
            errors.append(f"Could not read bundle: {exc}")

    if strict and errors:
        raise SessionBundleValidationError(path, errors)
    return errors

class _TeeStream:
    """Write to an original stream while also appending to a StringIO buffer."""

    def __init__(self, original: Any, buffer: io.StringIO) -> None:
        self._original = original
        self._buffer = buffer

    def write(self, data: str) -> int:
        self._original.write(data)
        self._buffer.write(data)
        return len(data)

    def flush(self) -> None:
        return self._original.flush()

    def __getattr__(self, name: str) -> Any:
        return getattr(self._original, name)


class _CapturingDisplayHook:
    """A display hook used while recording that captures expression results.

    The hook also preserves IPython's usual ``_``/``Out`` bookkeeping by
    delegating those updates to the shell's normal displayhook.
    """

    def __init__(self, shell: Any, outputs: list[dict[str, Any]]) -> None:
        self.shell = shell
        self.outputs = outputs
        self.exec_result = None

    def __call__(self, result: Any = None) -> None:
        if result is None:
            return
        try:
            self.shell.displayhook.update_user_ns(result)
        except Exception:
            pass
        try:
            self.shell.displayhook.fill_exec_result(result)
        except Exception:
            pass
        try:
            format_dict, md_dict = self.shell.display_formatter.format(result)
        except Exception:
            format_dict, md_dict = {}, {}
        self.outputs.append({"data": format_dict, "metadata": md_dict})

    def quiet(self) -> bool:
        return False


class _SessionRecorder:
    """State used while a session bundle is being recorded."""

    def __init__(self, shell: Any, path: Path, redact: list[str]) -> None:
        self.shell = shell
        self.path = path
        self.redact = list(redact)
        self.events: list[dict[str, Any]] = []
        self.seq = 0
        self.stdout_buffer = io.StringIO()
        self.stderr_buffer = io.StringIO()
        self.capture_outputs: list[dict[str, Any]] = []
        self.orig_stdout = sys.stdout
        self.orig_stderr = sys.stderr
        self.orig_hook = shell.display_trap.hook
        self.stdout_stream = _TeeStream(self.orig_stdout, self.stdout_buffer)
        self.stderr_stream = _TeeStream(self.orig_stderr, self.stderr_buffer)
        self.display_hook = _CapturingDisplayHook(shell, self.capture_outputs)
        self.in_cell = False
        self.skip_next_post = False

    def _reset_buffers(self) -> None:
        self.stdout_buffer.seek(0)
        self.stdout_buffer.truncate()
        self.stderr_buffer.seek(0)
        self.stderr_buffer.truncate()
        self.capture_outputs.clear()

    def on_pre_run_cell(self, info: Any) -> None:
        self.in_cell = True
        self._reset_buffers()

    def _skip_magic_cell(self, raw_cell: str) -> bool:
        return raw_cell.lstrip().startswith("%session_bundle")

    def on_post_run_cell(self, result: Any) -> None:
        self.in_cell = False
        if self.skip_next_post:
            self.skip_next_post = False
            self._reset_buffers()
            return
        raw_cell = getattr(result.info, "raw_cell", "") or ""
        if not raw_cell or raw_cell.isspace() or self._skip_magic_cell(raw_cell):
            self._reset_buffers()
            return

        stdout = self.stdout_buffer.getvalue()
        stderr = self.stderr_buffer.getvalue()

        execute_result: dict[str, Any] = {}
        if self.capture_outputs:
            text_parts = []
            for output in self.capture_outputs:
                data = output.get("data", {})
                if isinstance(data.get("text/plain"), str):
                    text_parts.append(data["text/plain"])
            if text_parts:
                execute_result = {"text/plain": "\n".join(text_parts)}

        event: dict[str, Any] = {
            "type": "cell",
            "seq": self.seq + 1,
            "recorded_at": _now_iso(),
            "execution_count": getattr(result, "execution_count", None),
            "code": raw_cell,
            "success": bool(getattr(result, "success", True)),
            "stdout": stdout,
            "stderr": stderr,
            "execute_result": execute_result,
        }

        if not event["success"]:
            exception = getattr(result, "error_before_exec", None)
            if exception is None:
                exception = getattr(result, "error_in_exec", None)
            if exception is not None:
                try:
                    error = self.shell._format_exception_for_storage(exception)
                except Exception:
                    import traceback as tb

                    error = {
                        "ename": type(exception).__name__,
                        "evalue": str(exception),
                        "traceback": tb.format_exception(
                            type(exception), exception, exception.__traceback__
                        ),
                    }
                event["error"] = error

        event = _redact_object(event, self.redact)
        self.events.append(event)
        self.seq += 1
        self._reset_buffers()


def start_session_bundle(
    shell: Any,
    path: Path | str,
    *,
    overwrite: bool = False,
    redact: Optional[list[str]] = None,
) -> str:
    """Start recording a session bundle on a running ``InteractiveShell``.

    Returns the final bundle path as a string.  Raises if a recording is
    already active, or if ``path`` exists and ``overwrite`` is ``False``.
    """
    existing = getattr(shell, "_session_bundle_recorder", None)
    if existing is not None:
        raise RuntimeError("A session bundle is already being recorded")

    path = _resolve_bundle_path(path)
    redact = list(redact or [])

    meta = {
        "format": FORMAT,
        "format_version": FORMAT_VERSION,
        "created_at": _now_iso(),
        "ipython_version": ipython_version,
        "python_version": platform.python_version(),
        "platform": platform.platform(),
        "redactions": redact,
        "event_count": 0,
    }
    save_session_bundle(path, meta, [], overwrite=overwrite)

    recorder = _SessionRecorder(shell, path, redact)
    shell.events.register("pre_run_cell", recorder.on_pre_run_cell)
    shell.events.register("post_run_cell", recorder.on_post_run_cell)
    sys.stdout = recorder.stdout_stream
    sys.stderr = recorder.stderr_stream
    shell.display_trap.hook = recorder.display_hook
    shell._session_bundle_recorder = recorder

    # If we are being started from inside a cell (e.g. via %session_bundle),
    # the currently running cell should not be recorded.
    if getattr(shell.display_trap, "is_active", False):
        recorder.skip_next_post = True

    return str(path)


def stop_session_bundle(shell: Any) -> str:
    """Stop recording and finalize the session bundle.

    Returns the final bundle path as a string.  Raises if no recording is
    active.
    """
    recorder = getattr(shell, "_session_bundle_recorder", None)
    if recorder is None:
        raise RuntimeError("No session bundle recording is active")

    path = recorder.path
    try:
        meta = {
            "format": FORMAT,
            "format_version": FORMAT_VERSION,
            "created_at": _now_iso(),
            "ipython_version": ipython_version,
            "python_version": platform.python_version(),
            "platform": platform.platform(),
            "redactions": recorder.redact,
            "event_count": len(recorder.events),
        }
        save_session_bundle(path, meta, recorder.events, overwrite=True)
    finally:
        sys.stdout = recorder.orig_stdout
        sys.stderr = recorder.orig_stderr
        shell.display_trap.hook = recorder.orig_hook
        shell.events.unregister("pre_run_cell", recorder.on_pre_run_cell)
        shell.events.unregister("post_run_cell", recorder.on_post_run_cell)
        shell._session_bundle_recorder = None
    return str(path)


def session_bundle_status(shell: Any) -> dict[str, Any]:
    """Return the current session-bundle recording status."""
    recorder = getattr(shell, "_session_bundle_recorder", None)
    if recorder is None:
        return {"recording": False, "path": None}
    return {"recording": True, "path": str(recorder.path)}


@contextmanager
def session_bundle_recorder(
    shell: Any,
    path: Path | str,
    *,
    overwrite: bool = False,
    redact: Optional[list[str]] = None,
) -> Iterator[None]:
    """Context manager that records the session bundle for its duration."""
    start_session_bundle(shell, path, overwrite=overwrite, redact=redact)
    try:
        yield
    finally:
        stop_session_bundle(shell)

def replay_session_bundle(
    shell: Any,
    path: Path | str,
    *,
    stop_on_error: bool = True,
    store_history: bool = True,
) -> None:

class SessionBundleMagics(Magics):
    """Magics for recording an IPython session to a bundle file."""

    @magic_arguments.magic_arguments()
    @magic_arguments.argument(
        "action",
        choices=("start", "status", "stop"),
        nargs="?",
        default="status",

    @magic_arguments.argument(

    )

        default=None,
        metavar="PATTERN",
        help="Literal string to redact from recorded events (may be repeated).",
    )
    @line_magic
    def session_bundle(self, line=""):
        """Record the current IPython session to a session bundle.

        Usage
        -----
        %session_bundle start <path> [--overwrite] [--redact PATTERN]...
        %session_bundle status
        %session_bundle stop

        Examples
        --------
        ::

            In [1]: %session_bundle start my_session
            In [2]: 1 + 2
            In [3]: %session_bundle stop
        """
        args = magic_arguments.parse_argstring(self.session_bundle, line)
        shell = self.shell

        if args.action == "start":
            if args.path is None:
                raise UsageError("Missing path for %session_bundle start")
            return shell.start_session_bundle(
                args.path,
                overwrite=args.overwrite,
                redact=args.redact,
            )
        elif args.action == "stop":
            return shell.stop_session_bundle()
        else:
            return shell.session_bundle_status()

class DisplayHook(Configurable):
    """The custom IPython displayhook to replace sys.displayhook.

    This class does many things, but the basic idea is that it is a callable
    that gets called anytime user code returns a value.
    """

    shell = Instance('IPython.core.interactiveshell.InteractiveShellABC',
                     allow_none=True)
    exec_result = Instance('IPython.core.interactiveshell.ExecutionResult',
                           allow_none=True)
    cull_fraction = Float(0.2)

    def __init__(self, shell=None, cache_size=1000, **kwargs):
        super(DisplayHook, self).__init__(shell=shell, **kwargs)
        self._is_active = False
        cache_size_min = 3
        if cache_size <= 0:
            self.do_full_cache = 0
            cache_size = 0
        elif cache_size < cache_size_min:
            self.do_full_cache = 0
            cache_size = 0
            warn('caching was disabled (min value for cache size is %s).' %
                 cache_size_min,stacklevel=3)
        else:
            self.do_full_cache = 1

        # these are deliberately global:
        to_user_ns = {'_':self._,'__':self.__,'___':self.___}
        self.shell.user_ns.update(to_user_ns)

    def check_for_underscore(self):
        """Check if the user has set the '_' variable by hand."""
        # If something injected a '_' variable in __builtin__, delete
        # ipython's automatic one so we don't clobber that.  gettext() in
        # particular uses _, so we need to stay away from it.
        if '_' in builtin_mod.__dict__:
            try:
                user_value = self.shell.user_ns['_']
                if user_value is not self._:
                    return
                del self.shell.user_ns['_']
            except KeyError:
                pass

    def quiet(self):
        """Should we silence the display hook because of ';'?"""
        # do not print output if input ends in ';'

        try:
            cell = self.shell.history_manager.input_hist_parsed[-1]
        except IndexError:
            # some uses of ipshellembed may fail here
            return False

        return self.semicolon_at_end_of_expression(cell)

    @staticmethod
    def semicolon_at_end_of_expression(expression):
        """Parse Python expression and detects whether last token is ';'"""

        sio = _io.StringIO(expression)
        tokens = list(tokenize.generate_tokens(sio.readline))

        for token in reversed(tokens):
            if token[0] in (tokenize.ENDMARKER, tokenize.NL, tokenize.NEWLINE, tokenize.COMMENT):
                continue
            if (token[0] == tokenize.OP) and (token[1] == ';'):
                return True
            else:
                return False

class _CapturingDisplayHook:
    """A display hook used while recording that captures expression results.

    The hook also preserves IPython's usual ``_``/``Out`` bookkeeping by
    delegating those updates to the shell's normal displayhook.
    """

    def __init__(
        self,
        shell: Any,
        outputs: list[dict[str, Any]],
        recorder: "_SessionRecorder",
    ) -> None:
        self.shell = shell
        self.outputs = outputs
        self.recorder = recorder
        self.exec_result = None

    def __call__(self, result: Any = None) -> None:
        if result is None:
            return
        # Respect IPython's ``;`` output suppression using the current cell
        # rather than history (which may not contain store_history=False cells).
        try:
            if self.shell.displayhook.semicolon_at_end_of_expression(
                self.recorder.current_raw_cell
            ):
                return
        except Exception:
            pass
        try:
            self.shell.displayhook.update_user_ns(result)
        except Exception:
            pass
        try:
            self.shell.displayhook.fill_exec_result(result)
        except Exception:
            pass
        try:
            format_dict, md_dict = self.shell.display_formatter.format(result)
        except Exception:
            format_dict, md_dict = {}, {}
        self.outputs.append({"data": format_dict, "metadata": md_dict})

    def quiet(self) -> bool:
        return False


class _SessionRecorder:
    """State used while a session bundle is being recorded."""

    def __init__(self, shell: Any, path: Path, redact: list[str]) -> None:
        self.shell = shell
        self.path = path
        self.redact = list(redact)
        self.events: list[dict[str, Any]] = []
        self.seq = 0
        self.stdout_buffer = io.StringIO()
        self.stderr_buffer = io.StringIO()
        self.capture_outputs: list[dict[str, Any]] = []
        self.orig_stdout = sys.stdout
        self.orig_stderr = sys.stderr
        self.orig_hook = shell.display_trap.hook
        self.stdout_stream = _TeeStream(self.orig_stdout, self.stdout_buffer)
        self.stderr_stream = _TeeStream(self.orig_stderr, self.stderr_buffer)
        self.display_hook = _CapturingDisplayHook(shell, self.capture_outputs, self)
        self.in_cell = False
        self.skip_next_post = False
        self.current_raw_cell = ""

    def _reset_buffers(self) -> None:
        self.stdout_buffer.seek(0)
        self.stdout_buffer.truncate()
        self.stderr_buffer.seek(0)
        self.stderr_buffer.truncate()
        self.capture_outputs.clear()

    def on_pre_run_cell(self, info: Any) -> None:
        # If a flag is still set when a new cell starts, the cell that started
        # recording was silent (so no post_run_cell event was emitted).  Clear
        # it so we do not skip this next real cell.
        self.skip_next_post = False
        self.in_cell = True
        self.current_raw_cell = getattr(info, "raw_cell", "") or ""
        self._reset_buffers()

    def _skip_magic_cell(self, raw_cell: str) -> bool:
        return raw_cell.lstrip().startswith("%session_bundle")

    def on_post_run_cell(self, result: Any) -> None:
        self.in_cell = False
        if self.skip_next_post:
            self.skip_next_post = False
            self._reset_buffers()
            return
        raw_cell = getattr(result.info, "raw_cell", "") or ""
        if not raw_cell or raw_cell.isspace() or self._skip_magic_cell(raw_cell):
            self._reset_buffers()
            return

        stdout = self.stdout_buffer.getvalue()
        stderr = self.stderr_buffer.getvalue()

        execute_result: dict[str, Any] = {}
        if self.capture_outputs:
            text_parts = []
            for output in self.capture_outputs:
                data = output.get("data", {})
                if isinstance(data.get("text/plain"), str):
                    text_parts.append(data["text/plain"])
            if text_parts:
                execute_result = {"text/plain": "\n".join(text_parts)}

        event: dict[str, Any] = {
            "type": "cell",
            "seq": self.seq + 1,
            "recorded_at": _now_iso(),
            "execution_count": getattr(result, "execution_count", None),
            "code": raw_cell,
            "success": bool(getattr(result, "success", True)),
            "stdout": stdout,
            "stderr": stderr,
            "execute_result": execute_result,
        }

        if not event["success"]:
            exception = getattr(result, "error_before_exec", None)
            if exception is None:
                exception = getattr(result, "error_in_exec", None)
            if exception is not None:
                try:
                    error = self.shell._format_exception_for_storage(exception)
                except Exception:
                    import traceback as tb

                    error = {
                        "ename": type(exception).__name__,
                        "evalue": str(exception),
                        "traceback": tb.format_exception(
                            type(exception), exception, exception.__traceback__
                        ),
                    }
                event["error"] = error

        event = _redact_object(event, self.redact)
        self.events.append(event)
        self.seq += 1
        self.current_raw_cell = ""
        self._reset_buffers()

    shell: Any,

    overwrite: bool = False,
    redact: Optional[list[str]] = None,
) -> str:
    """Start recording a session bundle on a running ``InteractiveShell``.

    already active, or if ``path`` exists and ``overwrite`` is ``False``.

        "created_at": _now_iso(),
        "ipython_version": ipython_version,
        "python_version": platform.python_version(),
        "platform": platform.platform(),
        "redactions": redact,
        "event_count": 0,
    }
    save_session_bundle(path, meta, [], overwrite=overwrite)

    recorder = _SessionRecorder(shell, path, redact)

    shell._session_bundle_recorder = recorder

    # the currently running cell should not be recorded.
    if getattr(shell.display_trap, "is_active", False):
        recorder.skip_next_post = True

    return str(path)

def start_session_bundle(
    shell: Any,
    path: Path | str,
    *,
    overwrite: bool = False,
    redact: Optional[list[str]] = None,
) -> str:
    """Start recording a session bundle on a running ``InteractiveShell``.

    Returns the final bundle path as a string.  Raises if a recording is
    already active, or if ``path`` exists and ``overwrite`` is ``False``.
    """
    existing = getattr(shell, "_session_bundle_recorder", None)
    if existing is not None:
        raise RuntimeError("A session bundle is already being recorded")

    path = _resolve_bundle_path(path)
    if isinstance(redact, str):
        redact = [redact]
    redact = list(redact or [])
    created_at = _now_iso()

    meta = {
        "format": FORMAT,
        "format_version": FORMAT_VERSION,
        "created_at": created_at,
        "ipython_version": ipython_version,
        "python_version": platform.python_version(),
        "platform": platform.platform(),
        "redactions": redact,
        "event_count": 0,
    }
    save_session_bundle(path, meta, [], overwrite=overwrite)

    recorder = _SessionRecorder(shell, path, redact, created_at)
    shell.events.register("pre_run_cell", recorder.on_pre_run_cell)
    shell.events.register("post_run_cell", recorder.on_post_run_cell)
    sys.stdout = recorder.stdout_stream
    sys.stderr = recorder.stderr_stream
    shell.display_trap.hook = recorder.display_hook
    shell._session_bundle_recorder = recorder

    # If we are being started from inside a cell (e.g. via %session_bundle),
    # the currently running cell should not be recorded.
    if getattr(shell.display_trap, "is_active", False):
        recorder.skip_next_post = True

    return str(path)


def stop_session_bundle(shell: Any) -> str:
    """Stop recording and finalize the session bundle.

    Returns the final bundle path as a string.  Raises if no recording is
    active.
    """
    recorder = getattr(shell, "_session_bundle_recorder", None)
    if recorder is None:
        raise RuntimeError("No session bundle recording is active")

    path = recorder.path
    try:
        meta = {
            "format": FORMAT,
            "format_version": FORMAT_VERSION,
            "created_at": recorder.started_at,
            "ipython_version": ipython_version,
            "python_version": platform.python_version(),
            "platform": platform.platform(),
            "redactions": recorder.redact,
            "event_count": len(recorder.events),
        }
        save_session_bundle(path, meta, recorder.events, overwrite=True)
    finally:
        sys.stdout = recorder.orig_stdout
        sys.stderr = recorder.orig_stderr
        shell.display_trap.hook = recorder.orig_hook
        shell.events.unregister("pre_run_cell", recorder.on_pre_run_cell)
        shell.events.unregister("post_run_cell", recorder.on_post_run_cell)
        shell._session_bundle_recorder = None
    return str(path)


def session_bundle_status(shell: Any) -> dict[str, Any]:
    """Return the current session-bundle recording status."""
    recorder = getattr(shell, "_session_bundle_recorder", None)
    if recorder is None:
        return {"recording": False, "path": None}
    return {"recording": True, "path": str(recorder.path)}


@contextmanager
def session_bundle_recorder(
    shell: Any,
    path: Path | str,
    *,
    overwrite: bool = False,
    redact: Optional[list[str]] = None,
) -> Iterator[str]:
    """Context manager that records the session bundle for its duration.

    Yields the final bundle path.
    """
    bundle_path = start_session_bundle(
        shell, path, overwrite=overwrite, redact=redact
    )
    try:
        yield bundle_path
    finally:
        stop_session_bundle(shell)

def replay_session_bundle(
    shell: Any,
    path: Path | str,
    *,
    stop_on_error: bool = True,
    store_history: bool = True,
) -> None:
    """Re-execute recorded cells from a session bundle in ``shell``.

    Parameters
    ----------
    shell : InteractiveShell
        Shell in which to replay the recorded cells.
    path : path-like
        Path to the bundle to replay.
    stop_on_error : bool
        If ``True``, stop replay and raise the first error encountered.  If
        ``False``, continue replaying remaining cells after an error.
    store_history : bool
        If ``True``, each replayed cell advances ``shell.execution_count``.
        If ``False``, the execution count is left untouched.
    """
    _, events = load_session_bundle(path)
    for event in events:
        if not isinstance(event, dict):
            continue
        result = shell.run_cell(
            event.get("code", ""),
            store_history=store_history,
            silent=False,
        )
        if stop_on_error and not result.success:
            if result.error_before_exec is not None:
                raise result.error_before_exec
            if result.error_in_exec is not None:
                raise result.error_in_exec

def arg_split(commandline: str, posix: bool = False, strict: bool = True) -> list[str]:
    """Split a command line's arguments in a shell-like manner.

    This is a modified version of the standard library's shlex.split()
    function, but with a default of posix=False for splitting, so that quotes
    in inputs are respected.

    if strict=False, then any errors shlex.split would raise will result in the
    unparsed remainder being the last element of the list, rather than raising.
    This is because we sometimes use arg_split to parse things other than
    command-line args.
    """

    lex = shlex.shlex(commandline, posix=posix)
    lex.whitespace_split = True
    # Extract tokens, ensuring that things like leaving open quotes
    # does not cause this to raise.  This is important, because we
    # sometimes pass Python source through this (e.g. %timeit f(" ")),
    # and it shouldn't raise an exception.
    # It may be a bad idea to parse things that are not command-line args
    # through this function, but we do, so let's be safe about it.
    lex.commenters='' #fix for GH-1269
    tokens = []
    while True:
        try:
            tokens.append(next(lex))
        except StopIteration:
            break
        except ValueError:
            if strict:
                raise
            # couldn't parse, get remaining blob as last token
            tokens.append(lex.token)
            break

    return tokens

        default=None,
        metavar="PATTERN",
        help="Literal string to redact from recorded events (may be repeated).",

        """Record the current IPython session to a session bundle.

        Usage
        -----
        %session_bundle start <path> [--overwrite] [--redact PATTERN]...
        %session_bundle status
        %session_bundle stop

        Examples
        --------
        ::

            In [1]: %session_bundle start my_session
            In [2]: 1 + 2
            In [3]: %session_bundle stop

                redact=args.redact,
            )

"""Session bundle recording and replay for IPython.

A session bundle is a ZIP archive (``.ipybundle``) containing two files:

- ``metadata.json``: information about the session and how it was recorded.

The module exposes helpers to save/load/validate bundles, to start/stop

class SessionBundleValidationError(Exception):
    """Raised when a session bundle fails validation in strict mode."""

    def __init__(self, bundle_path: Path | str, errors: list[str]) -> None:
        self.bundle_path = Path(bundle_path)
        self.errors = list(errors)
        super().__init__(
            "Invalid session bundle at {}: {}".format(
                self.bundle_path, "; ".join(self.errors)
            )

# Path helpers
# -----------------------------------------------------------------------------
def _resolve_bundle_path(path: Path | str) -> Path:
    """Normalize a bundle path, appending ``.ipybundle`` if needed."""
    path = Path(path)
    if path.suffix != BUNDLE_SUFFIX:
        path = Path(str(path) + BUNDLE_SUFFIX)
    return path


def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def _redact_text(text: str, patterns: list[str]) -> str:
    for pattern in patterns:
        if pattern:
            text = text.replace(pattern, REDACTED)
    return text


def _redact_object(obj: Any, patterns: list[str]) -> Any:
    """Recursively redact literal patterns from strings in ``obj``."""
    if isinstance(obj, str):
        return _redact_text(obj, patterns)
    if isinstance(obj, list):
        return [_redact_object(item, patterns) for item in obj]
    if isinstance(obj, dict):
        return {
            key: _redact_object(value, patterns)
            for key, value in obj.items()
        }
    return obj

    events = list(events)
    meta = dict(meta)
