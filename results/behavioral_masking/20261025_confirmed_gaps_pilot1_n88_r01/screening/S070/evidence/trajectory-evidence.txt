type Tree<T> = InnerNode<T> | Leaf<T>;
type Leaf<T> = [T];
type InnerNode<T> = [T, Record<string, Tree<T>>];

export type MinimisedTree<T> = Tree<T> | Record<string, Tree<T>> | undefined;

const enableLegacyPaths = (version: number) => version < 1;

function traverse<T>(
  tree: MinimisedTree<T>,
  walker: (v: T, path: string[]) => void,
  version: number,
  origin: string[] = []
): void {
  if (!tree) {
    return;
  }

  const legacyPaths = enableLegacyPaths(version);
  if (!isArray(tree)) {
    forEach(tree, (subtree, key) =>
      traverse(subtree, walker, version, [
        ...origin,
        ...parsePath(key, legacyPaths),
      ])
    );
    return;
  }

  const [nodeValue, children] = tree;
  if (children) {
    forEach(children, (child, key) => {
      traverse(child, walker, version, [
        ...origin,
        ...parsePath(key, legacyPaths),
      ]);
    });
  }

  walker(nodeValue, origin);
}

export function applyValueAnnotations(
  plain: any,
  annotations: MinimisedTree<TypeAnnotation>,

  plain: any,
  annotations: ReferentialEqualityAnnotations,
  version: number
) {
  const legacyPaths = enableLegacyPaths(version);
  function apply(identicalPaths: string[], path: string) {
    const object = getDeep(plain, parsePath(path, legacyPaths));

    identicalPaths
      .map(path => parsePath(path, legacyPaths))
      .forEach(identicalObjectPath => {
        plain = setDeep(plain, identicalObjectPath, () => object);
      });
  }

  if (isArray(annotations)) {
    const [root, other] = annotations;
    root.forEach(identicalPath => {
      plain = setDeep(
        plain,
        parsePath(identicalPath, legacyPaths),
        () => plain
      );
    });

    if (other) {
      forEach(other, apply);
    }
  } else {
    forEach(annotations, apply);
  }

  return plain;
}

export const transformValue = (
  value: any,
  superJson: SuperJSON
): { value: any; type: TypeAnnotation } | undefined => {
  const applicableCompositeRule = findArr(compositeRules, rule =>
    rule.isApplicable(value, superJson)

  );

      type: applicableSimpleRule.annotation,
    };
  }

  return undefined;
};

export const untransformValue = (
  json: any,
  type: TypeAnnotation,
  superJson: SuperJSON
) => {
  if (isArray(type)) {
    switch (type[0]) {
      case 'symbol':
        return symbolRule.untransform(json, type, superJson);
      case 'class':
        return classRule.untransform(json, type, superJson);
      case 'custom':
        return customRule.untransform(json, type, superJson);
      case 'typed-array':
        return typedArrayRule.untransform(json, type, superJson);
      default:
        throw new Error('Unknown transformation: ' + type);
    }
  } else {
    const transformation = simpleRulesByAnnotation[type];
    if (!transformation) {
      throw new Error('Unknown transformation: ' + type);
    }

    return transformation.untransform(json as never, superJson);
  }

const includeCausesModes: IncludeCausesMode[] = ['none', 'direct', 'deep'];

const DEFAULT_MAX_CAUSE_DEPTH = 16;

const isRecord = (input: unknown): input is Record<string, unknown> =>
  typeof input === 'object' && input !== null && !Array.isArray(input);

export function normalizeErrorStackOptions(
  input: unknown
): NormalizedErrorStackOptions | undefined {
  if (!isRecord(input)) {
    return undefined;
  }

  let mode: ErrorStackMode =
    input.mode === 'string' || input.mode === 'frames' ? input.mode : 'off';

  let maxStackLines: number | undefined;
  if (input.maxStackLines !== undefined) {
    if (Number.isInteger(input.maxStackLines) && (input.maxStackLines as number) > 0) {
      maxStackLines = input.maxStackLines as number;
    } else {
      mode = 'off';
    }
  }

  const normalizeNewlines = input.normalizeNewlines === true;
  const trimLeadingWhitespace = input.trimLeadingWhitespace !== false;

  const stripInternalFrames: StripInternalFramesMode = stripInternalFramesModes.includes(
    input.stripInternalFrames as StripInternalFramesMode
  )
    ? (input.stripInternalFrames as StripInternalFramesMode)
    : 'none';

  const redactPaths: RedactPathsMode = redactPathsModes.includes(
    input.redactPaths as RedactPathsMode
  )
    ? (input.redactPaths as RedactPathsMode)
    : 'none';

  let includeCauses: IncludeCausesMode = includeCausesModes.includes(
    input.includeCauses as IncludeCausesMode
  )
    ? (input.includeCauses as IncludeCausesMode)
    : 'none';

  if (input.maxCauseDepth !== undefined && !Number.isInteger(input.maxCauseDepth)) {
    includeCauses = 'none';
  }

  const maxCauseDepth =
    input.maxCauseDepth !== undefined && Number.isInteger(input.maxCauseDepth)
      ? (input.maxCauseDepth as number)
      : DEFAULT_MAX_CAUSE_DEPTH;

  const sanitizeMessage = input.sanitizeMessage === true;

  let classFilter: string[] | undefined;
  if (Array.isArray(input.classFilter)) {
    classFilter = input.classFilter.filter(
      (item): item is string => typeof item === 'string'
    );
    if (classFilter.length === 0) {
      classFilter = undefined;
    }
  }

  return {
    mode,
    normalizeNewlines,
    trimLeadingWhitespace,
    maxStackLines,
    stripInternalFrames,
    redactPaths,
    includeCauses,
    maxCauseDepth,
    sanitizeMessage,
    classFilter,
  };
}

  v: Error,
  superJson: SuperJSON,
  kind: ErrorSerializationKind

  const config = superJson.errorStackOptions;
  const baseError: any = {
    name: v.name,
    message: v.message,
  };

  const classMatches = classFilterMatches(config, v.name);
  if (config?.sanitizeMessage && classMatches) {
    baseError.message = sanitizeMessage(v.message);
  }

  if (config) {
    if (config.includeCauses !== 'none' && 'cause' in v) {
      const cause = (v as any).cause;
      if (cause instanceof Error) {
        if (config.includeCauses === 'direct') {
          baseError.cause = cause;
          superJson.setErrorCauseDepthLeft(cause, 0);
        } else {

            v,
            config.maxCauseDepth
          );
          if (remaining > 0) {
            baseError.cause = cause;
            superJson.setErrorCauseDepthLeft(cause, remaining - 1);
          }
        }
      }
    }
  } else if ('cause' in v) {
    baseError.cause = (v as any).cause;
  }

  if (
    typeof AggregateError !== 'undefined' &&
    v instanceof AggregateError
  ) {
    baseError.errors = v.errors;
  }

      config?.mode === 'off' &&

  v: Error,

  kind: ErrorSerializationKind

  const config = superJson.errorStackOptions;
  const baseError: any = {

    message: v.message,

  const classMatches = classFilterMatches(config, v.name);
  if (config?.sanitizeMessage && classMatches) {
    baseError.message = sanitizeMessage(v.message);

      const cause = (v as any).cause;

          config.includeCauses === 'direct' ? 1 : config.maxCauseDepth;
        const remaining = superJson.getErrorCauseDepthLeft(v, maxDepth);
        if (remaining > 0) {
          baseError.cause = cause;
          superJson.setErrorCauseDepthLeft(cause, remaining - 1);
        }
      }

  const hasErrors = Array.isArray(v.errors) && v.name === 'AggregateError';

  } else {

  }
  error.name = v.name;
  error.stack = v.stack;

    (error as any).stackFrames = v.stackFrames;

const stripInternalFramesModes: StripInternalFramesMode[] = [
  'none',
  'node',
  'superjson',
  'node_and_superjson',
];

const redactPathsModes: RedactPathsMode[] = ['none', 'basename', 'strip_cwd'];

const includeCausesModes: IncludeCausesMode[] = ['none', 'direct', 'deep'];

const DEFAULT_MAX_CAUSE_DEPTH = 16;

const isRecord = (input: unknown): input is Record<string, unknown> =>
  typeof input === 'object' && input !== null && !Array.isArray(input);

export function normalizeErrorStackOptions(
  input: unknown
): NormalizedErrorStackOptions | undefined {
  if (!isRecord(input)) {
    return undefined;
  }

  let mode: ErrorStackMode =
    input.mode === 'string' || input.mode === 'frames' ? input.mode : 'off';

  let maxStackLines: number | undefined;
  if (input.maxStackLines !== undefined) {
    if (Number.isInteger(input.maxStackLines) && (input.maxStackLines as number) > 0) {
      maxStackLines = input.maxStackLines as number;
    } else {
      mode = 'off';
    }
  }

  const normalizeNewlines = input.normalizeNewlines === true;
  const trimLeadingWhitespace = input.trimLeadingWhitespace !== false;

  const stripInternalFrames: StripInternalFramesMode = stripInternalFramesModes.includes(
    input.stripInternalFrames as StripInternalFramesMode
  )
    ? (input.stripInternalFrames as StripInternalFramesMode)
    : 'none';

  const redactPaths: RedactPathsMode = redactPathsModes.includes(
    input.redactPaths as RedactPathsMode
  )
    ? (input.redactPaths as RedactPathsMode)
    : 'none';

  let includeCauses: IncludeCausesMode = includeCausesModes.includes(
    input.includeCauses as IncludeCausesMode
  )
    ? (input.includeCauses as IncludeCausesMode)
    : 'none';

  if (input.maxCauseDepth !== undefined && !Number.isInteger(input.maxCauseDepth)) {
    includeCauses = 'none';
  }

  const maxCauseDepth =
    input.maxCauseDepth !== undefined && Number.isInteger(input.maxCauseDepth)
      ? (input.maxCauseDepth as number)
      : DEFAULT_MAX_CAUSE_DEPTH;

  const sanitizeMessage = input.sanitizeMessage === true;

  let classFilter: string[] | undefined;
  if (typeof input.classFilter === 'string') {
    classFilter = input.classFilter.length > 0 ? [input.classFilter] : undefined;
  } else if (Array.isArray(input.classFilter)) {
    classFilter = input.classFilter.filter(
      (item): item is string => typeof item === 'string'
    );
    if (classFilter.length === 0) {
      classFilter = undefined;
    }
  }

  return {
    mode,
    normalizeNewlines,
    trimLeadingWhitespace,
    maxStackLines,
    stripInternalFrames,
    redactPaths,
    includeCauses,
    maxCauseDepth,
    sanitizeMessage,
    classFilter,
  };
}

export function normalizeStackNewlines(stack: string): string {
  return stack.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
}

const NODE_INTERNAL_REGEX = /node:internal/;
const SUPERJSON_INTERNAL_REGEX = /src\/transformer\.ts|src\/plainer\.ts|src\/index\.ts/;

const isInternalFrame = (
  frame: string,
  mode: StripInternalFramesMode
): boolean => {
  switch (mode) {
    case 'node':
      return NODE_INTERNAL_REGEX.test(frame);
    case 'superjson':
      return SUPERJSON_INTERNAL_REGEX.test(frame);
    case 'node_and_superjson':
      return (
        NODE_INTERNAL_REGEX.test(frame) || SUPERJSON_INTERNAL_REGEX.test(frame)
      );
    default:
      return false;
  }
};

const PATH_WITH_LOCATION_REGEX =
  /((?:[A-Za-z]:[\\/]|\.\.?[\\/]|\/|node:|file:\/\/\/|webpack:\/\/\/|[a-zA-Z][a-zA-Z0-9+.-]*:\/\/|[\w@.-]+[\\/])[^\s:()]*?)(?::\d+)(?::\d+)?/g;

const getPathBasename = (path: string): string => {
  const parts = path.split(/[\\/]/);
  return parts[parts.length - 1] || path;
};

const stripCwd = (path: string): string => {
  const cwd = process.cwd();
  if (path.startsWith(cwd + '/')) {
    return path.slice(cwd.length + 1);
  }
  if (path.startsWith(cwd + '\\')) {
    return path.slice(cwd.length + 1);
  }
  if (path === cwd) {
    return path;
  }
  return path;
};

const redactPath = (line: string, mode: RedactPathsMode): string => {
  if (mode === 'none') {
    return line;
  }

  return line.replace(PATH_WITH_LOCATION_REGEX, (match, path: string) => {
    const suffix = match.slice(path.length);
    const redactedPath =
      mode === 'basename' ? getPathBasename(path) : stripCwd(path);
    return redactedPath + suffix;
  });
};

const splitStack = (stack: string): { header: string; frames: string[] } => {
  const lines = stack.split('\n');
  return {
    header: lines[0],
    frames: lines.slice(1),
  };
};

const trimFrames = (frames: string[]): string[] =>
  frames.map(frame => frame.trimStart());

const stripInternalFrames = (
  frames: string[],
  mode: StripInternalFramesMode
): string[] => frames.filter(frame => !isInternalFrame(frame, mode));

const redactPaths = (
  frames: string[],
  mode: RedactPathsMode
): string[] => frames.map(frame => redactPath(frame, mode));

const limitStack = (
  header: string,
  frames: string[],
  maxStackLines: number | undefined
): { header: string; frames: string[] } => {
  if (maxStackLines === undefined) {
    return { header, frames };
  }

  if (maxStackLines <= 0) {
    return { header, frames: [] };
  }
  const allLines = [header, ...frames];
  const limited = allLines.slice(0, maxStackLines);
  return {
    header: limited[0],
    frames: limited.slice(1),
  };
};

export function processStackString(
  stack: string,
  options?: NormalizedErrorStackOptions | ErrorStackOptions
): string {
  const normalized = normalizeOptions(options);
  let processed = stack;
  if (normalized.normalizeNewlines) {
    processed = normalizeStackNewlines(processed);
  }

  const { header, frames } = splitStack(processed);

  let resultFrames = frames;
  if (normalized.trimLeadingWhitespace) {
    resultFrames = trimFrames(resultFrames);
  }
  resultFrames = redactPaths(resultFrames, normalized.redactPaths);
  const limited = limitStack(header, resultFrames, normalized.maxStackLines);
  resultFrames = stripInternalFrames(
    limited.frames,
    normalized.stripInternalFrames
  );

  return [limited.header, ...resultFrames].join('\n');
}

export function processStackFrames(
  stack: string,
  options?: NormalizedErrorStackOptions | ErrorStackOptions
): StackFrame[] {
  const normalized = normalizeOptions(options);
  let processed = stack;
  if (normalized.normalizeNewlines) {
    processed = normalizeStackNewlines(processed);
  }

  const { header, frames } = splitStack(processed);

  let resultFrames = frames;
  if (normalized.trimLeadingWhitespace) {
    resultFrames = trimFrames(resultFrames);
  }
  resultFrames = stripInternalFrames(
    resultFrames,
    normalized.stripInternalFrames
  );
  resultFrames = redactPaths(resultFrames, normalized.redactPaths);
  const limited = limitStack(header, resultFrames, normalized.maxStackLines);

  return [
    { raw: limited.header },
    ...limited.frames.map(frame => ({ raw: frame })),
  ];
}

## Recipes

SuperJSON by default only supports built-in data types to keep bundle-size as low as possible.
Here are some recipes you can use to extend to non-default data types.

Place them in some central utility file and make sure they're executed before any other `SuperJSON` calls.
In a Next.js project, `_app.ts` would be a good spot for that.

### `Decimal.js` / `Prisma.Decimal`

```ts
import { Decimal } from 'decimal.js';

SuperJSON.registerCustom<Decimal, string>(
  {
    isApplicable: (v): v is Decimal => Decimal.isDecimal(v),
    serialize: v => v.toJSON(),
    deserialize: v => new Decimal(v),
  },
  'decimal.js'
);
```
