+package expr_test

+import (
+	"testing"

+func runCode(t *testing.T, code string, env any) (any, error) {
+	t.Helper()
+	program, err := expr.Compile(code, expr.AllowUndefinedVariables())
+	if err != nil {
+		return nil, err
+	}
+	return expr.Run(program, env)
+}
+
+func mustRun(t *testing.T, code string, env any) any {
+	t.Helper()
+	out, err := runCode(t, code, env)
+	require.NoError(t, err)
+	return out
+}
+
+func mustErr(t *testing.T, code string, env any) error {
+	t.Helper()
+	_, err := runCode(t, code, env)
+	require.Error(t, err)
+	return err
+}
+
+func TestTryFunctionForm(t *testing.T) {
+	require.Equal(t, 2, mustRun(t, `try(1 + 1, 42)`, nil))
+	require.Equal(t, 42, mustRun(t, `try(int("abc"), 42)`, nil))
+
+	// Fallback must be lazily evaluated.
+	require.Equal(t, 42, mustRun(t, `try(int("abc"), 42)`, nil))
+	require.Equal(t, 1, mustRun(t, `try(1, int("abc"))`, nil))
+
+	// Exactly two arguments are required.
+	_, err := runCode(t, `try(1)`, nil)
+	require.Error(t, err)
+	_, err = runCode(t, `try(1, 2, 3)`, nil)
+	require.Error(t, err)
+}
+
+func TestTryBlockForm(t *testing.T) {
+	require.Equal(t, 2, mustRun(t, `try { 1 + 1 } catch { 42 }`, nil))
+	require.Equal(t, 42, mustRun(t, `try { int("abc") } catch { 42 }`, nil))
+
+	// A try block requires a catch or finally clause.
+	_, err := runCode(t, `try { 1 }`, nil)
+	require.Error(t, err)
+}

+func TestCatchBindsError(t *testing.T) {
+	out := mustRun(t, `try { int("abc") } catch e { errtype(e) }`, nil)
+	require.Equal(t, "conversion", out)
+
+	out = mustRun(t, `try { throw("boom") } catch e { e }`, nil)
+	require.Error(t, out.(error))
+	require.Contains(t, out.(error).Error(), "boom")
+}
+
+func TestCatchIsSubstring(t *testing.T) {
+	require.Equal(t, "matched", mustRun(t, `try { throw("error happened") } catch e is "happened" { "matched" }`, nil))
+
+	// Non-matching substring propagates the original error.
+	err := mustErr(t, `try { throw("error happened") } catch e is "missing" { "matched" }`, nil)
+	require.Contains(t, err.Error(), "error happened")
+
+	// The substring expression must be a string.
+	_, err = runCode(t, `try { throw("x") } catch e is 42 { "matched" }`, nil)
+	require.Error(t, err)
+}

+func TestFinally(t *testing.T) {
+	// Finally runs after a successful try and its result is discarded.
+	require.Equal(t, 1, mustRun(t, `try { 1 } finally { 2 }`, nil))
+
+	// Finally runs after a handled catch.
+	require.Equal(t, 2, mustRun(t, `try { int("a") } catch { 2 } finally { 3 }`, nil))
+
+	// Finally runs even when the error is uncaught, then the error propagates.
+	log := []string{}
+	env := map[string]any{
+		"log":     &log,
+		"fail":    func() any { panic("boom") },
+		"cleanup": func() any { log = append(log, "cleanup"); return true },
+	}
+	err := mustErr(t, `try { fail() } finally { cleanup() }`, env)
+	require.Contains(t, err.Error(), "boom")
+	require.Equal(t, []string{"cleanup"}, log)
+
+	// If the finally body throws, that error overrides any prior result.
+	err = mustErr(t, `try { 1 } finally { int("boom") }`, nil)
+	require.Contains(t, err.Error(), "invalid operation: int(boom)")
+
+	err = mustErr(t, `try { throw("original") } catch { 2 } finally { int("boom") }`, nil)
+	require.Contains(t, err.Error(), "invalid operation: int(boom)")
+}

+func TestThrow(t *testing.T) {
+	// throw(value) uses string conversion of the value as the message.
+	out := mustRun(t, `try { throw(42) } catch e { e }`, nil)
+	require.Error(t, out.(error))
+	require.Equal(t, "42", out.(error).Error())
+
+	out = mustRun(t, `try { throw("custom") } catch e { errtype(e) }`, nil)
+	require.Equal(t, "custom", out)
+
+	// Requires exactly one argument.
+	_, err := runCode(t, `throw()`, nil)
+	require.Error(t, err)
+	_, err = runCode(t, `throw(1, 2)`, nil)
+	require.Error(t, err)
+
+	// Uncaught throw propagates out of the expression.
+	err = mustErr(t, `throw("boom")`, nil)
+	require.Contains(t, err.Error(), "boom")
+
+	// throw(e) of a caught error still produces a custom error.
+	out = mustRun(t, `try { try { throw("a") } catch e { throw(e) } } catch e2 { errtype(e2) }`, nil)
+	require.Equal(t, "custom", out)
+}

+func TestRetry(t *testing.T) {
+	var counter int
+	env := map[string]any{
+		"counter": &counter,
+		"fail": func() any {
+			counter++
+			if counter < 3 {
+				panic("not yet")
+			}
+			return counter
+		},
+	}
+	require.Equal(t, 3, mustRun(t, `try { fail() } catch { retry }`, env))
+	require.Equal(t, 3, counter)
+
+	// More than three retries raises a retry-exhaustion error.
+	counter = 0
+	env["fail"] = func() any { counter++; panic("always") }
+	err := mustErr(t, `try { fail() } catch { retry }`, env)
+	require.Contains(t, err.Error(), "retry")
+
+	// The exhaustion error is classified as "retry" by an outer catch.
+	out := mustRun(t, `try { try { fail() } catch { retry } } catch e { errtype(e) }`, env)
+	require.Equal(t, "retry", out)
+
+	// Using retry outside a catch block raises a runtime error.
+	err = mustErr(t, `retry`, nil)
+	require.Contains(t, err.Error(), "retry")
+}
+
+func TestRetryLimit(t *testing.T) {
+	// Exactly three retries are allowed before exhaustion.
+	var counter int
+	env := map[string]any{
+		"counter": &counter,
+		"fail": func() any {
+			counter++
+			if counter < 4 {
+				panic("not yet")
+			}
+			return counter
+		},
+	}
+	require.Equal(t, 4, mustRun(t, `try { fail() } catch { retry }`, env))
+	require.Equal(t, 4, counter)
+
+	// A fourth retry is rejected with a distinct retry error.
+	counter = 0
+	env["fail"] = func() any {
+		counter++
+		if counter < 5 {
+			panic("not yet")
+		}
+		return counter
+	}
+	err := mustErr(t, `try { fail() } catch { retry }`, env)
+	require.Contains(t, err.Error(), "retry")
+	require.Equal(t, 4, counter)
+}

+func TestErrType(t *testing.T) {
+	require.Equal(t, "index", mustRun(t, `try { [1,2,3][5] } catch e { errtype(e) }`, nil))
+	require.Equal(t, "conversion", mustRun(t, `try { int("abc") } catch e { errtype(e) }`, nil))
+
+	env := map[string]any{"s": "x", "n": nil}
+	require.Equal(t, "type", mustRun(t, `try { 1 + s } catch e { errtype(e) }`, env))
+	require.Equal(t, "nil", mustRun(t, `try { n.foo } catch e { errtype(e) }`, env))
+	require.Equal(t, "custom", mustRun(t, `try { throw("x") } catch e { errtype(e) }`, nil))
+	require.Equal(t, "none", mustRun(t, `errtype(nil)`, nil))
+
+	// errtype requires exactly one argument.
+	_, err := runCode(t, `errtype()`, nil)
+	require.Error(t, err)
+	_, err = runCode(t, `errtype(1, 2)`, nil)
+	require.Error(t, err)
+}

+func TestTryCatchNested(t *testing.T) {
+	// An error thrown from a catch body propagates to the outer handler.
+	out := mustRun(t, `try { try { throw("a") } catch { throw("b") } } catch e { e }`, nil)
+	require.Error(t, out.(error))
+	require.Equal(t, "b", out.(error).Error())
+
+	// Inner finally still runs when the inner catch body throws.
+	log := []string{}
+	env := map[string]any{
+		"log":     &log,
+		"cleanup": func() any { log = append(log, "cleanup"); return true },
+	}
+	out = mustRun(t, `try { try { throw("a") } catch { throw("b") } finally { cleanup() } } catch e { errtype(e) }`, env)
+	require.Equal(t, "custom", out)
+	require.Equal(t, []string{"cleanup"}, log)
+}

+func TestTryCatchOptimized(t *testing.T) {
+	// The optimizer should still produce a correct program.
+	program, err := expr.Compile(`try { int("abc") } catch e { errtype(e) }`)
+	require.NoError(t, err)
+	out, err := expr.Run(program, nil)
+	require.NoError(t, err)
+	require.Equal(t, "conversion", out)
