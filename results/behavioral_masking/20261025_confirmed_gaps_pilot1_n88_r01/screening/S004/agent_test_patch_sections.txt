+import { attest, contextualize } from "@ark/attest"
+import {
+	jsonSchemaToType,
+	writeJsonSchemaInvalidRefMessage,
+	writeJsonSchemaUnresolvableRefMessage
+} from "@ark/json-schema"
+
+contextualize(() => {
+	it("dependentRequired", () => {
+		const t = jsonSchemaToType({
+			type: "object",
+			dependentRequired: { credit_card: ["billing_address"] }
+		})
+		attest(t.allows({ credit_card: "x", billing_address: "y" })).equals(true)
+		attest(t.allows({ credit_card: "x" })).equals(false)
+		attest(t.allows({})).equals(true)
+	})
+
+	it("dependentSchemas", () => {
+		const t = jsonSchemaToType({
+			type: "object",
+			dependentSchemas: { foo: { required: ["bar"] } }
+		})
+		attest(t.allows({ foo: 1, bar: 2 })).equals(true)
+		attest(t.allows({ foo: 1 })).equals(false)
+		attest(t.allows({})).equals(true)
+	})
+
+	it("dependencies (array and schema forms)", () => {
+		const tArray = jsonSchemaToType({
+			type: "object",
+			dependencies: { a: ["b"] }
+		})
+		attest(tArray.allows({ a: 1, b: 2 })).equals(true)
+		attest(tArray.allows({ a: 1 })).equals(false)
+
+		const tSchema = jsonSchemaToType({
+			type: "object",
+			dependencies: { a: { required: ["b"] } }
+		})
+		attest(tSchema.allows({ a: 1, b: 2 })).equals(true)
+		attest(tSchema.allows({ a: 1 })).equals(false)
+	})
+
+	it("local $ref", () => {
+		const t = jsonSchemaToType({
+			$defs: { Name: { type: "string", minLength: 2 } },
+			$ref: "#/$defs/Name"
+		})
+		attest(t.allows("ab")).equals(true)
+		attest(t.allows("a")).equals(false)
+		attest(t.allows(5)).equals(false)
+	})
+
+	it("invalid $ref format", () => {
+		attest(() =>
+			jsonSchemaToType({ $ref: "https://example.com/foo" } as never)
+		).throws(writeJsonSchemaInvalidRefMessage())
+	})
+
+	it("non-existent $ref", () => {
+		const ref = "#/$defs/NonExistentDef"
+		attest(() => jsonSchemaToType({ $defs: {}, $ref: ref } as never)).throws(
+			writeJsonSchemaUnresolvableRefMessage(ref)
+		)
+	})
+
+	it("recursive $ref", () => {
+		const t = jsonSchemaToType({
+			$defs: {
+				Node: {
+					anyOf: [
+						{ type: "null" },
+						{
+							type: "object",
+							properties: {
+								value: { type: "string" },
+								next: { $ref: "#/$defs/Node" }
+							}
+						}
+					]
+				}
+			},
+			$ref: "#/$defs/Node"
+		})
+		attest(t.allows(null)).equals(true)
+		attest(t.allows({ value: "a", next: null })).equals(true)
+		attest(t.allows({ value: "a", next: { value: "b", next: null } })).equals(
+			true
+		)
+		attest(t.allows({ value: 1 })).equals(false)
+		attest(t.allows({ value: "a", next: 1 })).equals(false)
+	})
+
+	it("$ref in dependentSchemas", () => {
+		const t = jsonSchemaToType({
+			type: "object",
+			$defs: {
+				Node: {
+					type: "object",
+					properties: { v: { type: "string" } },
+					required: ["v"]
+				}
+			},
+			dependentSchemas: { node: { $ref: "#/$defs/Node" } }
+		})
+		attest(t.allows({ node: true, v: "x" })).equals(true)
+		attest(t.allows({ node: true })).equals(false)
+		attest(t.allows({})).equals(true)
+	})
+
+	it("enum deep equality", () => {
+		const t = jsonSchemaToType({
+			enum: [{ a: 1 }, { b: [1, 2] }]
+		})
+		attest(t.allows({ a: 1 })).equals(true)
+		attest(t.allows({ b: [1, 2] })).equals(true)
+		attest(t.allows({ a: 2 })).equals(false)
+		attest(t.allows({ b: [2, 1] })).equals(false)
+	})
+
+	it("if/then/else", () => {
+		const t = jsonSchemaToType({
+			type: "number",
+			if: { type: "number", minimum: 0 },
+			then: { type: "number", maximum: 10 },
+			else: { type: "number", minimum: -10 }
+		})
+		attest(t.allows(0)).equals(true)
+		attest(t.allows(10)).equals(true)
+		attest(t.allows(-5)).equals(true)
+		attest(t.allows(11)).equals(false)
+		attest(t.allows(-11)).equals(false)
+	})
+
+	it("if alone and then/else without if are no-ops", () => {
+		attest(jsonSchemaToType({ if: { type: "string" } }).allows(1)).equals(true)
+		attest(jsonSchemaToType({ then: { type: "string" } }).allows(1)).equals(
+			true
+		)
+	})
+
+	it("boolean if/then/else schemas", () => {
+		const tTrue = jsonSchemaToType({
+			if: true,
+			then: { type: "string" }
+		})
+		attest(tTrue.allows("a")).equals(true)
+		attest(tTrue.allows(1)).equals(false)
+
+		const tFalse = jsonSchemaToType({
+			if: false,
+			else: { type: "number" }
+		})
+		attest(tFalse.allows(1)).equals(true)
+		attest(tFalse.allows("a")).equals(false)
+	})
+
+	it("implicit object type", () => {
+		const t = jsonSchemaToType({
+			properties: { foo: { type: "string" } },
+			required: ["foo"]
+		})
+		attest(t.allows({ foo: "a" })).equals(true)
+		attest(t.allows({})).equals(false)
+		attest(t.allows({ foo: 1 })).equals(false)
+	})
+
+	it("recursive $ref inside anyOf composition", () => {
+		const t = jsonSchemaToType({
+			$defs: {
+				Tree: {
+					anyOf: [
+						{ type: "null" },
+						{
+							allOf: [
+								{ type: "object", properties: { value: { type: "string" } } },
+								{
+									type: "object",
+									properties: {
+										children: {
+											type: "array",
+											items: { $ref: "#/$defs/Tree" }
+										}
+									}
+								}
+							]
+						}
+					]
+				}
+			},
+			$ref: "#/$defs/Tree"
+		})
+		attest(t.allows(null)).equals(true)
+		attest(t.allows({ value: "a", children: [] })).equals(true)
+		attest(t.allows({ value: "a", children: [null, { value: "b" }] })).equals(
+			true
+		)
+		attest(t.allows({ value: 1 })).equals(false)
+	})
+})
