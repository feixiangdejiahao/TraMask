	FaintSeq     = "2"
	ItalicSeq    = "3"
	UnderlineSeq = "4"
	BlinkSeq     = "5"
	ReverseSeq   = "7"
	CrossOutSeq  = "9"
	OverlineSeq  = "53"

// Style is a string that various rendering styles can be applied to.
type Style struct {
	profile Profile
	string
	styles []string
}

// String returns a new Style.
func String(s ...string) Style {
	return Style{
		profile: ANSI,
		string:  strings.Join(s, " "),
	}
}

func (t Style) String() string {
	return t.Styled(t.string)
}

// Styled renders s with all applied styles.
func (t Style) Styled(s string) string {
	if t.profile == Ascii {
		return s
	}
	if len(t.styles) == 0 {
		return s
	}

	seq := strings.Join(t.styles, ";")
	if seq == "" {
		return s
	}

	return fmt.Sprintf("%s%sm%s%sm", CSI, seq, s, CSI+ResetSeq)
}

// Foreground sets a foreground color.
func (t Style) Foreground(c Color) Style {
	if c != nil {
		t.styles = append(t.styles, c.Sequence(false))
	}
	return t
}

// Background sets a background color.
func (t Style) Background(c Color) Style {
	if c != nil {
		t.styles = append(t.styles, c.Sequence(true))
	}
	return t
}

// Bold enables bold rendering.

	o := &Output{

// Foreground and Background sequence codes.
const (

// Color is an interface implemented by all colors that can be converted to an
// ANSI sequence.
type Color interface {
	// Sequence returns the ANSI Sequence for the color.
	Sequence(bg bool) string
}

// NoColor is a nop for terminals that don't support colors.
type NoColor struct{}

func (c NoColor) String() string {
	return ""
}

// ANSIColor is a color (0-15) as defined by the ANSI Standard.
type ANSIColor int

func (c ANSIColor) String() string {
	return ansiHex[c]
}

// ANSI256Color is a color (16-255) as defined by the ANSI Standard.
type ANSI256Color int

func (c ANSI256Color) String() string {
	return ansiHex[c]
}

// RGBColor is a hex-encoded color, e.g. "#abcdef".
type RGBColor string

// ConvertToRGB converts a Color to a colorful.Color.
func ConvertToRGB(c Color) colorful.Color {
	var hex string
	switch v := c.(type) {
	case RGBColor:
		hex = string(v)
	case ANSIColor:
		hex = ansiHex[v]
	case ANSI256Color:
		hex = ansiHex[v]
	}

	ch, _ := colorful.Hex(hex)
	return ch
}

// Sequence returns the ANSI Sequence for the color.
func (c NoColor) Sequence(_ bool) string {
	return ""
}

// Sequence returns the ANSI Sequence for the color.
//
//nolint:mnd
func (c ANSIColor) Sequence(bg bool) string {
	col := int(c)
	bgMod := func(c int) int {
		if bg {
			return c + 10
		}
		return c
	}

	if col < 8 {
		return fmt.Sprintf("%d", bgMod(col)+30) //nolint:mnd
	}

func xTermColor(s string) (RGBColor, error) {
	if len(s) < 24 || len(s) > 25 {
		return RGBColor(""), ErrInvalidColor
	}

	switch {
	case strings.HasSuffix(s, string(BEL)):
		s = strings.TrimSuffix(s, string(BEL))
	case strings.HasSuffix(s, string(ESC)):
		s = strings.TrimSuffix(s, string(ESC))
	case strings.HasSuffix(s, ST):
		s = strings.TrimSuffix(s, ST)
	default:
		return RGBColor(""), ErrInvalidColor
	}

	s = s[4:]

	prefix := ";rgb:"
	if !strings.HasPrefix(s, prefix) {
		return RGBColor(""), ErrInvalidColor
	}
	s = strings.TrimPrefix(s, prefix)

	h := strings.Split(s, "/")
	hex := fmt.Sprintf("#%s%s%s", h[0][:2], h[1][:2], h[2][:2])
	return RGBColor(hex), nil
}

func ansi256ToANSIColor(c ANSI256Color) ANSIColor {
	var r int
	md := math.MaxFloat64

	h, _ := colorful.Hex(ansiHex[c])
	for i := 0; i <= 15; i++ {
		hb, _ := colorful.Hex(ansiHex[i])
		d := h.DistanceHSLuv(hb)

		if d < md {
			md = d
			r = i
		}
	}

	return ANSIColor(r)
}

//nolint:mnd
func hexToANSI256Color(c colorful.Color) ANSI256Color {
	v2ci := func(v float64) int {
		if v < 48 {
			return 0
		}
		if v < 115 {
			return 1
		}
		return int((v - 35) / 40)
	}

	// Calculate the nearest 0-based color index at 16..231
	r := v2ci(c.R * 255.0) // 0..5 each
	g := v2ci(c.G * 255.0)
	b := v2ci(c.B * 255.0)
	ci := 36*r + 6*g + b /* 0..215 */

	// Calculate the represented colors back from the index
	i2cv := [6]int{0, 0x5f, 0x87, 0xaf, 0xd7, 0xff}
	cr := i2cv[r] // r/g/b, 0..255 each
	cg := i2cv[g]
	cb := i2cv[b]

	// Calculate the nearest 0-based gray index at 232..255
	var grayIdx int
	average := (r + g + b) / 3
	if average > 238 {
		grayIdx = 23
	} else {
		grayIdx = (average - 3) / 10 // 0..23
	}
	gv := 8 + 10*grayIdx // same value for r/g/b, 0..255

	// Return the one which is nearer to the original input rgb value
	c2 := colorful.Color{R: float64(cr) / 255.0, G: float64(cg) / 255.0, B: float64(cb) / 255.0}
	g2 := colorful.Color{R: float64(gv) / 255.0, G: float64(gv) / 255.0, B: float64(gv) / 255.0}
	colorDist := c.DistanceHSLuv(c2)
	grayDist := c.DistanceHSLuv(g2)

	if colorDist <= grayDist {
		return ANSI256Color(16 + ci)
	}
	return ANSI256Color(232 + grayIdx)
}

// Profile is a color profile: Ascii, ANSI, ANSI256, or TrueColor.
type Profile int

const (
	// TrueColor, 24-bit color profile.
	TrueColor = Profile(iota)
	// ANSI256, 8-bit color profile.
	ANSI256
	// ANSI, 4-bit color profile.
	ANSI
	// Ascii, uncolored profile.
	Ascii //nolint:revive
)

// Name returns the profile name as a string.
func (p Profile) Name() string {
	switch p {
	case Ascii:
		return "Ascii"
	case ANSI:
		return "ANSI"
	case ANSI256:
		return "ANSI256"
	case TrueColor:
		return "TrueColor"
	}
	return "Unknown"
}

// String returns a new Style.
func (p Profile) String(s ...string) Style {
	return Style{
		profile: p,
		string:  strings.Join(s, " "),
	}
}

// Convert transforms a given Color to a Color supported within the Profile.
func (p Profile) Convert(c Color) Color {
	if p == Ascii {
		return NoColor{}
	}

	switch v := c.(type) {
	case ANSIColor:
		return v

	case ANSI256Color:
		if p == ANSI {
			return ansi256ToANSIColor(v)
		}
		return v

	case RGBColor:
		h, err := colorful.Hex(string(v))
		if err != nil {
			return nil
		}
		if p != TrueColor {
			ac := hexToANSI256Color(h)
			if p == ANSI {
				return ansi256ToANSIColor(ac)
			}
			return ac
		}
		return v
	}

	return c
}

// Color creates a Color from a string. Valid inputs are hex colors, as well as
// ANSI color codes (0-15, 16-255).
func (p Profile) Color(s string) Color {
	if len(s) == 0 {
		return nil
	}

	var c Color
	if strings.HasPrefix(s, "#") {
		c = RGBColor(s)
	} else {
		i, err := strconv.Atoi(s)
		if err != nil {
			return nil
		}

		if i < 16 { //nolint:mnd
			c = ANSIColor(i)
		} else {
			c = ANSI256Color(i)
		}
	}

	return p.Convert(c)
}

// FromColor creates a Color from a color.Color.
func (p Profile) FromColor(c color.Color) Color {
	col, _ := colorful.MakeColor(c)
	return p.Color(col.Hex())
}

// Package ansi provides ANSI escape sequence helpers.
package ansi

import (
	"strconv"
	"strings"
	"unicode/utf8"

	"github.com/rivo/uniseg"
)

// ANSI escape sequence constants.
const (
	ESC = "\x1b"
	CSI = ESC + "["
	OSC = ESC + "]"
	ST  = ESC + "\\"
	BEL = "\a"
)

// TokenType represents a token type.
type TokenType int

// Token types.
const (
	// TokenText is a plain text token.
	TokenText TokenType = iota
	// TokenSGR is an ANSI SGR (Select Graphic Rendition) sequence.
	TokenSGR
	// TokenReset is an ANSI SGR reset sequence.
	TokenReset
	// TokenHyperlinkOpen is an OSC 8 hyperlink open sequence.
	TokenHyperlinkOpen
	// TokenHyperlinkClose is an OSC 8 hyperlink close sequence.
	TokenHyperlinkClose
)

// Token represents an ANSI token.
type Token struct {
	Type TokenType
	Raw  string
	Text string
}

// TruncateOptions contains options for TruncateANSI.
type TruncateOptions struct {
	Tail           string
	PreserveResets bool
}

// scanEscapeSequence returns the length of the ANSI escape sequence starting at
// the beginning of s. s must start with an escape character.
func scanEscapeSequence(s string) int {
	if len(s) == 0 || s[0] != '\x1b' {
		return 0
	}
	if len(s) == 1 {
		return 1
	}

	switch s[1] {
	case '[': // CSI
		for i := 2; i < len(s); i++ {
			if s[i] >= 0x40 && s[i] <= 0x7e {
				return i + 1
			}
		}
		return len(s)
	case ']': // OSC
		for i := 2; i < len(s); i++ {
			if s[i] == '\a' {
				return i + 1
			}
			if s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '\\' {
				return i + 2
			}
		}
		return len(s)
	default:
		// ESC + intermediate bytes + final byte, e.g. ESC ( B.
		if s[1] >= 0x20 && s[1] <= 0x2f {
			for i := 2; i < len(s); i++ {
				if s[i] >= 0x30 && s[i] <= 0x7e {
					return i + 1
				}
			}
			return len(s)
		}
		// ESC + a single character.
		return 2
	}
}

// isSGR reports whether s is an SGR sequence.
func isSGR(s string) bool {
	return strings.HasPrefix(s, CSI) && strings.HasSuffix(s, "m")
}

// isResetSGR reports whether s is an SGR sequence that resets the style.
func isResetSGR(s string) bool {
	if !isSGR(s) {
		return false
	}
	params := strings.TrimSuffix(strings.TrimPrefix(s, CSI), "m")
	if params == "" {
		return true
	}
	for _, p := range strings.Split(params, ";") {
		if i, err := strconv.Atoi(p); err == nil && i == 0 {
			return true
		}
	}
	return false
}

// hyperlinkBody returns the payload of an OSC 8 sequence, without the
// terminating sequence.
func hyperlinkBody(s string) (string, bool) {
	if !strings.HasPrefix(s, OSC+"8;") {
		return "", false
	}
	body := strings.TrimSuffix(s, BEL)
	body = strings.TrimSuffix(body, ST)
	return body[len(OSC):], true
}

// isHyperlinkOpen reports whether s is an OSC 8 hyperlink open sequence.
func isHyperlinkOpen(s string) bool {
	body, ok := hyperlinkBody(s)
	if !ok {
		return false
	}
	parts := strings.SplitN(body, ";", 3)
	return len(parts) == 3 && parts[2] != ""
}

// isHyperlinkClose reports whether s is an OSC 8 hyperlink close sequence.
func isHyperlinkClose(s string) bool {
	body, ok := hyperlinkBody(s)
	if !ok {
		return false
	}
	parts := strings.SplitN(body, ";", 3)
	return len(parts) == 3 && parts[2] == ""
}

func Tokenize(s string) []Token {
	var tokens []Token
	var text strings.Builder

	flushText := func() {
		if text.Len() == 0 {
			return
		}
		t := text.String()
		tokens = append(tokens, Token{
			Type: TokenText,
			Raw:  t,
			Text: t,
		})
		text.Reset()
	}

	for i := 0; i < len(s); {
		if s[i] != '\x1b' {
			text.WriteByte(s[i])
			i++
			continue
		}

		n := scanEscapeSequence(s[i:])
		seq := s[i : i+n]
		flushText()

		switch {
		case isSGR(seq):
			if isResetSGR(seq) {
				tokens = append(tokens, Token{Type: TokenReset, Raw: seq})
			} else {
				tokens = append(tokens, Token{Type: TokenSGR, Raw: seq})
			}
		case isHyperlinkOpen(seq):
			tokens = append(tokens, Token{Type: TokenHyperlinkOpen, Raw: seq})
		case isHyperlinkClose(seq):
			tokens = append(tokens, Token{Type: TokenHyperlinkClose, Raw: seq})
		default:
			text.WriteString(seq)
		}
		i += n
	}
	flushText()

	return tokens
}

// StripANSI removes all ANSI escape sequences from s.
func StripANSI(s string) string {
	var b strings.Builder
	for i := 0; i < len(s); {
		if s[i] == '\x1b' {
			i += scanEscapeSequence(s[i:])
			continue
		}
		b.WriteByte(s[i])
		i++
	}
	return b.String()
}

// ANSIWidth returns the visible width of s, ignoring ANSI escape sequences.
func ANSIWidth(s string) int {
	return uniseg.StringWidth(StripANSI(s))
}

// HasANSI reports whether s contains ANSI escape sequences.
func HasANSI(s string) bool {
	return strings.ContainsRune(s, '\x1b')
}

// cutVisible truncates s to the given visible width without splitting ANSI
// escape sequences.
func cutVisible(s string, width int) string {
	if width <= 0 {
		return ""
	}

	var b strings.Builder
	w := 0
	for i := 0; i < len(s); {
		if s[i] == '\x1b' {
			n := scanEscapeSequence(s[i:])
			b.WriteString(s[i : i+n])
			i += n
			continue
		}

		r, size := utf8.DecodeRuneInString(s[i:])
		rw := uniseg.StringWidth(string(r))
		if w+rw > width {
			break
		}
		b.WriteString(s[i : i+size])
		w += rw
		i += size
	}
	return b.String()
}

// resetSGRStyle returns the SGR sequence that remains active after a reset
// sequence which also sets non-zero parameters, e.g. ESC[0;31m.
func resetSGRStyle(seq string) string {
	if !isSGR(seq) {
		return ""
	}
	params := strings.TrimSuffix(strings.TrimPrefix(seq, CSI), "m")
	if params == "" {
		return ""
	}

	kept := make([]string, 0, len(strings.Split(params, ";")))
	for _, p := range strings.Split(params, ";") {
		if i, err := strconv.Atoi(p); err != nil || i != 0 {
			kept = append(kept, p)
		}
	}
	if len(kept) == 0 {
		return ""
	}
	return CSI + strings.Join(kept, ";") + "m"
}

// TruncateANSI truncates s to the given visible width, preserving ANSI escape
// sequences and never splitting CSI/OSC sequences.
func TruncateANSI(s string, width int, opts TruncateOptions) string {
	if width <= 0 {
		return ""
	}
	if ANSIWidth(s) <= width {
		return s
	}

	tail := opts.Tail
	remaining := width - ANSIWidth(tail)
	if remaining < 0 {
		remaining = 0
	}

	var b strings.Builder
	var active []string
	hyperlinkOpen := false
	w := 0
	truncated := false

	for _, tok := range Tokenize(s) {
		if truncated {
			break
		}

		switch tok.Type {
		case TokenText:
			tw := ANSIWidth(tok.Text)
			if tw <= remaining-w {
				b.WriteString(tok.Text)
				w += tw
				if w >= remaining {
					truncated = true
				}
				continue
			}
			cut := cutVisible(tok.Text, remaining-w)
			b.WriteString(cut)
			w += ANSIWidth(cut)
			truncated = true
		case TokenSGR:
			active = append(active, tok.Raw)
			b.WriteString(tok.Raw)
		case TokenReset:
			b.WriteString(tok.Raw)
			if opts.PreserveResets && len(active) > 0 {
				for _, seq := range active {
					b.WriteString(seq)
				}
				continue
			}
			active = nil
			if style := resetSGRStyle(tok.Raw); style != "" {
				active = append(active, style)
			}
		case TokenHyperlinkOpen:
			b.WriteString(tok.Raw)
			hyperlinkOpen = true
		case TokenHyperlinkClose:
			b.WriteString(tok.Raw)
			hyperlinkOpen = false
		}
	}

	if tail != "" {
		b.WriteString(tail)
	}
	if hyperlinkOpen {
		b.WriteString(OSC + "8;;" + ST)
	}
	if len(active) > 0 {
		b.WriteString(CSI + "0m")
	}

	return b.String()
}

func templateFuncs(p Profile, preserveResets bool) template.FuncMap {
	if p == Ascii {
		return asciiTemplateFuncs()
	}

	mkStyle := func(s string) Style {
		st := p.String(s)
		if preserveResets {
			st = st.PreserveResets()
		}
		return st
	}

	o := &Output{
		Profile:        p,
		preserveResets: preserveResets,
	}

	return template.FuncMap{
		"Color": func(values ...interface{}) string {
			s := mkStyle(values[len(values)-1].(string))

		},

			s := mkStyle(values[len(values)-1].(string))

const (
	ResetSeq     = "0"
	BoldSeq      = "1"
	FaintSeq     = "2"
	ItalicSeq    = "3"
	UnderlineSeq = "4"
	BlinkSeq     = "5"
	ReverseSeq   = "7"
	CrossOutSeq  = "9"
	OverlineSeq  = "53"
)

// Style is a string that various rendering styles can be applied to.
type Style struct {
	profile Profile
	string
	styles         []string
	preserveResets bool
}

// String returns a new Style.
func String(s ...string) Style {
	return Style{
		profile: ANSI,
		string:  strings.Join(s, " "),
	}
}

func (t Style) String() string {
	return t.Styled(t.string)
}

// PreserveResets ensures the style is reapplied after any reset sequence in
// the rendered text.
func (t Style) PreserveResets() Style {
	t.preserveResets = true
	return t
}

// Styled renders s with all applied styles.
func (t Style) Styled(s string) string {
	if t.profile == Ascii {
		return s
	}
	if len(t.styles) == 0 {
		return s
	}

	seq := strings.Join(t.styles, ";")
	if seq == "" {
		return s
	}
	if t.preserveResets {
		s = reopenAfterResets(s, seq)
	}

	return fmt.Sprintf("%s%sm%s%sm", CSI, seq, s, CSI+ResetSeq)
}

// Truncate truncates the styled text to the given visible width.
func (t Style) Truncate(width int, opts TruncateOptions) string {
	if t.profile == Ascii {
		return ansi.TruncateANSI(ansi.StripANSI(t.string), width, TruncateOptions{})
	}
	if t.preserveResets {
		// Styled() already reopens the style after reset sequences, so
		// avoid doing it twice in TruncateANSI.
		opts.PreserveResets = false
		if len(t.styles) == 0 {
			return ansi.TruncateANSI(t.string, width, TruncateOptions{Tail: opts.Tail, PreserveResets: true})
		}
		return ansi.TruncateANSI(t.Styled(t.string), width, opts)
	}

	return ansi.TruncateANSI(t.Styled(t.string), width, opts)
}

// reopenAfterResets reapplies the given SGR sequence after every reset
// sequence in s.
func reopenAfterResets(s, seq string) string {
	var b strings.Builder
	for _, tok := range ansi.Tokenize(s) {
		if tok.Type == ansi.TokenReset {
			b.WriteString(tok.Raw)
			b.WriteString(CSI + seq + "m")
		} else {
			b.WriteString(tok.Raw)
		}
	}
	return b.String()
}

	unsafe         bool
	cache          bool

	bgColor        Color

type Environ interface {
	Environ() []string
	Getenv(string) string
}

type osEnviron struct{}

func (oe *osEnviron) Environ() []string {
	return os.Environ()
}

func (oe *osEnviron) Getenv(key string) string {
	return os.Getenv(key)
}

// DefaultOutput returns the default global output.
func DefaultOutput() *Output {
	return output
}

// SetDefaultOutput sets the default global output.
func SetDefaultOutput(o *Output) {
	output = o
}

// NewOutput returns a new Output for the given writer.
func NewOutput(w io.Writer, opts ...OutputOption) *Output {
	o := &Output{
		w:       w,
		environ: &osEnviron{},
		Profile: -1,
		fgSync:  &sync.Once{},
		fgColor: NoColor{},
		bgSync:  &sync.Once{},
		bgColor: NoColor{},
	}

	if o.w == nil {
		o.w = os.Stdout
	}
	for _, opt := range opts {
		opt(o)
	}
	if o.Profile < 0 {
		o.Profile = o.EnvColorProfile()
	}

	return o
}

	return func(o *Output) {
		o.environ = environ

	return func(o *Output) {
		o.Profile = profile
	}
}

func WithColorCache(v bool) OutputOption {
	return func(o *Output) {
		o.cache = v

		_ = o.ForegroundColor()
		_ = o.BackgroundColor()
	}

	}

// hyperlinkBody returns the payload of an OSC 8 sequence, without the
// terminating sequence.
func hyperlinkBody(s string) (string, bool) {
	if !strings.HasPrefix(s, OSC+"8;") {
		return "", false
	}
	body := strings.TrimSuffix(s, BEL)
	body = strings.TrimSuffix(body, ST)
	return body[len(OSC):], true
}

// isHyperlinkOpen reports whether s is an OSC 8 hyperlink open sequence.
func isHyperlinkOpen(s string) bool {
	body, ok := hyperlinkBody(s)
	if !ok {
		return false
	}
	parts := strings.SplitN(body, ";", 3)
	return len(parts) == 3 && parts[2] != ""
}

// isHyperlinkClose reports whether s is an OSC 8 hyperlink close sequence.
func isHyperlinkClose(s string) bool {
	body, ok := hyperlinkBody(s)
	if !ok {
		return false
	}
	parts := strings.SplitN(body, ";", 3)
	return len(parts) == 3 && parts[2] == ""
}

// Tokenize splits s into ANSI tokens.
func Tokenize(s string) []Token {
	var tokens []Token
	var text strings.Builder

	flushText := func() {
		if text.Len() == 0 {
			return
		}
		t := text.String()
		tokens = append(tokens, Token{
			Type: TokenText,
			Raw:  t,
			Text: t,
		})
		text.Reset()
	}

	for i := 0; i < len(s); {
		if s[i] != '\x1b' {
			text.WriteByte(s[i])
			i++
			continue
		}

		n := scanEscapeSequence(s[i:])
		seq := s[i : i+n]
		flushText()

		switch {
		case isSGR(seq):
			if isResetSGR(seq) {
				tokens = append(tokens, Token{Type: TokenReset, Raw: seq})
			} else {
				tokens = append(tokens, Token{Type: TokenSGR, Raw: seq})
			}
		case isHyperlinkOpen(seq):
			tokens = append(tokens, Token{Type: TokenHyperlinkOpen, Raw: seq})
		case isHyperlinkClose(seq):
			tokens = append(tokens, Token{Type: TokenHyperlinkClose, Raw: seq})
		default:
			text.WriteString(seq)
		}
		i += n
	}
	flushText()

	return tokens
}

// StripANSI removes all ANSI escape sequences from s.
func StripANSI(s string) string {
	var b strings.Builder
	for i := 0; i < len(s); {
		if s[i] == '\x1b' {
			i += scanEscapeSequence(s[i:])
			continue
		}
		b.WriteByte(s[i])
		i++
	}
	return b.String()
}

// ANSIWidth returns the visible width of s, ignoring ANSI escape sequences.
func ANSIWidth(s string) int {
	return uniseg.StringWidth(StripANSI(s))
}

// HasANSI reports whether s contains ANSI escape sequences.
func HasANSI(s string) bool {
	return strings.ContainsRune(s, '\x1b')
}

// cutVisible truncates s to the given visible width without splitting ANSI
// escape sequences.
func cutVisible(s string, width int) string {
	var b strings.Builder
	w := 0
	for i := 0; i < len(s); {
		if s[i] == '\x1b' {
			n := scanEscapeSequence(s[i:])
			b.WriteString(s[i : i+n])
			i += n
			continue
		}
		if w >= width {
			break
		}

		r, size := utf8.DecodeRuneInString(s[i:])
		rw := uniseg.StringWidth(string(r))
		if w+rw > width {
			break
		}
		b.WriteString(s[i : i+size])
		w += rw
		i += size
	}
	return b.String()
}

// resetSGRStyle returns the SGR sequence that remains active after a reset
// sequence which also sets non-zero parameters, e.g. ESC[0;31m.
func resetSGRStyle(seq string) string {
	if !isSGR(seq) {
		return ""
	}
	params := strings.TrimSuffix(strings.TrimPrefix(seq, CSI), "m")
	if params == "" {
		return ""
	}

	kept := make([]string, 0, len(strings.Split(params, ";")))
	for _, p := range strings.Split(params, ";") {
		if i, err := strconv.Atoi(p); err != nil || i != 0 {
			kept = append(kept, p)
		}
	}
	if len(kept) == 0 {
		return ""
	}
	return CSI + strings.Join(kept, ";") + "m"
}

// TruncateANSI truncates s to the given visible width, preserving ANSI escape
// sequences and never splitting CSI/OSC sequences.
func TruncateANSI(s string, width int, opts TruncateOptions) string {
	if width <= 0 {
		return ""
	}
	if ANSIWidth(s) <= width {
		return s
	}

	tail := opts.Tail
	remaining := width - ANSIWidth(tail)
	if remaining < 0 {
		remaining = 0
	}

	var b strings.Builder
	var active []string
	hyperlinkOpen := false
	w := 0
	truncated := false

	for _, tok := range Tokenize(s) {
		if truncated {
			break
		}

		switch tok.Type {
		case TokenText:
			tw := ANSIWidth(tok.Text)
			if tw <= remaining-w {
				b.WriteString(tok.Text)
				w += tw
				continue
			}
			cut := cutVisible(tok.Text, remaining-w)
			b.WriteString(cut)
			w += ANSIWidth(cut)
			truncated = true
		case TokenSGR:
			active = append(active, tok.Raw)
			b.WriteString(tok.Raw)
		case TokenReset:
			b.WriteString(tok.Raw)
			if opts.PreserveResets && len(active) > 0 {
				for _, seq := range active {
					b.WriteString(seq)
				}
				continue
			}
			active = nil
			if style := resetSGRStyle(tok.Raw); style != "" {
				active = append(active, style)
			}
		case TokenHyperlinkOpen:
			b.WriteString(tok.Raw)
			hyperlinkOpen = true
		case TokenHyperlinkClose:
			b.WriteString(tok.Raw)
			hyperlinkOpen = false
		}
	}

	if tail != "" {
		b.WriteString(tail)
	}
	if hyperlinkOpen {
		b.WriteString(OSC + "8;;" + ST)
	}
	if len(active) > 0 {
		b.WriteString(CSI + "0m")
	}

	return b.String()
}

// scanEscapeSequence returns the length of the ANSI escape sequence starting at
// the beginning of s. s must start with an escape character.
func scanEscapeSequence(s string) int {
	if len(s) == 0 || s[0] != '\x1b' {
		return 0
	}
	if len(s) == 1 {
		return 1
	}

	switch s[1] {
	case '[': // CSI
		for i := 2; i < len(s); i++ {
			if s[i] >= 0x40 && s[i] <= 0x7e {
				return i + 1
			}
		}
		return len(s)
	case ']': // OSC
		for i := 2; i < len(s); i++ {
			if s[i] == '\a' {
				return i + 1
			}
			if s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '\\' {
				return i + 2
			}
		}
		return len(s)
	default:
		// ESC + intermediate bytes + final byte, e.g. ESC ( B.
		if s[1] >= 0x20 && s[1] <= 0x2f {
			for i := 2; i < len(s); i++ {
				if s[i] >= 0x30 && s[i] <= 0x7e {
					return i + 1
				}
			}
			return len(s)
		}
		// ESC + a single character.
		return 2
	}
}

// isSGR reports whether s is an SGR sequence.
func isSGR(s string) bool {
	return strings.HasPrefix(s, CSI) && strings.HasSuffix(s, "m")
}

// isResetSGR reports whether s is an SGR sequence that resets the style.
func isResetSGR(s string) bool {
	if !isSGR(s) {
		return false
	}
	params := strings.TrimSuffix(strings.TrimPrefix(s, CSI), "m")
	if params == "" {
		return true
	}
	for _, p := range strings.Split(params, ";") {
		if i, err := strconv.Atoi(p); err == nil && i == 0 {
			return true
		}
	}
	return false
}
